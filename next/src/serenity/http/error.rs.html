336#[derive(Debug)]
337#[non_exhaustive]
338pub enum HttpError {
339    /// When a non-successful status code was received for a request.
340    UnsuccessfulRequest(ErrorResponse),
341    /// When the decoding of a ratelimit header could not be properly decoded into an `i64` or
342    /// `f64`.
343    RateLimitI64F64,
344    /// When the decoding of a ratelimit header could not be properly decoded from UTF-8.
345    RateLimitUtf8,
346    /// When parsing a Webhook fails due to invalid input.
347    InvalidWebhook,
348    /// Header value contains invalid input.
349    InvalidHeader(InvalidHeaderValue),
350    /// Reqwest's Error contain information on why sending a request failed.
351    Request(ReqwestError),
352    /// When an application id was expected but missing.
353    ApplicationIdMissing,
354}
355
356impl HttpError {
357    /// Returns true when the error is caused by an unsuccessful request
358    #[must_use]
359    pub fn is_unsuccessful_request(&self) -> bool {
360        matches!(self, Self::UnsuccessfulRequest(_))
361    }
362
363    /// Returns true when the error is caused by an invalid header
364    #[must_use]
365    pub fn is_invalid_header(&self) -> bool {
366        matches!(self, Self::InvalidHeader(_))
367    }
368
369    /// Returns the status code if the error is an unsuccessful request
370    #[must_use]
371    pub fn status_code(&self) -> Option<StatusCode> {
372        match self {
373            Self::UnsuccessfulRequest(res) => Some(res.status_code),
374            _ => None,
375        }
376    }
377}
378
379impl From<ErrorResponse> for HttpError {
380    fn from(error: ErrorResponse) -> Self {

382    }
383}

397impl fmt::Display for HttpError {
398    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
399        match self {
400            Self::UnsuccessfulRequest(e) => {
401                f.write_str(&e.error.message)?;
402
403                // Return the HTTP response code with a JSON decode error
404                if e.error.code == JsonErrorCode::Unknown(1) {
405                    f.write_str(" (")?;
406                    // Return the verbose code and meaning in the alternate (`#`) format
407                    if f.alternate() {
408                        f.write_str(
409                            &HttpResponseCode::from(e.status_code).to_meaning().map_or_else(
410                                || {
411                                    let mut status_code = e.status_code.to_string();
412                                    status_code.make_ascii_lowercase();
413                                    Cow::Owned(status_code)
414                                },
415                                Cow::Borrowed,
416                            ),
417                        )?;
418                    } else {
419                        f.write_str("HTTP ")?;
420                        f.write_str(e.status_code.as_str())?;
421                    }
422                    f.write_str(")")?;
423
424                    // Append the JSON decode errors
425                    let mut errors_iter = e.error.errors.iter();
426                    if let Some(error) = errors_iter.next()
427                        && !error.message.is_empty()
428                    {
429                        f.write_str(": ")?;
430                        f.write_str(&error.message)?;
431                        // Include the JSON decode error source error in the alternate (`#`) format
432                        if f.alternate()
433                            && let Some(error) = errors_iter.next()
434                            && !error.message.is_empty()
435                        {
436                            f.write_str(": ")?;
437                            f.write_str(&error.message)?;
438                        }
439                    }
440                    return Ok(());
441                }
442
443                // Put Discord's human readable error explanations in parentheses
444                let mut errors_iter = e.error.errors.iter();
445                if let Some(error) = errors_iter.next() {
446                    f.write_str(" (")?;
447                    f.write_str(&error.path)?;
448                    f.write_str(": ")?;
449                    f.write_str(&error.message)?;
450                    for error in errors_iter {
451                        f.write_str(", ")?;
452                        f.write_str(&error.path)?;
453                        f.write_str(": ")?;
454                        f.write_str(&error.message)?;
455                    }
456                    f.write_str(")")?;
457                }
458
459                Ok(())
460            },
461            Self::RateLimitI64F64 => f.write_str("Error decoding a header into an i64 or f64"),
462            Self::RateLimitUtf8 => f.write_str("Error decoding a header from UTF-8"),
463            Self::InvalidWebhook => f.write_str("Provided URL is not a valid webhook."),
464            Self::InvalidHeader(_) => f.write_str("Provided value is an invalid header value."),
465            Self::Request(_) => f.write_str("Error while sending HTTP request."),
466            Self::ApplicationIdMissing => f.write_str("Application id was expected but missing."),
467        }
468    }
469}

513#[expect(clippy::missing_errors_doc)]
514pub fn deserialize_errors<'de, D: Deserializer<'de>>(
515    deserializer: D,
516) -> StdResult<FixedArray<DiscordJsonSingleError>, D::Error> {
517    let ErrorValue::Recurse(map) = ErrorValue::deserialize(deserializer)? else {
518        return Ok(FixedArray::new());
519    };
520
521    let mut errors = Vec::new();
522    let mut path = Vec::new();
523    loop_errors(map, &mut errors, &mut path).map_err(D::Error::custom)?;
524
525    Ok(errors.trunc_into())
526}

528fn make_error(
529    errors_to_process: Vec<RawDiscordJsonSingleError>,
530    errors: &mut Vec<DiscordJsonSingleError>,
531    path: &[&str],
532) {
533    let joined_path = Arc::from(path.join("."));
534    errors.extend(errors_to_process.into_iter().map(|raw| DiscordJsonSingleError {
535        code: raw.code,
536        message: raw.message,
537        path: Arc::clone(&joined_path),
538    }));
539}

541#[derive(serde::Deserialize)]
542#[serde(untagged)]
543enum ErrorValue<'a> {
544    Base(Vec<RawDiscordJsonSingleError>),
545    #[serde(borrow)]
546    Recurse(HashMap<&'a str, ErrorValue<'a>>),
547}
548
549fn loop_errors<'a>(
550    value: HashMap<&'a str, ErrorValue<'a>>,
551    errors: &mut Vec<DiscordJsonSingleError>,
552    path: &mut Vec<&'a str>,
553) -> Result<(), &'static str> {
554    for (key, value) in value {
555        if key == "_errors" {
556            let ErrorValue::Base(value) = value else { return Err("expected array, found map") };
557            make_error(value, errors, path);
558        } else {
559            let ErrorValue::Recurse(value) = value else { return Err("expected map, found array") };
560
561            path.push(key);
562            loop_errors(value, errors, path)?;
563            path.pop();
564        }
565    }
566
567    Ok(())
568}
