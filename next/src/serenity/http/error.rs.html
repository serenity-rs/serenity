281    pub status_code: StatusCode,
282    pub url: FixedString<u16>,

286impl ErrorResponse {
287    // We need a freestanding from-function since we cannot implement an async From-trait.
288    pub async fn from_response(r: Response, method: Method) -> Self {
289        ErrorResponse {
290            method,
291            status_code: r.status(),
292            url: FixedString::from_str_trunc(r.url().as_str()),
293            error: r.json().await.unwrap_or_else(|e| DiscordJsonError {
294                code: JsonErrorCode::Unknown(1),
295                errors: FixedArray::empty(),
296                message: format!("[Serenity] Could not decode json when receiving error response from discord:, {e}").trunc_into(),
297            }),
298        }
299    }
300}

302#[derive(Debug)]
303#[non_exhaustive]
304pub enum HttpError {
305    /// When a non-successful status code was received for a request.
306    UnsuccessfulRequest(ErrorResponse),
307    /// When the decoding of a ratelimit header could not be properly decoded into an `i64` or
308    /// `f64`.
309    RateLimitI64F64,
310    /// When the decoding of a ratelimit header could not be properly decoded from UTF-8.
311    RateLimitUtf8,
312    /// When parsing a Webhook fails due to invalid input.
313    InvalidWebhook,
314    /// Header value contains invalid input.
315    InvalidHeader(InvalidHeaderValue),
316    /// Reqwest's Error contain information on why sending a request failed.
317    Request(ReqwestError),
318    /// When an application id was expected but missing.
319    ApplicationIdMissing,
320}
321
322impl HttpError {
323    /// Returns true when the error is caused by an unsuccessful request
324    #[must_use]
325    pub fn is_unsuccessful_request(&self) -> bool {
326        matches!(self, Self::UnsuccessfulRequest(_))
327    }
328
329    /// Returns true when the error is caused by an invalid header
330    #[must_use]
331    pub fn is_invalid_header(&self) -> bool {
332        matches!(self, Self::InvalidHeader(_))
333    }
334
335    /// Returns the status code if the error is an unsuccessful request
336    #[must_use]
337    pub fn status_code(&self) -> Option<StatusCode> {
338        match self {
339            Self::UnsuccessfulRequest(res) => Some(res.status_code),
340            _ => None,
341        }
342    }
343}

345impl From<ErrorResponse> for HttpError {
346    fn from(error: ErrorResponse) -> Self {
347        Self::UnsuccessfulRequest(error)
348    }
349}

363impl fmt::Display for HttpError {
364    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
365        match self {
366            Self::UnsuccessfulRequest(e) => {
367                f.write_str(&e.error.message)?;
368
369                // Put Discord's human readable error explanations in parentheses
370                let mut errors_iter = e.error.errors.iter();
371                if let Some(error) = errors_iter.next() {
372                    f.write_str(" (")?;
373                    f.write_str(&error.path)?;
374                    f.write_str(": ")?;
375                    f.write_str(&error.message)?;
376                    for error in errors_iter {
377                        f.write_str(", ")?;
378                        f.write_str(&error.path)?;
379                        f.write_str(": ")?;
380                        f.write_str(&error.message)?;
381                    }
382                    f.write_str(")")?;
383                }
384
385                Ok(())
386            },
387            Self::RateLimitI64F64 => f.write_str("Error decoding a header into an i64 or f64"),
388            Self::RateLimitUtf8 => f.write_str("Error decoding a header from UTF-8"),
389            Self::InvalidWebhook => f.write_str("Provided URL is not a valid webhook."),
390            Self::InvalidHeader(_) => f.write_str("Provided value is an invalid header value."),
391            Self::Request(_) => f.write_str("Error while sending HTTP request."),
392            Self::ApplicationIdMissing => f.write_str("Application id was expected but missing."),
393        }
394    }
395}

406#[expect(clippy::missing_errors_doc)]
407pub fn deserialize_errors<'de, D: Deserializer<'de>>(
408    deserializer: D,
409) -> StdResult<FixedArray<DiscordJsonSingleError>, D::Error> {
410    let ErrorValue::Recurse(map) = ErrorValue::deserialize(deserializer)? else {
411        return Ok(FixedArray::new());
412    };
413
414    let mut errors = Vec::new();
415    let mut path = Vec::new();
416    loop_errors(map, &mut errors, &mut path).map_err(D::Error::custom)?;
417
418    Ok(errors.trunc_into())
419}

421fn make_error(
422    errors_to_process: Vec<RawDiscordJsonSingleError>,
423    errors: &mut Vec<DiscordJsonSingleError>,
424    path: &[&str],
425) {
426    let joined_path = Arc::from(path.join("."));
427    errors.extend(errors_to_process.into_iter().map(|raw| DiscordJsonSingleError {
428        code: raw.code,
429        message: raw.message,
430        path: Arc::clone(&joined_path),
431    }));
432}

434#[derive(serde::Deserialize)]
435#[serde(untagged)]
436enum ErrorValue<'a> {
437    Base(Vec<RawDiscordJsonSingleError>),
438    #[serde(borrow)]
439    Recurse(HashMap<&'a str, ErrorValue<'a>>),
440}
441
442fn loop_errors<'a>(
443    value: HashMap<&'a str, ErrorValue<'a>>,
444    errors: &mut Vec<DiscordJsonSingleError>,
445    path: &mut Vec<&'a str>,
446) -> Result<(), &'static str> {
447    for (key, value) in value {
448        if key == "_errors" {
449            let ErrorValue::Base(value) = value else { return Err("expected array, found map") };
450            make_error(value, errors, path);
451        } else {
452            let ErrorValue::Recurse(value) = value else { return Err("expected map, found array") };
453
454            path.push(key);
455            loop_errors(value, errors, path)?;
456            path.pop();
457        }
458    }
459
460    Ok(())
461}

470
471    #[tokio::test]
472    async fn test_error_response_into() {
473        let error = DiscordJsonError {
474            code: JsonErrorCode::Unknown(43121215),
475            errors: FixedArray::empty(),
476            message: FixedString::from_static_trunc("This is a Ferris error"),
477        };
478
479        let mut builder = Builder::new();
480        builder = builder.status(403);
481        builder = builder.url(String::from("https://ferris.crab").parse().unwrap());
482        let body_string = to_string(&error).unwrap();
483        let response = builder.body(body_string.into_bytes()).unwrap();
484
485        let reqwest_response: reqwest::Response = response.into();
486        let error_response = ErrorResponse::from_response(reqwest_response, Method::POST).await;
487
488        let known = ErrorResponse {
489            status_code: reqwest::StatusCode::from_u16(403).unwrap(),
490            url: FixedString::from_static_trunc("https://ferris.crab/"),
491            method: Method::POST,
492            error,
493        };
494
495        assert_eq!(error_response, known);
496    }
