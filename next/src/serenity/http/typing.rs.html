13/// A struct to start typing in a [`Channel`] for an indefinite period of time.
14///
15/// It indicates that the current user is currently typing in the channel.
16///
17/// Typing is started by using the [`Typing::start`] method and stopped by using the
18/// [`Typing::stop`] method. Note that on some clients, typing may persist for a few seconds after
19/// [`Typing::stop`] is called. Typing is also stopped when the struct is dropped.
20///
21/// If a message is sent while typing is triggered, the user will stop typing for a brief period of
22/// time and then resume again until either [`Typing::stop`] is called or the struct is dropped.
23///
24/// This should rarely be used for bots, although it is a good indicator that a long-running
25/// command is still being processed.
26///
27/// ## Examples
28///
29/// ```rust,no_run
30/// # use serenity::{http::{Http, Typing}, Result, model::prelude::*};
31/// # use std::sync::Arc;
32/// #
33/// # fn long_process() {}
34/// # fn main() {
35/// # let http: Http = unimplemented!();
36/// let channel_id = GenericChannelId::new(7);
37/// // Initiate typing (assuming `http` is bound)
38/// let typing = Typing::start(Arc::new(http), channel_id);
39///
40/// // Run some long-running process
41/// long_process();
42///
43/// // Stop typing
44/// typing.stop();
45/// # }
46/// ```
47///
48/// [`Channel`]: crate::model::channel::Channel
49#[derive(Debug)]
50pub struct Typing(Sender<()>);
51
52impl Typing {
53    /// Starts typing in the specified [`Channel`] for an indefinite period of time.
54    ///
55    /// Returns [`Typing`]. To stop typing, you must call the [`Typing::stop`] method on the
56    /// returned [`Typing`] object or wait for it to be dropped. Note that on some clients, typing
57    /// may persist for a few seconds after stopped.
58    ///
59    /// # Errors
60    ///
61    /// Returns an  [`Error::Http`] if there is an error.
62    ///
63    /// [`Channel`]: crate::model::channel::Channel
64    pub fn start(http: Arc<Http>, channel_id: GenericChannelId) -> Self {
65        let (sx, mut rx) = oneshot::channel();
66
67        spawn_named::<_, Result<_>>("typing::start", async move {
68            loop {
69                match rx.try_recv() {
70                    Ok(()) | Err(TryRecvError::Closed) => break,
71                    _ => (),
72                }
73
74                http.broadcast_typing(channel_id).await?;
75
76                // It is unclear for how long typing persists after this method is called.
77                // It is generally assumed to be 7 or 10 seconds, so we use 7 to be safe.
78                sleep(Duration::from_secs(7)).await;
79            }
80
81            Ok(())
82        });
83
84        Self(sx)
85    }
86
87    /// Stops typing in [`Channel`].
88    ///
89    /// This should be used to stop typing after it is started using [`Typing::start`]. Typing may
90    /// persist for a few seconds on some clients after this is called. Returns false if typing has
91    /// already stopped.
92    ///
93    /// [`Channel`]: crate::model::channel::Channel
94    #[expect(clippy::must_use_candidate)]
95    pub fn stop(self) -> bool {
96        self.0.send(()).is_ok()
97    }
98}
