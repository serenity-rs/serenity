9impl CreateAttachment<'_> {
10    async fn into_part(self) -> Result<Part> {
11        let mut part = match self.data {
12            AttachmentData::Bytes(bytes) => Part::stream(bytes),
13            AttachmentData::File(file) => Part::stream(file.try_clone().await?),
14            AttachmentData::Path(path) => Part::stream(File::open(path).await?),
15        };
16        part = guess_mime_str(part, &self.filename)?;
17        part = part.file_name(self.filename);
18        Ok(part)
19    }
20}

23pub enum MultipartUpload<'a> {
24    /// A file sent with the form data as an individual upload. For example, a sticker.
25    File(CreateAttachment<'a>),
26    /// Files sent with the form as message attachments.
27    Attachments(Vec<CreateAttachment<'a>>),
28}
29
30/// Holder for multipart body. Contains upload data, multipart fields, and payload_json for
31/// creating requests with attachments.
32#[derive(Clone, Debug)]
33pub struct Multipart<'a> {
34    pub upload: MultipartUpload<'a>,
35    /// Multipart text fields that are sent with the form data as individual fields. If a certain
36    /// endpoint does not support passing JSON body via `payload_json`, this must be used instead.
37    pub fields: Vec<(Cow<'static, str>, Cow<'static, str>)>,
38    /// JSON body that will set as the form value as `payload_json`.
39    pub payload_json: Option<String>,
40}
41
42impl Multipart<'_> {
43    pub(crate) async fn build_form(self) -> Result<Form> {
44        let mut multipart = Form::new();
45
46        match self.upload {
47            MultipartUpload::File(upload_file) => {
48                multipart = multipart.part("file", upload_file.into_part().await?);
49            },
50            MultipartUpload::Attachments(attachment_files) => {
51                for (idx, file) in attachment_files.into_iter().enumerate() {
52                    let part = file.into_part().await?;
53                    multipart = multipart.part(format!("files[{idx}]"), part);
54                }
55            },
56        }
57
58        for (name, value) in self.fields {
59            multipart = multipart.text(name, value);
60        }
61
62        if let Some(payload_json) = self.payload_json {
63            multipart = multipart.text("payload_json", payload_json);
64        }
65
66        Ok(multipart)
67    }
68}

70fn guess_mime_str(part: Part, filename: &str) -> Result<Part> {
71    // This is required for certain endpoints like create sticker, otherwise the Discord API will
72    // respond with a 500 Internal Server Error. The mime type chosen is the same as what reqwest
73    // does internally when using Part::file(), but it is not done for any of the other methods we
74    // use.
75    // https://datatracker.ietf.org/doc/html/rfc7578#section-4.4
76    let mime_type = mime_guess::from_path(filename).first_or_octet_stream();
77    part.mime_str(mime_type.essence_str()).map_err(Into::into)
78}
