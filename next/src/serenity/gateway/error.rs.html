6/// An error that occurred while attempting to deal with the gateway.
7///
8/// Note that - from a user standpoint - there should be no situation in which you manually handle
9/// these.
10#[derive(Debug)]
11#[non_exhaustive]
12pub enum Error {
13    /// There was an error building a URL.
14    BuildingUrl,
15    /// The connection closed, potentially uncleanly.
16    Closed(Option<CloseFrame>),
17    /// Expected a Hello during a handshake
18    ExpectedHello,
19    /// When there was an error sending a heartbeat.
20    HeartbeatFailed,
21    /// When invalid authentication (a bad token) was sent in the IDENTIFY.
22    InvalidAuthentication,
23    /// When an invalid API version was sent to the gateway.
24    InvalidApiVersion,
25    /// Expected a Ready or an InvalidateSession
26    InvalidHandshake,
27    /// When invalid sharding data was sent in the IDENTIFY.
28    ///
29    /// # Examples
30    ///
31    /// Sending a shard ID of 5 when sharding with 3 total is considered invalid.
32    InvalidShardData,
33    /// When no authentication was sent in the IDENTIFY.
34    NoAuthentication,
35    /// When a session Id was expected (for resuming), but was not present.
36    NoSessionId,
37    /// When a shard would have too many guilds assigned to it.
38    ///
39    /// # Examples
40    ///
41    /// When sharding 5500 guilds on 2 shards, at least one of the shards will have over the
42    /// maximum number of allowed guilds per shard.
43    ///
44    /// This limit is currently 2500 guilds per shard.
45    OverloadedShard,
46    /// Failed to reconnect after a number of attempts.
47    ReconnectFailure,
48    /// When undocumented gateway intents are provided.
49    InvalidGatewayIntents,
50    /// When disallowed gateway intents are provided.
51    ///
52    /// If an connection has been established but privileged gateway intents were provided without
53    /// enabling them prior.
54    DisallowedGatewayIntents,
55    #[cfg(feature = "transport_compression_zlib")]
56    /// A decompression error from the `flate2` crate.
57    DecompressZlib(flate2::DecompressError),
58    #[cfg(feature = "transport_compression_zstd")]
59    /// A decompression error from zstd.
60    DecompressZstd(usize),
61    /// When zstd decompression fails due to corrupted data.
62    #[cfg(feature = "transport_compression_zstd")]
63    DecompressZstdCorrupted,
64    /// When decompressed gateway data is not valid UTF-8.
65    DecompressUtf8(std::string::FromUtf8Error),
66}
67
68impl fmt::Display for Error {
69    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
70        match self {
71            Self::BuildingUrl => f.write_str("Error building url"),
72            Self::Closed(_) => f.write_str("Connection closed"),
73            Self::ExpectedHello => f.write_str("Expected a Hello"),
74            Self::HeartbeatFailed => f.write_str("Failed sending a heartbeat"),
75            Self::InvalidAuthentication => f.write_str("Sent invalid authentication"),
76            Self::InvalidApiVersion => f.write_str("Sent invalid API version"),
77            Self::InvalidHandshake => f.write_str("Expected a valid Handshake"),
78            Self::InvalidShardData => f.write_str("Sent invalid shard data"),
79            Self::NoAuthentication => f.write_str("Sent no authentication"),
80            Self::NoSessionId => f.write_str("No Session Id present when required"),
81            Self::OverloadedShard => f.write_str("Shard has too many guilds"),
82            Self::ReconnectFailure => f.write_str("Failed to Reconnect"),
83            Self::InvalidGatewayIntents => f.write_str("Invalid gateway intents were provided"),
84            Self::DisallowedGatewayIntents => {
85                f.write_str("Disallowed gateway intents were provided")
86            },
87            #[cfg(feature = "transport_compression_zlib")]
88            Self::DecompressZlib(inner) => fmt::Display::fmt(&inner, f),
89            #[cfg(feature = "transport_compression_zstd")]
90            Self::DecompressZstd(code) => write!(f, "Zstd decompression error: {code}"),
91            #[cfg(feature = "transport_compression_zstd")]
92            Self::DecompressZstdCorrupted => {
93                f.write_str("Zstd decompression error: corrupted data")
94            },
95            Self::DecompressUtf8(inner) => fmt::Display::fmt(&inner, f),
96        }
97    }
98}
