42/// A manager for handling the status of shards by starting them, restarting them, and stopping
43/// them when required.

46    /// A sender that is cloned and given out to each ShardRunner as it is created
47    manager_tx: Sender<ShardManagerMessage>,
48    manager_rx: Receiver<ShardManagerMessage>,

122    /// Retrieves a function which can be used to shut down the ShardManager later.
123    ///
124    /// This function will return `true` if the ShardManager has successfully been
125    /// notified to shut down, or false if it has already shut down and been dropped.
126    pub fn get_shutdown_trigger(&self) -> impl FnOnce() -> bool + Send + use<> {
127        let manager_tx = self.manager_tx.clone();
128        move || manager_tx.unbounded_send(ShardManagerMessage::Quit(Ok(()))).is_ok()
129    }
130
131    /// The main interface for starting the management of shards. Initializes the shards by
132    /// queueing them for starting, and then listens for [`ShardManagerMessage`]s in a loop.
133    ///
134    /// This function takes care of:
135    ///   1. Starting [`ShardRunner`]s and spawning tasks.
136    ///   2. Restarting shards (by shutting them down and spawning a new task) when requested.
137    ///   3. Quitting if a shard runner encounters a fatal gateway error.
138    ///
139    /// # Errors
140    ///
141    /// Returns a [`GatewayError`] if an unrecoverable error is received from the gateway. These
142    /// include the following:
143    ///   * [`GatewayError::InvalidAuthentication`]
144    ///   * [`GatewayError::InvalidApiVersion`]
145    ///   * [`GatewayError::InvalidGatewayIntents`]
146    ///   * [`GatewayError::DisallowedGatewayIntents`]
147    pub async fn run(
148        &mut self,
149        shard_index: u16,
150        shard_init: u16,
151        shard_total: NonZeroU16,
152    ) -> Result<(), GatewayError> {
153        self.initialize(shard_index, shard_init, shard_total);
154        loop {
155            let batch = self.queue.pop_batch();
156            self.checked_start(batch).await;
157
158            if let Ok(Some(msg)) =
159                timeout(self.wait_time_between_shard_start, self.manager_rx.next()).await
160            {
161                match msg {
162                    ShardManagerMessage::RestartShard(shard_id) => {
163                        #[cfg(feature = "voice")]
164                        if let Some(voice_manager) = &self.voice_manager {
165                            voice_manager.deregister_shard(shard_id.0).await;
166                        }
167                        self.queue_for_start(shard_id);
168                    },
169                    ShardManagerMessage::Quit(res) => return res,
170                }
171            }
172        }
173    }
174
175    /// Initializes all shards that the manager is responsible for.
176    ///
177    /// Note that this queues all shards but does not actually start them. To start the manager's
178    /// event loop and dispatch [`ShardRunner`]s as they get queued, call [`Self::run`] instead.
179    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
180    fn initialize(&mut self, shard_index: u16, shard_init: u16, shard_total: NonZeroU16) {
181        let shard_to = shard_index + shard_init;
182
183        self.shard_total = shard_total;
184        for shard_id in shard_index..shard_to {
185            self.queue_for_start(ShardId(shard_id));
186        }
187    }
188
189    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
190    fn queue_for_start(&mut self, shard_id: ShardId) {
191        info!("Queueing shard {shard_id} for starting");
192
193        self.queue.push_back(shard_id);
194    }
195
196    // This function assumes that each of the shard ids are bucketed separately according to
197    // `max_concurrency`. If this assumption is violated, you will likely get ratelimited.
198    //
199    // See: https://docs.discord.com/developers/events/gateway#sharding-max-concurrency
200    async fn checked_start(&mut self, shard_ids: Vec<ShardId>) {
201        if shard_ids.is_empty() {
202            return;
203        }
204
205        if shard_ids.len() > 1 {
206            debug!("[Shard Manager] Starting batch of {} shards", shard_ids.len());
207        }
208
209        // We must wait 5 seconds between IDENTIFYs to avoid session invalidations.
210        if let Some(instant) = self.last_start {
211            let elapsed = instant.elapsed();
212            if let Some(duration) = self.wait_time_between_shard_start.checked_sub(elapsed) {
213                sleep(duration).await;
214            }
215        }
216
217        for shard_id in shard_ids {
218            debug!("[Shard Manager] Starting shard {shard_id}");
219            if let Err(why) = self.start_runner(shard_id).await {
220                warn!("[Shard Manager] Err starting shard {shard_id}: {why:?}");
221                info!("[Shard Manager] Re-queueing shard {shard_id}");
222
223                // Re-queue at the *front* of the bucket in order to keep decent ordering
224                self.queue.push_front(shard_id);
225            }
226            self.last_start = Some(Instant::now());
227        }
228    }
229
230    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
231    async fn start_runner(&mut self, shard_id: ShardId) -> Result<()> {
232        let shard_info = ShardInfo {
233            id: shard_id,
234            total: self.shard_total,
235        };
236        let mut shard = Shard::new(
237            Arc::clone(&self.ws_url),
238            self.token.clone(),
239            shard_info,
240            self.intents,
241            self.presence.clone(),
242            self.compression,
243        )
244        .await?;
245
246        let cloned_http = Arc::clone(&self.http);
247        shard.set_application_id_callback(move |id| cloned_http.set_application_id(id));
248
249        let (runner_tx, runner_rx) = mpsc::unbounded();
250        let runner_info = Arc::new(RwLock::new(ShardRunnerInfo {
251            latency: None,
252            stage: ConnectionStage::Disconnected,
253        }));
254
255        self.runners.insert(shard_id, ShardRunnerMetadata {
256            info: Arc::clone(&runner_info),
257            tx: runner_tx.clone(),
258        });
259
260        let dispatcher = EventDispatcher {
261            context: Context {
262                data: Arc::clone(&self.data),
263                shard: runner_tx,
264                runner_info,
265                manager: self.manager_tx.clone(),
266                shard_id,
267                http: Arc::clone(&self.http),
268                #[cfg(feature = "cache")]
269                cache: Arc::clone(&self.cache),
270                #[cfg(feature = "collector")]
271                collectors: Arc::default(),
272            },
273            event_handler: self.event_handler.clone(),
274            raw_event_handler: self.raw_event_handler.clone(),
275            #[cfg(feature = "framework")]
276            framework: self.framework.get().cloned(),
277            #[cfg(feature = "voice")]
278            voice_manager: self.voice_manager.clone(),
279        };
280
281        let mut runner = ShardRunner {
282            manager_tx: self.manager_tx.clone(),
283            runner_rx,
284            shard,
285            dispatcher,
286        };
287        spawn_named("shard_runner::run", async move { runner.run().await });
288
289        Ok(())
290    }

299impl Drop for ShardManager {
300    /// A custom drop implementation to clean up after the manager.
301    ///
302    /// This shuts down all active [`ShardRunner`]s.
303    ///
304    /// [`ShardRunner`]: super::ShardRunner
305    fn drop(&mut self) {
306        info!("Shutting down all shards");
307
308        for entry in self.runners.iter() {
309            let (shard_id, runner) = entry.pair();
310            info!("Shutting down shard {}", shard_id);
311            if let Err(why) = runner.tx.unbounded_send(ShardRunnerMessage::Shutdown(1000)) {
312                warn!("Failed to send shutdown signal to shard {shard_id}: {why:?}");
313            }
314        }
315    }
316}

339/// A message indicating what action the [`ShardManager`] should take.
340pub enum ShardManagerMessage {
341    /// Indicates that the manager should attempt to restart a shard.
342    ///
343    /// Note that this makes use of the manager's shard queue for batching shards together, and if
344    /// the queue is operating in concurrent mode (in order to start multiple shards at once),
345    /// the shard is not guaranteed to immediately start, until more shards are queued.
346    RestartShard(ShardId),
347    /// Indicates that a shard runner encountered a fatal error and the shard manager should quit.
348    Quit(Result<(), GatewayError>),
349}
