1//! Mechanisms for configuring and managing sharded gateway connections.
2//!
3//! Sharding is a method for load-balancing bots across separate threads or processes. Sharding is
4//! enforced on bots by Discord once they reach a certain number of guilds (2500). Once this
5//! threshold is reached, a but must be sharded such that at most 2500 guilds are allocated per
6//! shard.
7//!
8//! The "recommended" number of guilds per shard is _around_ 1000. Sharding allows for bots to be
9//! distributed by handing shards off to separate processes or even separate machines in a
10//! distributed network (e.g. cloud workers). However, sometimes you may wish for all shards to
11//! share some global state. Serenity accomodates both of these usecases.
12//!
13//! See [Discord's documentation][docs] for more information.
14//!
15//! This module also provides some lower-level facilities for performing sharding manually:
16//!
17//! ### [`ShardManager`]
18//!
19//! The shard manager provides an interface for managing the starting, restarting, and shutdown of
20//! shards by spawning tasks, each containing a [`ShardRunner`], and communicating with each task
21//! using message passing.
22//!
23//! ### [`ShardRunner`]
24//!
25//! The shard runner is responsible for directly running a single shard and communicating with the
26//! gateway through its respective WebSocket client. It performs actions such as identifying,
27//! reconnecting, resuming, and sending presence updates to the gateway.
28//!
29//! [docs]: https://discordapp.com/developers/docs/topics/gateway#sharding

67/// An abstract handler for a websocket connection to Discord's gateway.
68///
69/// Allows a user to send and receive messages over said websocket, including:
70///   * setting the current activity
71///   * setting the current online status
72///   * receiving gateway events
73///   * connection management via heartbeating
74///
75/// Shard management (including starting, restarting, heartbeating), is performed by the [`Client`]
76/// automatically on the user's behalf.
77///
78/// # Stand-alone shards
79///
80/// You may instantiate a shard yourself - decoupled from the [`Client`] - by calling
81/// [`Shard::new`]. Most use cases will not necessitate this, and unless you're doing something
82/// really weird you can just let the client do it for you.
83///
84/// **Note**: You _really_ do not need to do this, especially if using multiple shards. Just call
85/// one of the appropriate methods on the [`Client`].
86///
87/// # Examples
88///
89/// See the documentation for [`Self::new`] on how to use this.
90///
91/// [`Client`]: crate::Client
92pub struct Shard {
93    pub client: WsClient,
94    presence: PresenceData,
95    application_id_callback: Option<Box<dyn FnOnce(ApplicationId) + Send + Sync>>,
96    seq: u64,
97    info: ShardInfo,
98    stage: ConnectionStage,
99    token: Token,
100    ws_url: Arc<str>,
101    heartbeater: HeartbeatMetadata,
102    resume_metadata: Option<ResumeMetadata>,
103    compression: TransportCompression,
104    pub intents: GatewayIntents,
105}

113    ///
114    /// Instantiating a new Shard manually for a bot with no shards, and then listening for events:
115    ///
116    /// ```rust,no_run
117    /// use std::num::NonZeroU16;
118    /// use std::sync::Arc;
119    ///
120    /// use serenity::gateway::{Shard, TransportCompression};
121    /// use serenity::model::gateway::{GatewayIntents, ShardInfo};
122    /// use serenity::model::id::ShardId;
123    /// use serenity::secrets::Token;
124    /// use tokio::sync::Mutex;
125    /// #
126    /// # use serenity::http::Http;
127    /// #
128    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
129    /// # let http: Arc<Http> = unimplemented!();
130    /// let token = Token::from_env("DISCORD_TOKEN")?;
131    /// let shard_info = ShardInfo {
132    ///     id: ShardId(0),
133    ///     total: NonZeroU16::MIN,
134    /// };
135    ///
136    /// // retrieve the gateway response, which contains the URL to connect to
137    /// let gateway = Arc::from(http.get_gateway().await?.url);
138    /// let shard = Shard::new(
139    ///     gateway,
140    ///     token,
141    ///     shard_info,
142    ///     GatewayIntents::all(),
143    ///     None,
144    ///     TransportCompression::None,
145    /// )
146    /// .await?;
147    ///
148    /// // at this point, you can create a `loop`, and receive events and match
149    /// // their variants
150    /// # Ok(())
151    /// # }
152    /// ```
153    ///
154    /// # Errors
155    ///
156    /// On Error, will return either [`Error::Gateway`], [`Error::Tungstenite`] or a Rustls/native
157    /// TLS error.
158    pub async fn new(
159        ws_url: Arc<str>,
160        token: Token,
161        info: ShardInfo,
162        intents: GatewayIntents,
163        presence: Option<PresenceData>,
164        compression: TransportCompression,
165    ) -> Result<Shard> {
166        let client = connect(&ws_url, compression).await?;
167
168        let presence = presence.unwrap_or_default();
169        let seq = 0;
170        let stage = ConnectionStage::Handshake;
171
172        Ok(Shard {
173            client,
174            presence,
175            application_id_callback: None,
176            seq,
177            stage,
178            token,
179            info,
180            ws_url,
181            heartbeater: HeartbeatMetadata::default(),
182            resume_metadata: None,
183            compression,
184            intents,
185        })
186    }

203    /// Sends a heartbeat to the gateway with the current sequence.
204    ///
205    /// This sets the last heartbeat time to now.
206    ///
207    /// # Errors
208    ///
209    /// Returns [`GatewayError::HeartbeatFailed`] if there was an error sending a heartbeat.
210    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
211    pub async fn heartbeat(&mut self) -> Result<()> {
212        match self.client.send_heartbeat(&self.info, Some(self.seq)).await {
213            Ok(()) => self.heartbeater.send(),
214            Err(why) => {
215                if let Error::Tungstenite(err) = &why
216                    && let TungsteniteError::Io(err) = &**err
217                    && err.raw_os_error() != Some(32)
218                {
219                    debug!("[{:?}] Err heartbeating: {:?}", self.info, err);
220                } else {
221                    warn!("[{:?}] Other err w/ keepalive: {:?}", self.info, why);
222                }
223                return Err(Error::Gateway(GatewayError::HeartbeatFailed));
224            },
225        }
226
227        Ok(())
228    }

265    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
266    fn handle_gateway_dispatch(
267        &mut self,
268        seq: u64,
269        event: DeserializedEvent,
270    ) -> Option<Box<Event>> {
271        if seq > self.seq + 1 {
272            warn!("[{:?}] Sequence off; them: {}, us: {}", self.info, seq, self.seq);
273        }
274
275        self.seq = seq;
276
277        let event = match event {
278            DeserializedEvent::Success(event) => event,
279            DeserializedEvent::Unknown(event) => {
280                debug!("Failed to deserialize event data: {}", event.err);
281                debug!("Raw event data: {}", event.data);
282                return None;
283            },
284        };
285
286        match &*event {
287            Event::Ready(ready) => {
288                debug!("[{:?}] Received Ready", self.info);
289
290                self.resume_metadata = Some(ResumeMetadata {
291                    session_id: ready.ready.session_id.clone(),
292                    resume_ws_url: ready.ready.resume_gateway_url.clone(),
293                });
294                self.stage = ConnectionStage::Connected;
295
296                if let Some(callback) = self.application_id_callback.take() {
297                    callback(ready.ready.application.id);
298                }
299            },
300            Event::Resumed(_) => {
301                info!("[{:?}] Resumed", self.info);
302
303                self.stage = ConnectionStage::Connected;
304                self.heartbeater.resume();
305            },
306            _ => {},
307        }
308
309        Some(event)
310    }

312    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
313    fn handle_gateway_closed(&mut self, data: Option<&CloseFrame>) -> Result<()> {
314        if let Some(code) = data.map(|d| d.code) {
315            match CloseCode(code.into()) {
316                CloseCode::UnknownError => warn!("[{:?}] Unknown gateway error.", self.info),
317                CloseCode::UnknownOpcode => warn!("[{:?}] Sent invalid opcode.", self.info),
318                CloseCode::DecodeError => warn!("[{:?}] Sent invalid message.", self.info),
319                CloseCode::NotAuthenticated => {
320                    warn!("[{:?}] Sent no authentication, or session invalidated.", self.info);
321                    return Err(Error::Gateway(GatewayError::NoAuthentication));
322                },
323                CloseCode::AuthenticationFailed => {
324                    error!(
325                        "[{:?}] Sent invalid authentication, please check the token.",
326                        self.info
327                    );
328
329                    return Err(Error::Gateway(GatewayError::InvalidAuthentication));
330                },
331                CloseCode::AlreadyAuthenticated => {
332                    warn!("[{:?}] Already authenticated.", self.info);
333                },
334                CloseCode::InvalidSequence => {
335                    warn!("[{:?}] Sent invalid seq: {}.", self.info, self.seq);
336                    self.seq = 0;
337                },
338                CloseCode::RateLimited => warn!("[{:?}] Gateway ratelimited.", self.info),
339                CloseCode::SessionTimeout => {
340                    info!("[{:?}] Invalid session.", self.info);
341                    self.resume_metadata = None;
342                },
343                CloseCode::InvalidShard => {
344                    warn!("[{:?}] Sent invalid shard data.", self.info);
345                    return Err(Error::Gateway(GatewayError::InvalidShardData));
346                },
347                CloseCode::ShardingRequired => {
348                    error!("[{:?}] Shard has too many guilds.", self.info);
349                    return Err(Error::Gateway(GatewayError::OverloadedShard));
350                },
351                CloseCode::InvalidApiVersion => {
352                    error!("[{:?}] Invalid gateway API version provided.", self.info);
353                    return Err(Error::Gateway(GatewayError::InvalidApiVersion));
354                },
355                CloseCode::InvalidGatewayIntents => {
356                    error!("[{:?}] Invalid gateway intents have been provided.", self.info);
357                    return Err(Error::Gateway(GatewayError::InvalidGatewayIntents));
358                },
359                CloseCode::DisallowedGatewayIntents => {
360                    error!("[{:?}] Disallowed gateway intents have been provided.", self.info);
361                    return Err(Error::Gateway(GatewayError::DisallowedGatewayIntents));
362                },
363                _ => warn!(
364                    "[{:?}] Unknown close code {}: {:?}",
365                    self.info,
366                    code,
367                    data.map(|d| &d.reason)
368                ),
369            }
370        }
371        Ok(())
372    }

374    /// Handles an event from the gateway over the receiver, requiring the receiver to be passed if
375    /// a reconnect needs to occur.
376    ///
377    /// The best case scenario is that one of two values is returned:
378    /// - `Ok(None)`: a heartbeat, late hello, or session invalidation was received;
379    /// - `Ok(Some((event, None)))`: an op0 dispatch was received, and the shard's voice state will
380    ///   be updated, _if_ the `voice` feature is enabled.
381    ///
382    /// # Errors
383    ///
384    /// Returns a [`GatewayError::InvalidAuthentication`] if invalid authentication was sent in the
385    /// IDENTIFY.
386    ///
387    /// Returns a [`GatewayError::InvalidShardData`] if invalid shard data was sent in the
388    /// IDENTIFY.
389    ///
390    /// Returns a [`GatewayError::NoAuthentication`] if no authentication was sent in the IDENTIFY.
391    ///
392    /// Returns a [`GatewayError::OverloadedShard`] if the shard would have too many guilds
393    /// assigned to it.
394    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
395    pub fn handle_event(&mut self, event: Result<GatewayEvent>) -> Result<Option<ShardAction>> {
396        match event {
397            Ok(GatewayEvent::Dispatch {
398                seq,
399                event,
400            }) => Ok(self.handle_gateway_dispatch(seq, event).map(ShardAction::Dispatch)),
401            Ok(GatewayEvent::Heartbeat) => {
402                info!("[{:?}] Received a request to heartbeat", self.info);
403                Ok(Some(ShardAction::Heartbeat))
404            },
405            Ok(GatewayEvent::HeartbeatAck) => {
406                self.heartbeater.ack();
407                trace!("[{:?}] Received heartbeat ack", self.info);
408                Ok(None)
409            },
410            Ok(GatewayEvent::Hello(interval)) => {
411                debug!("[{:?}] Received a Hello; interval: {}", self.info, interval);
412
413                if self.stage == ConnectionStage::Resuming {
414                    Ok(None)
415                } else {
416                    self.heartbeater.set_interval(interval);
417                    let action = if self.stage == ConnectionStage::Handshake {
418                        ShardAction::Identify
419                    } else {
420                        debug!("[{:?}] Received late Hello; autoreconnecting", self.info);
421                        ShardAction::Reconnect
422                    };
423
424                    Ok(Some(action))
425                }
426            },
427            Ok(GatewayEvent::InvalidateSession(resumable)) => {
428                info!("[{:?}] Received session invalidation", self.info);
429                if !resumable {
430                    self.resume_metadata = None;
431                }
432
433                Ok(Some(ShardAction::Reconnect))
434            },
435            Ok(GatewayEvent::Reconnect) => Ok(Some(ShardAction::Reconnect)),
436            Err(Error::Gateway(GatewayError::Closed(data))) => {
437                self.handle_gateway_closed(data.as_ref())?;
438                Ok(Some(ShardAction::Reconnect))
439            },
440            Err(Error::Tungstenite(why)) => {
441                info!("[{:?}] Websocket error: {:?}", self.info, why);
442                info!("[{:?}] Will attempt to auto-reconnect", self.info);
443
444                Ok(Some(ShardAction::Reconnect))
445            },
446            Err(why) => {
447                warn!("[{:?}] Unhandled error: {:?}", self.info, why);
448                Ok(None)
449            },
450        }
451    }

453    /// Does a heartbeat if needed. Returns false if something went wrong and the shard should be
454    /// restarted.
455    ///
456    /// `true` is returned under one of the following conditions:
457    /// - the heartbeat interval has not elapsed
458    /// - a heartbeat was successfully sent
459    /// - there is no known heartbeat interval yet
460    ///
461    /// `false` is returned under one of the following conditions:
462    /// - a heartbeat acknowledgement was not received in time
463    /// - an error occurred while heartbeating
464    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
465    pub async fn do_heartbeat(&mut self) -> bool {
466        // Check if we already acked the last heartbeat, or didn't receive a heartbeat ACK in time.
467        // Otherwise, we're good to heartbeat.
468        if let Some(acked) = self.heartbeater.last_heartbeat_acked() {
469            if !acked {
470                debug!("[{:?}] Last heartbeat not acknowledged", self.info);
471            }
472            acked
473        } else if let Err(why) = self.heartbeat().await {
474            warn!("[{:?}] Err heartbeating: {:?}", self.info, why);
475            false
476        } else {
477            trace!("[{:?}] Heartbeat", self.info);
478            true
479        }
480    }

584    /// Reinitializes an existing WebSocket client, replacing it.
585    ///
586    /// This will set the stage of the shard before and after instantiation of the client.
587    ///
588    /// # Errors
589    ///
590    /// Errors if unable to establish a websocket connection.
591    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
592    pub async fn reinitialize(&mut self) -> Result<WsClient> {
593        debug!("[{:?}] Initializing.", self.info);
594
595        // Reconnect to the resume URL if possible, otherwise use the generic URL.
596        let ws_url = self
597            .resume_metadata
598            .as_ref()
599            .map_or(self.ws_url.as_ref(), |m| m.resume_ws_url.as_ref());
600
601        // We need to do two, sort of three things here:
602        // - set the stage of the shard as opening the websocket connection
603        // - open the websocket connection
604        // - if successful, set the current stage as Handshaking
605        //
606        // This is used to accurately assess whether the state of the shard is accurate when a
607        // Hello is received.
608        self.stage = ConnectionStage::Connecting;
609        self.heartbeater.start();
610        let client = connect(ws_url, self.compression).await?;
611        self.stage = ConnectionStage::Handshake;
612
613        Ok(client)
614    }

616    /// # Errors
617    ///
618    /// Errors if unable to re-establish a websocket connection.
619    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
620    pub async fn resume(&mut self) -> Result<()> {
621        debug!("[{:?}] Attempting to resume", self.info);
622
623        self.client = self.reinitialize().await?;
624        self.stage = ConnectionStage::Resuming;
625
626        if let Some(m) = &self.resume_metadata {
627            self.client
628                .send_resume(&self.info, &m.session_id, self.seq, self.token.expose_secret())
629                .await
630        } else {
631            Err(Error::Gateway(GatewayError::NoSessionId))
632        }
633    }

644async fn connect(base_url: &str, compression: TransportCompression) -> Result<WsClient> {
645    let url = Url::parse(&aformat!(
646        "{}?v={}{}",
647        CapStr::<64>(base_url),
648        constants::GATEWAY_VERSION,
649        compression.query_param()
650    ))
651    .map_err(|why| {
652        warn!("Error building gateway URL with base `{base_url}`: {why:?}");
653        Error::Gateway(GatewayError::BuildingUrl)
654    })?;
655
656    WsClient::connect(url, compression).await
657}

659struct HeartbeatMetadata {
660    /// Instant when the shard was started.
661    started: Instant,
662    /// When the last heartbeat was sent by the client.
663    last_sent: Option<Instant>,
664    /// When the last heartbeat was acknowledged by the server.
665    last_ack: Option<Instant>,
666    /// The heartbeat interval dictated by Discord, if the Hello packet has been received.
667    interval: Option<Duration>,
668}
669
670impl HeartbeatMetadata {
671    fn start(&mut self) {
672        self.started = Instant::now();
673    }
674
675    fn resume(&mut self) {
676        self.last_sent = Some(Instant::now());
677        self.last_ack = None;
678    }
679
680    fn send(&mut self) {
681        self.last_sent = Some(Instant::now());
682    }
683
684    fn ack(&mut self) {
685        self.last_ack = Some(Instant::now());
686    }
687
688    fn set_interval(&mut self, millis: u64) {
689        self.interval = Some(Duration::from_millis(millis));
690    }
691
692    fn last_heartbeat_acked(&self) -> Option<bool> {
693        let Some(interval) = self.interval else {
694            // No Hello received yet, wait a bit longer before reconnecting.
695            return Some(self.started.elapsed() < Duration::from_secs(15));
696        };
697
698        // If the heartbeat interval has not yet passed, then we don't need to perform a keepalive
699        // or attempt to reconnect. Similarly, if we've just resumed then simply keep waiting for
700        // the first heartbeat ACK.
701        if let Some(last_sent) = self.last_sent {
702            if last_sent.elapsed() <= interval {
703                return Some(true);
704            } else if let Some(last_ack) = self.last_ack
705                && last_ack <= last_sent
706            {
707                return Some(false);
708            }
709        }
710
711        None
712    }
713
714    fn latency(&self) -> Option<Duration> {
715        if let Some(sent) = self.last_sent
716            && let Some(ack) = self.last_ack
717            && ack > sent
718        {
719            return Some(ack - sent);
720        }
721
722        None
723    }
724}
725
726impl Default for HeartbeatMetadata {
727    fn default() -> Self {
728        Self {
729            started: Instant::now(),
730            last_sent: None,
731            last_ack: None,
732            interval: None,
733        }
734    }
735}

770/// The transport compression method to use.
771#[derive(Copy, Clone, Debug, PartialEq, Eq)]
772#[non_exhaustive]
773pub enum TransportCompression {
774    /// No transport compression. Payload compression will be used instead.
775    None,
776
777    #[cfg(feature = "transport_compression_zlib")]
778    /// Use zlib-stream transport compression.
779    Zlib,
780
781    #[cfg(feature = "transport_compression_zstd")]
782    /// Use zstd-stream transport compression.
783    Zstd,
784}
785
786impl TransportCompression {
787    fn query_param(self) -> ArrayString<21> {
788        #[cfg_attr(
789            not(any(
790                feature = "transport_compression_zlib",
791                feature = "transport_compression_zstd"
792            )),
793            expect(unused_mut)
794        )]
795        let mut res = ArrayString::new();
796        match self {
797            Self::None => {},
798            #[cfg(feature = "transport_compression_zlib")]
799            Self::Zlib => aformat_into!(res, "&compress=zlib-stream"),
800            #[cfg(feature = "transport_compression_zstd")]
801            Self::Zstd => aformat_into!(res, "&compress=zstd-stream"),
802        }
803
804        res
805    }
