1//! A module for [`Client`] and supporting types.
2//!
3//! The [`Client`] contains information about a bot's token, as well as event handlers. Dispatching
4//! events to handlers and starting sharded gateway connections is handled directly by the client.
5//! In addition, the client automatically handles caching via the [`Cache`] struct.
6//!
7//! # Sharding
8//!
9//! If you do not require sharding - such as for a small bot - then use [`Client::start`]. If you
10//! don't know what sharding is, refer to the [`sharding`] module documentation.
11//!
12//! There are a few methods of sharding available:
13//! - [`Client::start_autosharded`]: retrieves the number of shards Discord recommends using from
14//!   the API, and then automatically starts that number of shards.
15//! - [`Client::start_shard`]: starts a single shard for use in the instance, handled by the
16//!   instance of the Client. Use this if you only want 1 shard handled by this instance.
17//! - [`Client::start_shards`]: starts all shards in this instance. This is best for when you want a
18//!   completely shared State.
19//! - [`Client::start_shard_range`]: start a range of shards within this instance. This should be
20//!   used when you, for example, want to split 10 shards across 3 instances.
21//!
22//! Click [here][Client#examples] for an example on how to use a [`Client`].
23//!
24//! [`sharding`]: crate::gateway::sharding

206    ///

279impl IntoFuture for ClientBuilder {
280    type Output = Result<Client>;
281
282    type IntoFuture = BoxFuture<'static, Result<Client>>;
283
284    fn into_future(self) -> Self::IntoFuture {
285        let data = self.data.unwrap_or(Arc::new(()));
286        #[cfg(feature = "framework")]
287        let framework = self.framework;
288        let intents = self.intents;
289        let presence = self.presence;
290        let http = self.http;
291
292        if let Some(ratelimiter) = &http.ratelimiter
293            && let Some(event_handler) = &self.event_handler
294        {
295            let event_handler = Arc::clone(event_handler);
296            ratelimiter.set_ratelimit_callback(Box::new(move |info| {
297                let event_handler = Arc::clone(&event_handler);
298                spawn_named("ratelimit::dispatch", async move {
299                    event_handler.ratelimit(info).await;
300                });
301            }));
302        }
303
304        #[cfg(feature = "cache")]
305        let cache = Arc::new(Cache::new_with_settings(self.cache_settings));
306
307        Box::pin(async move {
308            let (ws_url, shard_total, max_concurrency) = match http.get_bot_gateway().await {
309                Ok(response) => (
310                    Arc::from(response.url),
311                    response.shards,
312                    response.session_start_limit.max_concurrency,
313                ),
314                Err(err) => {
315                    tracing::warn!("HTTP request to get gateway URL failed: {err}");
316                    (Arc::from("wss://gateway.discord.gg"), NonZeroU16::MIN, NonZeroU16::MIN)
317                },
318            };
319
320            #[cfg(feature = "framework")]
321            let framework_cell = Arc::new(OnceLock::new());
322
323            let shard_manager = ShardManager::new(ShardManagerOptions {
324                token: self.token,
325                data: Arc::clone(&data),
326                event_handler: self.event_handler,
327                raw_event_handler: self.raw_event_handler,
328                #[cfg(feature = "framework")]
329                framework: Arc::clone(&framework_cell),
330                #[cfg(feature = "voice")]
331                voice_manager: self.voice_manager.clone(),
332                ws_url: Arc::clone(&ws_url),
333                compression: self.compression,
334                shard_total,
335                max_concurrency,
336                #[cfg(feature = "cache")]
337                cache: Arc::clone(&cache),
338                http: Arc::clone(&http),
339                intents,
340                presence: Some(presence),
341                wait_time_between_shard_start: self.wait_time_between_shard_start,
342            });
343
344            let client = Client {
345                data,
346                shard_manager,
347                #[cfg(feature = "voice")]
348                voice_manager: self.voice_manager,
349                ws_url,
350                #[cfg(feature = "cache")]
351                cache,
352                http,
353            };
354            #[cfg(feature = "framework")]
355            if let Some(mut framework) = framework {
356                framework.init(&client).await;
357                if let Err(_existing) = framework_cell.set(framework.into()) {
358                    tracing::warn!("overwrote existing contents of framework OnceLock");
359                }
360            }
361            Ok(client)
362        })
363    }
364}

366/// A high-level client that abstracts over the REST API as well as Discord's gateway.
367///
368/// It enables the user to start sending authenticated HTTP requests, plus also initialize a
369/// WebSocket connection to the gateway through [`Shard`]s. Refer to the [documentation on using
370/// sharding][super::sharding] for more information.
371///
372/// # Event Handlers
373///
374/// Event handlers can be configured. For example, the event handler will be dispatched to
375/// whenever a [`Event::MessageCreate`] is received over the connection.
376///
377/// Note that you do not need to manually handle events, as they are handled internally and then
378/// dispatched to your event handler.
379///
380/// [`Shard`]: crate::gateway::Shard
381/// [`Event::MessageCreate`]: crate::model::event::Event::MessageCreate
382pub struct Client {
383    data: Arc<dyn std::any::Any + Send + Sync>,
384    /// The shard manager for the client.
385    ///
386    /// This is the brains, managing shards (websocket connections) and bot lifecycle.
387    pub shard_manager: ShardManager,
388    /// The voice manager for the client.
389    ///
390    /// This is an ergonomic structure for interfacing over shards' voice
391    /// connections.
392    #[cfg(feature = "voice")]
393    pub voice_manager: Option<Arc<dyn VoiceGatewayManager + 'static>>,
394    /// URL that the client's shards will use to connect to the gateway.
395    pub ws_url: Arc<str>,
396    /// The cache for the client.
397    #[cfg(feature = "cache")]
398    pub cache: Arc<Cache>,
399    /// An HTTP client.
400    pub http: Arc<Http>,
401}

426    ///
427    /// This will start receiving events in a loop and start dispatching the events to your
428    /// registered handlers.
429    ///
430    /// Note that this should be used only for users and for bots which are in less than 2500
431    /// guilds. If you have a reason for sharding and/or are in more than 2500 guilds, use one of

438    ///

440    ///

470    ///
471    /// This will start receiving events in a loop and start dispatching the events to your

508    pub async fn start_autosharded(&mut self) -> Result<()> {
509        let (end, total) = {
510            let res = self.http.get_bot_gateway().await?;
511            (res.shards.get() - 1, res.shards)
512        };
513
514        self.start_connection(0, end, total).await
515    }

625    /// This will create and handle all shards within a given range within this single process. If
626    /// you only need to start a single shard within the process, or all shards within the process,
627    /// use [`Self::start_shard`] or [`Self::start_shards`], respectively.
628    ///
629    /// Refer to the [Gateway documentation][gateway docs] for more information on effectively
630    /// using sharding.
631    ///
632    /// # Examples
633    ///
634    /// For a bot using a total of 10 shards, initialize shards 4 through 7:
635    ///
636    /// ```rust,no_run
637    /// # use std::error::Error;
638    /// # use serenity::prelude::*;
639    /// use serenity::Client;
640    ///
641    /// # async fn run() -> Result<(), Box<dyn Error>> {
642    /// let token = Token::from_env("DISCORD_TOKEN")?;
643    /// let mut client = Client::builder(token, GatewayIntents::default()).await?;
644    ///
645    /// if let Err(why) = client.start_shard_range(4..7, 10).await {
646    ///     println!("Err with client: {:?}", why);
647    /// }
648    /// # Ok(())
649    /// # }
650    /// ```
651    ///
652    /// # Errors
653    ///
654    /// Returns [`Error::Gateway`] when all shards have shutdown due to an error.
655    /// Returns [`Error::Http`] if fetching the current User fails when initialising a voice
656    /// manager.
657    ///
658    /// [Gateway docs]: crate::gateway#sharding
659    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
660    pub async fn start_shard_range(&mut self, range: Range<u16>, total_shards: u16) -> Result<()> {
661        self.start_connection(range.start, range.end, check_shard_total(total_shards)).await
662    }
663
664    #[cfg_attr(feature = "tracing_instrument", instrument(skip(self)))]
665    async fn start_connection(
666        &mut self,
667        start_shard: u16,
668        end_shard: u16,
669        total_shards: NonZeroU16,
670    ) -> Result<()> {
671        #[cfg(feature = "voice")]
672        if let Some(voice_manager) = &self.voice_manager {
673            #[cfg(feature = "cache")]
674            let cache_user_id = {
675                let cache_user = self.cache.current_user();
676                if cache_user.id == UserId::default() { None } else { Some(cache_user.id) }
677            };
678
679            #[cfg(not(feature = "cache"))]
680            let cache_user_id: Option<UserId> = None;
681
682            let user_id = match cache_user_id {
683                Some(u) => u,
684                None => self.http.get_current_user().await?.id,
685            };
686
687            voice_manager.initialise(total_shards, user_id).await;
688        }
689
690        let init = end_shard - start_shard + 1;
691
692        debug!("Initializing shard info: {} - {}/{}", start_shard, init, total_shards);
693
694        self.shard_manager.run(start_shard, init, total_shards).await.map_err(Error::Gateway)
695    }
696}
697
698fn check_shard_total(total_shards: u16) -> NonZeroU16 {
699    NonZeroU16::new(total_shards).unwrap_or_else(|| {
700        warn!("Invalid shard total provided ({total_shards}), defaulting to 1");
701        NonZeroU16::MIN
702    })
