14#[async_trait]
15pub trait EventHandler: Send + Sync {
16    /// Checks if the `event` should be dispatched or ignored. Returns `true` by default.
17    ///
18    /// Returning `false` will drop an event and prevent it being dispatched by any
19    /// frameworks and will exclude it from any collectors.
20    ///
21    /// ## Warning
22    ///
23    /// Similar to [`RawEventHandler`], this method runs synchronously to the [`ShardRunner`], keep
24    /// runtime complexity low.
25    fn filter_event(&self, _context: &Context, _event: &Event) -> bool {
26        true
27    }
28
29    /// Dispatches an event through this handler, allowing for event matching and handling
30    /// based on individual event variants.
31    async fn dispatch(&self, _context: &Context, _event: &FullEvent) {}
32
33    /// Dispatched when an HTTP rate limit is hit.
34    async fn ratelimit(&self, _data: RatelimitInfo) {}
35}

37macro_rules! full_event {
38    ( $(
39        $( #[doc = $doc:literal] )*
40        $( #[deprecated = $deprecated:literal] )?
41        $( #[cfg(feature = $feature:literal)] )?
42        $variant_name:ident { $( $arg_name:ident: $arg_type:ty ),* };
43    )* ) => {
44        /// This enum stores every possible event that an [`EventHandler`] can receive.
45        #[cfg_attr(not(feature = "unstable"), non_exhaustive)]
46        #[derive(Clone, Debug, VariantNames, IntoStaticStr, EnumCount, Serialize)]
47        #[strum(serialize_all = "SCREAMING_SNAKE_CASE")]
48        pub enum FullEvent {
49            $(
50                $( #[doc = $doc] )*
51                $( #[deprecated = $deprecated] )?
52                $( #[cfg(feature = $feature)] )?
53                #[cfg_attr(not(feature = "unstable"), non_exhaustive)]
54                $variant_name {
55                    $( $arg_name: $arg_type ),*
56                },
57            )*
58        }
59    }
60}

432/// An event handler that receives raw `dispatch` events.
433///
434/// ## Warning
435/// As this is a low level trait, the methods of this trait are run on the same tokio task as the
436/// [`ShardRunner`].
437///
438/// This means that if any of these methods take too long to return, the shard may drop events or be
439/// disconnected entirely.
440///
441/// It is recommended to clone the fields needed out of [`Event`], then spawn a task to run
442/// concurrently to the shard loop.
443#[async_trait]
444pub trait RawEventHandler: Send + Sync {
445    /// Dispatched when any event occurs
446    async fn raw_event(&self, _ctx: Context, _ev: &Event) {}
447
448    /// Checks if the `event` should be dispatched or ignored. Returns `true` by default.
449    ///
450    /// Returning `false` will drop an event and prevent it being dispatched by any frameworks and
451    /// will exclude it from any collectors.
452    fn filter_event(&self, _context: &Context, _event: &Event) -> bool {
453        // Suppress unused argument warnings
454        true
455    }
456}
