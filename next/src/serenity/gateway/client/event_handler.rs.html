9#[async_trait]
10pub trait EventHandler: Send + Sync {
11    /// Checks if the `event` should be dispatched or ignored. Returns Some(event) by default.
12    ///
13    /// Returning `None` will drop the event and never process it.
14    ///
15    /// Returning a mutated `Some` is a supported use case, but may cause contradictions to
16    /// documented behaviour or otherwise cause unexpected errors within serenity/library
17    /// consumers. Mutate with extreme caution.
18    ///
19    /// ## Warning
20    ///
21    /// Similar to [`RawEventHandler`], this method runs synchronously to the [`ShardRunner`], keep
22    /// runtime complexity low.
23    fn filter_event(&self, _context: &Context, event: Box<Event>) -> Option<Box<Event>> {
24        Some(event)
25    }
26
27    /// Dispatches an event through this handler, allowing for event matching and handling
28    /// based on individual event variants.
29    async fn dispatch(&self, _context: &Context, _event: &FullEvent) {}
30
31    /// Dispatched when an HTTP rate limit is hit.
32    async fn ratelimit(&self, _data: RatelimitInfo) {}
33}

35/// An event handler that receives raw `dispatch` events.
36///
37/// ## Warning
38/// As this is a low level trait, the methods of this trait are run on the same tokio task as the
39/// [`ShardRunner`].
40///
41/// This means that if any of these methods take too long to return, the shard may drop events or be
42/// disconnected entirely.
43///
44/// It is recommended to clone the fields needed out of [`Event`], then spawn a task to run
45/// concurrently to the shard loop.
46#[async_trait]
47pub trait RawEventHandler: Send + Sync {
48    /// Dispatched when any event occurs
49    async fn raw_event(&self, _ctx: Context, _ev: &Event) {}
50
51    /// Checks if the `event` should be dispatched or ignored. Returns Some(event) by default.
52    ///
53    /// Returning `None` will drop the event and never process it.
54    ///
55    /// Returning a mutated `Some` is a supported use case, but may cause contradictions to
56    /// documented behaviour or otherwise cause unexpected errors within serenity/library
57    /// consumers. Mutate with extreme caution.
58    fn filter_event(&self, _context: &Context, event: Box<Event>) -> Option<Box<Event>> {
59        Some(event)
60    }
61}
