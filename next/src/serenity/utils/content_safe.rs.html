7/// Struct that allows to alter [`content_safe`]'s behaviour.
8#[bool_to_bitflags::bool_to_bitflags(
9    getter_prefix = "get_",
10    setter_prefix = "",
11    private_getters,
12    document_setters,
13    owning_setters
14)]
15#[derive(Copy, Clone, Debug)]
16pub struct ContentSafeOptions {
17    /// [`content_safe`] will replace role mentions (`<@&{id}>`) with its name prefixed with `@`
18    /// (`@rolename`) or with `@deleted-role` if the identifier is invalid.
19    pub clean_role: bool,
20    /// If set to true, [`content_safe`] will replace user mentions (`<@!{id}>` or `<@{id}>`) with
21    /// the user's name prefixed with `@` (`@username`) or with `@invalid-user` if the identifier
22    /// is invalid.
23    pub clean_user: bool,
24    /// If set to true, [`content_safe`] will replace channel mentions (`<#{id}>`) with the
25    /// channel's name prefixed with `#` (`#channelname`) or with `#deleted-channel` if the
26    /// identifier is invalid.
27    pub clean_channel: bool,
28    /// If set, [`content_safe`] will replace `@here` with a non-pinging alternative.
29    pub clean_here: bool,
30    /// If set, [`content_safe`] will replace `@everyone` with a non-pinging alternative.
31    pub clean_everyone: bool,
32}
33
34impl ContentSafeOptions {
35    #[must_use]
36    pub fn new() -> Self {
37        ContentSafeOptions::default()
38    }
39}

41impl Default for ContentSafeOptions {
42    /// Instantiates with all options set to `true`.
43    fn default() -> Self {
44        ContentSafeOptions {
45            __generated_flags: ContentSafeOptionsGeneratedFlags::all(),
46        }
47    }
48}

50/// Transforms role, channel, user, `@everyone` and `@here` mentions into raw text by using the
51/// Guild and the users passed in with `users`.
52///
53/// [`ContentSafeOptions`] decides what kind of mentions should be filtered and how the raw-text
54/// will be displayed.
55///
56/// # Examples
57///
58/// Sanitise an `@everyone` mention.
59///
60/// ```rust
61/// # let cache = serenity::cache::Cache::default();
62/// # let guild = serenity::model::guild::Guild::default();
63/// use serenity::utils::{ContentSafeOptions, content_safe};
64///
65/// let with_mention = "@everyone";
66/// let without_mention = content_safe(&guild, &with_mention, ContentSafeOptions::default(), &[]);
67///
68/// assert_eq!("@\u{200B}everyone", without_mention);
69/// ```
70///
71/// Filtering out mentions from a message.
72///
73/// ```rust
74/// use serenity::cache::Cache;
75/// use serenity::model::channel::Message;
76/// use serenity::utils::{ContentSafeOptions, content_safe};
77///
78/// fn filter_message(cache: &Cache, message: &Message) -> String {
79///     if let Some(guild) = message.guild(cache) {
80///         content_safe(&guild, &message.content, ContentSafeOptions::default(), &message.mentions)
81///     } else {
82///         // We don't need to clean messages in DMs
83///         message.content.to_string()
84///     }
85/// }
86/// ```
87#[must_use]
88pub fn content_safe(guild: &Guild, s: &str, options: ContentSafeOptions, users: &[User]) -> String {
89    let mut content = clean_mentions(guild, s, options, users);
90
91    if options.get_clean_here() {
92        content = content.replace("@here", "@\u{200B}here");
93    }
94
95    if options.get_clean_everyone() {
96        content = content.replace("@everyone", "@\u{200B}everyone");
97    }
98
99    content
100}

102fn clean_mentions(guild: &Guild, s: &str, options: ContentSafeOptions, users: &[User]) -> String {
103    let mut content = String::with_capacity(s.len());
104    let mut brackets = s.match_indices(['<', '>']).peekable();
105    let mut progress = 0;
106    while let Some((idx1, b1)) = brackets.next() {
107        // Find inner-most pairs of angle brackets
108        if b1 == "<"
109            && let Some(&(idx2, b2)) = brackets.peek()
110            && b2 == ">"
111        {
112            content.push_str(&s[progress..idx1]);
113            let mention_str = &s[idx1..=idx2];
114
115            // Don't waste time parsing if we're not going to clean the mention anyway
116            // NOTE: Emoji mentions aren't cleaned.
117            let mut chars = mention_str.chars();
118            chars.next();
119            let should_parse = match chars.next() {
120                Some('#') => options.get_clean_channel(),
121                Some('@') => {
122                    if let Some('&') = chars.next() {
123                        options.get_clean_role()
124                    } else {
125                        options.get_clean_user()
126                    }
127                },
128                _ => false,
129            };
130
131            // NOTE: numeric strings that are too large to fit into u64 will not parse correctly
132            // and will be left unchanged.
133            let cleaned_mention = if should_parse && let Ok(mention) = mention_str.parse() {
134                &clean_mention(guild, mention, users)
135            } else {
136                mention_str
137            };
138            content.push_str(cleaned_mention);
139            progress = idx2 + 1;
140        }
141    }
142    content.push_str(&s[progress..]);
143    content
144}

146fn clean_mention(guild: &Guild, mention: Mention, users: &[User]) -> Cow<'static, str> {
147    match mention {
148        Mention::Channel(id) => {
149            if let Some(channel) = guild.channel(id) {
150                format!("#{}", channel.base().name).into()
151            } else {
152                "#deleted-channel".into()
153            }
154        },
155        Mention::Role(id) => guild
156            .roles
157            .get(&id)
158            .map_or(Cow::Borrowed("@deleted-role"), |role| format!("@{}", role.name).into()),
159        Mention::User(id) => {
160            if let Some(member) = guild.members.get(&id) {
161                format!("@{}", member.display_name()).into()
162            } else if let Some(user) = users.iter().find(|u| u.id == id) {
163                format!("@{}", user.name).into()
164            } else {
165                "@invalid-user".into()
166            }
167        },
168    }
169}

173mod tests {
174    use small_fixed_array::FixedString;

181    #[test]
182    fn test_content_safe() {
183        let user = User {
184            id: UserId::new(100000000000000000),
185            name: FixedString::from_static_trunc("Crab"),
186            ..Default::default()
187        };
188
189        let no_member_guild = Guild {
190            id: GuildId::new(381880193251409931),
191            ..Default::default()
192        };
193
194        let mut guild = no_member_guild.clone();
195
196        let member = Member {
197            nick: Some(FixedString::from_static_trunc("Ferris")),
198            user: user.clone(),
199            ..Default::default()
200        };
201
202        let role = Role {
203            id: RoleId::new(333333333333333333),
204            name: FixedString::from_static_trunc("ferris-club-member"),
205            ..Default::default()
206        };
207
208        let channel = GuildChannel {
209            id: ChannelId::new(111880193700067777),
210            base: BaseGuildChannel {
211                name: FixedString::from_static_trunc("general"),
212                ..Default::default()
213            },
214            ..Default::default()
215        };
216
217        guild.channels.insert(channel.clone());
218        guild.members.insert(member.clone());
219        guild.roles.insert(role);
220
221        let with_user_mentions = "<@!100000000000000000> <@!000000000000000000> <@123> <@!123> \
222        <@!123123123123123123123> <@123> <@123123123123123123> <@!invalid> \
223        <@invalid> <@日本語 한국어$§)[/__#\\(/&2032$§#> \
224        <@!i)/==(<<>z/9080)> <@!1231invalid> <@invalid123> \
225        <@123invalid> <@> <@ ";
226
227        let without_user_mentions = "@Ferris @invalid-user @invalid-user @invalid-user \
228        <@!123123123123123123123> @invalid-user @invalid-user <@!invalid> \
229        <@invalid> <@日本語 한국어$§)[/__#\\(/&2032$§#> \
230        <@!i)/==(<<>z/9080)> <@!1231invalid> <@invalid123> \
231        <@123invalid> <@> <@ ";
232
233        // User mentions
234        let options = ContentSafeOptions::default();
235        assert_eq!(without_user_mentions, content_safe(&guild, with_user_mentions, options, &[]));
236
237        assert_eq!(
238            "@invalid-user",
239            content_safe(&no_member_guild, "<@100000000000000001>", options, &[])
240        );
241
242        let mut options = ContentSafeOptions::default();
243        assert_eq!(
244            format!("@{}", user.name),
245            content_safe(
246                &no_member_guild,
247                "<@!100000000000000000>",
248                options,
249                std::slice::from_ref(&user)
250            )
251        );
252
253        assert_eq!(
254            "@invalid-user",
255            content_safe(&no_member_guild, "<@!100000000000000000>", options, &[])
256        );
257
258        assert_eq!(
259            format!("@{}", member.nick.as_ref().unwrap()),
260            content_safe(&guild, "<@100000000000000000>", options, &[])
261        );
262
263        options = options.clean_user(false);
264        assert_eq!(with_user_mentions, content_safe(&guild, with_user_mentions, options, &[]));
265
266        // Channel mentions

268        #unsafe-club <#111880193700067777> <#ferrisferrisferris> \
269        <#000000000000000001>";
270
271        let without_channel_mentions = "<#> <#deleted-channel> #deleted-channel \
272        #deleted-channel #unsafe-club #general <#ferrisferrisferris> \
273        #deleted-channel";
274
275        assert_eq!(
276            without_channel_mentions,
277            content_safe(&guild, with_channel_mentions, options, &[])
278        );
279
280        options = options.clean_channel(false);
281        assert_eq!(
282            with_channel_mentions,
283            content_safe(&guild, with_channel_mentions, options, &[])
284        );
285
286        // Role mentions
287        let with_role_mentions = "<@&> @deleted-role <@&9829> \
288        <@&333333333333333333> <@&000000000000000001> \
289        <@&111111111111111111111111111111> <@&<@&1234>";
290
291        let without_role_mentions = "<@&> @deleted-role @deleted-role \
292        @ferris-club-member @deleted-role \
293        <@&111111111111111111111111111111> <@&@deleted-role";
294
295        assert_eq!(without_role_mentions, content_safe(&guild, with_role_mentions, options, &[]));
296
297        options = options.clean_role(false);
298        assert_eq!(with_role_mentions, content_safe(&guild, with_role_mentions, options, &[]));
299
300        // Everyone mentions

302        let without_everyone_mention = "@\u{200B}everyone";
303
304        assert_eq!(
305            without_everyone_mention,
306            content_safe(&guild, with_everyone_mention, options, &[])
307        );
308
309        options = options.clean_everyone(false);
310        assert_eq!(
311            with_everyone_mention,
312            content_safe(&guild, with_everyone_mention, options, &[])
313        );
314
315        // Here mentions

318        let without_here_mention = "@\u{200B}here";

324    }
