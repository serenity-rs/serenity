21/// Retrieves the "code" part of an invite out of a URL.
22///
23/// # Examples
24///
25/// Two formats of [invite][`Invite`] codes are supported, both regardless of protocol prefix.
26/// Some examples:
27///
28/// 1. Retrieving the code from the URL `"https://discord.gg/0cDvIgU2voY8RSYL"`:
29///
30/// ```rust
31/// use serenity::utils;
32///
33/// let url = "https://discord.gg/0cDvIgU2voY8RSYL";
34///
35/// assert_eq!(utils::parse_invite(url), "0cDvIgU2voY8RSYL");
36/// ```
37///
38/// 2. Retrieving the code from the URL `"http://discord.com/invite/0cDvIgU2voY8RSYL"`:
39///
40/// ```rust
41/// use serenity::utils;
42///
43/// let url = "http://discord.com/invite/0cDvIgU2voY8RSYL";
44///
45/// assert_eq!(utils::parse_invite(url), "0cDvIgU2voY8RSYL");
46/// ```
47///
48/// [`Invite`]: crate::model::invite::Invite
49#[must_use]
50pub fn parse_invite(code: &str) -> &str {
51    let code = code.trim_start_matches("http://").trim_start_matches("https://");
52    let lower = code.to_lowercase();
53    if lower.starts_with("discord.gg/") {
54        &code[11..]
55    } else if lower.starts_with("discord.com/invite/") {
56        &code[19..]
57    } else {
58        code
59    }
60}

62/// Retrieves the username and discriminator out of a user tag (`name#discrim`).
63/// In order to accomodate next gen Discord usernames, this will also accept `name` style tags.
64///
65/// If the user tag is invalid, None is returned.
66///
67/// # Examples
68/// ```rust
69/// use std::num::NonZeroU16;
70///
71/// use serenity::utils::parse_user_tag;
72///
73/// assert_eq!(parse_user_tag("kangalioo#9108"), Some(("kangalioo", NonZeroU16::new(9108))));
74/// assert_eq!(parse_user_tag("kangalioo#10108"), None);
75/// assert_eq!(parse_user_tag("kangalioo"), Some(("kangalioo", None)));
76/// ```
77#[must_use]
78pub fn parse_user_tag(s: &str) -> Option<(&str, Option<NonZeroU16>)> {
79    if let Some((name, discrim)) = s.split_once('#') {
80        let discrim: u16 = discrim.parse().ok()?;
81        if discrim > 9999 {
82            return None;
83        }
84        Some((name, NonZeroU16::new(discrim)))
85    } else {
86        Some((s, None))
87    }
88}

90/// Retrieves an Id from a user mention.
91///
92/// If the mention is invalid, then [`None`] is returned.
93///
94/// # Examples
95///
96/// Retrieving an Id from a valid [`User`] mention:
97///
98/// ```rust
99/// use serenity::model::id::UserId;
100/// use serenity::utils::parse_user_mention;
101///
102/// // regular username mention
103/// assert_eq!(parse_user_mention("<@114941315417899012>"), Some(UserId::new(114941315417899012)));
104///
105/// // nickname mention
106/// assert_eq!(parse_user_mention("<@!114941315417899012>"), Some(UserId::new(114941315417899012)));
107/// ```
108///
109/// Asserting that an invalid username or nickname mention returns [`None`]:
110///
111/// ```rust
112/// use serenity::utils::parse_user_mention;
113///
114/// assert!(parse_user_mention("<@1149413154aa17899012").is_none());
115/// assert!(parse_user_mention("<@!11494131541789a90b1c2").is_none());
116/// ```
117///
118/// [`User`]: crate::model::user::User
119#[must_use]
120pub fn parse_user_mention(mention: &str) -> Option<UserId> {
121    if mention.len() < 4 {
122        return None;
123    }
124
125    let len = mention.len() - 1;
126    if mention.starts_with("<@!") {
127        mention[3..len].parse().ok()
128    } else if mention.starts_with("<@") {
129        mention[2..len].parse().ok()
130    } else {
131        None
132    }
133}

215/// Retrieves the animated state, name and Id from an emoji mention, in the form of an
216/// [`EmojiIdentifier`].
217///
218/// If the emoji usage is invalid, then [`None`] is returned.
219///
220/// # Examples
221///
222/// Ensure that a valid [`Emoji`] usage is correctly parsed:
223///
224/// ```rust
225/// use serenity::model::id::{EmojiId, GuildId};
226/// use serenity::model::misc::EmojiIdentifier;
227/// use serenity::utils::parse_emoji;
228///
229/// let emoji = parse_emoji("<:smugAnimeFace:302516740095606785>").unwrap();
230/// assert_eq!(emoji.animated, false);
231/// assert_eq!(emoji.id, EmojiId::new(302516740095606785));
232/// assert_eq!(&*emoji.name, "smugAnimeFace");
233/// ```
234///
235/// Asserting that an invalid emoji usage returns [`None`]:
236///
237/// ```rust
238/// use serenity::utils::parse_emoji;
239///
240/// assert!(parse_emoji("<:smugAnimeFace:302516740095606785").is_none());
241/// ```
242///
243/// [`Emoji`]: crate::model::guild::Emoji
244#[must_use]
245pub fn parse_emoji(mention: &str) -> Option<EmojiIdentifier> {
246    let len = mention.len();
247    if !(6..=56).contains(&len) {
248        return None;
249    }
250
251    if (mention.starts_with("<:") || mention.starts_with("<a:")) && mention.ends_with('>') {
252        let mut name = String::default();
253        let mut id = String::default();
254        let animated = &mention[1..3] == "a:";
255
256        let start = if animated { 3 } else { 2 };
257
258        for (i, x) in mention[start..].chars().enumerate() {
259            if x == ':' {
260                let from = i + start + 1;
261
262                for y in mention[from..].chars() {
263                    if y == '>' {
264                        break;
265                    }
266                    id.push(y);
267                }
268
269                break;
270            }
271            name.push(x);
272        }
273
274        id.parse().ok().map(|id| EmojiIdentifier {
275            name: name.trunc_into(),
276            animated,
277            id,
278        })
279    } else {
280        None
281    }
282}

284/// Turns a string into a vector of string arguments, splitting by spaces, but parsing content
285/// within quotes as one individual argument.
286///
287/// # Examples
288///
289/// Parsing two quoted commands:
290///
291/// ```rust
292/// use serenity::utils::parse_quotes;
293///
294/// let command = r#""this is the first" "this is the second""#;
295/// let expected = vec!["this is the first".to_string(), "this is the second".to_string()];
296///
297/// assert_eq!(parse_quotes(command), expected);
298/// ```
299///
300/// ```rust
301/// use serenity::utils::parse_quotes;
302///
303/// let command = r#""this is a quoted command that doesn't have an ending quotation"#;
304/// let expected =
305///     vec!["this is a quoted command that doesn't have an ending quotation".to_string()];
306///
307/// assert_eq!(parse_quotes(command), expected);
308/// ```
309#[must_use]
310pub fn parse_quotes(s: &str) -> Vec<String> {
311    let mut args = vec![];
312    let mut in_string = false;
313    let mut escaping = false;
314    let mut current_str = String::default();
315
316    for x in s.chars() {
317        if in_string {
318            if x == '\\' && !escaping {
319                escaping = true;
320            } else if x == '"' && !escaping {
321                if !current_str.is_empty() {
322                    args.push(current_str);
323                }
324
325                current_str = String::default();
326                in_string = false;
327            } else {
328                current_str.push(x);
329                escaping = false;
330            }
331        } else if x == ' ' {
332            if !current_str.is_empty() {
333                args.push(current_str.clone());
334            }
335
336            current_str = String::default();
337        } else if x == '"' {
338            if !current_str.is_empty() {
339                args.push(current_str.clone());
340            }
341
342            in_string = true;
343            current_str = String::default();
344        } else {
345            current_str.push(x);
346        }
347    }
348
349    if !current_str.is_empty() {
350        args.push(current_str);
351    }
352
353    args
354}

356/// Discord's official domains. This is used in [`parse_webhook`] and in its corresponding test.
357const DOMAINS: [&str; 6] = [
358    "discord.com",
359    "canary.discord.com",
360    "ptb.discord.com",
361    "discordapp.com",
362    "canary.discordapp.com",
363    "ptb.discordapp.com",
364];
365
366const MAX_DOMAIN_LEN: usize = {
367    let mut max_len = 0;
368    let mut i = 0;
369
370    while i < DOMAINS.len() {
371        let cur_len = DOMAINS[i].len();
372        if cur_len > max_len {
373            max_len = cur_len;
374        }
375
376        i += 1;
377    }
378
379    max_len
380};

382/// Parses the id and token from a webhook url. Expects a [`url::Url`] rather than a [`&str`].
383///
384/// # Examples
385///
386/// ```rust
387/// use serenity::utils;
388///
389/// let url_str = "https://discord.com/api/webhooks/245037420704169985/ig5AO-wdVWpCBtUUMxmgsWryqgsW3DChbKYOINftJ4DCrUbnkedoYZD0VOH1QLr-S3sV";
390/// let url = url_str.parse().unwrap();
391/// let (id, token) = utils::parse_webhook(&url).unwrap();
392///
393/// assert_eq!(id, 245037420704169985);
394/// assert_eq!(token, "ig5AO-wdVWpCBtUUMxmgsWryqgsW3DChbKYOINftJ4DCrUbnkedoYZD0VOH1QLr-S3sV");
395/// ```
396#[must_use]
397pub fn parse_webhook(url: &Url) -> Option<(WebhookId, &str)> {
398    let (webhook_id, token) = url.path().strip_prefix("/api/webhooks/")?.split_once('/')?;
399    if !["http", "https"].contains(&url.scheme())
400        || !DOMAINS.contains(&url.domain()?)
401        || !(17..=20).contains(&webhook_id.len())
402        || !(60..=68).contains(&token.len())
403    {
404        return None;
405    }
406    Some((webhook_id.parse().ok()?, token))
407}

409/// Retrieves IDs from "{channel ID}-{message ID}" (retrieved by shift-clicking on "Copy ID").
410///
411/// If the string is invalid, None is returned.
412///
413/// # Examples
414/// ```rust
415/// use serenity::model::prelude::*;
416/// use serenity::utils::parse_message_id_pair;
417///
418/// assert_eq!(
419///     parse_message_id_pair("673965002805477386-842482646604972082"),
420///     Some((GenericChannelId::new(673965002805477386), MessageId::new(842482646604972082))),
421/// );
422/// assert_eq!(
423///     parse_message_id_pair("673965002805477386-842482646604972082-472029906943868929"),
424///     None,
425/// );
426/// ```
427#[must_use]
428pub fn parse_message_id_pair(s: &str) -> Option<(GenericChannelId, MessageId)> {
429    let mut parts = s.splitn(2, '-');
430    let channel_id = parts.next()?.parse().ok()?;
431    let message_id = parts.next()?.parse().ok()?;
432    Some((channel_id, message_id))
433}

435/// Retrieves guild, channel, and message ID from a message URL.
436///
437/// If the URL is malformed, None is returned.
438///
439/// # Examples
440/// ```rust
441/// use serenity::model::prelude::*;
442/// use serenity::utils::parse_message_url;
443///
444/// assert_eq!(
445///     parse_message_url(
446///         "https://discord.com/channels/381880193251409931/381880193700069377/806164913558781963"
447///     ),
448///     Some((
449///         GuildId::new(381880193251409931),
450///         GenericChannelId::new(381880193700069377),
451///         MessageId::new(806164913558781963),
452///     )),
453/// );
454/// assert_eq!(
455///     parse_message_url(
456///         "https://canary.discord.com/channels/381880193251409931/381880193700069377/806164913558781963"
457///     ),
458///     Some((
459///         GuildId::new(381880193251409931),
460///         GenericChannelId::new(381880193700069377),
461///         MessageId::new(806164913558781963),
462///     )),
463/// );
464/// assert_eq!(parse_message_url("https://google.com"), None);
465/// ```
466#[must_use]
467pub fn parse_message_url(s: &str) -> Option<(GuildId, GenericChannelId, MessageId)> {
468    use aformat::{CapStr, aformat};
469
470    for domain in DOMAINS {
471        let prefix = aformat!("https://{}/channels/", CapStr::<MAX_DOMAIN_LEN>(domain));
472        if let Some(parts) = s.strip_prefix(prefix.as_str()) {
473            let mut parts = parts.splitn(3, '/');
474
475            let guild_id = parts.next()?.parse().ok()?;
476            let channel_id = parts.next()?.parse().ok()?;
477            let message_id = parts.next()?.parse().ok()?;
478            return Some((guild_id, channel_id, message_id));
479        }
480    }
481    None
482}

484/// Retrieves guild, and channel ID from a channel URL.
485///
486/// If the URL is malformed, None is returned.
487///
488/// # Examples
489/// ```rust
490/// use serenity::model::prelude::*;
491/// use serenity::utils::parse_channel_url;

501///     Some((GuildId::new(381880193251409931), GenericChannelId::new(381880193700069377),)),
502/// );
503/// assert_eq!(parse_channel_url("https://google.com"), None);
504/// ```
505#[must_use]
506pub fn parse_channel_url(s: &str) -> Option<(GuildId, GenericChannelId)> {
507    use aformat::{CapStr, aformat};
508
509    for domain in DOMAINS {
510        let prefix = aformat!("https://{}/channels/", CapStr::<MAX_DOMAIN_LEN>(domain));
511        if let Some(parts) = s.strip_prefix(prefix.as_str()) {
512            let mut parts = parts.splitn(2, '/');
513
514            let guild_id = parts.next()?.parse().ok()?;
515            let channel_id = parts.next()?.parse().ok()?;
516            return Some((guild_id, channel_id));
517        }
518    }
519    None
520}

522/// Calculates the Id of the shard responsible for a guild, given its Id and total number of shards
523/// used.
524///
525/// # Examples
526///
527/// Retrieve the Id of the shard for a guild with Id `81384788765712384`, using 17 shards:
528///
529/// ```rust
530/// use serenity::model::id::GuildId;
531/// use serenity::utils;
532///
533/// let guild_id = GuildId::new(81384788765712384);
534/// let shard_total = std::num::NonZeroU16::new(17).unwrap();
535///
536/// assert_eq!(utils::shard_id(guild_id, shard_total), 7);
537/// ```
538#[must_use]
539pub fn shard_id(guild_id: GuildId, shard_count: NonZeroU16) -> u16 {
540    ((guild_id.get() >> 22) % u64::from(shard_count.get())) as u16
541}
