29enum_number! {
30    /// An enum representing the gateway opcodes.
31    ///
32    /// [Discord docs](https://docs.discord.com/developers/topics/opcodes-and-status-codes#gateway-gateway-opcodes).
33    #[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd, Deserialize, Serialize)]
34    #[non_exhaustive]
35    pub enum Opcode {
36        /// Dispatches an event.
37        Dispatch = 0,
38        /// Used for ping checking.
39        Heartbeat = 1,
40        /// Used for client handshake.
41        Identify = 2,
42        /// Used to update the client status.
43        PresenceUpdate = 3,
44        /// Used to join/move/leave voice channels.
45        VoiceStateUpdate = 4,
46        /// Used for voice ping checking.
47        VoiceServerPing = 5,
48        /// Used to resume a closed connection.
49        Resume = 6,
50        /// Used to tell clients to reconnect to the gateway.
51        Reconnect = 7,
52        /// Used to request guild members.
53        RequestGuildMembers = 8,
54        /// Used to notify clients that they have an invalid session Id.
55        InvalidSession = 9,
56        /// Sent immediately after connection, contains heartbeat + server info.
57        Hello = 10,
58        /// Sent immediately following a client heartbeat that was received.
59        HeartbeatAck = 11,
60        /// Used to request soundboard sounds from a list of guilds.
61        RequestSoundboardSounds = 31,
62        /// Used to request ephemeral channel data for channels in a guild.
63        RequestChannelInfo = 43,
64        _ => Unknown(u8),
65    }
66}

68enum_number! {
69    /// An enum representing the gateway close codes.
70    ///
71    /// [Discord docs](https://docs.discord.com/developers/topics/opcodes-and-status-codes#gateway-gateway-close-event-codes)
72    #[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd, Deserialize, Serialize)]
73    #[non_exhaustive]
74    pub enum CloseCode {
75        /// Unknown error; try reconnecting.
76        UnknownError = 4000,
77        /// Invalid gateway opcode.
78        UnknownOpcode = 4001,
79        /// An invalid payload was sent.
80        DecodeError = 4002,
81        /// A payload was sent prior to identifying, or the session was invalidated.
82        NotAuthenticated = 4003,
83        /// The account token sent with the identify payload was incorrect.
84        AuthenticationFailed = 4004,
85        /// More than one identify payload was sent.
86        AlreadyAuthenticated = 4005,
87        /// The sequence sent when resuming the session was invalid.
88        InvalidSequence = 4007,
89        /// Payloads were being sent too quickly.
90        RateLimited = 4008,
91        /// The gateway session timed out, and a new one must be started.
92        SessionTimeout = 4009,
93        /// An invalid shard was sent when identifying.
94        InvalidShard = 4010,
95        /// The session would have handled too many guilds; you must use sharding to connect.
96        ShardingRequired = 4011,
97        /// An invalid gateway API version was sent.
98        InvalidApiVersion = 4012,
99        /// An invalid gateway intent was sent.
100        InvalidGatewayIntents = 4013,
101        /// A disallowed gateway intent was sent; you may have it disabled or may not be approved
102        /// to use it.
103        DisallowedGatewayIntents = 4014,
104
105        _ => Unknown(u16),
106    }
