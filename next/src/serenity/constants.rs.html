38enum_number! {
39    /// An enum representing the gateway opcodes.
40    ///
41    /// [Discord docs](https://docs.discord.com/developers/topics/opcodes-and-status-codes#gateway-gateway-opcodes).
42    #[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd, Deserialize, Serialize)]
43    #[non_exhaustive]
44    pub enum Opcode {
45        /// Dispatches an event.
46        Dispatch = 0,
47        /// Used for ping checking.
48        Heartbeat = 1,
49        /// Used for client handshake.
50        Identify = 2,
51        /// Used to update the client status.
52        PresenceUpdate = 3,
53        /// Used to join/move/leave voice channels.
54        VoiceStateUpdate = 4,
55        /// Used for voice ping checking.
56        VoiceServerPing = 5,
57        /// Used to resume a closed connection.
58        Resume = 6,
59        /// Used to tell clients to reconnect to the gateway.
60        Reconnect = 7,
61        /// Used to request guild members.
62        RequestGuildMembers = 8,
63        /// Used to notify clients that they have an invalid session Id.
64        InvalidSession = 9,
65        /// Sent immediately after connection, contains heartbeat + server info.
66        Hello = 10,
67        /// Sent immediately following a client heartbeat that was received.
68        HeartbeatAck = 11,
69        /// Used to request soundboard sounds from a list of guilds.
70        RequestSoundboardSounds = 31,
71        _ => Unknown(u8),
72    }
73}

75enum_number! {
76    /// An enum representing the gateway close codes.
77    ///
78    /// [Discord docs](https://docs.discord.com/developers/topics/opcodes-and-status-codes#gateway-gateway-close-event-codes)
79    #[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd, Deserialize, Serialize)]
80    #[non_exhaustive]
81    pub enum CloseCode {
82        /// Unknown error; try reconnecting.
83        UnknownError = 4000,
84        /// Invalid gateway opcode.
85        UnknownOpcode = 4001,
86        /// An invalid payload was sent.
87        DecodeError = 4002,
88        /// A payload was sent prior to identifying, or the session was invalidated.
89        NotAuthenticated = 4003,
90        /// The account token sent with the identify payload was incorrect.
91        AuthenticationFailed = 4004,
92        /// More than one identify payload was sent.
93        AlreadyAuthenticated = 4005,
94        /// The sequence sent when resuming the session was invalid.
95        InvalidSequence = 4007,
96        /// Payloads were being sent too quickly.
97        RateLimited = 4008,
98        /// The gateway session timed out, and a new one must be started.
99        SessionTimeout = 4009,
100        /// An invalid shard was sent when identifying.
101        InvalidShard = 4010,
102        /// The session would have handled too many guilds; you must use sharding to connect.
103        ShardingRequired = 4011,
104        /// An invalid gateway API version was sent.
105        InvalidApiVersion = 4012,
106        /// An invalid gateway intent was sent.
107        InvalidGatewayIntents = 4013,
108        /// A disallowed gateway intent was sent; you may have it disabled or may not be approved
109        /// to use it.
110        DisallowedGatewayIntents = 4014,
111
112        _ => Unknown(u16),
113    }
