22/// A builder for creating a new attachment from a file path, file data, or URL.
23///
24/// [Discord docs](https://docs.discord.com/developers/resources/message#attachment-object-attachment-structure).
25#[derive(Clone, Debug)]
26#[non_exhaustive]
27#[must_use]
28pub struct CreateAttachment<'a> {
29    pub filename: Cow<'static, str>,
30    pub description: Option<Cow<'a, str>>,
31    pub spoiler: bool,
32    pub data: AttachmentData<'a>,
33}
34
35impl<'a> CreateAttachment<'a> {
36    /// Builds an [`CreateAttachment`] from the raw attachment data.
37    pub fn bytes(data: impl Into<Bytes>, filename: impl Into<Cow<'static, str>>) -> Self {
38        CreateAttachment {
39            filename: filename.into(),
40            description: None,
41            spoiler: false,
42            data: AttachmentData::Bytes(data.into()),
43        }
44    }
45
46    /// Builds an [`CreateAttachment`] by reading a local file.
47    ///
48    /// # Errors
49    ///
50    /// Returns [`Error::Io`] if the path is not a valid file path.
51    pub fn path(path: &'a Path) -> Result<Self> {
52        let filename = path
53            .file_name()
54            .ok_or_else(|| std::io::Error::other("attachment path must not be a directory"))?
55            .to_string_lossy()
56            .into_owned();
57        Ok(CreateAttachment {
58            filename: filename.into(),
59            description: None,
60            spoiler: false,
61            data: AttachmentData::Path(path),
62        })
63    }
64
65    /// Builds an [`CreateAttachment`] by reading from a file handler.
66    pub fn file(file: &'a File, filename: impl Into<Cow<'static, str>>) -> Self {
67        CreateAttachment {
68            filename: filename.into(),
69            description: None,
70            spoiler: false,
71            data: AttachmentData::File(file),
72        }
73    }
74
75    /// Builds an [`CreateAttachment`] by downloading attachment data from a URL.
76    ///
77    /// # Errors
78    ///
79    /// Returns [`Error::Http`] if downloading the data fails.
80    #[cfg(feature = "http")]
81    pub async fn url(
82        http: &Http,
83        url: impl reqwest::IntoUrl,
84        filename: impl Into<Cow<'static, str>>,
85    ) -> Result<Self> {
86        let response = http.client.get(url).send().await?;
87        let data = response.bytes().await?;
88
89        Ok(CreateAttachment::bytes(data, filename))
90    }
91
92    /// Returns the underlying data for the attachment.
93    ///
94    /// # Errors
95    ///
96    /// Returns an error if fetching the data failed in some way. If the attachment is a
97    /// [`CreateAttachment::path`], then the file at the specified path was unable to be read. If
98    /// instead it's [`CreateAttachment::file`], then cloning the handle to the file failed, likely
99    /// due to hitting the system's limit on number of open file handles.
100    pub async fn get_data(&self) -> Result<Bytes> {
101        match &self.data {
102            AttachmentData::Bytes(bytes) => Ok(bytes.clone()),
103            AttachmentData::Path(path) => {
104                let mut file = File::open(path).await?;
105                let mut data = Vec::new();
106                file.read_to_end(&mut data).await?;
107                Ok(data.into())
108            },
109            AttachmentData::File(file) => {
110                let mut data = Vec::new();
111                file.try_clone().await?.read_to_end(&mut data).await?;
112                Ok(data.into())
113            },
114        }
115    }
116
117    /// Converts the attachment data to a base64-encoded data URI.
118    ///
119    /// # Errors
120    ///
121    /// See [`CreateAttachment::get_data`] for details.
122    pub async fn encode(&self, mimetype: &str) -> Result<DataUri<'_>> {
123        use base64::engine::{Config, Engine};
124
125        let prefix = format!("data:{mimetype};base64,");
126        let data = self.get_data().await?;
127
128        let engine = base64::prelude::BASE64_STANDARD;
129        let encoded_size = base64::encoded_len(data.len(), engine.config().encode_padding())
130            .and_then(|len| len.checked_add(prefix.len()))
131            .expect("buffer capacity overflow");
132
133        let mut encoded = String::with_capacity(encoded_size);
134        encoded.push_str(&prefix);
135        engine.encode_string(&data, &mut encoded);
136        Ok(DataUri(encoded.into()))
137    }
138
139    /// Sets a description for the file (max 1024 characters).
140    pub fn description(mut self, description: impl Into<Cow<'a, str>>) -> Self {
141        self.description = Some(description.into());
142        self
143    }

145    /// Whether the attachment should be marked as a spoiler and blurred until clicked.
146    pub fn spoiler(mut self, spoiler: bool) -> Self {
147        self.spoiler = spoiler;
148        self
149    }

152/// A wrapper around some base64-encoded data. Used when an endpoint expects a base64 payload
153/// directly as part of the JSON body, instead of as a multipart upload.
154#[derive(Clone, Debug, Serialize)]
155#[serde(transparent)]
156pub struct DataUri<'a>(Cow<'a, str>);
157
158impl<'a> DataUri<'a> {
159    /// Accesses the stored base64-encoded data.
160    #[must_use]
161    pub fn as_base64(&self) -> &str {
162        &self.0
163    }
164
165    /// Constructs a [`DataUri`] from a base64-encoded blob of data. The string must be a valid data
166    /// URI, for example:
167    ///
168    /// ```
169    /// use serenity::builder::DataUri;
170    ///
171    /// let s = "data:image/png;base64,R0lGODlhAQABAIAAAP///wAAACwAAAAAAQABAAACAkQBADs=";
172    /// assert!(DataUri::from_base64(s).is_ok());
173    /// ```
174    ///
175    /// # Errors
176    ///
177    /// Returns a [`Error::Url`] if the string is not a valid data URI. See the [Discord
178    /// docs](https://docs.discord.com/developers/reference#image-data).
179    pub fn from_base64(s: impl Into<Cow<'a, str>>) -> Result<Self> {
180        let s = s.into();
181        if let Some(("data", tail)) = s.split_once(':')
182            && let Some((mimetype, encoding)) = tail.split_once(';')
183            && mimetype.split_once('/').is_some()
184            && encoding.starts_with("base64,")
185        {
186            Ok(Self(s))
187        } else {
188            Err(Error::Url(UrlError::InvalidDataURI))
189        }
190    }
191}

230#[derive(Clone, Debug)]
231enum EditAttachmentsInner<'a> {
232    New(CreateAttachment<'a>),
233    Existing(EditAttachment<'a>),
234}
235
236/// You can add new attachments and edit existing ones using this builder.
237///
238/// When this builder is _not_ supplied in a message edit, Discord keeps the attachments intact.
239/// However, as soon as a builder is supplied, Discord removes all attachments from the message. If
240/// you want to keep old attachments, you must specify this either using [`Self::keep_all`], or
241/// individually for each attachment using [`Self::keep`].
242///
243/// # Examples
244///
245/// ## Removing all attachments
246///
247/// ```rust,no_run
248/// # use serenity::all::*;
249/// # async fn foo_(ctx: Http, mut msg: Message) -> Result<(), Error> {
250/// msg.edit(ctx, EditMessage::new().attachments(EditAttachments::new())).await?;
251/// # Ok(()) }
252/// ```
253///
254/// ## Adding a new attachment without deleting existing attachments
255///
256/// ```rust,no_run
257/// # use serenity::all::*;
258/// # async fn foo_(ctx: Http, mut msg: Message, my_attachment: CreateAttachment<'_>) -> Result<(), Error> {
259/// msg.edit(ctx, EditMessage::new().attachments(
260///     EditAttachments::keep_all(&msg).add(my_attachment)
261/// )).await?;
262/// # Ok(()) }
263/// ```
264///
265/// ## Updating an existing attachment without deleting existing attachments
266///
267/// ```rust,no_run
268/// # use serenity::all::*;
269/// # async fn foo_(ctx: Http, mut msg: Message, my_attachment: CreateAttachment<'_>) -> Result<(), Error> {
270/// msg.edit(ctx, EditMessage::new().attachments(
271///     EditAttachments::keep_all(&msg).update(
272///         EditAttachment::new(msg.attachments[0].id)
273///             .description("updated attachment")
274///             .spoiler(true),
275///     ),
276/// )).await?;
277/// # Ok(()) }
278/// ```
279///
280/// ## Delete all but the first attachment
281///
282/// ```rust,no_run
283/// # use serenity::all::*;
284/// # async fn foo_(ctx: Http, mut msg: Message, my_attachment: CreateAttachment<'_>) -> Result<(), Error> {
285/// msg.edit(ctx, EditMessage::new().attachments(
286///     EditAttachments::new().keep(msg.attachments[0].id)
287/// )).await?;
288/// # Ok(()) }
289/// ```
290///
291/// ## Delete all but the first attachment, add a description, and mark it as a spoiler
292///
293/// ```rust,no_run
294/// # use serenity::all::*;
295/// # async fn foo_(ctx: Http, mut msg: Message, my_attachment: CreateAttachment<'_>) -> Result<(), Error> {
296/// msg.edit(ctx, EditMessage::new().attachments(
297///     EditAttachments::new().keep_and_update(
298///         EditAttachment::new(msg.attachments[0].id)
299///             .description("updated attachment")
300///             .spoiler(true),
301///     ),
302/// )).await?;
303/// # Ok(()) }
304/// ```
305///
306/// ## Delete only the first attachment
307///
308/// ```rust,no_run
309/// # use serenity::all::*;
310/// # async fn foo_(ctx: Http, mut msg: Message, my_attachment: CreateAttachment<'_>) -> Result<(), Error> {
311/// msg.edit(ctx, EditMessage::new().attachments(
312///     EditAttachments::keep_all(&msg).remove(msg.attachments[0].id)
313/// )).await?;
314/// # Ok(()) }
315/// ```
316///
317/// # Notes
318///
319/// Internally, this type is used not just for message editing endpoints, but also for message
320/// creation endpoints.
321#[derive(Default, Debug, Clone)]
322#[must_use]
323pub struct EditAttachments<'a> {
324    inner: Vec<EditAttachmentsInner<'a>>,
325}
326
327impl<'a> EditAttachments<'a> {
328    /// An empty attachments builder.
329    ///
330    /// Existing attachments are not kept by default, either. See [`Self::keep_all()`] or
331    /// [`Self::keep()`].
332    pub fn new() -> Self {
333        Self::default()
334    }
335
336    /// Creates a new attachments builder that keeps all existing attachments.
337    ///
338    /// Shorthand for [`Self::new()`] and calling [`Self::keep()`] for every [`AttachmentId`] in
339    /// [`Message::attachments`].
340    ///
341    /// This method may be paired with [`Self::update()`] to update the `description` (alt text)
342    /// and/or [`IS_SPOILER`] flag of individual existing attachments being kept.
343    ///
344    /// If you only want to keep a subset of attachments from the message, either implement this
345    /// method manually, or use [`Self::remove()`].
346    ///
347    /// **Note: this EditAttachments must be run on the same message as is supplied here, or else
348    /// Discord will throw an error!**
349    ///
350    /// [`IS_SPOILER`]: crate::model::channel::AttachmentFlags::IS_SPOILER
351    pub fn keep_all(msg: &Message) -> Self {
352        Self {
353            inner: msg
354                .attachments
355                .iter()
356                .map(|a| {
357                    EditAttachmentsInner::Existing(EditAttachment {
358                        id: a.id,
359                        description: None,
360                        spoiler: None,
361                    })
362                })
363                .collect(),
364        }
365    }
366
367    /// This method adds an existing attachment to the list of attachments that are kept after
368    /// editing.
369    ///
370    /// Opposite of [`Self::remove`].
371    pub fn keep(mut self, id: AttachmentId) -> Self {
372        self.inner.push(EditAttachmentsInner::Existing(EditAttachment {
373            id,
374            description: None,
375            spoiler: None,
376        }));
377        self
378    }
379
380    /// This method updates the `description` (alt text) and/or [`IS_SPOILER`] flag of an
381    /// existing attachment.
382    ///
383    /// This will also add the existing attachment to the list of attachments that are kept
384    /// after editing if not already done explicitly via [`Self::keep_all()`] or [`Self::keep()`],
385    /// or implicitly via [`Self::remove()`]. However, when keeping and updating a single
386    /// attachment, [`Self::keep_and_update()`] should be the preferred method.
387    ///
388    /// [`IS_SPOILER`]: crate::model::channel::AttachmentFlags::IS_SPOILER
389    pub fn update(mut self, attachment: EditAttachment<'a>) -> Self {
390        for inner in &mut self.inner {
391            if let EditAttachmentsInner::Existing(existing) = inner
392                && existing.id == attachment.id
393            {
394                existing.description = attachment.description;
395                existing.spoiler = attachment.spoiler;
396                return self;
397            }
398        }
399        self.inner.push(EditAttachmentsInner::Existing(EditAttachment {
400            id: attachment.id,
401            description: attachment.description,
402            spoiler: attachment.spoiler,
403        }));
404        self
405    }
406
407    /// This method adds an existing attachment to the list of attachments that are kept after
408    /// editing, and optionally updates the `description` (alt text) and/or [`IS_SPOILER`] flag.
409    ///
410    /// [`IS_SPOILER`]: crate::model::channel::AttachmentFlags::IS_SPOILER
411    pub fn keep_and_update(mut self, attachment: EditAttachment<'a>) -> Self {
412        self.inner.push(EditAttachmentsInner::Existing(EditAttachment {
413            id: attachment.id,
414            description: attachment.description,
415            spoiler: attachment.spoiler,
416        }));
417        self
418    }
419
420    /// This method removes an existing attachment from the list of attachments that are kept after
421    /// editing.
422    ///
423    /// Opposite of [`Self::keep`].
424    pub fn remove(mut self, id: AttachmentId) -> Self {
425        self.inner.retain(|a| match a {
426            EditAttachmentsInner::Existing(attachment) => attachment.id != id,
427            EditAttachmentsInner::New(_) => true,
428        });
429        self
430    }
431
432    /// Adds a new attachment to the attachment list.
433    #[expect(clippy::should_implement_trait)] // Clippy thinks add == std::ops::Add::add
434    pub fn add(mut self, attachment: CreateAttachment<'a>) -> Self {
435        self.inner.push(EditAttachmentsInner::New(attachment));
436        self
437    }
438
439    /// Clones all new attachments into a new Vec, keeping only data and filename, because those
440    /// are needed for the multipart form data.
441    #[cfg(feature = "http")]
442    pub(crate) fn new_attachments(&self) -> Vec<CreateAttachment<'a>> {
443        self.inner
444            .iter()
445            .filter_map(|attachment| {
446                if let EditAttachmentsInner::New(attachment) = &attachment {
447                    Some(attachment.clone())
448                } else {
449                    None
450                }
451            })
452            .collect()
453    }
454}

456impl Serialize for EditAttachments<'_> {
457    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
458        #[derive(Serialize)]
459        struct NewAttachment<'a> {
460            id: u64,
461            filename: &'a Cow<'static, str>,
462            description: &'a Option<Cow<'a, str>>,
463            is_spoiler: bool,
464        }
465
466        #[derive(Serialize)]
467        struct ExistingAttachment<'a> {
468            id: AttachmentId,
469            #[serde(skip_serializing_if = "Option::is_none")]
470            description: &'a Option<Cow<'a, str>>,
471            #[serde(skip_serializing_if = "Option::is_none")]
472            is_spoiler: Option<bool>,
473        }
474
475        // Instead of an `AttachmentId`, the `id` field for new attachments corresponds to the
476        // index of the new attachment in the multipart payload. The attachment data will be
477        // labeled with `files[{id}]` in the multipart body. See `Multipart::build_form`.
478        let mut id = 0;
479        let mut seq = serializer.serialize_seq(Some(self.inner.len()))?;
480        for attachment in &self.inner {
481            match attachment {
482                EditAttachmentsInner::New(new_attachment) => {
483                    let attachment = NewAttachment {
484                        id,
485                        filename: &new_attachment.filename,
486                        description: &new_attachment.description,
487                        is_spoiler: new_attachment.spoiler,
488                    };
489                    id += 1;
490                    seq.serialize_element(&attachment)?;
491                },
492                EditAttachmentsInner::Existing(attachment) => {
493                    seq.serialize_element(&ExistingAttachment {
494                        id: attachment.id,
495                        description: &attachment.description,
496                        is_spoiler: attachment.spoiler,
497                    })?;
498                },
499            }
500        }
501        seq.end()
502    }
