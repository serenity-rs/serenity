1use std::borrow::Cow;
2
3use crate::model::prelude::*;
4
5/// A builder to create or edit a [`Soundboard`] for use with [`GuildId::edit_soundboard`].
6///
7/// [Discord docs](https://docs.discord.com/developers/resources/soundboard#modify-guild-soundboard-sound)
8#[derive(Clone, Debug, Default, Serialize)]
9#[must_use]
10pub struct EditSoundboard<'a> {
11    #[serde(skip_serializing_if = "Option::is_none")]
12    name: Option<Cow<'a, str>>,
13    #[serde(skip_serializing_if = "Option::is_none")]
14    volume: Option<f64>,
15    #[serde(skip_serializing_if = "Option::is_none")]
16    emoji_id: Option<EmojiId>,
17    #[serde(skip_serializing_if = "Option::is_none")]
18    emoji_name: Option<Cow<'a, str>>,
19
20    #[serde(skip)]
21    audit_log_reason: Option<&'a str>,
22}
23
24impl<'a> EditSoundboard<'a> {
25    /// Equivalent to [`Self::default`].
26    pub fn new() -> Self {
27        Self::default()
28    }
29
30    /// The name of the soundboard sound to set.
31    ///
32    /// **Note**: Must be between 2 and 32 characters long.
33    pub fn name(mut self, name: impl Into<Cow<'a, str>>) -> Self {
34        self.name = Some(name.into());
35        self
36    }
37
38    /// Set the volume of the soundboard sound.
39    ///
40    /// **Note**: Must be between 0 to 1.
41    pub fn volume(mut self, volume: f64) -> Self {
42        self.volume = volume.into();
43        self
44    }
45
46    /// Set the ID of the custom emoji.
47    pub fn emoji_id(mut self, id: EmojiId) -> Self {
48        self.emoji_id = Some(id);
49        self
50    }
51
52    /// Set the Unicode character of the custom emoji.
53    pub fn emoji_name(mut self, name: impl Into<Cow<'a, str>>) -> Self {
54        self.emoji_name = Some(name.into());
55        self
56    }
57
58    /// Sets the request's audit log reason.
59    pub fn audit_log_reason(mut self, reason: &'a str) -> Self {
60        self.audit_log_reason = Some(reason);
61        self
62    }
63
64    /// Edits the soundboard sound.
65    ///
66    /// **Note**: If the soundboard sound was created by the current user, requires either the
67    /// [Create Guild Expressions] or the [Manage Guild Expressions] permission. Otherwise, the
68    /// [Manage Guild Expressions] permission is required.
69    ///
70    /// # Errors
71    ///
72    /// Returns [`Error::Http`] if the current user lacks permission, or if invalid data is given.
73    ///
74    /// [Create Guild Expressions]: Permissions::CREATE_GUILD_EXPRESSIONS
75    /// [Manage Guild Expressions]: Permissions::MANAGE_GUILD_EXPRESSIONS
76    #[cfg(feature = "http")]
77    pub async fn execute(
78        self,
79        cache_http: impl CacheHttp,
80        guild_id: GuildId,
81        sound_id: SoundId,
82    ) -> Result<Soundboard> {
83        cache_http
84            .http()
85            .edit_guild_soundboard(guild_id, sound_id, &self, self.audit_log_reason)
86            .await
87    }
88}
