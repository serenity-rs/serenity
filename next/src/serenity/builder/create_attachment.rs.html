17    Bytes(Bytes),
18    File(&'a File),
19    Path(&'a Path),
20}

22/// A builder for creating a new attachment from a file path, file data, or URL.
23///
24/// [Discord docs](https://docs.discord.com/developers/resources/message#attachment-object-attachment-structure).
25#[derive(Clone, Debug)]
26#[non_exhaustive]
27#[must_use]
28pub struct CreateAttachment<'a> {
29    pub filename: Cow<'static, str>,
30    pub description: Option<Cow<'a, str>>,
31    pub data: AttachmentData<'a>,
32}
33
34impl<'a> CreateAttachment<'a> {
35    /// Builds an [`CreateAttachment`] from the raw attachment data.
36    pub fn bytes(data: impl Into<Bytes>, filename: impl Into<Cow<'static, str>>) -> Self {
37        CreateAttachment {
38            filename: filename.into(),
39            description: None,
40            data: AttachmentData::Bytes(data.into()),
41        }
42    }
43
44    /// Builds an [`CreateAttachment`] by reading a local file.
45    ///
46    /// # Errors
47    ///
48    /// Returns [`Error::Io`] if the path is not a valid file path.
49    pub fn path(path: &'a Path) -> Result<Self> {
50        let filename = path
51            .file_name()
52            .ok_or_else(|| std::io::Error::other("attachment path must not be a directory"))?
53            .to_string_lossy()
54            .into_owned();
55        Ok(CreateAttachment {
56            filename: filename.into(),
57            description: None,
58            data: AttachmentData::Path(path),
59        })
60    }
61
62    /// Builds an [`CreateAttachment`] by reading from a file handler.
63    pub fn file(file: &'a File, filename: impl Into<Cow<'static, str>>) -> Self {
64        CreateAttachment {
65            filename: filename.into(),
66            description: None,
67            data: AttachmentData::File(file),
68        }
69    }
70
71    /// Builds an [`CreateAttachment`] by downloading attachment data from a URL.
72    ///
73    /// # Errors
74    ///
75    /// Returns [`Error::Http`] if downloading the data fails.
76    #[cfg(feature = "http")]
77    pub async fn url(
78        http: &Http,
79        url: impl reqwest::IntoUrl,
80        filename: impl Into<Cow<'static, str>>,
81    ) -> Result<Self> {
82        let response = http.client.get(url).send().await?;
83        let data = response.bytes().await?;
84
85        Ok(CreateAttachment::bytes(data, filename))
86    }
87
88    /// Returns the underlying data for the attachment.
89    ///
90    /// # Errors
91    ///
92    /// Returns an error if fetching the data failed in some way. If the attachment is a
93    /// [`CreateAttachment::path`], then the file at the specified path was unable to be read. If
94    /// instead it's [`CreateAttachment::file`], then cloning the handle to the file failed, likely
95    /// due to hitting the system's limit on number of open file handles.
96    pub async fn get_data(&self) -> Result<Bytes> {
97        match &self.data {
98            AttachmentData::Bytes(bytes) => Ok(bytes.clone()),
99            AttachmentData::Path(path) => {
100                let mut file = File::open(path).await?;
101                let mut data = Vec::new();
102                file.read_to_end(&mut data).await?;
103                Ok(data.into())
104            },
105            AttachmentData::File(file) => {
106                let mut data = Vec::new();
107                file.try_clone().await?.read_to_end(&mut data).await?;
108                Ok(data.into())
109            },
110        }
111    }
112
113    /// Converts the attachment data to a base64-encoded data URI.
114    ///
115    /// # Errors
116    ///
117    /// See [`CreateAttachment::get_data`] for details.
118    pub async fn encode(&self, mimetype: &str) -> Result<DataUri<'_>> {
119        use base64::engine::{Config, Engine};
120
121        let prefix = format!("data:{mimetype};base64,");
122        let data = self.get_data().await?;
123
124        let engine = base64::prelude::BASE64_STANDARD;
125        let encoded_size = base64::encoded_len(data.len(), engine.config().encode_padding())
126            .and_then(|len| len.checked_add(prefix.len()))
127            .expect("buffer capacity overflow");
128
129        let mut encoded = String::with_capacity(encoded_size);
130        encoded.push_str(&prefix);
131        engine.encode_string(&data, &mut encoded);
132        Ok(DataUri(encoded.into()))
133    }
134
135    /// Sets a description for the file (max 1024 characters).
136    pub fn description(mut self, description: impl Into<Cow<'a, str>>) -> Self {
137        self.description = Some(description.into());
138        self
139    }
140}

142/// A wrapper around some base64-encoded data. Used when an endpoint expects a base64 payload
143/// directly as part of the JSON body, instead of as a multipart upload.
144#[derive(Clone, Debug, Serialize)]
145#[serde(transparent)]
146pub struct DataUri<'a>(Cow<'a, str>);
147
148impl<'a> DataUri<'a> {
149    /// Accesses the stored base64-encoded data.
150    #[must_use]
151    pub fn as_base64(&self) -> &str {
152        &self.0
153    }
154
155    /// Constructs a [`DataUri`] from a base64-encoded blob of data. The string must be a valid data
156    /// URI, for example:
157    ///
158    /// ```
159    /// use serenity::builder::DataUri;
160    ///
161    /// let s = "data:image/png;base64,R0lGODlhAQABAIAAAP///wAAACwAAAAAAQABAAACAkQBADs=";
162    /// assert!(DataUri::from_base64(s).is_ok());
163    /// ```
164    ///
165    /// # Errors
166    ///
167    /// Returns a [`Error::Url`] if the string is not a valid data URI. See the [Discord
168    /// docs](https://docs.discord.com/developers/reference#image-data).
169    pub fn from_base64(s: impl Into<Cow<'a, str>>) -> Result<Self> {
170        let s = s.into();
171        if let Some(("data", tail)) = s.split_once(':')
172            && let Some((mimetype, encoding)) = tail.split_once(';')
173            && mimetype.split_once('/').is_some()
174            && encoding.starts_with("base64,")
175        {
176            Ok(Self(s))
177        } else {
178            Err(Error::Url(UrlError::InvalidDataURI))
179        }
180    }
181}

183#[derive(Clone, Debug)]
184enum EditAttachmentsInner<'a> {
185    New(CreateAttachment<'a>),
186    Existing(AttachmentId),
187}
188
189/// You can add new attachments and edit existing ones using this builder.
190///
191/// When this builder is _not_ supplied in a message edit, Discord keeps the attachments intact.
192/// However, as soon as a builder is supplied, Discord removes all attachments from the message. If
193/// you want to keep old attachments, you must specify this either using [`Self::keep_all`], or
194/// individually for each attachment using [`Self::keep`].
195///
196/// # Examples
197///
198/// ## Removing all attachments
199///
200/// ```rust,no_run
201/// # use serenity::all::*;
202/// # async fn foo_(ctx: Http, mut msg: Message) -> Result<(), Error> {
203/// msg.edit(ctx, EditMessage::new().attachments(EditAttachments::new())).await?;
204/// # Ok(()) }
205/// ```
206///
207/// ## Adding a new attachment without deleting existing attachments
208///
209/// ```rust,no_run
210/// # use serenity::all::*;
211/// # async fn foo_(ctx: Http, mut msg: Message, my_attachment: CreateAttachment<'_>) -> Result<(), Error> {
212/// msg.edit(ctx, EditMessage::new().attachments(
213///     EditAttachments::keep_all(&msg).add(my_attachment)
214/// )).await?;
215/// # Ok(()) }
216/// ```
217///
218/// ## Delete all but the first attachment
219///
220/// ```rust,no_run
221/// # use serenity::all::*;
222/// # async fn foo_(ctx: Http, mut msg: Message, my_attachment: CreateAttachment<'_>) -> Result<(), Error> {
223/// msg.edit(ctx, EditMessage::new().attachments(
224///     EditAttachments::new().keep(msg.attachments[0].id)
225/// )).await?;
226/// # Ok(()) }
227/// ```
228///
229/// ## Delete only the first attachment
230///
231/// ```rust,no_run
232/// # use serenity::all::*;
233/// # async fn foo_(ctx: Http, mut msg: Message, my_attachment: CreateAttachment<'_>) -> Result<(), Error> {
234/// msg.edit(ctx, EditMessage::new().attachments(
235///     EditAttachments::keep_all(&msg).remove(msg.attachments[0].id)
236/// )).await?;
237/// # Ok(()) }
238/// ```
239///
240/// # Notes
241///
242/// Internally, this type is used not just for message editing endpoints, but also for message
243/// creation endpoints.
244#[derive(Default, Debug, Clone)]
245#[must_use]
246pub struct EditAttachments<'a> {
247    inner: Vec<EditAttachmentsInner<'a>>,
248}
249
250impl<'a> EditAttachments<'a> {
251    /// An empty attachments builder.
252    ///
253    /// Existing attachments are not kept by default, either. See [`Self::keep_all()`] or
254    /// [`Self::keep()`].
255    pub fn new() -> Self {
256        Self::default()
257    }
258
259    /// Creates a new attachments builder that keeps all existing attachments.
260    ///
261    /// Shorthand for [`Self::new()`] and calling [`Self::keep()`] for every [`AttachmentId`] in
262    /// [`Message::attachments`].
263    ///
264    /// If you only want to keep a subset of attachments from the message, either implement this
265    /// method manually, or use [`Self::remove()`].
266    ///
267    /// **Note: this EditAttachments must be run on the same message as is supplied here, or else
268    /// Discord will throw an error!**
269    pub fn keep_all(msg: &Message) -> Self {
270        Self {
271            inner: msg.attachments.iter().map(|a| EditAttachmentsInner::Existing(a.id)).collect(),
272        }
273    }
274
275    /// This method adds an existing attachment to the list of attachments that are kept after
276    /// editing.
277    ///
278    /// Opposite of [`Self::remove`].
279    pub fn keep(mut self, id: AttachmentId) -> Self {
280        self.inner.push(EditAttachmentsInner::Existing(id));
281        self
282    }
283
284    /// This method removes an existing attachment from the list of attachments that are kept after
285    /// editing.
286    ///
287    /// Opposite of [`Self::keep`].
288    pub fn remove(mut self, id: AttachmentId) -> Self {
289        self.inner.retain(|a| match a {
290            EditAttachmentsInner::Existing(existing_id) => *existing_id != id,
291            EditAttachmentsInner::New(_) => true,
292        });
293        self
294    }
295
296    /// Adds a new attachment to the attachment list.
297    #[expect(clippy::should_implement_trait)] // Clippy thinks add == std::ops::Add::add
298    pub fn add(mut self, attachment: CreateAttachment<'a>) -> Self {
299        self.inner.push(EditAttachmentsInner::New(attachment));
300        self
301    }
302
303    /// Clones all new attachments into a new Vec, keeping only data and filename, because those
304    /// are needed for the multipart form data.
305    #[cfg(feature = "http")]
306    pub(crate) fn new_attachments(&self) -> Vec<CreateAttachment<'a>> {
307        self.inner
308            .iter()
309            .filter_map(|attachment| {
310                if let EditAttachmentsInner::New(attachment) = &attachment {
311                    Some(attachment.clone())
312                } else {
313                    None
314                }
315            })
316            .collect()
317    }
318}

320impl Serialize for EditAttachments<'_> {
321    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
322        #[derive(Serialize)]
323        struct NewAttachment<'a> {
324            id: u64,
325            filename: &'a Cow<'static, str>,
326            description: &'a Option<Cow<'a, str>>,
327        }
328
329        #[derive(Serialize)]
330        struct ExistingAttachment {
331            id: AttachmentId,
332        }
333
334        // Instead of an `AttachmentId`, the `id` field for new attachments corresponds to the
335        // index of the new attachment in the multipart payload. The attachment data will be
336        // labeled with `files[{id}]` in the multipart body. See `Multipart::build_form`.
337        let mut id = 0;
338        let mut seq = serializer.serialize_seq(Some(self.inner.len()))?;
339        for attachment in &self.inner {
340            match attachment {
341                EditAttachmentsInner::New(new_attachment) => {
342                    let attachment = NewAttachment {
343                        id,
344                        filename: &new_attachment.filename,
345                        description: &new_attachment.description,
346                    };
347                    id += 1;
348                    seq.serialize_element(&attachment)?;
349                },
350                EditAttachmentsInner::Existing(id) => {
351                    seq.serialize_element(&ExistingAttachment {
352                        id: *id,
353                    })?;
354                },
355            }
356        }
357        seq.end()
358    }
