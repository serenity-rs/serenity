9/// A builder to create an [`Invite`] for use via [`ChannelId::create_invite`].
10///
11/// This is a structured and cleaner way of creating an invite, as all parameters are optional.
12///
13/// # Examples
14///
15/// Create an invite with a max age of 3600 seconds and 10 max uses:
16///
17/// ```rust,no_run
18/// # use serenity::{prelude::*, model::prelude::*};
19/// use serenity::builder::CreateInvite;
20/// use serenity::http::Http;
21/// # async fn run(http: &Http, channel_id: ChannelId) -> Result<(), Box<dyn std::error::Error>> {
22/// let builder = CreateInvite::new().max_age(3600).max_uses(10);
23/// let creation = channel_id.create_invite(http, builder).await?;
24/// # Ok(())
25/// # }
26/// ```
27///
28/// [Discord docs](https://docs.discord.com/developers/resources/channel#create-channel-invite)
29#[derive(Clone, Debug, Default, Serialize)]
30#[must_use]
31pub struct CreateInvite<'a> {
32    #[serde(skip_serializing_if = "Option::is_none")]
33    max_age: Option<u32>,
34    #[serde(skip_serializing_if = "Option::is_none")]
35    max_uses: Option<u8>,
36    #[serde(skip_serializing_if = "Option::is_none")]
37    temporary: Option<bool>,
38    #[serde(skip_serializing_if = "Option::is_none")]
39    unique: Option<bool>,
40    #[serde(skip_serializing_if = "Option::is_none")]
41    target_type: Option<InviteTargetType>,
42    #[serde(skip_serializing_if = "Option::is_none")]
43    target_user_id: Option<UserId>,
44    #[serde(skip_serializing_if = "Option::is_none")]
45    target_application_id: Option<ApplicationId>,
46    #[serde(skip_serializing_if = "Option::is_none")]
47    role_ids: Option<Cow<'a, [RoleId]>>,
48
49    #[serde(skip)]
50    audit_log_reason: Option<&'a str>,
51}
52
53impl<'a> CreateInvite<'a> {
54    /// Equivalent to [`Self::default`].
55    pub fn new() -> Self {
56        Self::default()
57    }
58
59    /// The duration of the invite in seconds before expiry.
60    ///
61    /// Between `0` (never) and `604800` (7 days). Defaults to `86400` (24 hours).
62    pub fn max_age(mut self, max_age: u32) -> Self {
63        self.max_age = Some(max_age);
64        self
65    }
66
67    /// The maximum number of uses for the invite.
68    ///
69    /// Between `0` (unlimited) and `100`. Defaults to `0`.
70    pub fn max_uses(mut self, max_uses: u8) -> Self {
71        self.max_uses = Some(max_uses);
72        self
73    }
74
75    /// Whether the invite only grants temporary membership.
76    ///
77    /// Defaults to `false`.
78    pub fn temporary(mut self, temporary: bool) -> Self {
79        self.temporary = Some(temporary);
80        self
81    }
82
83    /// Whether or not to try to reuse a similar invite.
84    ///
85    /// Defaults to `false`.
86    pub fn unique(mut self, unique: bool) -> Self {
87        self.unique = Some(unique);
88        self
89    }
90
91    /// The type of target for this voice channel invite.
92    pub fn target_type(mut self, target_type: InviteTargetType) -> Self {
93        self.target_type = Some(target_type);
94        self
95    }
96
97    /// The [`UserId`] of the user whose stream to display for this invite.
98    ///
99    /// Required if `target_type` is `Stream`. The user must be streaming in the channel.
100    pub fn target_user_id(mut self, target_user_id: UserId) -> Self {
101        self.target_user_id = Some(target_user_id);
102        self
103    }
104
105    /// The ID of the embedded application to open for this invite.
106    ///
107    /// Required if `target_type` is `EmmbeddedApplication`. The application must have the
108    /// [`ApplicationFlags::EMBEDDED`] flag.
109    ///
110    /// Some examples of popular embedded applications:
111    ///
112    /// Watch Together: `880218394199220334`
113    /// Wordle: `1211781489931452447`
114    /// Poker Night: `755827207812677713`
115    /// Chess in the Park: `832012774040141894`
116    pub fn target_application_id(mut self, target_application_id: ApplicationId) -> Self {
117        self.target_application_id = Some(target_application_id);
118        self
119    }
120
121    /// The [`RoleId`]s for roles in the guild given to the users that accept this invite.
122    pub fn role_ids(mut self, role_ids: impl Into<Cow<'a, [RoleId]>>) -> Self {
123        self.role_ids = Some(role_ids.into());
124        self
125    }
126
127    /// Sets the request's audit log reason.
128    pub fn audit_log_reason(mut self, reason: &'a str) -> Self {
129        self.audit_log_reason = Some(reason);
130        self
131    }
132
133    /// Creates an invite for the given channel.
134    ///
135    /// **Note**: Requires the [Create Instant Invite] permission.
136    ///
137    /// # Errors
138    ///
139    /// /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is
140    /// given.
141    ///
142    /// [Create Instant Invite]: Permissions::CREATE_INSTANT_INVITE
143    #[cfg(feature = "http")]
144    pub async fn execute(self, http: &Http, channel_id: ChannelId) -> Result<Invite> {
145        http.create_invite(channel_id, &self, self.audit_log_reason).await
146    }
147}
