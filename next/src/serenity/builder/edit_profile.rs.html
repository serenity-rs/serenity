9/// A builder to edit the current user's settings, to be used in conjunction with
10/// [`CurrentUser::edit`].
11///
12/// [Discord docs](https://docs.discord.com/developers/resources/user#modify-current-user)
13#[derive(Clone, Debug, Default, Serialize)]
14#[must_use]
15pub struct EditProfile<'a> {
16    #[serde(skip_serializing_if = "Option::is_none")]
17    username: Option<Cow<'a, str>>,
18    #[serde(skip_serializing_if = "Option::is_none")]
19    avatar: Option<Option<DataUri<'a>>>,
20    #[serde(skip_serializing_if = "Option::is_none")]
21    banner: Option<Option<DataUri<'a>>>,
22}
23
24impl<'a> EditProfile<'a> {
25    /// Equivalent to [`Self::default`].
26    pub fn new() -> Self {
27        Self::default()
28    }
29
30    /// Set the avatar of the current user.
31    ///
32    /// # Examples
33    ///
34    /// ```rust,no_run
35    /// # use serenity::builder::{EditProfile, CreateAttachment};
36    /// # use serenity::prelude::*;
37    /// # use serenity::model::prelude::*;
38    /// # use serenity::http::Http;
39    /// #
40    /// # #[cfg(feature = "http")]
41    /// # async fn run() -> Result<(), SerenityError> {
42    /// # let http: Http = unimplemented!();
43    /// # let mut user = CurrentUser::default();
44    /// let avatar = CreateAttachment::path("./my_image.jpg".as_ref())?.encode("image/jpeg").await?;
45    /// let builder = EditProfile::new().avatar(avatar);
46    /// user.edit(&http, builder).await?;
47    /// # Ok(())
48    /// # }
49    /// ```
50    pub fn avatar(mut self, avatar: DataUri<'a>) -> Self {
51        self.avatar = Some(Some(avatar));
52        self
53    }
54
55    /// Delete the current user's avatar, resetting it to the default logo.
56    pub fn delete_avatar(mut self) -> Self {
57        self.avatar = Some(None);
58        self
59    }
60
61    /// Modifies the current user's username.
62    ///
63    /// When modifying the username, if another user has the same _new_ username and current
64    /// discriminator, a new unique discriminator will be assigned. If there are no available
65    /// discriminators with the requested username, an error will occur.
66    pub fn username(mut self, username: impl Into<Cow<'a, str>>) -> Self {
67        self.username = Some(username.into());
68        self
69    }
70
71    /// Sets the banner of the current user.
72    pub fn banner(mut self, banner: DataUri<'a>) -> Self {
73        self.banner = Some(Some(banner));
74        self
75    }
76
77    /// Deletes the current user's banner, resetting it to the default.
78    pub fn delete_banner(mut self) -> Self {
79        self.banner = Some(None);
80        self
81    }
82
83    pub fn into_owned(self) -> EditProfile<'static> {
84        let Self {
85            username,
86            avatar,
87            banner,
88        } = self;
89        EditProfile {
90            username: username.map(|u| u.into_owned().into()),
91            avatar: avatar.map(|a| a.map(DataUri::into_owned)),
92            banner: banner.map(|b| b.map(DataUri::into_owned)),
93        }
94    }
95
96    /// Edit the current user's profile with the fields set.
97    ///
98    /// # Errors
99    ///
100    /// Returns an [`Error::Http`] if an invalid value is set. May also return an [`Error::Json`]
101    /// if there is an error in deserializing the API response.
102    #[cfg(feature = "http")]
103    pub async fn execute(self, http: &Http) -> Result<CurrentUser> {
104        http.edit_profile(&self).await
105    }
106}
