11/// A builder to edit the current user's settings, to be used in conjunction with
12/// [`CurrentUser::edit`].
13///
14/// [Discord docs](https://docs.discord.com/developers/resources/user#modify-current-user)
15#[derive(Clone, Debug, Default, Serialize)]
16#[must_use]
17pub struct EditProfile<'a> {
18    #[serde(skip_serializing_if = "Option::is_none")]
19    username: Option<Cow<'a, str>>,
20    #[serde(skip_serializing_if = "Option::is_none")]
21    avatar: Option<Option<DataUri<'a>>>,
22    #[serde(skip_serializing_if = "Option::is_none")]
23    banner: Option<Option<DataUri<'a>>>,
24}
25
26impl<'a> EditProfile<'a> {
27    /// Equivalent to [`Self::default`].
28    pub fn new() -> Self {
29        Self::default()
30    }
31
32    /// Set the avatar of the current user.
33    ///
34    /// # Examples
35    ///
36    /// ```rust,no_run
37    /// # use serenity::builder::{EditProfile, CreateAttachment};
38    /// # use serenity::prelude::*;
39    /// # use serenity::model::prelude::*;
40    /// # use serenity::http::Http;
41    /// #
42    /// # #[cfg(feature = "http")]
43    /// # async fn run() -> Result<(), SerenityError> {
44    /// # let http: Http = unimplemented!();
45    /// # let mut user = CurrentUser::default();
46    /// let avatar = CreateAttachment::path("./my_image.jpg".as_ref())?.encode("image/jpeg").await?;
47    /// let builder = EditProfile::new().avatar(avatar);
48    /// user.edit(&http, builder).await?;
49    /// # Ok(())
50    /// # }
51    /// ```
52    pub fn avatar(mut self, avatar: DataUri<'a>) -> Self {
53        self.avatar = Some(Some(avatar));
54        self
55    }
56
57    /// Delete the current user's avatar, resetting it to the default logo.
58    pub fn delete_avatar(mut self) -> Self {
59        self.avatar = Some(None);
60        self
61    }
62
63    /// Modifies the current user's username.
64    ///
65    /// When modifying the username, if another user has the same _new_ username and current
66    /// discriminator, a new unique discriminator will be assigned. If there are no available
67    /// discriminators with the requested username, an error will occur.
68    pub fn username(mut self, username: impl Into<Cow<'a, str>>) -> Self {
69        self.username = Some(username.into());
70        self
71    }
72
73    /// Sets the banner of the current user.
74    pub fn banner(mut self, banner: DataUri<'a>) -> Self {
75        self.banner = Some(Some(banner));
76        self
77    }
78
79    /// Deletes the current user's banner, resetting it to the default.
80    pub fn delete_banner(mut self) -> Self {
81        self.banner = Some(None);
82        self
83    }
84
85    /// Edit the current user's profile with the fields set.
86    ///
87    /// # Errors
88    ///
89    /// Returns an [`Error::Http`] if an invalid value is set. May also return an [`Error::Json`]
90    /// if there is an error in deserializing the API response.
91    #[cfg(feature = "http")]
92    pub async fn execute(self, http: &Http) -> Result<CurrentUser> {
93        http.edit_profile(&self).await
94    }
95}
