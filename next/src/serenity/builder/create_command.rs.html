9/// A builder for creating a new [`CommandOption`].
10///
11/// [`Self::kind`], [`Self::name`], and [`Self::description`] are required fields.
12///
13/// [`CommandOption`]: crate::model::application::CommandOption
14///
15/// [Discord docs](https://docs.discord.com/developers/interactions/application-commands#application-command-object-application-command-option-structure).
16#[derive(Clone, Debug, Serialize)]
17#[must_use]
18pub struct CreateCommandOption<'a> {
19    #[serde(rename = "type")]
20    kind: CommandOptionType,
21    name: Cow<'a, str>,
22    #[serde(skip_serializing_if = "Option::is_none")]
23    name_localizations: Option<HashMap<Cow<'a, str>, Cow<'a, str>>>,
24    description: Cow<'a, str>,
25    #[serde(skip_serializing_if = "Option::is_none")]
26    description_localizations: Option<HashMap<Cow<'a, str>, Cow<'a, str>>>,
27    #[serde(default)]
28    required: bool,
29    #[serde(default)]
30    choices: Cow<'a, [CreateCommandOptionChoice<'a>]>,
31    #[serde(default)]
32    options: Cow<'a, [CreateCommandOption<'a>]>,
33    #[serde(default)]
34    channel_types: Cow<'a, [ChannelType]>,
35    #[serde(default)]
36    min_value: Option<serde_json::Number>,
37    #[serde(default)]
38    max_value: Option<serde_json::Number>,
39    #[serde(default)]
40    min_length: Option<u16>,
41    #[serde(default)]
42    max_length: Option<u16>,
43    #[serde(default)]
44    autocomplete: bool,
45}

88    /// Specifies a localized name of the option.
89    ///
90    /// ```rust
91    /// # use serenity::builder::CreateCommandOption;
92    /// # use serenity::model::application::CommandOptionType;
93    /// # CreateCommandOption::new(CommandOptionType::Integer, "", "")
94    /// .name("age")
95    /// .name_localized("zh-CN", "岁数")
96    /// # ;

99        mut self,

101        name: impl Into<Cow<'a, str>>,

103        let map = self.name_localizations.get_or_insert_with(Default::default);

321/// A builder for creating a new [`Command`].
322///
323/// [`Command`]: crate::model::application::Command
324///
325/// Discord docs:
326/// - [global command](https://docs.discord.com/developers/interactions/application-commands#create-global-application-command)
327/// - [guild command](https://docs.discord.com/developers/interactions/application-commands#create-guild-application-command)
328#[derive(Clone, Debug, Serialize)]
329#[must_use]
330pub struct CreateCommand<'a> {
331    #[serde(skip_serializing_if = "Option::is_none")]
332    #[serde(rename = "type")]
333    kind: Option<CommandType>,
334    #[serde(skip_serializing_if = "Option::is_none")]
335    handler: Option<EntryPointHandlerType>,
336
337    #[serde(flatten)]
338    fields: EditCommand<'a>,
339}
340
341impl<'a> CreateCommand<'a> {
342    /// Creates a new builder with the given name, leaving all other fields empty.
343    pub fn new(name: impl Into<Cow<'a, str>>) -> Self {
344        Self {
345            kind: None,
346            handler: None,
347
348            fields: EditCommand::new().name(name),
349        }
350    }
351
352    /// Specifies the name of the application command, replacing the current value as set in
353    /// [`Self::new`].
354    ///
355    /// **Note**: Must be between 1 and 32 lowercase characters, matching `r"^[\w-]{1,32}$"`. Two
356    /// global commands of the same app cannot have the same name. Two guild-specific commands of
357    /// the same app cannot have the same name.
358    pub fn name(mut self, name: impl Into<Cow<'a, str>>) -> Self {
359        self.fields = self.fields.name(name);
360        self
361    }
362
363    /// Specifies a localized name of the application command.
364    ///
365    /// ```rust
366    /// # serenity::builder::CreateCommand::new("birthday")
367    /// .name_localized("zh-CN", "生日")
368    /// .name_localized("el", "γενέθλια")
369    /// # ;
370    /// ```
371    pub fn name_localized(
372        mut self,
373        locale: impl Into<Cow<'a, str>>,
374        name: impl Into<Cow<'a, str>>,
375    ) -> Self {
376        self.fields = self.fields.name_localized(locale, name);
377        self
378    }
379
380    /// Specifies the type of the application command.
381    pub fn kind(mut self, kind: CommandType) -> Self {
382        self.kind = Some(kind);
383        self
384    }
385
386    /// Specifies the default permissions required to execute the command.
387    pub fn default_member_permissions(mut self, permissions: Permissions) -> Self {
388        self.fields = self.fields.default_member_permissions(permissions);
389        self
390    }
391
392    /// Specifies if the command is available in DMs.
393    #[cfg(not(feature = "unstable"))]
394    pub fn dm_permission(mut self, enabled: bool) -> Self {
395        self.fields = self.fields.dm_permission(enabled);
396        self
397    }
398
399    /// Specifies the description of the application command.
400    ///
401    /// **Note**: Must be between 1 and 100 characters long.
402    pub fn description(mut self, description: impl Into<Cow<'a, str>>) -> Self {
403        self.fields = self.fields.description(description);
404        self
405    }
406
407    /// Specifies a localized description of the application command.
408    ///
409    /// ```rust
410    /// # serenity::builder::CreateCommand::new("birthday")
411    /// .description("Wish a friend a happy birthday")
412    /// .description_localized("zh-CN", "祝你朋友生日快乐")
413    /// # ;
414    /// ```
415    pub fn description_localized(
416        mut self,
417        locale: impl Into<Cow<'a, str>>,
418        description: impl Into<Cow<'a, str>>,
419    ) -> Self {
420        self.fields = self.fields.description_localized(locale, description);
421        self
422    }
423
424    /// Adds an application command option for the application command.
425    ///
426    /// **Note**: Application commands can have up to 25 options.
427    pub fn add_option(mut self, option: CreateCommandOption<'a>) -> Self {
428        self.fields = self.fields.add_option(option);
429        self
430    }

432    /// Sets all the application command options for the application command.
433    ///
434    /// **Note**: Application commands can have up to 25 options.

436        self.fields = self.fields.set_options(options);
437        self
438    }

440    /// Adds an installation context that this application command can be used in.

479    /// Create a [`Command`], overwriting an existing one with the same name if it exists.
480    ///
481    /// Providing a [`GuildId`] will create a command in the corresponding [`Guild`]. Otherwise, a
482    /// global command will be created.
483    ///
484    /// # Errors
485    ///
486    /// Returns [`Error::Http`] if invalid data is given. See [Discord's docs] for more details.
487    ///
488    /// May also return [`Error::Json`] if there is an error in deserializing the API response.
489    ///
490    /// [Discord's docs]: https://docs.discord.com/developers/interactions/application-commands#slash-commands
491    #[cfg(feature = "http")]
492    pub async fn execute(self, http: &Http, guild_id: Option<GuildId>) -> Result<Command> {
493        match guild_id {
494            Some(guild_id) => http.create_guild_command(guild_id, &self).await,
495            None => http.create_global_command(&self).await,
496        }
497    }
