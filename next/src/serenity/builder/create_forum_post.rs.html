10/// [Discord docs](https://docs.discord.com/developers/resources/channel#start-thread-in-forum-or-media-channel).
11#[derive(Clone, Debug, Serialize)]
12#[must_use]
13pub struct CreateForumPost<'a> {
14    name: Cow<'a, str>,
15    #[serde(skip_serializing_if = "Option::is_none")]
16    auto_archive_duration: Option<AutoArchiveDuration>,
17    #[serde(skip_serializing_if = "Option::is_none")]
18    rate_limit_per_user: Option<NonMaxU16>,
19    message: CreateMessage<'a>,
20    #[serde(skip_serializing_if = "<[_]>::is_empty")]
21    applied_tags: Cow<'a, [ForumTagId]>,
22
23    #[serde(skip)]
24    audit_log_reason: Option<&'a str>,
25}
26
27impl<'a> CreateForumPost<'a> {
28    /// Creates a builder with the given name and message content, leaving all other fields empty.
29    pub fn new(name: impl Into<Cow<'a, str>>, message: CreateMessage<'a>) -> Self {
30        Self {
31            name: name.into(),
32            message,
33            auto_archive_duration: None,
34            rate_limit_per_user: None,
35            applied_tags: Cow::default(),
36            audit_log_reason: None,
37        }
38    }
39
40    /// The name of the forum post. Replaces the current value as set in [`Self::new`].
41    ///
42    /// **Note**: Must be between 2 and 100 characters long.
43    pub fn name(mut self, name: impl Into<Cow<'a, str>>) -> Self {
44        self.name = name.into();
45        self
46    }
47
48    /// The contents of the first message in the forum post.
49    ///
50    /// See [`CreateMessage`] for restrictions around message size.
51    pub fn message(mut self, message: CreateMessage<'a>) -> Self {
52        self.message = message;
53        self
54    }
55
56    /// Duration in minutes to automatically archive the forum post after recent activity.
57    pub fn auto_archive_duration(mut self, duration: AutoArchiveDuration) -> Self {
58        self.auto_archive_duration = Some(duration);
59        self
60    }
61
62    /// How many seconds must a user wait before sending another message.
63    ///
64    /// Bots, or users with the [`MANAGE_MESSAGES`] and/or [`MANAGE_CHANNELS`] permissions are
65    /// exempt from this restriction.
66    ///
67    /// **Note**: Must be between 0 and 21600 seconds (360 minutes or 6 hours).
68    ///
69    /// [`MANAGE_MESSAGES`]: crate::model::permissions::Permissions::MANAGE_MESSAGES
70    /// [`MANAGE_CHANNELS`]: crate::model::permissions::Permissions::MANAGE_CHANNELS
71    #[doc(alias = "slowmode")]
72    pub fn rate_limit_per_user(mut self, seconds: NonMaxU16) -> Self {
73        self.rate_limit_per_user = Some(seconds);
74        self
75    }
76
77    pub fn add_applied_tag(mut self, applied_tag: ForumTagId) -> Self {
78        self.applied_tags.to_mut().push(applied_tag);
79        self
80    }
81
82    pub fn set_applied_tags(mut self, applied_tags: impl Into<Cow<'a, [ForumTagId]>>) -> Self {
83        self.applied_tags = applied_tags.into();
84        self
85    }
86
87    /// Sets the request's audit log reason.
88    pub fn audit_log_reason(mut self, reason: &'a str) -> Self {
89        self.audit_log_reason = Some(reason);
90        self
91    }
92
93    /// Creates a forum post in the given channel.
94    ///
95    /// # Errors
96    ///
97    /// Returns [`Error::Http`] if the current user lacks permission, or if invalid data is given.
98    #[cfg(feature = "http")]
99    pub async fn execute(self, http: &Http, channel_id: ChannelId) -> Result<GuildThread> {
100        let files = self.message.attachments.new_attachments();
101        http.create_forum_post(channel_id, &self, files, self.audit_log_reason).await
102    }
