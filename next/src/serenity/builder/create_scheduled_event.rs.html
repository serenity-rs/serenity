8/// [Discord docs](https://docs.discord.com/developers/resources/guild-scheduled-event#create-guild-scheduled-event)
9#[derive(Clone, Debug, Serialize)]
10#[must_use]
11pub struct CreateScheduledEvent<'a> {
12    #[serde(skip_serializing_if = "Option::is_none")]
13    channel_id: Option<ChannelId>,
14    #[serde(skip_serializing_if = "Option::is_none")]
15    entity_metadata: Option<CreateScheduledEventMetadata<'a>>,
16    name: Cow<'a, str>,
17    privacy_level: ScheduledEventPrivacyLevel,
18    scheduled_start_time: Timestamp,
19    #[serde(skip_serializing_if = "Option::is_none")]
20    scheduled_end_time: Option<Timestamp>,
21    #[serde(skip_serializing_if = "Option::is_none")]
22    description: Option<Cow<'a, str>>,
23    entity_type: ScheduledEventType,
24    #[serde(skip_serializing_if = "Option::is_none")]
25    image: Option<DataUri<'a>>,
26
27    #[serde(skip)]
28    audit_log_reason: Option<&'a str>,
29}
30
31impl<'a> CreateScheduledEvent<'a> {
32    /// Creates a builder with the provided kind, name, and start time, leaving all other fields
33    /// empty.
34    pub fn new(
35        kind: ScheduledEventType,
36        name: impl Into<Cow<'a, str>>,
37        scheduled_start_time: impl Into<Timestamp>,
38    ) -> Self {
39        Self {
40            name: name.into(),
41            entity_type: kind,
42            scheduled_start_time: scheduled_start_time.into(),
43
44            image: None,
45            channel_id: None,
46            description: None,
47            entity_metadata: None,
48            scheduled_end_time: None,
49
50            // Set the privacy level to `GUILD_ONLY`. As this is the only possible value of this
51            // field, it's onlyu used at event creation, and we don't even parse it into the
52            // `ScheduledEvent` struct.
53            privacy_level: ScheduledEventPrivacyLevel::GuildOnly,
54
55            audit_log_reason: None,
56        }
57    }
58
59    /// Sets the channel id of the scheduled event. Required if [`Self::kind`] is
60    /// [`ScheduledEventType::StageInstance`] or [`ScheduledEventType::Voice`].
61    pub fn channel_id(mut self, channel_id: ChannelId) -> Self {
62        self.channel_id = Some(channel_id);
63        self
64    }
65
66    /// Sets the name of the scheduled event, replacing the current value as set in [`Self::new`].
67    pub fn name(mut self, name: impl Into<Cow<'a, str>>) -> Self {
68        self.name = name.into();
69        self
70    }
71
72    /// Sets the description of the scheduled event.
73    pub fn description(mut self, description: impl Into<Cow<'a, str>>) -> Self {
74        self.description = Some(description.into());
75        self
76    }
77
78    /// Sets the start time of the scheduled event, replacing the current value as set in
79    /// [`Self::new`].
80    pub fn start_time(mut self, timestamp: impl Into<Timestamp>) -> Self {
81        self.scheduled_start_time = timestamp.into();
82        self
83    }
84
85    /// Sets the end time of the scheduled event. Required if [`Self::kind`] is
86    /// [`ScheduledEventType::External`].
87    pub fn end_time(mut self, timestamp: impl Into<Timestamp>) -> Self {
88        self.scheduled_end_time = Some(timestamp.into());
89        self
90    }
91
92    /// Sets the entity type of the scheduled event, replacing the current value as set in
93    /// [`Self::new`].
94    pub fn kind(mut self, kind: ScheduledEventType) -> Self {
95        self.entity_type = kind;
96        self
97    }
98
99    /// Sets the location of the scheduled event. Required to be set and non-empty if
100    /// [`Self::kind`] is [`ScheduledEventType::External`].
101    ///
102    /// [`External`]: ScheduledEventType::External
103    pub fn location(mut self, location: impl Into<Cow<'a, str>>) -> Self {
104        self.entity_metadata = Some(CreateScheduledEventMetadata {
105            location: Some(location.into()),
106        });
107        self
108    }
109
110    /// Sets the cover image for the scheduled event.
111    pub fn image(mut self, image: DataUri<'a>) -> Self {
112        self.image = Some(image);
113        self
114    }
115
116    /// Sets the request's audit log reason.
117    pub fn audit_log_reason(mut self, reason: &'a str) -> Self {
118        self.audit_log_reason = Some(reason);
119        self
120    }
121
122    /// Creates a new scheduled event in the guild with the data set, if any.
123    ///
124    /// **Note**: Requires the [Create Events] permission.
125    ///
126    /// # Errors
127    ///
128    /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is given.
129    ///
130    /// [Create Events]: Permissions::CREATE_EVENTS
131    #[cfg(feature = "http")]
132    pub async fn execute(self, http: &Http, guild_id: GuildId) -> Result<ScheduledEvent> {
133        http.create_scheduled_event(guild_id, &self, self.audit_log_reason).await
134    }
135}
