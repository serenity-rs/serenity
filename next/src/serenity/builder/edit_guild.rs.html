8/// A builder to optionally edit certain fields of a guild
9///
10/// [Discord docs](https://docs.discord.com/developers/resources/guild#modify-guild).
11#[derive(Clone, Debug, Default, Serialize)]
12#[must_use]
13pub struct EditGuild<'a> {
14    #[serde(skip_serializing_if = "Option::is_none")]
15    name: Option<Cow<'a, str>>,
16    // [Omitting region because Discord deprecated it]
17    #[serde(skip_serializing_if = "Option::is_none")]
18    verification_level: Option<VerificationLevel>,
19    #[serde(skip_serializing_if = "Option::is_none")]
20    default_message_notifications: Option<Option<DefaultMessageNotificationLevel>>,
21    #[serde(skip_serializing_if = "Option::is_none")]
22    explicit_content_filter: Option<Option<ExplicitContentFilter>>,
23    #[serde(skip_serializing_if = "Option::is_none")]
24    afk_channel_id: Option<Option<ChannelId>>,
25    #[serde(skip_serializing_if = "Option::is_none")]
26    afk_timeout: Option<AfkTimeout>,
27    #[serde(skip_serializing_if = "Option::is_none")]
28    icon: Option<Option<DataUri<'a>>>,
29    #[serde(skip_serializing_if = "Option::is_none")]
30    owner_id: Option<UserId>,
31    #[serde(skip_serializing_if = "Option::is_none")]
32    splash: Option<Option<DataUri<'a>>>,
33    #[serde(skip_serializing_if = "Option::is_none")]
34    discovery_splash: Option<Option<DataUri<'a>>>,
35    #[serde(skip_serializing_if = "Option::is_none")]
36    banner: Option<Option<DataUri<'a>>>,
37    #[serde(skip_serializing_if = "Option::is_none")]
38    system_channel_id: Option<Option<ChannelId>>,
39    #[serde(skip_serializing_if = "Option::is_none")]
40    system_channel_flags: Option<SystemChannelFlags>,
41    #[serde(skip_serializing_if = "Option::is_none")]
42    rules_channel_id: Option<Option<ChannelId>>,
43    #[serde(skip_serializing_if = "Option::is_none")]
44    public_updates_channel_id: Option<Option<ChannelId>>,
45    #[serde(skip_serializing_if = "Option::is_none")]
46    preferred_locale: Option<Option<Cow<'a, str>>>,
47    #[serde(skip_serializing_if = "Option::is_none")]
48    features: Option<Cow<'a, [Cow<'a, str>]>>,
49    #[serde(skip_serializing_if = "Option::is_none")]

54    #[serde(skip)]
55    audit_log_reason: Option<Cow<'a, str>>,
56}

64    /// Set the "AFK voice channel" that users are to move to if they have been AFK for an amount

78    /// Set the icon of the guild. Pass [`None`] to remove the icon.
79    ///
80    /// # Examples
81    ///
82    /// Using the utility builder - [`CreateAttachment`] - to read an image and encode it in
83    /// base64, to then set as the guild icon.
84    ///
85    /// ```rust,no_run
86    /// # use serenity::builder::{EditGuild, CreateAttachment};
87    /// # use serenity::{http::Http, model::id::GuildId};
88    /// #
89    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
90    /// # let http: Http = unimplemented!();
91    /// # let mut guild = GuildId::new(1).to_partial_guild(&http).await?;
92    /// let icon = CreateAttachment::path("./guild_icon.png".as_ref())?.encode("image/png").await?;
93    ///
94    /// // assuming a `guild` has already been bound
95    /// let builder = EditGuild::new().icon(Some(icon));
96    /// guild.edit(&http, builder).await?;
97    /// # Ok(())
98    /// # }
99    /// ```
100    ///
101    /// [`CreateAttachment`]: super::CreateAttachment
102    pub fn icon(mut self, icon: Option<DataUri<'a>>) -> Self {
103        self.icon = Some(icon);
104        self
105    }

305    pub fn into_owned(self) -> EditGuild<'static> {
306        let Self {
307            name,
308            verification_level,
309            default_message_notifications,
310            explicit_content_filter,
311            afk_channel_id,
312            afk_timeout,
313            icon,
314            owner_id,
315            splash,
316            discovery_splash,
317            banner,
318            system_channel_id,
319            system_channel_flags,
320            rules_channel_id,
321            public_updates_channel_id,
322            preferred_locale,
323            features,
324            description,
325            premium_progress_bar_enabled,
326            audit_log_reason,
327        } = self;
328        EditGuild {
329            name: name.map(|n| n.into_owned().into()),
330            verification_level,
331            default_message_notifications,
332            explicit_content_filter,
333            afk_channel_id,
334            afk_timeout,
335            icon: icon.map(|i| i.map(DataUri::into_owned)),
336            owner_id,
337            splash: splash.map(|s| s.map(DataUri::into_owned)),
338            discovery_splash: discovery_splash.map(|s| s.map(DataUri::into_owned)),
339            banner: banner.map(|b| b.map(DataUri::into_owned)),
340            system_channel_id,
341            system_channel_flags,
342            rules_channel_id,
343            public_updates_channel_id,
344            preferred_locale: preferred_locale.map(|l| l.map(|l| l.into_owned().into())),
345            features: features.map(|f| {
346                Cow::Owned(f.into_owned().into_iter().map(|f| Cow::Owned(f.into_owned())).collect())
347            }),
348            description: description.map(|d| d.into_owned().into()),
349            premium_progress_bar_enabled,
350            audit_log_reason: audit_log_reason.map(|r| Cow::Owned(r.into_owned())),
351        }
352    }

360    /// Edits the given guild.
361    ///
362    /// **Note**: Requires the [Manage Guild] permission.
363    ///
364    /// # Errors
365    ///
366    /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is given.
367    ///
368    /// [Manage Guild]: Permissions::MANAGE_GUILD
369    #[cfg(feature = "http")]
370    pub async fn execute(self, http: &Http, guild_id: GuildId) -> Result<PartialGuild> {
371        http.edit_guild(guild_id, &self, self.audit_log_reason.as_deref()).await
372    }
