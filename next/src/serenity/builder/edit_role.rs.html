8/// A builder to create or edit a [`Role`] for use via a number of model methods.
9///
10/// These are:
11///
12/// - [`GuildId::create_role`]
13/// - [`GuildId::edit_role`]
14/// - [`Role::edit`]
15///
16/// Defaults are provided for each parameter on role creation.
17///
18/// # Examples
19///
20/// Create a hoisted, mentionable role named `"a test role"`:
21///
22/// ```rust,no_run
23/// # use serenity::builder::EditRole;
24/// # use serenity::http::Http;
25/// # use serenity::model::id::GuildId;
26/// # use std::sync::Arc;
27/// #
28/// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
29/// # let http: Arc<Http> = unimplemented!();
30/// # let guild_id: GuildId = unimplemented!();
31/// #
32/// // assuming a `guild_id` has been bound
33/// let builder = EditRole::new().name("a test role").hoist(true).mentionable(true);
34/// let role = guild_id.create_role(&http, builder).await?;
35/// # Ok(())
36/// # }
37/// ```
38///
39/// [Discord docs](https://docs.discord.com/developers/resources/guild#modify-guild-role)
40#[derive(Clone, Debug, Default, Serialize)]
41#[must_use]
42pub struct EditRole<'a> {
43    #[serde(skip_serializing_if = "Option::is_none")]
44    name: Option<Cow<'a, str>>,
45    #[serde(skip_serializing_if = "Option::is_none")]
46    permissions: Option<u64>,
47    #[serde(skip_serializing_if = "Option::is_none")]
48    #[serde(rename = "color")]
49    colour: Option<Colour>,
50    #[serde(skip_serializing_if = "Option::is_none")]
51    #[serde(rename = "colors")]
52    colours: Option<CreateRoleColours>,
53    #[serde(skip_serializing_if = "Option::is_none")]
54    hoist: Option<bool>,
55    #[serde(skip_serializing_if = "Option::is_none")]
56    icon: Option<Option<DataUri<'a>>>,
57    #[serde(skip_serializing_if = "Option::is_none")]
58    unicode_emoji: Option<Option<Cow<'a, str>>>,
59
60    #[serde(skip_serializing_if = "Option::is_none")]
61    mentionable: Option<bool>,
62
63    #[serde(skip)]
64    position: Option<i16>,
65    #[serde(skip)]
66    audit_log_reason: Option<&'a str>,
67}

69impl<'a> EditRole<'a> {
70    /// Equivalent to [`Self::default`].

75    /// Creates a new builder with the values of the given [`Role`].
76    pub fn from_role(role: &'a Role) -> Self {
77        EditRole {

79            mentionable: Some(role.mentionable()),
80            name: Some(Cow::Borrowed(&role.name)),
81            permissions: Some(role.permissions.bits()),
82            position: Some(role.position),
83            colour: Some(role.colour),
84            unicode_emoji: role.unicode_emoji.as_ref().map(|v| Some(Cow::Borrowed(v.as_str()))),
85            audit_log_reason: None,
86            colours: Some(role.colours.into()),
87            // TODO: Do we want to download role.icon?
88            icon: None,
89        }
90    }
91
92    /// Set the colour of the role.
93    pub fn colour(mut self, colour: impl Into<Colour>) -> Self {
94        self.colour = Some(colour.into());
95        self
96    }
97
98    /// Sets the colours of the role. Supports gradient and holographic role colours.
99    pub fn colours(mut self, colours: impl Into<CreateRoleColours>) -> Self {

103
104    /// Whether or not to hoist the role above lower-positioned roles in the user list.

107        self
108    }
109
110    /// Whether or not to make the role mentionable, upon which users with that role will be
111    /// notified.
112    pub fn mentionable(mut self, mentionable: bool) -> Self {
113        self.mentionable = Some(mentionable);
114        self
115    }
116
117    /// Set the role's name.
118    pub fn name(mut self, name: impl Into<Cow<'a, str>>) -> Self {
119        self.name = Some(name.into());
120        self
121    }
122
123    /// Set the role's permissions.
124    pub fn permissions(mut self, permissions: Permissions) -> Self {
125        self.permissions = Some(permissions.bits());
126        self
127    }
128
129    /// Set the role's position in the role list. This correlates to the role's position in the
130    /// user list.
131    pub fn position(mut self, position: i16) -> Self {
132        self.position = Some(position);
133        self
134    }
135
136    /// Set the role icon to a unicode emoji.
137    pub fn unicode_emoji(mut self, unicode_emoji: Option<Cow<'a, str>>) -> Self {
138        self.unicode_emoji = Some(unicode_emoji);
139        self.icon = Some(None);
140        self
141    }
142
143    /// Set the role icon to a custom image.
144    pub fn icon(mut self, icon: Option<DataUri<'a>>) -> Self {
145        self.icon = Some(icon);
146        self.unicode_emoji = Some(None);
147        self
148    }
149
150    /// Sets the request's audit log reason.
151    pub fn audit_log_reason(mut self, reason: &'a str) -> Self {
152        self.audit_log_reason = Some(reason);
153        self
154    }
155
156    /// Edits the role.
157    ///
158    /// **Note**: Requires the [Manage Roles] permission.
159    ///
160    /// # Errors
161    ///
162    /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is given.
163    ///
164    /// [Manage Roles]: Permissions::MANAGE_ROLES
165    #[cfg(feature = "http")]
166    pub async fn execute(
167        self,
168        http: &Http,
169        guild_id: GuildId,
170        role_id: Option<RoleId>,
171    ) -> Result<Role> {
172        let role = match role_id {
173            Some(role_id) => {
174                http.edit_role(guild_id, role_id, &self, self.audit_log_reason).await?
175            },
176            None => http.create_role(guild_id, &self, self.audit_log_reason).await?,
177        };
178
179        if let Some(position) = self.position {
180            guild_id
181                .edit_role_positions(http, [(role.id, position)], self.audit_log_reason)
182                .await?;
183        }
184        Ok(role)
185    }
186}

188/// The colours of a Discord role, secondary_colour and tertiary_colour may only be set if
189/// the [Guild] has the `ENHANCED_ROLE_COLORS` feature.
190///
191/// Note: 2024-07-05 - tertiary_colour is currently enforced to be set with a specific pair of
192/// primary and secondary colours, for current validation see
193/// [Discord docs](https://docs.discord.com/developers/topics/permissions#role-object-role-colors-object).
