9/// A builder to create a guild sticker
10///
11/// [Discord docs](https://docs.discord.com/developers/resources/sticker#create-guild-sticker)
12#[derive(Clone, Debug)]
13#[must_use]
14pub struct CreateSticker<'a> {
15    name: Cow<'static, str>,
16    description: Cow<'static, str>,
17    tags: Cow<'static, str>,
18    file: CreateAttachment<'a>,
19    audit_log_reason: Option<&'a str>,
20}
21
22impl<'a> CreateSticker<'a> {
23    /// Creates a new builder with the given data. All of this builder's fields are required.
24    pub fn new(name: impl Into<Cow<'static, str>>, file: CreateAttachment<'a>) -> Self {
25        Self {
26            name: name.into(),
27            tags: Cow::default(),
28            description: Cow::default(),
29            file,
30            audit_log_reason: None,
31        }
32    }
33
34    /// Set the name of the sticker, replacing the current value as set in [`Self::new`].
35    ///
36    /// **Note**: Must be between 2 and 30 characters long.
37    pub fn name(mut self, name: impl Into<Cow<'static, str>>) -> Self {
38        self.name = name.into();
39        self
40    }
41
42    /// Set the description of the sticker.
43    ///
44    /// **Note**: Must be empty or 2-100 characters.
45    pub fn description(mut self, description: impl Into<Cow<'static, str>>) -> Self {
46        self.description = description.into();
47        self
48    }
49
50    /// The Discord name of a unicode emoji representing the sticker's expression.
51    ///
52    /// **Note**: Max 200 characters long.
53    pub fn tags(mut self, tags: impl Into<Cow<'static, str>>) -> Self {
54        self.tags = tags.into();
55        self
56    }
57
58    /// Set the sticker file. Replaces the current value as set in [`Self::new`].
59    ///
60    /// **Note**: Must be a PNG, APNG, or Lottie JSON file, max 500 KB.
61    pub fn file(mut self, file: CreateAttachment<'a>) -> Self {
62        self.file = file;
63        self
64    }
65
66    /// Sets the request's audit log reason.
67    pub fn audit_log_reason(mut self, reason: &'a str) -> Self {
68        self.audit_log_reason = Some(reason);
69        self
70    }
71
72    /// Creates a new sticker in the guild with the data set, if any.
73    ///
74    /// **Note**: Requires the [Create Guild Expressions] permission.
75    ///
76    /// # Errors
77    ///
78    /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is given.
79    ///
80    /// [Create Guild Expressions]: Permissions::CREATE_GUILD_EXPRESSIONS
81    #[cfg(feature = "http")]
82    pub async fn execute(self, http: &Http, guild_id: GuildId) -> Result<Sticker> {
83        let map = vec![
84            ("name".into(), self.name),
85            ("tags".into(), self.tags),
86            ("description".into(), self.description),
87        ];
88
89        http.create_sticker(guild_id, map, self.file, self.audit_log_reason).await
90    }
91}
