1use std::borrow::Cow;

3#[cfg(feature = "http")]

7/// A builder to specify the fields to edit in a [`GuildWidget`].
8///
9/// [Discord docs](https://docs.discord.com/developers/resources/guild#modify-guild-widget)
10#[derive(Clone, Debug, Default, Serialize)]
11#[must_use]
12pub struct EditGuildWidget<'a> {
13    #[serde(skip_serializing_if = "Option::is_none")]
14    enabled: Option<bool>,
15    #[serde(skip_serializing_if = "Option::is_none")]
16    channel_id: Option<ChannelId>,
17
18    #[serde(skip)]
19    audit_log_reason: Option<Cow<'a, str>>,
20}
21
22impl<'a> EditGuildWidget<'a> {
23    /// Equivalent to [`Self::default`].
24    pub fn new() -> Self {
25        Self::default()
26    }
27
28    /// Whether the widget is enabled or not.
29    pub fn enabled(mut self, enabled: bool) -> Self {
30        self.enabled = Some(enabled);
31        self
32    }
33
34    /// The server description shown in the welcome screen.
35    pub fn channel_id(mut self, id: ChannelId) -> Self {
36        self.channel_id = Some(id);
37        self
38    }
39
40    /// Sets the request's audit log reason.
41    pub fn audit_log_reason(mut self, reason: impl Into<Cow<'a, str>>) -> Self {
42        self.audit_log_reason = Some(reason.into());
43        self
44    }
45
46    pub fn into_owned(self) -> EditGuildWidget<'static> {
47        let Self {
48            enabled,
49            channel_id,
50            audit_log_reason,
51        } = self;
52        EditGuildWidget {
53            enabled,
54            channel_id,
55            audit_log_reason: audit_log_reason.map(|r| Cow::Owned(r.into_owned())),
56        }
57    }
58
59    /// Edits the guild's widget.
60    ///
61    /// **Note**: Requires the [Manage Guild] permission.
62    ///
63    /// # Errors
64    ///
65    /// Returns [`Error::Http`] if the current user lacks permission.
66    ///
67    /// [Manage Guild]: Permissions::MANAGE_GUILD
68    #[cfg(feature = "http")]
69    pub async fn execute(self, http: &Http, guild_id: GuildId) -> Result<GuildWidget> {
70        http.edit_guild_widget(guild_id, &self, self.audit_log_reason.as_deref()).await
71    }
72}
