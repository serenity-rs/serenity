9/// A builder which edits the properties of the bot's [`Member`], to be used in conjunction with
10/// [`GuildId::edit_current_member`].
11///
12/// [Discord docs](https://docs.discord.com/developers/resources/guild#modify-current-member)
13#[derive(Clone, Debug, Default, Serialize)]
14#[must_use]
15pub struct EditCurrentMember<'a> {
16    #[serde(skip_serializing_if = "Option::is_none")]
17    nick: Option<Option<Cow<'a, str>>>,
18    #[serde(skip_serializing_if = "Option::is_none")]
19    banner: Option<Option<DataUri<'a>>>,
20    #[serde(skip_serializing_if = "Option::is_none")]
21    avatar: Option<Option<DataUri<'a>>>,
22    #[serde(skip_serializing_if = "Option::is_none")]
23    bio: Option<Option<Cow<'a, str>>>,
24
25    #[serde(skip)]
26    audit_log_reason: Option<Cow<'a, str>>,
27}
28
29impl<'a> EditCurrentMember<'a> {
30    /// Equivalent to [`Self::default`].
31    pub fn new() -> Self {
32        Self::default()
33    }
34
35    /// Changes the bots's nickname. Pass `None` or an empty string to reset the nickname.
36    ///
37    /// **Note**: Requires the [Change Nickname] permission.
38    ///
39    /// [Change Nickname]: Permissions::CHANGE_NICKNAME
40    pub fn nickname(mut self, nick: Option<Cow<'a, str>>) -> Self {
41        self.nick = Some(nick);
42        self
43    }
44
45    /// Changes the bot's guild-specific base64 encoded banner image.
46    ///
47    /// The `banner` must be base64-encoded 16:9 png/jpeg image data.
48    pub fn banner(mut self, banner: Option<DataUri<'a>>) -> Self {
49        self.banner = Some(banner);
50        self
51    }
52
53    /// Changes the bot's guild-specific base64 encoded avatar image.
54    ///
55    /// The `avatar` must be base64-encoded png/jpeg image data.
56    pub fn avatar(mut self, avatar: Option<DataUri<'a>>) -> Self {
57        self.avatar = Some(avatar);
58        self
59    }
60
61    /// Changes the bot's bio (about me) in the guild.
62    pub fn bio(mut self, bio: Option<Cow<'a, str>>) -> Self {
63        self.bio = Some(bio);
64        self
65    }
66
67    /// Sets the request's audit log reason.
68    pub fn audit_log_reason(mut self, reason: impl Into<Cow<'a, str>>) -> Self {
69        self.audit_log_reason = Some(reason.into());
70        self
71    }
72
73    pub fn into_owned(self) -> EditCurrentMember<'static> {
74        let Self {
75            nick,
76            banner,
77            avatar,
78            bio,
79            audit_log_reason,
80        } = self;
81        EditCurrentMember {
82            nick: nick.map(|n| n.map(|n| n.into_owned().into())),
83            banner: banner.map(|b| b.map(DataUri::into_owned)),
84            avatar: avatar.map(|a| a.map(DataUri::into_owned)),
85            bio: bio.map(|b| b.map(|b| b.into_owned().into())),
86            audit_log_reason: audit_log_reason.map(|r| Cow::Owned(r.into_owned())),
87        }
88    }
89
90    /// Edits the properties of the application's guild member.
91    ///
92    /// For details on permissions requirements, refer to each specific method.
93    ///
94    /// # Errors
95    ///
96    /// Returns [`Error::Http`] if the current user lacks permission, or if invalid data is given.
97    #[cfg(feature = "http")]
98    pub async fn execute(self, http: &Http, guild_id: GuildId) -> Result<Member> {
99        http.edit_current_member(guild_id, &self, self.audit_log_reason.as_deref()).await
100    }
