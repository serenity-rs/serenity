3#[cfg(feature = "http")]

8/// A builder for creating or editing guild AutoMod rules.
9///
10/// # Examples
11///
12/// See [`GuildId::create_automod_rule`] for details.
13///
14/// [Discord docs](https://docs.discord.com/developers/resources/auto-moderation#modify-auto-moderation-rule)
15#[derive(Clone, Debug, Serialize)]
16#[must_use]
17pub struct EditAutoModRule<'a> {
18    #[serde(skip_serializing_if = "Option::is_none")]
19    name: Option<Cow<'a, str>>,
20    event_type: EventType,
21    #[serde(flatten, skip_serializing_if = "Option::is_none")]
22    trigger: Option<Trigger>,
23    #[serde(skip_serializing_if = "Option::is_none")]
24    actions: Option<Cow<'a, [Action]>>,
25    #[serde(skip_serializing_if = "Option::is_none")]
26    enabled: Option<bool>,
27    #[serde(skip_serializing_if = "Option::is_none")]
28    exempt_roles: Option<Cow<'a, [RoleId]>>,
29    #[serde(skip_serializing_if = "Option::is_none")]
30    exempt_channels: Option<Cow<'a, [ChannelId]>>,
31
32    #[serde(skip)]
33    audit_log_reason: Option<Cow<'a, str>>,
34}
35
36impl<'a> EditAutoModRule<'a> {
37    /// Equivalent to [`Self::default`].
38    pub fn new() -> Self {
39        Self::default()
40    }
41
42    /// The display name of the rule.
43    pub fn name(mut self, name: impl Into<Cow<'a, str>>) -> Self {
44        self.name = Some(name.into());
45        self
46    }
47
48    /// Set the event context the rule should be checked.
49    pub fn event_type(mut self, event_type: EventType) -> Self {
50        self.event_type = event_type;
51        self
52    }
53
54    /// Set the type of content which can trigger the rule.
55    ///
56    /// **None**: The trigger type can't be edited after creation. Only its values.
57    pub fn trigger(mut self, trigger: Trigger) -> Self {
58        self.trigger = Some(trigger);
59        self
60    }
61
62    /// Set the actions which will execute when the rule is triggered.
63    pub fn actions(mut self, actions: impl Into<Cow<'a, [Action]>>) -> Self {
64        self.actions = Some(actions.into());
65        self
66    }
67
68    /// Set whether the rule is enabled.
69    pub fn enabled(mut self, enabled: bool) -> Self {
70        self.enabled = Some(enabled);
71        self
72    }
73
74    /// Set roles that should not be affected by the rule.
75    ///
76    /// Maximum of 20.
77    pub fn exempt_roles(mut self, roles: impl Into<Cow<'a, [RoleId]>>) -> Self {
78        self.exempt_roles = Some(roles.into());
79        self
80    }
81
82    /// Set channels that should not be affected by the rule.
83    ///
84    /// Maximum of 50.
85    pub fn exempt_channels(mut self, channels: impl Into<Cow<'a, [ChannelId]>>) -> Self {
86        self.exempt_channels = Some(channels.into());
87        self
88    }
89
90    /// Sets the request's audit log reason.
91    pub fn audit_log_reason(mut self, reason: impl Into<Cow<'a, str>>) -> Self {
92        self.audit_log_reason = Some(reason.into());
93        self
94    }
95
96    pub fn into_owned(self) -> EditAutoModRule<'static> {
97        let Self {
98            name,
99            event_type,
100            trigger,
101            actions,
102            enabled,
103            exempt_roles,
104            exempt_channels,
105            audit_log_reason,
106        } = self;
107        EditAutoModRule {
108            name: name.map(|n| n.into_owned().into()),
109            event_type,
110            trigger,
111            actions: actions.map(|a| Cow::Owned(a.into_owned())),
112            enabled,
113            exempt_roles: exempt_roles.map(|r| Cow::Owned(r.into_owned())),
114            exempt_channels: exempt_channels.map(|c| Cow::Owned(c.into_owned())),
115            audit_log_reason: audit_log_reason.map(|r| Cow::Owned(r.into_owned())),
116        }
117    }
118
119    /// Creates or edits an [`AutoModRule`] in a guild.
120    ///
121    /// Providing a [`RuleId`] will edit that corresponding rule, otherwise a new rule will be
122    /// created.
123    ///
124    /// **Note**: Requires the [Manage Guild] permission.
125    ///
126    /// # Errors
127    ///
128    /// Returns [`Error::Http`] if the current user lacks permission, or if invalid data is given.
129    ///
130    /// [Manage Guild]: Permissions::MANAGE_GUILD
131    #[cfg(feature = "http")]
132    pub async fn execute(
133        self,
134        http: &Http,
135        guild_id: GuildId,
136        rule_id: Option<RuleId>,
137    ) -> Result<AutoModRule> {
138        match rule_id {
139            Some(id) => {
140                http.edit_automod_rule(guild_id, id, &self, self.audit_log_reason.as_deref()).await
141            },
142            // Automod Rule creation has required fields, whereas modifying a rule does not.
143            // TODO: Enforce these fields (maybe with a separate CreateAutoModRule builder).
144            None => {
145                http.create_automod_rule(guild_id, &self, self.audit_log_reason.as_deref()).await
146            },
147        }
148    }
149}
150
151impl Default for EditAutoModRule<'_> {

153        Self {

155            trigger: None,
156            actions: None,
157            enabled: None,
158            exempt_roles: None,
159            exempt_channels: None,
160            event_type: EventType::MessageSend,
161            audit_log_reason: None,
162        }
163    }
164}
