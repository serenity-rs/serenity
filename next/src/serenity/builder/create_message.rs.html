16/// A builder to specify the contents of an send message request, primarily meant for use
17/// through [`GenericChannelId::send_message`].
18///
19/// There are three situations where different field requirements are present:
20///
21/// 1. When sending a message without embeds or stickers, [`Self::content`] is the only required
22///    field that is required to be set.
23/// 2. When sending an [`Self::embed`], no other field is required.
24/// 3. When sending stickers with [`Self::sticker_id`] or other sticker methods, no other field is
25///    required.
26///
27/// Note that if you only need to send the content of a message, without specifying other fields,
28/// then [`GenericChannelId::say`] may be a more preferable option.
29///
30/// # Examples
31///
32/// Sending a message with a content of `"test"` and applying text-to-speech:
33///
34/// ```rust,no_run
35/// use serenity::builder::{CreateEmbed, CreateMessage};
36/// use serenity::model::id::GenericChannelId;
37/// # use serenity::http::Http;
38/// # use std::sync::Arc;
39/// #
40/// # async fn run() {
41/// # let http: Arc<Http> = unimplemented!();
42/// # let channel_id = GenericChannelId::new(7);
43/// let embed = CreateEmbed::new().title("This is an embed").description("With a description");
44/// let builder = CreateMessage::new().content("test").tts(true).embed(embed);
45/// let _ = channel_id.send_message(&http, builder).await;
46/// # }
47/// ```
48///
49/// [Discord docs](https://docs.discord.com/developers/resources/message#create-message)
50#[derive(Clone, Debug, Default, Serialize)]
51#[must_use]
52pub struct CreateMessage<'a> {
53    #[serde(skip_serializing_if = "Option::is_none")]
54    content: Option<Cow<'a, str>>,
55    #[serde(skip_serializing_if = "Option::is_none")]
56    nonce: Option<Nonce>,
57    tts: bool,
58    embeds: Cow<'a, [CreateEmbed<'a>]>,
59    #[serde(skip_serializing_if = "Option::is_none")]
60    allowed_mentions: Option<CreateAllowedMentions<'a>>,
61    #[serde(skip_serializing_if = "Option::is_none")]
62    message_reference: Option<MessageReference>,
63    #[serde(skip_serializing_if = "Option::is_none")]
64    components: Option<Cow<'a, [CreateComponent<'a>]>>,
65    sticker_ids: Cow<'a, [StickerId]>,
66    #[serde(skip_serializing_if = "Option::is_none")]
67    flags: Option<MessageFlags>,
68    pub(crate) attachments: EditAttachments<'a>,
69    enforce_nonce: bool,
70    #[serde(skip_serializing_if = "Option::is_none")]
71    poll: Option<CreatePoll<'a, Ready>>,
72}

159    /// Sets a list of files to include in the message.
160    ///
161    /// Calling this multiple times will overwrite the file list. To append files, call

264    /// Send a message to the channel.
265    ///
266    /// **Note**: Requires the [Send Messages] permission. Additionally, attaching files requires
267    /// the [Attach Files] permission.
268    ///
269    /// **Note**: Message contents must be under 2000 unicode code points, and embeds must be under
270    /// 6000 code points.
271    ///
272    /// # Errors
273    ///
274    /// Returns a [`ModelError::TooLarge`] if the message contents are over the above limits.
275    ///
276    /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is given.
277    ///
278    /// [Send Messages]: Permissions::SEND_MESSAGES
279    /// [Attach Files]: Permissions::ATTACH_FILES
280    #[cfg(feature = "http")]
281    pub async fn execute(mut self, http: &Http, channel_id: GenericChannelId) -> Result<Message> {
282        self.check_length()?;
283
284        let files = self.attachments.new_attachment_data();
285        if self.allowed_mentions.is_none() {
286            self.allowed_mentions.clone_from(&http.default_allowed_mentions);
287        }
288
289        http.send_message(channel_id, files, &self).await
290    }
