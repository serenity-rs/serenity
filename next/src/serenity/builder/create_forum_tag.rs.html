1use std::borrow::Cow;
2
3use crate::model::prelude::*;
4
5/// [Discord docs](https://docs.discord.com/developers/resources/channel#forum-tag-object-forum-tag-structure)
6///
7/// Contrary to the [`ForumTag`] struct, only the name field is required.
8#[must_use]
9#[derive(Clone, Debug, Serialize)]
10pub struct CreateForumTag<'a> {
11    name: Cow<'a, str>,
12    moderated: bool,
13    emoji_id: Option<EmojiId>,
14    emoji_name: Option<Cow<'a, str>>,
15}
16
17impl<'a> CreateForumTag<'a> {
18    pub fn new(name: impl Into<Cow<'a, str>>) -> Self {
19        Self {
20            name: name.into(),
21            moderated: false,
22            emoji_id: None,
23            emoji_name: None,
24        }
25    }
26
27    pub fn moderated(mut self, moderated: bool) -> Self {
28        self.moderated = moderated;
29        self
30    }
31
32    pub fn emoji(mut self, emoji: impl Into<ReactionType>) -> Self {
33        match emoji.into() {
34            ReactionType::Custom {
35                id, ..
36            } => {
37                self.emoji_id = Some(id);
38                self.emoji_name = None;
39            },
40            ReactionType::Unicode(unicode_emoji) => {
41                self.emoji_id = None;
42                self.emoji_name = Some(unicode_emoji.into());
43            },
44        }
45        self
46    }
47}
