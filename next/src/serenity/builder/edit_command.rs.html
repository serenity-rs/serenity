9/// A builder for editing an existing [`Command`].
10///
11/// [`Command`]: crate::model::application::Command
12///
13/// Discord docs:
14/// - [global command](https://docs.discord.com/developers/interactions/application-commands#edit-global-application-command)
15/// - [guild command](https://docs.discord.com/developers/interactions/application-commands#edit-guild-application-command)
16#[derive(Clone, Debug, Default, Serialize)]
17#[must_use]
18pub struct EditCommand<'a> {
19    name: Option<Cow<'a, str>>,
20    name_localizations: HashMap<Cow<'a, str>, Cow<'a, str>>,
21    #[serde(skip_serializing_if = "Option::is_none")]
22    description: Option<Cow<'a, str>>,
23    description_localizations: HashMap<Cow<'a, str>, Cow<'a, str>>,
24    options: Cow<'a, [CreateCommandOption<'a>]>,
25    #[serde(skip_serializing_if = "Option::is_none")]
26    default_member_permissions: Option<Permissions>,
27    #[serde(skip_serializing_if = "Option::is_none")]
28    #[cfg(not(feature = "unstable"))]
29    dm_permission: Option<bool>,
30    #[serde(skip_serializing_if = "Option::is_none")]
31    integration_types: Option<Cow<'a, [InstallationContext]>>,
32    #[serde(skip_serializing_if = "Option::is_none")]
33    contexts: Option<Cow<'a, [InteractionContext]>>,
34    nsfw: bool,
35}
36
37impl<'a> EditCommand<'a> {
38    /// Equivalent to [`Self::default`].
39    pub fn new() -> Self {
40        Self::default()
41    }
42
43    /// Specifies the name of the application command.
44    ///
45    /// **Note**: Must be between 1 and 32 lowercase characters, matching `r"^[\w-]{1,32}$"`. Two
46    /// global commands of the same app cannot have the same name. Two guild-specific commands of
47    /// the same app cannot have the same name.
48    pub fn name(mut self, name: impl Into<Cow<'a, str>>) -> Self {
49        self.name = Some(name.into());
50        self
51    }
52
53    /// Specifies a localized name of the application command.
54    ///
55    /// ```rust
56    /// # serenity::builder::EditCommand::new()
57    /// .name("birthday")
58    /// .name_localized("zh-CN", "生日")
59    /// .name_localized("el", "γενέθλια")
60    /// # ;
61    /// ```
62    pub fn name_localized(
63        mut self,
64        locale: impl Into<Cow<'a, str>>,
65        name: impl Into<Cow<'a, str>>,
66    ) -> Self {
67        self.name_localizations.insert(locale.into(), name.into());
68        self
69    }
70
71    /// Specifies the default permissions required to execute the command.
72    pub fn default_member_permissions(mut self, permissions: Permissions) -> Self {
73        self.default_member_permissions = Some(permissions);
74        self
75    }
76
77    /// Specifies if the command is available in DMs.
78    #[cfg(not(feature = "unstable"))]
79    pub fn dm_permission(mut self, enabled: bool) -> Self {
80        self.dm_permission = Some(enabled);
81        self
82    }
83
84    /// Specifies the description of the application command.
85    ///
86    /// **Note**: Must be between 1 and 100 characters long.
87    pub fn description(mut self, description: impl Into<Cow<'a, str>>) -> Self {
88        self.description = Some(description.into());
89        self
90    }
91
92    /// Specifies a localized description of the application command.
93    ///
94    /// ```rust
95    /// # serenity::builder::CreateCommand::new("")
96    /// .description("Wish a friend a happy birthday")
97    /// .description_localized("zh-CN", "祝你朋友生日快乐")
98    /// # ;
99    /// ```
100    pub fn description_localized(
101        mut self,
102        locale: impl Into<Cow<'a, str>>,
103        description: impl Into<Cow<'a, str>>,
104    ) -> Self {
105        self.description_localizations.insert(locale.into(), description.into());
106        self
107    }
108
109    /// Adds an application command option for the application command.
110    ///
111    /// **Note**: Application commands can have up to 25 options.
112    pub fn add_option(mut self, option: CreateCommandOption<'a>) -> Self {
113        self.options.to_mut().push(option);
114        self
115    }
116
117    /// Sets all the application command options for the application command.
118    ///
119    /// **Note**: Application commands can have up to 25 options.
120    pub fn set_options(mut self, options: impl Into<Cow<'a, [CreateCommandOption<'a>]>>) -> Self {
121        self.options = options.into();
122        self
123    }
124
125    /// Adds an installation context that this application command can be used in.
126    pub fn add_integration_type(mut self, integration_type: InstallationContext) -> Self {
127        self.integration_types.get_or_insert_default().to_mut().push(integration_type);
128        self
129    }
130
131    /// Sets the installation contexts that this application command can be used in.
132    pub fn integration_types(mut self, integration_types: Cow<'a, [InstallationContext]>) -> Self {
133        self.integration_types = Some(integration_types);
134        self
135    }
136
137    /// Adds an interaction context that this application command can be used in.
138    pub fn add_context(mut self, context: InteractionContext) -> Self {
139        self.contexts.get_or_insert_default().to_mut().push(context);
140        self
141    }
142
143    /// Sets the interaction contexts that this application command can be used in.
144    pub fn contexts(mut self, contexts: Cow<'a, [InteractionContext]>) -> Self {
145        self.contexts = Some(contexts);
146        self
147    }
148
149    /// Whether this command is marked NSFW (age-restricted)
150    pub fn nsfw(mut self, nsfw: bool) -> Self {
151        self.nsfw = nsfw;
152        self
153    }
154
155    /// Edit a [`Command`], overwriting an existing one with the same name if it exists.
156    ///
157    /// Providing a [`GuildId`] will edit a command in the corresponding [`Guild`]. Otherwise, a
158    /// global command will be edited.
159    ///
160    /// # Errors
161    ///
162    /// Returns [`Error::Http`] if invalid data is given. See [Discord's docs] for more details.
163    ///
164    /// May also return [`Error::Json`] if there is an error in deserializing the API response.
165    ///
166    /// [Discord's docs]: https://docs.discord.com/developers/interactions/application-commands#slash-commands
167    #[cfg(feature = "http")]
168    pub async fn execute(
169        self,
170        http: &Http,
171        command_id: CommandId,
172        guild_id: Option<GuildId>,
173    ) -> Result<Command> {
174        match guild_id {
175            Some(guild_id) => http.edit_guild_command(guild_id, command_id, &self).await,
176            None => http.edit_global_command(command_id, &self).await,
177        }
178    }
