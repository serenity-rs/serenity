20/// The common result type between most library functions.
21///
22/// The library exposes functions which, for a result type, exposes only one type, rather than the
23/// usual 2 (`Result<T, Error>`). This is because all functions that return a result return
24/// serenity's [`Error`], so this is implied, and a "simpler" result is used.
25pub type Result<T, E = Error> = StdResult<T, E>;
26
27/// A common error enum returned by most of the library's functionality within a custom [`Result`].
28#[derive(Debug)]
29#[non_exhaustive]
30pub enum Error {
31    /// An [`std::io`] error.
32    Io(IoError),
33    /// An error from the [`serde_json`] crate.
34    Json(serde_json::Error),
35    /// An error from the [`model`] module.
36    ///
37    /// [`model`]: crate::model
38    Model(ModelError),
39    /// An error from the [`gateway`] module.
40    ///
41    /// [`gateway`]: crate::gateway
42    #[cfg(feature = "gateway")]
43    Gateway(GatewayError),
44    /// An error from the [`http`] module.
45    ///
46    /// [`http`]: crate::http
47    #[cfg(feature = "http")]
48    Http(HttpError),
49    /// An error from the `tungstenite` crate.
50    #[cfg(feature = "gateway")]
51    Tungstenite(Box<TungsteniteError>),
52    /// An error from the [`secrets`] module.
53    ///
54    /// [`secrets`]: crate::secrets
55    Token(TokenError),
56    /// When parsing a URL failed due to invalid input.
57    Url(UrlError),
58}

60#[derive(Debug)]
61#[non_exhaustive]
62pub enum UrlError {
63    Parsing(url::ParseError),
64    InvalidDataURI,
65}
66
67impl fmt::Display for UrlError {
68    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
69        match self {
70            Self::Parsing(inner) => fmt::Display::fmt(&inner, f),
71            Self::InvalidDataURI => f.write_str("Provided string is not a valid data URI"),
72        }
73    }
74}
75
76impl StdError for UrlError {
77    fn source(&self) -> Option<&(dyn StdError + 'static)> {
78        match self {
79            Self::Parsing(inner) => Some(inner),
80            _ => None,
81        }
82    }
83}
