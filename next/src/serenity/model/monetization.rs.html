7/// A premium offering that can be made available to an application's users and guilds.
8///
9/// [Discord docs](https://docs.discord.com/developers/resources/sku#sku-object).
10#[derive(Clone, Debug, Serialize, Deserialize)]
11pub struct Sku {
12    /// The unique ID of the SKU.
13    pub id: SkuId,
14    /// The class of the SKU.
15    #[serde(rename = "type")]
16    pub kind: SkuKind,
17    /// Id of the SKU's parent application.
18    pub application_id: ApplicationId,
19    /// The customer-facing name of the premium offering.
20    pub name: String,
21    /// A system-generated URL slug based on the SKU.
22    pub slug: String,
23    /// Flags indicating the type of subscription the SKU represents.
24    pub flags: SkuFlags,
25}
26
27impl Sku {
28    /// Returns the store url for this SKU. If included in a message, will render as a rich embed.
29    /// See the [Discord docs] for details.
30    ///
31    /// [Discord docs]: https://docs.discord.com/developers/monetization/managing-skus#linking-to-a-specific-sku
32    #[must_use]
33    pub fn url(&self) -> String {
34        format!(
35            "https://discord.com/application-directory/{}/store/{}",
36            self.application_id, self.id
37        )
38    }
39}

42    /// Differentiates between SKU classes.
43    ///
44    /// [Discord docs](https://docs.discord.com/developers/resources/sku#sku-object-sku-types).
45    #[derive(Clone, Debug, Serialize, Deserialize)]
46    #[non_exhaustive]
47    pub enum SkuKind {
48        /// A durable one-time purchase.
49        Durable = 2,
50        /// A consumable one-time purchase.
51        Consumable = 3,
52        /// Represents a recurring subscription.
53        Subscription = 5,
54        /// A system-generated group for each SKU created of type [`SkuKind::Subscription`].
55        SubscriptionGroup = 6,
56        _ => Unknown(u8),
57    }
58}

60bitflags! {
61    /// Differentates between user and server subscriptions.
62    ///
63    /// [Discord docs](https://docs.discord.com/developers/resources/sku#sku-object-sku-flags).
64    #[derive(Copy, Clone, Debug, Eq, Hash, PartialEq)]
65    pub struct SkuFlags: u16 {
66        /// SKU is available for purchase.
67        const AVAILABLE = 1 << 2;
68        /// Recurring SKU that can be purchased by a user and applied to a single server. Grants
69        /// access to every user in that server.
70        const GUILD_SUBSCRIPTION = 1 << 7;
71        /// Recurring SKU purchased by a user for themselves. Grants access to the purchasing user
72        /// in every server.
73        const USER_SUBSCRIPTION = 1 << 8;
74    }
75}

77/// Represents that a user or guild has access to a premium offering in the application.
78///
79/// [Discord docs](https://docs.discord.com/developers/resources/entitlement#entitlement-object-entitlement-structure).
80#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
81#[derive(Clone, Debug, Serialize, Deserialize)]
82pub struct Entitlement {
83    /// The ID of the entitlement.
84    pub id: EntitlementId,
85    /// The ID of the corresponding SKU.
86    pub sku_id: SkuId,
87    /// The ID of the parent application.
88    pub application_id: ApplicationId,
89    /// The ID of the user that is granted access to the SKU.
90    pub user_id: Option<UserId>,
91    /// The type of the entitlement.
92    #[serde(rename = "type")]
93    pub kind: EntitlementKind,
94    /// Whether the entitlement has been deleted or not. Entitlements are not deleted when they
95    /// expire.
96    pub deleted: bool,
97    /// Start date after which the entitlement is valid. Not present when using test entitlements.
98    pub starts_at: Option<Timestamp>,
99    /// End date after which the entitlement is no longer valid. Not present when using test
100    /// entitlements.
101    pub ends_at: Option<Timestamp>,
102    /// The ID of the guild that is granted access to the SKU.
103    pub guild_id: Option<GuildId>,
104    /// For consumable items, whether or not the entitlement has been consumed.
105    pub consumed: Option<bool>,
106}
107
108impl Entitlement {
109    /// Returns a link to the SKU corresponding to this entitlement. See [`Sku::url`] for details.
110    #[must_use]
111    pub fn sku_url(&self) -> String {
112        format!(
113            "https://discord.com/application-directory/{}/store/{}",
114            self.application_id, self.sku_id
115        )
116    }
117
118    /// For a one-time purchase consumable SKU (of kind [`Consumable`]), marks the entitlement as
119    /// consumed. On success, the [`consumed`] field will be set to `Some(true)`.
120    ///
121    /// # Errors
122    ///
123    /// Will fail if the corresponding SKU is not of kind [`Consumable`].
124    ///
125    /// [`Consumable`]: SkuKind::Consumable
126    /// [`consumed`]: Entitlement::consumed
127    #[cfg(feature = "model")]
128    pub async fn consume(&mut self, http: &Http) -> Result<()> {
129        http.consume_entitlement(self.id).await?;
130        self.consumed = Some(true);
131        Ok(())
132    }
133
134    /// Returns all entitlements for the current application, active and expired.
135    ///
136    /// # Errors
137    ///
138    /// May error due to an invalid response from discord, or network error.
139    #[cfg(feature = "model")]
140    pub async fn list(http: &Http, builder: GetEntitlements<'_>) -> Result<Vec<Entitlement>> {
141        builder.execute(http).await
142    }
143}
