6#[derive(Copy, Clone, Debug, Eq, Hash, PartialEq)]
7#[non_exhaustive]
8pub enum Maximum {
9    EmbedLength,
10    EmbedCount,
11    MessageLength,
12    StickerCount,
13    WebhookName,
14    AuditLogReason,
15    BulkDeleteAmount,
16}
17
18#[cfg(feature = "http")]
19impl Maximum {
20    pub(crate) fn check_overflow(self, value: usize) -> Result<(), Error> {
21        let max = self.value();
22        if value > max {
23            Err(Error::TooLarge {
24                maximum: self,
25                value,
26            })
27        } else {
28            Ok(())
29        }
30    }
31
32    pub(crate) fn value(self) -> usize {
33        match self {
34            Self::EmbedCount => crate::constants::EMBED_MAX_COUNT,
35            Self::EmbedLength => crate::constants::EMBED_MAX_LENGTH,
36            Self::MessageLength => crate::constants::MESSAGE_CODE_LIMIT,
37            Self::StickerCount => crate::constants::STICKER_MAX_COUNT,
38            Self::WebhookName | Self::BulkDeleteAmount => 100,
39            Self::AuditLogReason => 512,
40        }
41    }
42}
43
44impl fmt::Display for Maximum {
45    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
46        match self {
47            Self::EmbedCount => f.write_str("Embed count"),
48            Self::EmbedLength => f.write_str("Embed length"),
49            Self::MessageLength => f.write_str("Message length"),
50            Self::StickerCount => f.write_str("Sticker count"),
51            Self::WebhookName => f.write_str("Webhook name"),
52            Self::AuditLogReason => f.write_str("Audit log reason"),
53            Self::BulkDeleteAmount => f.write_str("Message bulk delete count"),
54        }
55    }
56}

60pub enum Minimum {

62    BulkDeleteAmount,

65#[cfg(feature = "http")]
66impl Minimum {

68        let min = self.value();
69        if value < min {
70            Err(Error::TooSmall {
71                minimum: self,
72                value,
73            })
74        } else {
75            Ok(())
76        }
77    }

96/// An error returned from the [`model`] module.
97///
98/// This is always wrapped within the library's [`Error::Model`] variant.
99///
100/// [`Error::Model`]: crate::Error::Model
101/// [`model`]: crate::model
102#[derive(Clone, Debug, Eq, Hash, PartialEq)]
103#[non_exhaustive]
104pub enum Error {
105    /// Indicates that the `minimum` has been missed by the `value`.
106    TooSmall { minimum: Minimum, value: usize },
107    /// Indicates that the `maximum` has been exceeded by the `value`.
108    TooLarge { maximum: Maximum, value: usize },
109    /// An indication that a [`Guild`] could not be found by [Id][`GuildId`] in the [`Cache`].
110    ///
111    /// [`Guild`]: super::guild::Guild
112    /// [`GuildId`]: super::id::GuildId
113    /// [`Cache`]: crate::cache::Cache
114    GuildNotFound,
115    /// An indication that a [`Role`] could not be found by [Id][`RoleId`] in the [`Cache`].
116    ///
117    /// [`Role`]: super::guild::Role
118    /// [`RoleId`]: super::id::RoleId
119    /// [`Cache`]: crate::cache::Cache
120    RoleNotFound,
121    /// An indication that a [`Member`] could not be found by [Id][`UserId`] in the [`Cache`].
122    ///
123    /// [`Member`]: super::guild::Member
124    /// [`UserId`]: super::id::UserId
125    /// [`Cache`]: crate::cache::Cache
126    MemberNotFound,
127    /// An indication that a [`Channel`] could not be found by [Id][`ChannelId`] in the [`Cache`].
128    ///
129    /// [`Channel`]: super::channel::Channel
130    /// [`ChannelId`]: super::id::ChannelId
131    /// [`Cache`]: crate::cache::Cache
132    ChannelNotFound,
133    /// An indication that a [`Message`] has already been crossposted, and cannot be crossposted
134    /// twice.
135    ///
136    /// [`Message`]: super::channel::Message
137    MessageAlreadyCrossposted,
138    /// An indication that you cannot crosspost a [`Message`].
139    ///
140    /// For instance, you cannot crosspost a system message or a message coming from the crosspost
141    /// feature.
142    ///
143    /// [`Message`]: super::channel::Message
144    CannotCrosspostMessage,
145    /// Indicates that there are hierarchy problems restricting an action.
146    ///
147    /// For example, when banning a user, if the other user has a role with an equal to or higher
148    /// position, then they can not be banned.
149    ///
150    /// When editing a role, if the role is higher in position than the current user's highest
151    /// role, then the role can not be edited.
152    Hierarchy,
153    /// An indicator that the [current user] cannot perform an action.
154    ///
155    /// [current user]: super::user::CurrentUser
156    InvalidUser,
157    /// An indicator that an item is missing from the [`Cache`], and the action can not be
158    /// continued.
159    ///
160    /// [`Cache`]: crate::cache::Cache
161    ItemMissing,
162    /// Indicates that the current user is attempting to Direct Message another bot user, which is
163    /// disallowed by the API.
164    MessagingBot,
165    /// An indicator that the [`ChannelType`] cannot perform an action.
166    ///
167    /// [`ChannelType`]: super::channel::ChannelType
168    InvalidChannelType,
169    /// Indicates that the webhook token is missing.
170    NoTokenSet,
171    /// When attempting to delete a built in nitro sticker instead of a guild sticker.
172    DeleteNitroSticker,
173    /// When attempting to edit a voice message.
174    CannotEditVoiceMessage,
175}
176
177impl Error {
178    /// Return `true` if the model error is related to an item missing in the cache.
179    #[must_use]
180    pub const fn is_cache_err(&self) -> bool {
181        matches!(
182            self,
183            Self::ItemMissing
184                | Self::ChannelNotFound
185                | Self::RoleNotFound
186                | Self::GuildNotFound
187                | Self::MemberNotFound
188        )
189    }
190}
191
192impl fmt::Display for Error {
193    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
194        match self {
195            Self::TooSmall {
196                minimum,
197                value,
198            } => write!(f, "The {minimum} minimum has been missed by {value}"),
199            Self::TooLarge {
200                maximum,
201                value,
202            } => write!(f, "The {maximum} maximum has been overflowed by {value}"),
203            Self::GuildNotFound => f.write_str("Guild not found in the cache."),
204            Self::RoleNotFound => f.write_str("Role not found in the cache."),
205            Self::MemberNotFound => f.write_str("Member not found in the cache."),
206            Self::ChannelNotFound => f.write_str("Channel not found in the cache."),
207            Self::Hierarchy => f.write_str("Role hierarchy prevents this action."),
208            Self::InvalidChannelType => f.write_str("The channel cannot perform the action."),
209            Self::InvalidUser => f.write_str("The current user cannot perform the action."),
210            Self::ItemMissing => f.write_str("The required item is missing from the cache."),
211            Self::MessageAlreadyCrossposted => f.write_str("Message already crossposted."),
212            Self::CannotCrosspostMessage => f.write_str("Cannot crosspost this message type."),
213            Self::MessagingBot => f.write_str("Attempted to message another bot user."),
214            Self::NoTokenSet => f.write_str("Token is not set."),
215            Self::DeleteNitroSticker => f.write_str("Cannot delete an official sticker."),
216            Self::CannotEditVoiceMessage => f.write_str("Cannot edit voice message."),
217        }
218    }
219}
