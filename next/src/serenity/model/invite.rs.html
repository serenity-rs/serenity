11/// Information about an invite code.
12///
13/// [Discord docs](https://docs.discord.com/developers/resources/invite#invite-object).
14#[derive(Clone, Debug, Deserialize, Serialize)]
15#[non_exhaustive]
16pub struct Invite {
17    /// The type of invite.
18    #[serde(rename = "type")]
19    pub kind: InviteType,
20    /// The unique code for the invite.
21    pub code: FixedString,
22    /// A representation of the minimal amount of information needed about the [`Guild`] being
23    /// invited to.
24    pub guild: Option<InviteGuild>,
25    /// A representation of the minimal amount of information needed about the [`GuildChannel`]
26    /// being invited to.
27    pub channel: InviteChannel,
28    /// The [`User`] who created the invite.
29    ///
30    /// Can be [`None`] for invites created by Discord, e.g., server widgets or vanity invite
31    /// links.
32    pub inviter: Option<User>,
33    /// The type of target for this voice channel invite.
34    pub target_type: Option<InviteTargetType>,
35    /// The [`User`] whose stream to display for this voice channel stream invite.
36    ///
37    /// Only shows up if `target_type` is `Stream`.
38    pub target_user: Option<User>,
39    /// The embedded application to open for this voice channel embedded application invite.
40    ///
41    /// Only shows up if `target_type` is `EmmbeddedApplication`.
42    pub target_application: Option<MessageApplication>,
43    /// The approximate number of [`Member`]s with an active session in the related [`Guild`].
44    ///
45    /// An active session is defined as an open, heartbeating WebSocket connection.
46    /// These include [invisible][`OnlineStatus::Invisible`] members.
47    ///
48    /// Only included when retrieving a single invite with `member_counts` set to `true`.
49    pub approximate_presence_count: Option<NonMaxU64>,
50    /// The approximate number of [`Member`]s in the related [`Guild`].
51    ///
52    /// Only included when retrieving a single invite with `member_counts` set to `true`.
53    pub approximate_member_count: Option<NonMaxU64>,
54    /// The expiration date of this invite.
55    pub expires_at: Option<Timestamp>,
56    /// Guild scheduled event data.
57    ///
58    /// Only included when retrieving a single invite with a valid [`ScheduledEventId`] provided
59    /// for `event_id`.
60    #[serde(rename = "guild_scheduled_event")]
61    pub scheduled_event: Option<ScheduledEvent>,
62    /// Guild invite flags for guild invites.
63    pub flags: Option<GuildInviteFlags>,
64    /// A representation of the minimal amount of information needed about the [`Role`]s assigned
65    /// to a user upon accepting the invite.
66    pub roles: Option<FixedArray<InviteRole>>,
67    /// Extra information about an invite.
68    ///
69    /// Only included when retrieving guild invites with the
70    /// [`MANAGE_GUILD`][Permissions::MANAGE_GUILD] permission or channel invites with the
71    /// [`MANAGE_CHANNELS`][Permissions::MANAGE_CHANNELS] permission.
72    #[serde(flatten)]
73    pub metadata: Option<InviteMetadata>,
74}

88#[cfg(feature = "model")]
89impl Invite {
90    /// Creates an invite for the given channel.
91    ///
92    /// **Note**: Requires the [Create Instant Invite] permission.
93    ///
94    /// # Errors
95    ///
96    /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is given.
97    ///
98    /// [Create Instant Invite]: Permissions::CREATE_INSTANT_INVITE
99    pub async fn create(
100        http: &Http,
101        channel_id: ChannelId,
102        builder: CreateInvite<'_>,
103    ) -> Result<Invite> {
104        channel_id.create_invite(http, builder).await
105    }
106
107    /// Deletes the invite.
108    ///
109    /// **Note**: Requires the [Manage Channels] permission on the channel this invite belongs to,
110    /// or [Manage Guild] to remove any invite across the guild.
111    ///
112    /// # Errors
113    ///
114    /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is given.
115    ///
116    /// [Manage Guild]: Permissions::MANAGE_GUILD
117    /// [Manage Channels]: Permissions::MANAGE_CHANNELS
118    pub async fn delete(&self, http: &Http, reason: Option<&str>) -> Result<Invite> {
119        http.delete_invite(&self.code, reason).await
120    }
121
122    /// Gets information about an invite.
123    ///
124    /// # Arguments
125    /// * `code` - The invite code.
126    /// * `member_counts` - Whether to include information about the current number of members in
127    ///   the server that the invite belongs to.
128    /// * `event_id` - An optional server event ID to include with the invite.
129    ///
130    /// More information about these arguments can be found on Discord's
131    /// [API documentation](https://docs.discord.com/developers/resources/invite#get-invite).
132    ///
133    /// # Errors
134    ///
135    /// May return an [`Error::Http`] if the invite is invalid. Can also return an [`Error::Json`]
136    /// if there is an error deserializing the API response.
137    pub async fn get(
138        http: &Http,
139        code: &str,
140        member_counts: bool,
141        event_id: Option<ScheduledEventId>,
142    ) -> Result<Invite> {
143        http.get_invite(code, member_counts, event_id).await
144    }
145
146    /// Returns a URL to use for the invite.
147    ///
148    /// # Examples
149    ///
150    /// Retrieve the URL for an invite with the code `WxZumR`:
151    ///
152    /// ```rust
153    /// # use serde_json::{json, from_value};
154    /// # use serenity::model::prelude::*;
155    /// #
156    /// # fn main() {
157    /// # let invite = from_value::<Invite>(json!({
158    /// #     "type": 0,
159    /// #     "code": "WxZumR",
160    /// #     "guild": {
161    /// #         "id": GuildId::new(2),
162    /// #         "icon": None::<String>,
163    /// #         "name": "bar",
164    /// #         "splash_hash": None::<String>,
165    /// #         "text_channel_count": 7,
166    /// #         "voice_channel_count": 3,
167    /// #         "features": ["NEWS", "DISCOVERABLE"],
168    /// #         "verification_level": 2,
169    /// #         "nsfw_level": 0,
170    /// #     },
171    /// #     "channel": {
172    /// #         "id": ChannelId::new(1),
173    /// #         "name": "foo",
174    /// #         "type": ChannelType::Text,
175    /// #     },
176    /// #     "inviter": {
177    /// #         "id": UserId::new(3),
178    /// #         "username": "foo",
179    /// #         "discriminator": "1234",
180    /// #         "avatar": "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa"
181    /// #     },
182    /// #     "approximate_presence_count": Some(717),
183    /// #     "approximate_member_count": Some(1812),
184    /// # })).unwrap();
185    /// #
186    /// assert_eq!(invite.url(), "https://discord.gg/WxZumR");
187    /// # }
188    /// ```
189    #[must_use]
190    pub fn url(&self) -> String {
191        format!("https://discord.gg/{}", self.code)
192    }
193}

243/// Extra information about an invite. Extends [`Invite`].
244///
245/// Retrieving this information requires the [Manage Guild] permission when retrieving guild
246/// invites or the [Manage Channels] permission when retrieving channel invites.
247///
248/// [Manage Guild]: Permissions::MANAGE_GUILD
249/// [Manage Channels]: Permissions::MANAGE_CHANNELS
250///
251/// [Discord docs](https://docs.discord.com/developers/resources/invite#invite-metadata-object).
252#[derive(Clone, Debug, Deserialize, Serialize)]
253#[non_exhaustive]
254pub struct InviteMetadata {
255    /// The number of times the invite has been used.
256    pub uses: u64,
257    /// The maximum number of times the invite can be used.
258    ///
259    /// If the value is `0`, then the invite has unlimited uses.
260    pub max_uses: u8,
261    /// The duration (in seconds) after which the invite expires.
262    ///
263    /// If the value is `0`, then the invite never expires.
264    pub max_age: u32,
265    /// Whether the invite only grants temporary membership.
266    pub temporary: bool,
267    /// When the invite was created.
268    pub created_at: Timestamp,
269}
