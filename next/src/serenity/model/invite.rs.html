11/// Information about an invite code.
12///
13/// Information can not be accessed for guilds the current user is banned from.
14///
15/// [Discord docs](https://docs.discord.com/developers/resources/invite#invite-object).
16#[derive(Clone, Debug, Deserialize, Serialize)]
17#[non_exhaustive]
18pub struct Invite {
19    /// The approximate number of [`Member`]s in the related [`Guild`].
20    pub approximate_member_count: Option<NonMaxU64>,
21    /// The approximate number of [`Member`]s with an active session in the related [`Guild`].
22    ///
23    /// An active session is defined as an open, heartbeating WebSocket connection.
24    /// These include [invisible][`OnlineStatus::Invisible`] members.
25    pub approximate_presence_count: Option<NonMaxU64>,
26    /// The unique code for the invite.
27    pub code: FixedString,
28    /// A representation of the minimal amount of information needed about the [`GuildChannel`]
29    /// being invited to.
30    pub channel: InviteChannel,
31    /// A representation of the minimal amount of information needed about the [`Guild`] being
32    /// invited to.
33    pub guild: Option<InviteGuild>,
34    /// A representation of the minimal amount of information needed about the [`User`] that
35    /// created the invite.
36    ///
37    /// This can be [`None`] for invites created by Discord such as invite-widgets or vanity invite
38    /// links.
39    pub inviter: Option<User>,
40    /// The type of target for this voice channel invite.
41    pub target_type: Option<InviteTargetType>,
42    /// The user whose stream to display for this voice channel stream invite.
43    ///
44    /// Only shows up if `target_type` is `Stream`.
45    pub target_user: Option<UserId>,
46    /// The embedded application to open for this voice channel embedded application invite.
47    ///
48    /// Only shows up if `target_type` is `EmmbeddedApplication`.
49    pub target_application: Option<ApplicationId>,
50    /// The expiration date of this invite, returned from `Http::get_invite` when `with_expiration`
51    /// is true.
52    pub expires_at: Option<Timestamp>,
53    /// The Stage instance data if there is a public Stage instance in the Stage channel this
54    /// invite is for.
55    pub stage_instance: Option<InviteStageInstance>,
56    /// Guild scheduled event data, only included if guild_scheduled_event_id contains a valid
57    /// guild scheduled event id (according to Discord docs, whatever that means).
58    #[serde(rename = "guild_scheduled_event")]
59    pub scheduled_event: Option<ScheduledEvent>,
60}

62#[cfg(feature = "model")]
63impl Invite {
64    /// Creates an invite for the given channel.
65    ///
66    /// **Note**: Requires the [Create Instant Invite] permission.
67    ///
68    /// # Errors
69    ///
70    /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is given.
71    ///
72    /// [Create Instant Invite]: Permissions::CREATE_INSTANT_INVITE
73    pub async fn create(
74        http: &Http,
75        channel_id: ChannelId,
76        builder: CreateInvite<'_>,
77    ) -> Result<RichInvite> {
78        channel_id.create_invite(http, builder).await
79    }
80
81    /// Deletes the invite.
82    ///
83    /// **Note**: Requires the [Manage Guild] permission.
84    ///
85    /// # Errors
86    ///
87    /// Returns [`Error::Http`] if the current user lacks permission or if the invite is
88    /// invalid.
89    ///
90    /// [Manage Guild]: Permissions::MANAGE_GUILD
91    /// [permission]: super::permissions
92    pub async fn delete(&self, http: &Http, reason: Option<&str>) -> Result<Invite> {
93        http.delete_invite(&self.code, reason).await
94    }
95
96    /// Gets information about an invite.
97    ///
98    /// # Arguments
99    /// * `code` - The invite code.
100    /// * `member_counts` - Whether to include information about the current number of members in
101    ///   the server that the invite belongs to.
102    /// * `expiration` - Whether to include information about when the invite expires.
103    /// * `event_id` - An optional server event ID to include with the invite.
104    ///
105    /// More information about these arguments can be found on Discord's
106    /// [API documentation](https://docs.discord.com/developers/resources/invite#get-invite).
107    ///
108    /// # Errors
109    ///
110    /// May return an [`Error::Http`] if the invite is invalid. Can also return an [`Error::Json`]
111    /// if there is an error deserializing the API response.
112    pub async fn get(
113        http: &Http,
114        code: &str,
115        member_counts: bool,
116        expiration: bool,
117        event_id: Option<ScheduledEventId>,
118    ) -> Result<Invite> {
119        http.get_invite(code, member_counts, expiration, event_id).await
120    }
121
122    /// Returns a URL to use for the invite.
123    ///
124    /// # Examples
125    ///
126    /// Retrieve the URL for an invite with the code `WxZumR`:
127    ///
128    /// ```rust
129    /// # use serde_json::{json, from_value};
130    /// # use serenity::model::prelude::*;
131    /// #
132    /// # fn main() {
133    /// # let invite = from_value::<Invite>(json!({
134    /// #     "approximate_member_count": Some(1812),
135    /// #     "approximate_presence_count": Some(717),
136    /// #     "code": "WxZumR",
137    /// #     "channel": {
138    /// #         "id": ChannelId::new(1),
139    /// #         "name": "foo",
140    /// #         "type": ChannelType::Text,
141    /// #     },
142    /// #     "guild": {
143    /// #         "id": GuildId::new(2),
144    /// #         "icon": None::<String>,
145    /// #         "name": "bar",
146    /// #         "splash_hash": None::<String>,
147    /// #         "text_channel_count": 7,
148    /// #         "voice_channel_count": 3,
149    /// #         "features": ["NEWS", "DISCOVERABLE"],
150    /// #         "verification_level": 2,
151    /// #         "nsfw_level": 0,
152    /// #     },
153    /// #     "inviter": {
154    /// #         "id": UserId::new(3),
155    /// #         "username": "foo",
156    /// #         "discriminator": "1234",
157    /// #         "avatar": "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa"
158    /// #     },
159    /// # })).unwrap();
160    /// #
161    /// assert_eq!(invite.url(), "https://discord.gg/WxZumR");
162    /// # }
163    /// ```
164    #[must_use]
165    pub fn url(&self) -> String {
166        format!("https://discord.gg/{}", self.code)
167    }
168}

201/// Detailed information about an invite.
202///
203/// This information can only be retrieved by anyone with the [Manage Guild] permission. Otherwise,
204/// a minimal amount of information can be retrieved via the [`Invite`] struct.
205///
206/// [Manage Guild]: Permissions::MANAGE_GUILD
207///
208/// [Discord docs](https://docs.discord.com/developers/resources/invite#invite-metadata-object) (extends [`Invite`] fields).
209#[derive(Clone, Debug, Deserialize, Serialize)]
210#[non_exhaustive]
211pub struct RichInvite {
212    /// A representation of the minimal amount of information needed about the channel being
213    /// invited to.
214    pub channel: InviteChannel,
215    /// The unique code for the invite.
216    pub code: FixedString,
217    /// When the invite was created.
218    pub created_at: Timestamp,
219    /// A representation of the minimal amount of information needed about the [`Guild`] being
220    /// invited to.
221    pub guild: Option<InviteGuild>,
222    /// The user that created the invite.
223    pub inviter: Option<User>,
224    /// The maximum age of the invite in seconds, from when it was created.
225    pub max_age: u32,
226    /// The maximum number of times that an invite may be used before it expires.
227    ///
228    /// Note that this does not supersede the [`Self::max_age`] value, if the value of
229    /// [`Self::temporary`] is `true`. If the value of `temporary` is `false`, then the invite
230    /// _will_ self-expire after the given number of max uses.
231    ///
232    /// If the value is `0`, then the invite is permanent.
233    pub max_uses: u8,
234    /// Indicator of whether the invite self-expires after a certain amount of time or uses.
235    pub temporary: bool,
236    /// The amount of times that an invite has been used.
237    pub uses: u64,
238}
