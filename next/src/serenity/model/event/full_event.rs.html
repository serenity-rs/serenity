11macro_rules! full_event {
12    ( $(
13        $( #[doc = $doc:literal] )*
14        $( #[deprecated = $deprecated:literal] )?
15        $( #[cfg(feature = $feature:literal)] )?
16        $variant_name:ident { $( $arg_name:ident: $arg_type:ty ),* };
17    )* ) => {
18        /// This enum stores every possible event that an `EventHandler` can receive.
19        #[cfg_attr(not(feature = "unstable"), non_exhaustive)]
20        #[derive(Clone, Debug, VariantNames, IntoStaticStr, EnumCount, Serialize)]
21        #[strum(serialize_all = "SCREAMING_SNAKE_CASE")]
22        pub enum FullEvent {
23            $(
24                $( #[doc = $doc] )*
25                $( #[deprecated = $deprecated] )?
26                $( #[cfg(feature = $feature)] )?
27                #[cfg_attr(not(feature = "unstable"), non_exhaustive)]
28                $variant_name {
29                    $( $arg_name: $arg_type ),*
30                },
31            )*
32        }
33    }
34}

406impl FullEvent {
407    pub fn from_event(
408        event: Box<Event>,
409        extra_event: &mut Option<Self>,
410        #[cfg(feature = "cache")] cache: &Cache,
411    ) -> Self {
412        match *event {
413            Event::CommandPermissionsUpdate(event) => Self::CommandPermissionsUpdate {
414                permission: event.permission,
415            },
416            Event::AutoModRuleCreate(event) => Self::AutoModRuleCreate {
417                rule: event.rule,
418            },
419            Event::AutoModRuleUpdate(event) => Self::AutoModRuleUpdate {
420                rule: event.rule,
421            },
422            Event::AutoModRuleDelete(event) => Self::AutoModRuleDelete {
423                rule: event.rule,
424            },
425            Event::AutoModActionExecution(event) => Self::AutoModActionExecution {
426                execution: event.execution,
427            },
428            Event::ChannelCreate(event) => {
429                #[cfg(feature = "cache")]
430                event.update(cache);
431
432                let channel = event.channel;
433                if channel.base.kind == ChannelType::Category {
434                    Self::CategoryCreate {
435                        category: channel,
436                    }
437                } else {
438                    Self::ChannelCreate {
439                        channel,
440                    }
441                }
442            },
443            Event::ChannelDelete(event) => {
444                #[cfg(feature = "cache")]
445                let cached_messages = event.update(cache);
446                #[cfg(not(feature = "cache"))]
447                let cached_messages = None;
448
449                let channel = event.channel;
450                if channel.base.kind == ChannelType::Category {
451                    Self::CategoryDelete {
452                        category: channel,
453                    }
454                } else {
455                    Self::ChannelDelete {
456                        channel,
457                        messages: cached_messages,
458                    }
459                }
460            },
461            Event::ChannelPinsUpdate(event) => Self::ChannelPinsUpdate {
462                pin: event,
463            },
464            Event::ChannelUpdate(event) => {
465                #[cfg(feature = "cache")]
466                let old_channel = event.update(cache);
467                #[cfg(not(feature = "cache"))]
468                let old_channel = None;
469
470                Self::ChannelUpdate {
471                    old: old_channel,
472                    new: event.channel,
473                }
474            },
475            Event::GuildAuditLogEntryCreate(event) => Self::GuildAuditLogEntryCreate {
476                entry: event.entry,
477                guild_id: event.guild_id,
478            },
479            Event::GuildBanAdd(event) => Self::GuildBanAddition {
480                guild_id: event.guild_id,
481                banned_user: event.user,
482            },
483            Event::GuildBanRemove(event) => Self::GuildBanRemoval {
484                guild_id: event.guild_id,
485                unbanned_user: event.user,
486            },
487            Event::GuildCreate(event) => {
488                #[cfg(feature = "cache")]
489                let is_new = Some(is_guild_new(cache, event.guild.id));
490                #[cfg(not(feature = "cache"))]
491                let is_new = None;
492
493                #[cfg(feature = "cache")]
494                if let Some(guilds) = event.update(cache) {
495                    *extra_event = Some(Self::CacheReady {
496                        guilds,
497                    });
498                }
499
500                Self::GuildCreate {
501                    guild: event.guild,
502                    is_new,
503                }
504            },
505            Event::GuildDelete(event) => {
506                #[cfg(feature = "cache")]
507                let full = event.update(cache);
508                #[cfg(not(feature = "cache"))]
509                let full = None;
510
511                Self::GuildDelete {
512                    incomplete: event.guild,
513                    full,
514                }
515            },
516            Event::GuildEmojisUpdate(event) => {
517                #[cfg(feature = "cache")]
518                event.update(cache);
519
520                Self::GuildEmojisUpdate {
521                    guild_id: event.guild_id,
522                    current_state: event.emojis,
523                }
524            },
525            Event::GuildIntegrationsUpdate(event) => Self::GuildIntegrationsUpdate {
526                guild_id: event.guild_id,
527            },
528            Event::GuildMemberAdd(event) => {
529                #[cfg(feature = "cache")]
530                event.update(cache);
531
532                Self::GuildMemberAddition {
533                    new_member: event.member,
534                }
535            },
536            Event::GuildMemberRemove(event) => {
537                #[cfg(feature = "cache")]
538                let member = event.update(cache);
539                #[cfg(not(feature = "cache"))]
540                let member = None;
541
542                Self::GuildMemberRemoval {
543                    guild_id: event.guild_id,
544                    user: event.user,
545                    member_data_if_available: member,
546                }
547            },
548            Event::GuildMemberUpdate(event) => {
549                #[cfg(feature = "cache")]
550                let (before, after) = (
551                    event.update(cache),
552                    cache
553                        .guild(event.guild_id)
554                        .and_then(|g| g.members.get(&event.user.id).cloned()),
555                );
556                #[cfg(not(feature = "cache"))]
557                let (before, after) = (None, None);
558
559                Self::GuildMemberUpdate {
560                    old_if_available: before,
561                    new: after,
562                    event,
563                }
564            },
565            Event::GuildMembersChunk(event) => {
566                #[cfg(feature = "cache")]
567                event.update(cache);
568
569                Self::GuildMembersChunk {
570                    chunk: event,
571                }
572            },
573            Event::GuildRoleCreate(event) => {
574                #[cfg(feature = "cache")]
575                event.update(cache);
576
577                Self::GuildRoleCreate {
578                    new: event.role,
579                }
580            },
581            Event::GuildRoleDelete(event) => {
582                #[cfg(feature = "cache")]
583                let role = event.update(cache);
584                #[cfg(not(feature = "cache"))]
585                let role = None;
586
587                Self::GuildRoleDelete {
588                    guild_id: event.guild_id,
589                    removed_role_id: event.role_id,
590                    removed_role_data_if_available: role,
591                }
592            },
593            Event::GuildRoleUpdate(event) => {
594                #[cfg(feature = "cache")]
595                let before = event.update(cache);
596                #[cfg(not(feature = "cache"))]
597                let before = None;
598
599                Self::GuildRoleUpdate {
600                    old_data_if_available: before,
601                    new: event.role,
602                }
603            },
604            Event::GuildStickersUpdate(event) => {
605                #[cfg(feature = "cache")]
606                event.update(cache);
607
608                Self::GuildStickersUpdate {
609                    guild_id: event.guild_id,
610                    current_state: event.stickers,
611                }
612            },
613            Event::GuildUpdate(event) => {
614                #[cfg(feature = "cache")]
615                let before = cache.guild(event.guild.id).map(|g| g.clone());
616                #[cfg(not(feature = "cache"))]
617                let before = None;
618
619                Self::GuildUpdate {
620                    old_data_if_available: before,
621                    new_data: event.guild,
622                }
623            },
624            Event::InviteCreate(event) => Self::InviteCreate {
625                data: event,
626            },
627            Event::InviteDelete(event) => Self::InviteDelete {
628                data: event,
629            },
630            Event::MessageCreate(event) => {
631                #[cfg(feature = "cache")]
632                event.update(cache);
633
634                Self::Message {
635                    new_message: event.message,
