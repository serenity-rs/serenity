16/// An interaction triggered by a message component.
17///
18/// [Discord docs](https://docs.discord.com/developers/interactions/receiving-and-responding#interaction-object-interaction-structure).
19#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
20#[derive(Clone, Debug, Deserialize, Serialize)]
21#[serde(remote = "Self")]
22#[non_exhaustive]
23pub struct ComponentInteraction {
24    /// Id of the interaction.
25    pub id: InteractionId,
26    /// Id of the application this interaction is for.
27    pub application_id: ApplicationId,
28    /// The data of the interaction which was triggered.
29    pub data: ComponentInteractionData,
30    /// The guild Id this interaction was sent from, if there is one.
31    #[serde(skip_serializing_if = "Option::is_none")]
32    pub guild_id: Option<GuildId>,
33    /// Channel that the interaction was sent from.
34    pub channel: Option<GenericInteractionChannel>,
35    /// The channel Id this interaction was sent from.
36    pub channel_id: GenericChannelId,
37    /// The `member` data for the invoking user.
38    ///
39    /// **Note**: It is only present if the interaction is triggered in a guild.
40    #[serde(skip_serializing_if = "Option::is_none")]
41    pub member: Option<Member>,
42    /// The `user` object for the invoking user.
43    #[serde(default)]
44    pub user: User,
45    /// A continuation token for responding to the interaction.
46    pub token: FixedString,
47    /// Always `1`.
48    pub version: u8,
49    /// The message this interaction was triggered by, if it is a component.
50    pub message: Box<Message>,
51    /// Permissions the app or bot has within the channel the interaction was sent from.
52    pub app_permissions: Permissions,
53    /// The selected language of the invoking user.
54    pub locale: FixedString,
55    /// The guild's preferred locale.
56    pub guild_locale: Option<FixedString>,
57    /// For monetized applications, any entitlements of the invoking user.
58    pub entitlements: FixedArray<Entitlement>,
59    /// The owners of the applications that authorized the interaction, such as a guild or user.
60    #[serde(default)]
61    pub authorizing_integration_owners: AuthorizingIntegrationOwners,
62    /// The context where the interaction was triggered from.
63    pub context: Option<InteractionContext>,
64    /// Attachment size limit in bytes.
65    pub attachment_size_limit: u32,
66}

70    /// Gets the interaction response.
71    ///
72    /// # Errors
73    ///
74    /// Returns an [`Error::Http`] if there is no interaction response.
75    pub async fn get_response(&self, http: &Http) -> Result<Message> {
76        http.get_original_interaction_response(&self.token).await
77    }
78
79    /// Creates a response to the interaction received.
80    ///
81    /// **Note**: Message contents must be under 2000 unicode code points.
82    ///
83    /// # Errors
84    ///
85    /// Returns an [`Error::Model`] if the message content is too long. May also return an
86    /// [`Error::Http`] if the API returns an error, or an [`Error::Json`] if there is an error in
87    /// deserializing the API response.
88    pub async fn create_response(
89        &self,
90        http: &Http,
91        builder: CreateInteractionResponse<'_>,
92    ) -> Result<()> {
93        builder.execute(http, self.id, &self.token).await
94    }
95
96    /// Edits the initial interaction response.
97    ///
98    /// **Note**: Message contents must be under 2000 unicode code points.
99    ///
100    /// # Errors
101    ///
102    /// Returns an [`Error::Model`] if the message content is too long. May also return an
103    /// [`Error::Http`] if the API returns an error, or an [`Error::Json`] if there is an error in
104    /// deserializing the API response.
105    pub async fn edit_response(
106        &self,
107        http: &Http,
108        builder: EditInteractionResponse<'_>,
109    ) -> Result<Message> {
110        builder.execute(http, &self.token).await
111    }
112
113    /// Deletes the initial interaction response.
114    ///
115    /// Does not work on ephemeral messages.
116    ///
117    /// # Errors
118    ///
119    /// May return [`Error::Http`] if the API returns an error. Such as if the response was already
120    /// deleted.
121    pub async fn delete_response(&self, http: &Http) -> Result<()> {
122        http.delete_original_interaction_response(&self.token).await
123    }
124
125    /// Creates a followup response to the response sent.
126    ///
127    /// **Note**: Message contents must be under 2000 unicode code points.
128    ///
129    /// # Errors
130    ///
131    /// Returns [`Error::Model`] if the content is too long. May also return [`Error::Http`] if the
132    /// API returns an error, or [`Error::Json`] if there is an error in deserializing the
133    /// response.
134    pub async fn create_followup(
135        &self,
136        http: &Http,
137        builder: CreateInteractionResponseFollowup<'_>,
138    ) -> Result<Message> {
139        builder.execute(http, None, &self.token).await
140    }
141
142    /// Edits a followup response to the response sent.
143    ///
144    /// **Note**: Message contents must be under 2000 unicode code points.
145    ///
146    /// # Errors
147    ///
148    /// Returns [`Error::Model`] if the content is too long. May also return [`Error::Http`] if the
149    /// API returns an error, or [`Error::Json`] if there is an error in deserializing the
150    /// response.
151    pub async fn edit_followup(
152        &self,
153        http: &Http,
154        message_id: MessageId,
155        builder: CreateInteractionResponseFollowup<'_>,
156    ) -> Result<Message> {
157        builder.execute(http, Some(message_id), &self.token).await
158    }
159
160    /// Deletes a followup message.
161    ///
162    /// # Errors
163    ///
164    /// May return [`Error::Http`] if the API returns an error. Such as if the response was already
165    /// deleted.
166    pub async fn delete_followup(&self, http: &Http, message_id: MessageId) -> Result<()> {
167        http.delete_followup_message(&self.token, message_id).await
168    }
169
170    /// Gets a followup message.
171    ///
172    /// # Errors
173    ///
174    /// May return [`Error::Http`] if the API returns an error. Such as if the response was
175    /// deleted.
176    pub async fn get_followup(&self, http: &Http, message_id: MessageId) -> Result<Message> {
177        http.get_followup_message(&self.token, message_id).await
178    }
179
180    /// Helper function to defer an interaction.
181    ///
182    /// # Errors
183    ///
184    /// Returns an [`Error::Http`] if the API returns an error, or an [`Error::Json`] if there is
185    /// an error in deserializing the API response.
186    pub async fn defer(&self, http: &Http) -> Result<()> {
187        self.create_response(http, CreateInteractionResponse::Acknowledge).await
188    }
189
190    /// Helper function to defer an interaction ephemerally
191    ///
192    /// # Errors
193    ///
194    /// May also return an [`Error::Http`] if the API returns an error, or an [`Error::Json`] if
195    /// there is an error in deserializing the API response.
196    pub async fn defer_ephemeral(&self, http: &Http) -> Result<()> {
197        let builder = CreateInteractionResponse::Defer(
198            CreateInteractionResponseMessage::new().ephemeral(true),
199        );
200        self.create_response(http, builder).await
201    }
202}

205impl<'de> Deserialize<'de> for ComponentInteraction {
206    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> StdResult<Self, D::Error> {
207        // calls #[serde(remote)]-generated inherent method
208        let mut interaction = Self::deserialize(deserializer)?;
209        if let (Some(guild_id), Some(member)) = (interaction.guild_id, &mut interaction.member) {
210            member.guild_id = guild_id;
211            // If `member` is present, `user` wasn't sent and is still filled with default data
212            interaction.user = member.user.clone();
213        }
214        Ok(interaction)
215    }
216}

226#[derive(Clone, Debug)]
227pub enum ComponentInteractionDataKind {
228    Button,
229    StringSelect { values: FixedArray<String> },
230    UserSelect { values: FixedArray<UserId> },
231    RoleSelect { values: FixedArray<RoleId> },
232    MentionableSelect { values: FixedArray<GenericId> },
233    ChannelSelect { values: FixedArray<ChannelId> },
234    Unknown(u8),
235}
236
237// Manual impl needed to emulate integer enum tags
238impl<'de> Deserialize<'de> for ComponentInteractionDataKind {
239    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> StdResult<Self, D::Error> {
240        #[derive(Deserialize)]
241        struct Json {
242            component_type: ComponentType,
243            values: Option<Value>,
244        }
245        let json = Json::deserialize(deserializer)?;
246
247        macro_rules! parse_values {
248            () => {
249                from_value(json.values.ok_or_else(|| D::Error::missing_field("values"))?)
250                    .map_err(D::Error::custom)?
251            };
252        }
253
254        Ok(match json.component_type {
255            ComponentType::Button => Self::Button,
256            ComponentType::StringSelect => Self::StringSelect {
257                values: parse_values!(),
258            },
259            ComponentType::UserSelect => Self::UserSelect {
260                values: parse_values!(),
261            },
262            ComponentType::RoleSelect => Self::RoleSelect {
263                values: parse_values!(),
264            },
265            ComponentType::MentionableSelect => Self::MentionableSelect {
266                values: parse_values!(),
267            },
268            ComponentType::ChannelSelect => Self::ChannelSelect {
269                values: parse_values!(),
270            },
271            x @ (ComponentType::ActionRow | ComponentType::InputText) => {
272                return Err(D::Error::custom(format_args!(
273                    "invalid message component type in this context: {x:?}",
274                )));
275            },
276            ComponentType(x) => Self::Unknown(x),
277        })
278    }
279}

281impl Serialize for ComponentInteractionDataKind {
282    #[rustfmt::skip] // Remove this for horror.
283    fn serialize<S: serde::Serializer>(&self, serializer: S) -> StdResult<S::Ok, S::Error> {
284        let mut map = serializer.serialize_map(Some(2))?;
285        map.serialize_entry("component_type", &match self {
286            Self::Button { .. } => 2,
287            Self::StringSelect { .. } => 3,
288            Self::UserSelect { .. } => 5,
289            Self::RoleSelect { .. } => 6,
290            Self::MentionableSelect { .. } => 7,
291            Self::ChannelSelect { .. } => 8,
292            Self::Unknown(x) => *x,
293        })?;
294
295        match self {
296            Self::StringSelect { values } => map.serialize_entry("values", values)?,
297            Self::UserSelect { values } => map.serialize_entry("values", values)?,
298            Self::RoleSelect { values } => map.serialize_entry("values", values)?,
299            Self::MentionableSelect { values } => map.serialize_entry("values", values)?,
300            Self::ChannelSelect { values } => map.serialize_entry("values", values)?,
301            Self::Button | Self::Unknown(_) => map.serialize_entry("values", &None::<()>)?,
302        }
303
304        map.end()
305    }
