16/// An interaction triggered by a modal submit.
17///
18/// [Discord docs](https://docs.discord.com/developers/interactions/receiving-and-responding#interaction-object).
19#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
20#[derive(Clone, Debug, Deserialize, Serialize)]
21#[serde(remote = "Self")]
22#[non_exhaustive]
23pub struct ModalInteraction {
24    /// Id of the interaction.
25    pub id: InteractionId,
26    /// Id of the application this interaction is for.
27    pub application_id: ApplicationId,
28    /// The data of the interaction which was triggered.
29    pub data: ModalInteractionData,
30    /// The guild Id this interaction was sent from, if there is one.
31    #[serde(skip_serializing_if = "Option::is_none")]
32    pub guild_id: Option<GuildId>,
33    /// Channel that the interaction was sent from.
34    pub channel: Option<GenericInteractionChannel>,
35    /// The channel Id this interaction was sent from.
36    pub channel_id: GenericChannelId,
37    /// The `member` data for the invoking user.
38    ///
39    /// **Note**: It is only present if the interaction is triggered in a guild.
40    #[serde(skip_serializing_if = "Option::is_none")]
41    pub member: Option<Member>,
42    /// The `user` object for the invoking user.
43    #[serde(default)]
44    pub user: User,
45    /// A continuation token for responding to the interaction.
46    pub token: FixedString,
47    /// Always `1`.
48    pub version: u8,
49    /// The message this interaction was triggered by
50    ///
51    /// **Note**: Does not exist if the modal interaction originates from an application command
52    /// interaction
53    #[serde(skip_serializing_if = "Option::is_none")]
54    pub message: Option<Box<Message>>,
55    /// Permissions the app or bot has within the channel the interaction was sent from.
56    pub app_permissions: Permissions,
57    /// The selected language of the invoking user.
58    pub locale: FixedString,
59    /// The guild's preferred locale.
60    pub guild_locale: Option<FixedString>,
61    /// For monetized applications, any entitlements of the invoking user.
62    pub entitlements: FixedArray<Entitlement>,
63    /// Attachment size limit in bytes.
64    pub attachment_size_limit: u32,
65}

68impl ModalInteraction {
69    /// Gets the interaction response.
70    ///
71    /// # Errors
72    ///
73    /// Returns an [`Error::Http`] if there is no interaction response.
74    pub async fn get_response(&self, http: &Http) -> Result<Message> {
75        http.get_original_interaction_response(&self.token).await
76    }
77
78    /// Creates a response to the interaction received.
79    ///
80    /// **Note**: Message contents must be under 2000 unicode code points.
81    ///
82    /// # Errors
83    ///
84    /// Returns an [`Error::Model`] if the message content is too long. May also return an
85    /// [`Error::Http`] if the API returns an error, or an [`Error::Json`] if there is an error in
86    /// deserializing the API response.
87    pub async fn create_response(
88        &self,
89        http: &Http,
90        builder: CreateInteractionResponse<'_>,
91    ) -> Result<()> {
92        builder.execute(http, self.id, &self.token).await
93    }
94
95    /// Edits the initial interaction response.
96    ///
97    /// **Note**: Message contents must be under 2000 unicode code points.
98    ///
99    /// # Errors
100    ///
101    /// Returns an [`Error::Model`] if the message content is too long. May also return an
102    /// [`Error::Http`] if the API returns an error, or an [`Error::Json`] if there is an error in
103    /// deserializing the API response.
104    pub async fn edit_response(
105        &self,
106        http: &Http,
107        builder: EditInteractionResponse<'_>,
108    ) -> Result<Message> {
109        builder.execute(http, &self.token).await
110    }
111
112    /// Deletes the initial interaction response.
113    ///
114    /// Does not work on ephemeral messages.
115    ///
116    /// # Errors
117    ///
118    /// May return [`Error::Http`] if the API returns an error. Such as if the response was already
119    /// deleted.
120    pub async fn delete_response(&self, http: &Http) -> Result<()> {
121        http.delete_original_interaction_response(&self.token).await
122    }
123
124    /// Creates a followup response to the response sent.
125    ///
126    /// **Note**: Message contents must be under 2000 unicode code points.
127    ///
128    /// # Errors
129    ///
130    /// Returns [`Error::Model`] if the content is too long. May also return [`Error::Http`] if the
131    /// API returns an error, or [`Error::Json`] if there is an error in deserializing the
132    /// response.
133    pub async fn create_followup(
134        &self,
135        http: &Http,
136        builder: CreateInteractionResponseFollowup<'_>,
137    ) -> Result<Message> {
138        builder.execute(http, None, &self.token).await
139    }
140
141    /// Edits a followup response to the response sent.
142    ///
143    /// **Note**: Message contents must be under 2000 unicode code points.
144    ///
145    /// # Errors
146    ///
147    /// Returns [`Error::Model`] if the content is too long. May also return [`Error::Http`] if the
148    /// API returns an error, or [`Error::Json`] if there is an error in deserializing the
149    /// response.
150    pub async fn edit_followup(
151        &self,
152        http: &Http,
153        message_id: MessageId,
154        builder: CreateInteractionResponseFollowup<'_>,
155    ) -> Result<Message> {
156        builder.execute(http, Some(message_id), &self.token).await
157    }
158
159    /// Deletes a followup message.
160    ///
161    /// # Errors
162    ///
163    /// May return [`Error::Http`] if the API returns an error. Such as if the response was already
164    /// deleted.
165    pub async fn delete_followup(&self, http: &Http, message_id: MessageId) -> Result<()> {
166        http.delete_followup_message(&self.token, message_id).await
167    }
168
169    /// Helper function to defer an interaction.
170    ///
171    /// # Errors
172    ///
173    /// Returns an [`Error::Http`] if the API returns an error, or an [`Error::Json`] if there is
174    /// an error in deserializing the API response.
175    pub async fn defer(&self, http: &Http) -> Result<()> {
176        self.create_response(http, CreateInteractionResponse::Acknowledge).await
177    }
178
179    /// Helper function to defer an interaction ephemerally
180    ///
181    /// # Errors
182    ///
183    /// May also return an [`Error::Http`] if the API returns an error, or an [`Error::Json`] if
184    /// there is an error in deserializing the API response.
185    pub async fn defer_ephemeral(&self, http: &Http) -> Result<()> {
186        let builder = CreateInteractionResponse::Defer(
187            CreateInteractionResponseMessage::new().ephemeral(true),
188        );
189        self.create_response(http, builder).await
190    }
191}

194impl<'de> Deserialize<'de> for ModalInteraction {
195    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> StdResult<Self, D::Error> {
196        let mut interaction = Self::deserialize(deserializer)?; // calls #[serde(remote)]-generated inherent method
197        if let (Some(guild_id), Some(member)) = (interaction.guild_id, &mut interaction.member) {
198            member.guild_id = guild_id;
199            // If `member` is present, `user` wasn't sent and is still filled with default data
200            interaction.user = member.user.clone();
201        }
202        Ok(interaction)
203    }
204}

228/// A component which can appear inside a modal form submission.
229///
230/// [Discord docs](https://docs.discord.com/developers/interactions/receiving-and-responding#interaction-object-component-interaction-response-structures)
231#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
232#[derive(Clone, Debug, Serialize)]
233#[non_exhaustive]
234pub enum ModalComponent {
235    TextDisplay(TextDisplay),
236    Label(Label),
237}
238
239impl<'de> Deserialize<'de> for ModalComponent {
240    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
241        #[derive(Deserialize)]
242        struct ModalComponentRaw {
243            #[serde(rename = "type")]
244            kind: ComponentType,
245        }
246
247        let raw_data = <&RawValue>::deserialize(deserializer)?;
248        let raw = ModalComponentRaw::deserialize(raw_data).map_err(DeError::custom)?;
249
250        match raw.kind {
251            ComponentType::TextDisplay => Deserialize::deserialize(raw_data).map(Self::TextDisplay),
252            ComponentType::Label => Deserialize::deserialize(raw_data).map(Self::Label),
253            ComponentType(i) => {
254                return Err(DeError::custom(format_args!("Unknown modal component type {i}")));
255            },
256        }
257        .map_err(DeError::custom)
258    }
