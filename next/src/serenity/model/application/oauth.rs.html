1use std::fmt;
2
3use serde::{Deserialize, Serialize};

5/// The available OAuth2 Scopes.
6///
7/// [Discord docs](https://docs.discord.com/developers/topics/oauth2#shared-resources-oauth2-scopes).
8#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd, Deserialize, Serialize)]
9#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
10#[non_exhaustive]
11pub enum Scope {
12    /// For oauth2 bots, this puts the bot in the user's selected guild by default.
13    #[serde(rename = "bot")]
14    Bot,
15    /// Allows your app to use Slash Commands in a guild.
16    #[serde(rename = "applications.commands")]
17    ApplicationsCommands,
18    /// Allows your app to update its Slash Commands via this bearer token - client credentials
19    /// grant only.
20    #[serde(rename = "applications.commands.update")]
21    ApplicationsCommandsUpdate,
22    /// Allows your app to update permissions for its commands in a guild a user has permissions
23    /// to.
24    #[serde(rename = "applications.commands.permissions.update")]
25    ApplicationsCommandsPermissionsUpdate,
26    /// Allows `/users/@me` without [`Self::Email`].
27    #[serde(rename = "identify")]
28    Identify,
29    /// Allows your app to read a user’s Nitro subscription type as defined by
30    /// [`User::premium_type`] - only available to approved partners.
31    ///
32    /// [`User::premium_type`]: crate::model::user::User::premium_type
33    #[serde(rename = "identify.premium")]
34    IdentifyPremium,
35    /// Enables `/users/@me` to return an `email` field.
36    #[serde(rename = "email")]
37    Email,
38    /// Allows `/users/@me/connections` to return linked third-party accounts.
39    #[serde(rename = "connections")]
40    Connections,
41    /// Allows `/users/@me/guilds` to return basic information about all of a user's guilds.
42    #[serde(rename = "guilds")]
43    Guilds,
44    /// Allows `/guilds/{guild.id}/members/{user.id}` to be used for joining users to a guild.
45    #[serde(rename = "guilds.join")]
46    GuildsJoin,
47    /// Allows `/users/@me/guilds/{guild.id}/member` to return a user's member information in a
48    /// guild.
49    #[serde(rename = "guilds.members.read")]
50    GuildsMembersRead,
51    /// Allows your app to join users to a group dm.
52    #[serde(rename = "gdm.join")]
53    GdmJoin,
54    /// For local rpc server access, this allows you to control a user's local Discord client -
55    /// requires Discord approval.
56    #[serde(rename = "rpc")]
57    Rpc,
58    /// For local rpc server api access, this allows you to receive notifications pushed out to the
59    /// user - requires Discord approval.
60    #[serde(rename = "rpc.notifications.read")]
61    RpcNotificationsRead,
62    #[serde(rename = "rpc.voice.read")]
63    RpcVoiceRead,
64    #[serde(rename = "rpc.voice.write")]
65    RpcVoiceWrite,
66    #[serde(rename = "rpc.activities.write")]
67    RpcActivitiesWrite,
68    /// This generates a webhook that is returned in the oauth token response for authorization
69    /// code grants.
70    #[serde(rename = "webhook.incoming")]
71    WebhookIncoming,
72    /// For local rpc server api access, this allows you to read messages from all client channels
73    /// (otherwise restricted to channels/guilds your app creates).
74    #[serde(rename = "messages.read")]
75    MessagesRead,
76    /// Allows your app to upload/update builds for a user's applications - requires Discord
77    /// approval.
78    #[serde(rename = "applications.builds.upload")]
79    ApplicationsBuildsUpload,
80    /// Allows your app to read build data for a user's applications.
81    #[serde(rename = "applications.builds.read")]
82    ApplicationsBuildsRead,
83    /// Allows your app to read and update store data (SKUs, store listings, achievements, etc.)
84    /// for a user's applications.
85    #[serde(rename = "applications.store.update")]
86    ApplicationsStoreUpdate,
87    /// Allows your app to read entitlements for a user's applications.
88    #[serde(rename = "applications.entitlements")]
89    ApplicationsEntitlements,
90    /// Allows your app to fetch data from a user's "Now Playing/Recently Played" list - requires
91    /// Discord approval.
92    #[serde(rename = "activities.read")]
93    ActivitiesRead,
94    /// Allows your app to update a user's activity - requires Discord approval (Not required for
95    /// gamesdk activity manager!).
96    #[serde(rename = "activities.write")]
97    ActivitiesWrite,
98    /// Allows your app to know a user's friends and implicit relationships - requires Discord
99    /// approval.
100    #[serde(rename = "relationships.read")]
101    RelationshipsRead,
102    /// Allows your app to see information about the user's DMs and group DMs - requires Discord
103    /// approval.
104    #[serde(rename = "dm_channels.read")]
105    DmChannelsRead,
106    /// Allows your app to connect to voice on user's behalf and see all the voice members -
107    /// requires Discord approval.
108    #[serde(rename = "voice")]
109    Voice,
110    /// Allows your app to update a user's connection and metadata for the app.
111    #[serde(rename = "role_connections.write")]
112    RoleConnectionsWrite,
113}
114
115impl fmt::Display for Scope {
116    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
117        self.serialize(f)
118    }
