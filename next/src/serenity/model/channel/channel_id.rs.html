46    /// Fetches a channel from the cache, falling back to HTTP/temp cache.
47    ///
48    /// It is highly recommended to pass the `guild_id` parameter as otherwise this may perform many
49    /// HTTP requests.
50    ///
51    /// # Errors
52    ///
53    /// Errors if the HTTP fallback fails, or if the channel does not come from the guild passed.
54    pub async fn to_guild_channel(
55        self,
56        cache_http: impl CacheHttp,
57        guild_id: Option<GuildId>,
58    ) -> Result<GuildChannel> {
59        #[cfg(feature = "cache")]
60        if let Some(cache) = cache_http.cache() {
61            if let Some(guild_id) = guild_id
62                && let Some(guild) = cache.guild(guild_id)
63                && let Some(channel) = guild.channels.get(&self)
64            {
65                return Ok(channel.clone());
66            }
67
68            #[cfg(feature = "temp_cache")]
69            if let Some(temp_channel) = cache.temp_channels.get(&self) {
70                if guild_id.is_some_and(|id| temp_channel.base.guild_id != id) {
71                    return Err(Error::Model(ModelError::ChannelNotFound));
72                }
73
74                return Ok(GuildChannel::clone(&temp_channel));
75            }
76        }
77
78        let channel = cache_http.http().get_channel(self.widen()).await?;
79        let guild_channel = channel.guild().ok_or(ModelError::InvalidChannelType)?;
80
81        #[cfg(all(feature = "cache", feature = "temp_cache"))]
82        if let Some(cache) = cache_http.cache() {
83            let cached_channel = MaybeOwnedArc::new(guild_channel.clone());
84            cache.temp_channels.insert(self, cached_channel);
85        }
86
87        if guild_id.is_some_and(|id| guild_channel.base.guild_id != id) {
88            return Err(Error::Model(ModelError::ChannelNotFound));
89        }
90
91        Ok(guild_channel)
92    }

516    /// Deletes messages by Ids from the given vector in the given channel.
517    ///
518    /// The Discord API supports deleting between 2 and 100 messages at once. This function
519    /// also handles the case of a single message by calling `delete_message` internally.
520    ///
521    /// Requires the [Manage Messages] permission.
522    ///
523    /// **Note**: Messages that are older than 2 weeks can't be deleted using this method.
524    ///
525    /// # Errors
526    ///
527    /// Returns [`ModelError::TooSmall`] or [`ModelError::TooLarge`] if an attempt was made to
528    /// delete either 0 or more than 100 messages.
529    ///
530    /// Also will return [`Error::Http`] if the current user lacks permission to delete messages.
531    ///
532    /// [Manage Messages]: Permissions::MANAGE_MESSAGES
533    pub async fn delete_messages(
534        self,
535        http: &Http,
536        message_ids: &[MessageId],
537        reason: Option<&str>,
538    ) -> Result<()> {
539        use crate::model::error::{Maximum, Minimum};
540
541        #[derive(serde::Serialize)]
542        struct DeleteMessages<'a> {
543            messages: &'a [MessageId],
544        }
545
546        Minimum::BulkDeleteAmount.check_underflow(message_ids.len())?;
547        Maximum::BulkDeleteAmount.check_overflow(message_ids.len())?;
548
549        if message_ids.len() == 1 {
550            self.delete_message(http, message_ids[0], reason).await
551        } else {
552            let req = DeleteMessages {
553                messages: message_ids,
554            };
555
556            http.delete_messages(self, &req, reason).await
557        }
558    }

642    /// Attempts to retrieve the channel from the guild cache, otherwise from HTTP/temp cache.
643    ///
644    /// # Errors
645    ///
646    /// Returns [`Error::Http`] if the channel retrieval request failed.
647    #[cfg_attr(not(feature = "cache"), expect(unused_variables))]
648    pub async fn to_channel(
649        self,
650        cache_http: impl CacheHttp,
651        guild_id: Option<GuildId>,
652    ) -> Result<Channel> {
653        #[cfg_attr(not(feature = "temp_cache"), expect(unused_variables))]
654        let (channel_id, thread_id) = self.split();
655
656        #[cfg(feature = "cache")]
657        if let Some(cache) = cache_http.cache() {
658            match guild_id.and_then(|id| cache.guild(id)).as_ref().and_then(|g| g.channel(self)) {
659                Some(GenericGuildChannelRef::Channel(chan)) => {
660                    return Ok(Channel::Guild(chan.clone()));
661                },
662                Some(GenericGuildChannelRef::Thread(th)) => {
663                    return Ok(Channel::GuildThread(th.clone()));
664                },
665                None => {},
666            }
667
668            #[cfg(feature = "temp_cache")]
669            {
670                if let Some(channel) = cache.temp_channels.get(&channel_id) {
671                    return Ok(Channel::Guild(GuildChannel::clone(&*channel)));
672                }
673
674                if let Some(thread) = cache.temp_threads.get(&thread_id) {
675                    return Ok(Channel::GuildThread(GuildThread::clone(&*thread)));
676                }
677            }
678        }
679
680        let channel = cache_http.http().get_channel(self).await?;
681
682        #[cfg(all(feature = "cache", feature = "temp_cache"))]
683        if let Some(cache) = cache_http.cache() {
684            match &channel {
685                Channel::Guild(guild_channel) => {
686                    let cached_channel = MaybeOwnedArc::new(guild_channel.clone());
687                    cache.temp_channels.insert(channel_id, cached_channel);
688                },
689                Channel::GuildThread(guild_thread) => {
690                    let cached_thread = MaybeOwnedArc::new(guild_thread.clone());
691                    cache.temp_threads.insert(thread_id, cached_thread);
692                },
693                // No access to `UserId`, so this can't cache.

1122/// A helper class returned by [`GenericChannelId::messages_iter`]
1123#[derive(Clone, Debug)]
1124#[cfg(feature = "model")]
1125pub struct MessagesIter<'a> {
1126    http: &'a Http,
1127    #[cfg(feature = "cache")]
1128    cache: Option<&'a Arc<Cache>>,
1129    channel_id: GenericChannelId,
1130    buffer: Vec<Message>,
1131    before: Option<MessageId>,
1132    tried_fetch: bool,
1133}
1134
1135#[cfg(feature = "model")]
1136impl<'a> MessagesIter<'a> {
1137    fn new(cache_http: &'a impl CacheHttp, channel_id: GenericChannelId) -> MessagesIter<'a> {
1138        MessagesIter {
1139            http: cache_http.http(),
1140            #[cfg(feature = "cache")]
1141            cache: cache_http.cache(),
1142            channel_id,
1143            buffer: Vec::new(),
1144            before: None,
1145            tried_fetch: false,
1146        }
1147    }
1148
1149    #[cfg(not(feature = "cache"))]
1150    fn cache_http(&self) -> impl CacheHttp + '_ {
1151        self.http
1152    }
1153
1154    #[cfg(feature = "cache")]
1155    fn cache_http(&self) -> impl CacheHttp + '_ {
1156        (self.cache, self.http)
1157    }
1158
1159    /// Fills the `self.buffer` cache with [`Message`]s.
1160    ///
1161    /// This drops any messages that were currently in the buffer. Ideally, it should only be
1162    /// called when `self.buffer` is empty. Additionally, this updates `self.before` so that the
1163    /// next call does not return duplicate items.
1164    ///
1165    /// If there are no more messages to be fetched, then this sets `self.before` as [`None`],
1166    /// indicating that no more calls ought to be made.
1167    ///
1168    /// If this method is called with `self.before` as None, the last 100 (or lower) messages sent
1169    /// in the channel are added in the buffer.
1170    ///
1171    /// The messages are sorted such that the newest message is the first element of the buffer and
1172    /// the newest message is the last.
1173    ///
1174    /// [`Message`]: crate::model::channel::Message
1175    async fn refresh(&mut self) -> Result<()> {
1176        // Number of messages to fetch.
1177        let grab_size = 100;
1178
1179        // If `self.before` is not set yet, we can use `.messages` to fetch the last message after
1180        // very first fetch from last.
1181        let mut builder = GetMessages::new().limit(grab_size);
1182        if let Some(before) = self.before {
1183            builder = builder.before(before);
1184        }
1185        self.buffer = self.channel_id.messages(self.cache_http(), builder).await?;
1186
1187        self.buffer.reverse();
1188
1189        self.before = self.buffer.first().map(|m| m.id);
1190
1191        self.tried_fetch = true;
1192
1193        Ok(())
1194    }

1196    /// Streams over all the messages in a channel.
1197    ///
1198    /// This is accomplished and equivalent to repeated calls to [`GenericChannelId::messages`]. A
1199    /// buffer of at most 100 messages is used to reduce the number of calls necessary.
1200    ///
1201    /// The stream returns the newest message first, followed by older messages.
1202    ///
1203    /// # Examples
1204    ///
1205    /// ```rust,no_run
1206    /// # use serenity::model::id::GenericChannelId;
1207    /// # use serenity::http::Http;
1208    /// #
1209    /// # async fn run() {
1210    /// # let channel_id = GenericChannelId::new(1);
1211    /// # let http: Http = unimplemented!();
1212    /// use serenity::futures::StreamExt;
1213    /// use serenity::model::channel::MessagesIter;
1214    ///
1215    /// let mut messages = MessagesIter::stream(&http, channel_id).boxed();
1216    /// while let Some(message_result) = messages.next().await {
1217    ///     match message_result {
1218    ///         Ok(message) => println!("{} said \"{}\"", message.author.name, message.content,),
1219    ///         Err(error) => eprintln!("Uh oh! Error: {}", error),
1220    ///     }
1221    /// }
1222    /// # }
1223    /// ```
1224    pub fn stream(
1225        cache_http: &'a impl CacheHttp,
1226        channel_id: GenericChannelId,
1227    ) -> impl Stream<Item = Result<Message>> + 'a {
1228        let init_state = MessagesIter::new(cache_http, channel_id);
1229
1230        futures::stream::unfold(init_state, |mut state| async {
1231            if (state.buffer.is_empty() && state.before.is_some() || !state.tried_fetch)
1232                && let Err(error) = state.refresh().await
1233            {
1234                return Some((Err(error), state));
1235            }
1236
1237            // the resultant stream goes from newest to oldest.
1238            state.buffer.pop().map(|entry| (Ok(entry), state))
1239        })
1240    }
