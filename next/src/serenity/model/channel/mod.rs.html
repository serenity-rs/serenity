47impl GenericChannelId {
48    /// Copies this ID into a [`ChannelId`] and a [`ThreadId`].
49    ///
50    /// It is only correct to use this when you use both returned values,
51    /// otherwise use [`Self::expect_channel`] or [`Self::expect_thread`].
52    #[must_use]
53    pub fn split(self) -> (ChannelId, ThreadId) {
54        (self.expect_channel(), self.expect_thread())
55    }
56
57    /// Converts the type of this Id to [`ChannelId`].
58    ///
59    /// This converts the type without changing the inner value, therefore,
60    /// is only correct when you have knowledge which is not in the type system.
61    ///
62    /// This should be used as rarely as [`Option::expect`].
63    #[must_use]
64    pub fn expect_channel(self) -> ChannelId {
65        ChannelId::new(self.get())
66    }
67
68    /// Converts the type of this Id to [`ThreadId`].
69    ///
70    /// This converts the type without changing the inner value, therefore,
71    /// is only correct when you have knowledge which is not in the type system.
72    ///
73    /// This should be used as rarely as [`Option::expect`].
74    #[must_use]
75    pub fn expect_thread(self) -> ThreadId {
76        ThreadId::new(self.get())
77    }

109/// A container for any channel.
110#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
111#[derive(Clone, Debug, Serialize)]
112#[serde(untagged)]
113#[non_exhaustive]
114pub enum Channel {
115    /// A channel within a [`Guild`].
116    Guild(GuildChannel),
117    /// A thread inside a [`Guild`].
118    GuildThread(GuildThread),
119    /// A private channel to another [`User`] (Direct Message). No other users may access the
120    /// channel.
121    Private(PrivateChannel),
122}

124#[cfg(feature = "model")]
125impl Channel {
126    /// If this is a guild channel, returns it.
127    #[must_use]
128    pub fn guild(self) -> Option<GuildChannel> {
129        match self {
130            Self::Guild(channel) => Some(channel),
131            _ => None,
132        }
133    }
134
135    /// If this is a guild thread, returns it.
136    #[must_use]
137    pub fn thread(self) -> Option<GuildThread> {
138        match self {
139            Self::GuildThread(thread) => Some(thread),
140            _ => None,
141        }
142    }
143
144    /// If this is a private channel, returns it.
145    #[must_use]
146    pub fn private(self) -> Option<PrivateChannel> {
147        match self {
148            Self::Private(channel) => Some(channel),
149            _ => None,
150        }
151    }
152
153    /// If this is a category channel, returns it.
154    #[must_use]
155    pub fn category(self) -> Option<GuildChannel> {
156        match self {
157            Self::Guild(c) if c.base.kind == ChannelType::Category => Some(c),
158            _ => None,
159        }
160    }
161
162    /// Deletes the inner channel.
163    ///
164    /// # Errors
165    ///
166    /// Returns [`Error::Http`] if the current user lacks permission.
167    pub async fn delete(&self, http: &Http, reason: Option<&str>) -> Result<()> {
168        match self {
169            Self::Guild(public_channel) => {
170                public_channel.delete(http, reason).await?;
171            },
172            Self::GuildThread(thread) => {
173                thread.delete(http, reason).await?;
174            },
175            Self::Private(private_channel) => {
176                private_channel.delete(http).await?;
177            },
178        }
179
180        Ok(())
181    }
182
183    /// Retrieves the inner Id.
184    #[must_use]
185    pub fn id(&self) -> GenericChannelId {
186        match self {
187            Self::Guild(ch) => ch.id.widen(),
188            Self::GuildThread(ch) => ch.id.widen(),
189            Self::Private(ch) => ch.id.widen(),
190        }
191    }
192
193    /// If this is a guild channel or guild thread, returns the corresponding guild's Id.
194    #[must_use]
195    pub fn guild_id(&self) -> Option<GuildId> {
196        match self {
197            Channel::GuildThread(thread) => Some(thread.base.guild_id),
198            Channel::Guild(channel) => Some(channel.base.guild_id),
199            Channel::Private(_) => None,
200        }
201    }
202}

204fn extract_type<'de, D>(deserializer: D) -> StdResult<(u64, &'de RawValue), D::Error>
205where
206    D: Deserializer<'de>,
207{
208    #[derive(Deserialize)]
209    struct ChannelRaw {
210        #[serde(rename = "type")]
211        kind: u64,
212    }
213
214    let raw_data = <&RawValue>::deserialize(deserializer)?;
215    let raw = ChannelRaw::deserialize(raw_data).map_err(DeError::custom)?;
216
217    Ok((raw.kind, raw_data))
218}
219
220// Manual impl needed to emulate integer enum tags
221impl<'de> Deserialize<'de> for Channel {
222    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> StdResult<Self, D::Error> {
223        let (kind, raw_data) = extract_type(deserializer)?;
224
225        match kind {
226            0 | 2 | 4 | 5 | 13 | 14 | 15 => Deserialize::deserialize(raw_data).map(Channel::Guild),
227            10..=12 => Deserialize::deserialize(raw_data).map(Channel::GuildThread),
228            1 => Deserialize::deserialize(raw_data).map(Channel::Private),
229            _ => return Err(DeError::custom("Unknown channel type")),
230        }
231        .map_err(DeError::custom)
232    }
233}

235impl fmt::Display for Channel {
236    /// Formats the channel into a "mentioned" string.
237    ///
238    /// This will return a different format for each type of channel:
239    /// - [`PrivateChannel`]s: the recipient's name;
240    /// - [`GuildChannel`]s: a string mentioning the channel that users who can see the channel can
241    ///   click on.
242    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
243        match self {
244            Self::Guild(ch) => fmt::Display::fmt(&ch.id.widen().mention(), f),
245            Self::GuildThread(ch) => fmt::Display::fmt(&ch.id.widen().mention(), f),
246            Self::Private(ch) => fmt::Display::fmt(&ch.recipient.name, f),
247        }
248    }
249}

324impl std::fmt::Display for InvalidPermissionOverwriteType {
325    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
326        write!(f, "Invalid Permission Overwrite Type: {}", self.0)
327    }

330impl std::convert::TryFrom<PermissionOverwriteData> for PermissionOverwrite {
331    type Error = InvalidPermissionOverwriteType;
332
333    fn try_from(data: PermissionOverwriteData) -> StdResult<Self, Self::Error> {
334        let kind = match data.kind {
335            0 => PermissionOverwriteType::Role(data.id.get().into()),
336            1 => PermissionOverwriteType::Member(data.id.into()),
337            raw => return Err(InvalidPermissionOverwriteType(raw)),
338        };
339
340        Ok(PermissionOverwrite {
341            allow: data.allow,
342            deny: data.deny,
343            kind,
344        })
345    }
346}
347
348impl From<PermissionOverwrite> for PermissionOverwriteData {
349    fn from(data: PermissionOverwrite) -> Self {
350        let (kind, id) = match data.kind {
351            PermissionOverwriteType::Role(id) => (0, id.get().into()),
352            PermissionOverwriteType::Member(id) => (1, id.into()),
353        };
354
355        PermissionOverwriteData {
356            allow: data.allow,
357            deny: data.deny,
358            kind,
359            id,
360        }
361    }
362}

500#[derive(Deserialize)]
501struct RawForumEmoji {
502    emoji_id: Option<EmojiId>,
503    emoji_name: Option<FixedString>,
504}
505
506impl serde::Serialize for ForumEmoji {
507    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
508        let mut map = serializer.serialize_map(Some(2))?;
509        match self {
510            Self::Id(id) => {
511                map.serialize_entry("emoji_id", id)?;
512                map.serialize_entry("emoji_name", &None::<()>)?;
513            },
514            Self::Name(name) => {
515                map.serialize_entry("emoji_id", &None::<()>)?;
516                map.serialize_entry("emoji_name", name)?;
517            },
518        }
519
520        map.end()
521    }
522}
523
524impl<'de> serde::Deserialize<'de> for ForumEmoji {
525    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
526        let helper = RawForumEmoji::deserialize(deserializer)?;
527        match (helper.emoji_id, helper.emoji_name) {
528            (Some(id), None) => Ok(ForumEmoji::Id(id)),
529            (None, Some(name)) => Ok(ForumEmoji::Name(name)),
530            (None, None) => {
531                Err(serde::de::Error::custom("expected emoji_name or emoji_id, found neither"))
532            },
533            (Some(_), Some(_)) => {
534                Err(serde::de::Error::custom("expected emoji_name or emoji_id, found both"))
535            },
536        }
537    }
538}
