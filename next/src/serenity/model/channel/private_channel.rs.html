8/// A Direct Message text channel with another user.
9///
10/// [Discord docs](https://docs.discord.com/developers/resources/channel#channel-object).
11#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
12#[derive(Clone, Debug, Default, Deserialize, Serialize)]
13#[non_exhaustive]
14pub struct PrivateChannel {
15    /// The unique Id of the private channel.
16    ///
17    /// Can be used to calculate the first message's creation date.
18    pub id: ChannelId,
19    /// The Id of the last message sent.
20    pub last_message_id: Option<MessageId>,
21    /// Timestamp of the last time a [`Message`] was pinned.
22    pub last_pin_timestamp: Option<Timestamp>,
23    /// Indicator of the type of channel this is.
24    ///
25    /// This should always be [`ChannelType::Private`].
26    #[serde(rename = "type")]
27    pub kind: ChannelType,
28    /// The recipient to the private channel.
29    #[serde(with = "single_recipient", rename = "recipients")]
30    pub recipient: User,
31}
32
33#[cfg(feature = "model")]
34impl PrivateChannel {
35    /// Deletes the channel. This does not delete the contents of the channel, and is equivalent to
36    /// closing a private channel on the client, which can be re-opened.
37    #[expect(clippy::missing_errors_doc)]
38    pub async fn delete(&self, http: &Http) -> Result<PrivateChannel> {
39        let resp = self.id.widen().delete(http, None).await?;
40        resp.private().ok_or(Error::Model(ModelError::InvalidChannelType))
41    }
42}
43
44impl fmt::Display for PrivateChannel {
45    /// Formats the private channel, displaying the recipient's username.
46    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
47        f.write_str(&self.recipient.name)
48    }
49}
