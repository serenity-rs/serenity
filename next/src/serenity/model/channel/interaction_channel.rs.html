5/// Represents the shared fields between a [`InteractionChannel`] and a [`InteractionGuildThread`].
6#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
7#[derive(Clone, Debug, Deserialize, Serialize)]
8#[non_exhaustive]
9pub struct BaseInteractionChannel {
10    /// The channel type.
11    #[serde(rename = "type")]
12    pub kind: ChannelType,
13    /// The Id of the guild the channel is located in.
14    #[serde(default)]
15    pub guild_id: GuildId,
16    /// The channel name.
17    pub name: Option<FixedString>,
18    /// The Id of the last message sent in the channel (or thread for forum channels).
19    ///
20    /// **Note:** May not point to an existing or valid message or thread.
21    pub last_message_id: Option<MessageId>,
22    /// Amount of seconds a user has to wait before sending another message.
23    ///
24    /// Bots, as well as users with the [`BYPASS_SLOWMODE`] permission, are unaffected.
25    ///
26    /// [`BYPASS_SLOWMODE`]: Permissions::BYPASS_SLOWMODE
27    #[doc(alias = "slowmode")]
28    #[serde(default)]
29    pub rate_limit_per_user: Option<NonMaxU16>,
30    /// The timestamp of the time the last pinned message was pinned.
31    pub last_pin_timestamp: Option<Timestamp>,
32    /// Computed permissions for the invoking user in the channel, including overwrites.
33    pub permissions: Option<Permissions>,
34    /// Computed permissions for the bot user in the channel, including overwrites.
35    pub app_permissions: Option<Permissions>,
36    /// Extra information about the channel.
37    #[serde(default)]
38    pub flags: ChannelFlags,
39}

41/// Represents a partial channel from an interaction.
42///
43/// [Discord docs](https://docs.discord.com/developers/resources/channel#channel-object),
44/// [subset specification](https://docs.discord.com/developers/interactions/receiving-and-responding#interaction-object-resolved-data-structure).
45#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
46#[derive(Clone, Debug, Deserialize, Serialize)]
47#[non_exhaustive]
48pub struct InteractionChannel {
49    /// The shared fields between [`InteractionChannel`] and [`InteractionGuildThread`].
50    #[serde(flatten)]
51    pub base: BaseInteractionChannel,
52    /// The channel Id.
53    pub id: ChannelId,
54    /// Sorting position of the channel. Channels with the same position are sorted by Id.
55    ///
56    /// The default text channel will _almost always_ have a position of `0`.
57    #[serde(default)]
58    pub position: u16,
59    /// The topic of the channel.
60    pub topic: Option<FixedString<u16>>,
61    /// Whether the channel is age-restricted.
62    #[serde(default)]
63    pub nsfw: bool,
64    /// The Id of the parent category.
65    ///
66    /// **Note**: This is only available for channels in a category.
67    pub parent_id: Option<ChannelId>,
68}

91#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
92#[derive(Clone, Debug, Serialize)]
93#[serde(untagged)]
94pub enum GenericInteractionChannel {
95    Channel(InteractionChannel),
96    Thread(InteractionGuildThread),
97}
98
99impl GenericInteractionChannel {
100    #[must_use]
101    pub fn id(&self) -> GenericChannelId {
102        match self {
103            Self::Channel(ch) => ch.id.widen(),
104            Self::Thread(th) => th.id.widen(),
105        }
106    }
107
108    #[must_use]
109    pub fn base(&self) -> &BaseInteractionChannel {
110        match self {
111            Self::Channel(ch) => &ch.base,
112            Self::Thread(th) => &th.base,
113        }
114    }
115}
116
117impl ExtractKey<GenericChannelId> for GenericInteractionChannel {
118    fn extract_key(&self) -> &GenericChannelId {
119        match self {
120            Self::Channel(channel) => GenericChannelId::cast_from(&channel.id.0),
121            Self::Thread(thread) => GenericChannelId::cast_from(&thread.id.0),
122        }
123    }
124}

126impl<'de> serde::Deserialize<'de> for GenericInteractionChannel {
127    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
128        let (kind, data) = super::extract_type(deserializer)?;
129
130        match kind {
131            10..=12 => Deserialize::deserialize(data).map(Self::Thread),
132            _ => Deserialize::deserialize(data).map(Self::Channel),
133        }
134        .map_err(serde::de::Error::custom)
135    }
