5/// Represents a rich embed which allows using richer markdown, multiple fields and more. This was
6/// heavily inspired by [slack's attachments].
7///
8/// You can include an attachment in your own message by a user or a bot, or in a webhook.
9///
10/// **Note**: Maximum amount of characters you can put is 256 in a field name,
11/// 1024 in a field value, and 4096 in a description.
12///
13/// [Discord docs](https://docs.discord.com/developers/resources/message#embed-object).
14///
15/// [slack's attachments]: https://api.slack.com/docs/message-attachments
16#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
17#[derive(Clone, Default, Debug, Deserialize, Serialize)]
18#[non_exhaustive]
19pub struct Embed {
20    /// The title of the embed.
21    #[serde(skip_serializing_if = "Option::is_none")]
22    pub title: Option<FixedString<u16>>,
23    /// The type of the embed. For embeds not generated by Discord's backend, this will always be
24    /// "rich".
25    #[serde(rename = "type")]
26    #[serde(skip_serializing_if = "Option::is_none")]
27    pub kind: Option<FixedString<u8>>,
28    /// The description of the embed.
29    ///
30    /// The maximum value for this field is 2048 unicode codepoints.
31    #[serde(skip_serializing_if = "Option::is_none")]
32    pub description: Option<FixedString<u16>>,
33    /// The URL of the embed.
34    #[serde(skip_serializing_if = "Option::is_none")]
35    pub url: Option<FixedString>,
36    /// Timestamp information.
37    #[serde(skip_serializing_if = "Option::is_none")]
38    pub timestamp: Option<Timestamp>,
39    /// The colour code of the embed.
40    #[serde(rename = "color")]
41    #[serde(skip_serializing_if = "Option::is_none")]
42    pub colour: Option<Colour>,
43    /// Footer information for the embed.
44    #[serde(skip_serializing_if = "Option::is_none")]
45    pub footer: Option<EmbedFooter>,
46    /// Image information of the embed.
47    #[serde(skip_serializing_if = "Option::is_none")]
48    pub image: Option<EmbedImage>,
49    /// Thumbnail information of the embed.
50    #[serde(skip_serializing_if = "Option::is_none")]
51    pub thumbnail: Option<EmbedImage>,
52    /// The embed's video information.
53    ///
54    /// This is present if the [`Self::kind`] is `"video"`.
55    #[serde(skip_serializing_if = "Option::is_none")]
56    pub video: Option<EmbedVideo>,
57    /// Provider information for the embed.
58    ///
59    /// For example, if the embed [`Self::kind`] is `"video"`, the provider might contain
60    /// information about YouTube.
61    #[serde(skip_serializing_if = "Option::is_none")]
62    pub provider: Option<EmbedProvider>,
63    /// Information about the author of the embed.
64    #[serde(skip_serializing_if = "Option::is_none")]
65    pub author: Option<EmbedAuthor>,
66    /// The array of fields.
67    ///
68    /// The maximum number of fields is 25.
69    #[serde(default)]
70    #[serde(skip_serializing_if = "FixedArray::is_empty")]
71    pub fields: FixedArray<EmbedField, u8>,
72}

96/// A field object in an embed.
97///
98/// [Discord docs](https://docs.discord.com/developers/resources/message#embed-object-embed-field-structure).
99#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
100#[derive(Clone, Debug, Deserialize, Serialize)]
101#[non_exhaustive]
102pub struct EmbedField {
103    /// The name of the field.
104    ///
105    /// The maximum length of this field is 512 unicode codepoints.
106    pub name: FixedString<u16>,
107    /// The value of the field.
108    ///
109    /// The maximum length of this field is 1024 unicode codepoints.
110    pub value: FixedString<u16>,
111    /// Indicator of whether the field should display as inline.
112    #[serde(default)]
113    pub inline: bool,
114}
115
116impl EmbedField {
117    /// Creates a new embed field.
118    ///
119    /// **Note**: Refer to the [`Self::name`] and [`Self::value`] documentation for maximum
120    /// lengths.
121    pub fn new<T, U>(name: T, value: U, inline: bool) -> Self
122    where
123        T: Into<String>,
124        U: Into<String>,
125    {
126        Self::new_(name.into().trunc_into(), value.into().trunc_into(), inline)
127    }
128
129    pub(crate) const fn new_(
130        name: FixedString<u16>,
131        value: FixedString<u16>,
132        inline: bool,
133    ) -> Self {
134        Self {
135            name,
136            value,
137            inline,
138        }
139    }
140}

180    /// [media type]: https://en.wikipedia.org/wiki/Media_type

183    ///

186    /// The version of the placeholder.
187    pub placeholder_version: Option<NonMaxU32>,
188    /// The description (alt text) for the image.
189    pub description: Option<FixedString>,
190    /// The embed media flags for the image.
191    pub flags: Option<EmbedMediaFlags>,
