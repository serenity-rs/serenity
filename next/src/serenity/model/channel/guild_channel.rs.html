20/// Represents the shared fields between [`GuildChannel`] and [`GuildThread`].
21///
22/// [Discord docs](https://docs.discord.com/developers/topics/threads#thread-fields)
23#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
24#[derive(Clone, Debug, Default, Deserialize, Serialize)]
25#[non_exhaustive]
26pub struct BaseGuildChannel {
27    /// The Id of the guild the channel is located in.
28    #[serde(default)]
29    pub guild_id: GuildId,
30    /// The type of the channel.
31    #[serde(rename = "type")]
32    pub kind: ChannelType,
33    /// The name of the channel. (1-100 characters)
34    pub name: FixedString<u16>,
35    /// The Id of the last message sent in the channel.
36    pub last_message_id: Option<MessageId>,
37    /// The timestamp of the time a pin was most recently made.
38    pub last_pin_timestamp: Option<Timestamp>,
39    /// A rate limit that applies per user and excludes bots.
40    ///
41    /// **Note**: This is only available for text channels excluding news channels.
42    #[doc(alias = "slowmode")]
43    #[serde(default)]
44    pub rate_limit_per_user: Option<NonMaxU16>,
45}

243        let channel = builder.execute(http, self.id).await?;

248    /// Edits the voice state of a given user in a stage channel.
249    ///
250    /// **Note**: Requires the [Request to Speak] permission. Also requires the [Mute Members]
251    /// permission to suppress another user or unsuppress the current user. This is not required if
252    /// suppressing the current user.
253    ///
254    /// # Example
255    ///
256    /// Invite a user to speak.
257    ///
258    /// ```rust
259    /// # #[cfg(feature = "cache")]
260    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
261    /// # use std::sync::Arc;
262    /// # use serenity::{cache::Cache, http::Http, model::id::{GuildId, ChannelId, UserId}};
263    /// #
264    /// # let http: Http = unimplemented!();
265    /// # let cache = Cache::default();
266    /// # let (guild_id, channel_id, user_id) = (GuildId::new(1), ChannelId::new(1), UserId::new(1));
267    /// use serenity::builder::EditVoiceState;
268    /// use serenity::model::ModelError;
269    ///
270    /// let channel = {
271    ///     let guild = cache.guild(guild_id).ok_or(ModelError::ItemMissing)?;
272    ///     guild.viewable_channels.get(&channel_id).ok_or(ModelError::ItemMissing)?.clone()
273    /// };
274    ///
275    /// let builder = EditVoiceState::new().suppress(false);
276    /// channel.edit_voice_state(&http, user_id, builder).await?;
277    /// # Ok(())
278    /// # }
279    /// ```
280    ///
281    /// # Errors
282    ///
283    /// Returns a [`ModelError::InvalidChannelType`] if the channel is not a stage channel.
284    ///
285    /// Returns [`Error::Http`] if the user lacks permission, or if invalid data is given.
286    ///
287    /// [Request to Speak]: Permissions::REQUEST_TO_SPEAK
288    /// [Mute Members]: Permissions::MUTE_MEMBERS
289    pub async fn edit_voice_state(
290        &self,
291        http: &Http,
292        user_id: UserId,
293        builder: EditVoiceState,
294    ) -> Result<()> {
295        if self.base.kind != ChannelType::Stage {
296            return Err(Error::from(ModelError::InvalidChannelType));
297        }
298
299        builder.execute(http, self.base.guild_id, self.id, Some(user_id)).await
300    }

304    /// **Note**: Requires the [Request to Speak] permission. The [Mute Members] permission is

309    /// Send a request to speak, then clear the request.

367    /// Retrieves [`Member`]s from the current channel.
368    ///
369    /// [`ChannelType::Voice`] and [`ChannelType::Stage`] returns [`Member`]s using the channel.
370    ///
371    /// [`ChannelType::Text`] and [`ChannelType::News`] return [`Member`]s that can read the
372    /// channel.
373    ///
374    /// # Errors
375    ///
376    /// Other [`ChannelType`]s lack the concept of [`Member`]s and will return:
377    /// [`ModelError::InvalidChannelType`].
378    #[cfg(feature = "cache")]
379    pub fn members(&self, cache: &Cache) -> Result<Vec<Member>> {
380        let guild = cache.guild(self.base.guild_id).ok_or(ModelError::GuildNotFound)?;
381
382        match self.base.kind {
383            ChannelType::Voice | ChannelType::Stage => Ok(guild
384                .voice_states
385                .iter()
386                .filter_map(|v| {
387                    v.channel_id.and_then(|c| {
388                        if c == self.id { guild.members.get(&v.user_id).cloned() } else { None }
389                    })
390                })
391                .collect()),
392            ChannelType::News | ChannelType::Text => Ok(guild
393                .members
394                .iter()
395                .filter(|member| {
396                    guild.user_permissions_in(self, member).contains(Permissions::VIEW_CHANNEL)
397                })
398                .cloned()
399                .collect::<Vec<Member>>()),
400            _ => Err(Error::from(ModelError::InvalidChannelType)),
401        }
402    }

592/// A container for a guild channel that might be obfuscated.
593#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
594#[derive(Clone, Debug, Serialize)]
595pub enum MaybeObfuscated {
596    Viewable(GuildChannel),
597    Obfuscated(ObfuscatedChannel),
598}
599
600impl MaybeObfuscated {
601    /// Returns the [`ChannelId`] of [`MaybeObfuscated`].
602    #[must_use]
603    pub fn id(&self) -> ChannelId {
604        match self {
605            MaybeObfuscated::Viewable(gc) => gc.id,
606            MaybeObfuscated::Obfuscated(oc) => oc.id,
607        }
608    }
609
610    /// Returns the [`ChannelType`] of [`MaybeObfuscated`].
611    #[must_use]
612    pub fn kind(&self) -> ChannelType {
613        match self {
614            MaybeObfuscated::Viewable(gc) => gc.base.kind,
615            MaybeObfuscated::Obfuscated(oc) => oc.kind,
616        }
617    }
618
619    /// Returns the name of [`MaybeObfuscated`], if viewable.
620    ///
621    /// **Note**: If the channel is obfuscated, this will return `None`.
622    #[must_use]
623    pub fn name(&self) -> Option<&str> {
624        match self {
625            MaybeObfuscated::Viewable(gc) => Some(gc.base.name.as_str()),
626            MaybeObfuscated::Obfuscated(_) => None,
627        }
628    }
629
630    /// Returns the [`ChannelId`] of the parent category [`MaybeObfuscated`] belongs to.
631    ///
632    /// **Note**: If the channel does not belong to a category, this will return `None`.
633    #[must_use]
