20/// Represents the shared fields between [`GuildChannel`] and [`GuildThread`].
21///
22/// [Discord docs](https://docs.discord.com/developers/topics/threads#thread-fields)
23#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
24#[derive(Clone, Debug, Default, Deserialize, Serialize)]
25#[non_exhaustive]
26pub struct BaseGuildChannel {
27    /// The Id of the guild the channel is located in.
28    #[serde(default)]
29    pub guild_id: GuildId,
30    /// The type of the channel.
31    #[serde(rename = "type")]
32    pub kind: ChannelType,
33    /// The name of the channel. (1-100 characters)
34    pub name: FixedString<u16>,
35    /// The Id of the last message sent in the channel.
36    pub last_message_id: Option<MessageId>,
37    /// The timestamp of the time a pin was most recently made.
38    pub last_pin_timestamp: Option<Timestamp>,
39    /// A rate limit that applies per user and excludes bots.
40    ///
41    /// **Note**: This is only available for text channels excluding news channels.
42    #[doc(alias = "slowmode")]
43    #[serde(default)]
44    pub rate_limit_per_user: Option<NonMaxU16>,
45}

56/// Represents a channel in a [`Guild`], excluding thread information.
57///
58/// [Discord docs](https://docs.discord.com/developers/resources/channel#channel-object).
59#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
60#[derive(Clone, Debug, Default, Deserialize, Serialize)]
61#[non_exhaustive]
62pub struct GuildChannel {
63    /// The shared fields between [`GuildChannel`] and [`GuildThread`].
64    #[serde(flatten)]
65    pub base: BaseGuildChannel,
66    /// The unique ID of the channel.
67    pub id: ChannelId,
68    /// The Id of the parent category for a channel.
69    ///
70    /// **Note**: This is only available for channels in a category.
71    // Technically shared, but for different purposes.
72    pub parent_id: Option<ChannelId>,
73    /// The bitrate of the channel.
74    ///
75    /// **Note**: This is only available for voice and stage channels.
76    pub bitrate: Option<NonMaxU32>,
77    /// Permission overwrites for [`Member`]s and for [`Role`]s.
78    #[serde(default)]
79    pub permission_overwrites: FixedArray<PermissionOverwrite>,
80    /// The position of the channel.
81    ///
82    /// The default text channel will _almost always_ have a position of `0`.
83    #[serde(default)]
84    pub position: u16,
85    /// The topic of the channel.
86    ///
87    /// **Note**: This is only available for text, forum and stage channels.
88    pub topic: Option<FixedString<u16>>,
89    /// The maximum number of members allowed in the channel.
90    ///
91    /// This is max 99 for voice channels and 10,000 for stage channels (0 refers to no limit).
92    pub user_limit: Option<NonMaxU16>,
93    /// Used to tell if the channel is not safe for work.
94    // This field can or can not be present sometimes, but if it isn't default to `false`.
95    #[serde(default)]
96    pub nsfw: bool,
97    /// The region override.
98    ///
99    /// **Note**: This is only available for voice and stage channels. [`None`] for voice and stage
100    /// channels means automatic region selection.
101    pub rtc_region: Option<FixedString<u8>>,
102    /// The video quality mode for a voice channel.
103    pub video_quality_mode: Option<VideoQualityMode>,
104    /// Default duration for newly created threads, in minutes, to automatically archive the thread
105    /// after recent activity.
106    pub default_auto_archive_duration: Option<AutoArchiveDuration>,
107    /// Computed permissions for the invoking user in the channel, including overwrites.
108    ///
109    /// Only included inside [`CommandDataResolved`].
110    pub permissions: Option<Permissions>,
111    /// Extra information about the channel
112    ///
113    /// **Note**: This is only available in forum channels.
114    #[serde(default)]
115    pub flags: ChannelFlags,
116    /// The set of available tags.
117    ///
118    /// **Note**: This is only available in forum channels.
119    #[serde(default)]
120    pub available_tags: FixedArray<ForumTag>,
121    /// The emoji to show in the add reaction button
122    ///
123    /// **Note**: This is only available in a forum.
124    pub default_reaction_emoji: Option<ForumEmoji>,
125    /// The initial `rate_limit_per_user` to set on newly created threads in a channel. This field
126    /// is copied to the thread at creation time and does not live update.
127    ///
128    /// **Note**: This is only available in a forum or text channel.
129    pub default_thread_rate_limit_per_user: Option<NonMaxU16>,
130    /// The status of a voice channel.
131    ///
132    /// **Note**: This is only available in voice channels.
133    pub status: Option<FixedString<u16>>,
134    /// The default sort order type used to order posts
135    ///
136    /// **Note**: This is only available in a forum.
137    pub default_sort_order: Option<SortOrder>,
138    /// The default forum layout view used to display posts in a forum. Defaults to 0, which
139    /// indicates a layout view has not been set by a channel admin.
140    ///
141    /// **Note**: This is only available in a forum.
142    pub default_forum_layout: Option<ForumLayoutType>,
143}

165    /// Whether or not this channel is text-based, meaning that it is possible to send messages.
166    #[must_use]
167    pub fn is_text_based(&self) -> bool {
168        matches!(
169            self.base.kind,
170            ChannelType::Text
171                | ChannelType::News
172                | ChannelType::Voice
173                | ChannelType::Stage
174                | ChannelType::PublicThread
175                | ChannelType::PrivateThread
176                | ChannelType::NewsThread
177        )
178    }

180    /// Deletes this channel, returning the channel on a successful deletion.
181    ///
182    /// **Note**: Requires the [Manage Channels] permission.
183    ///
184    /// # Errors
185    ///
186    /// Returns [`Error::Http`] if the current user lacks permission.
187    ///
188    /// [Manage Channels]: Permissions::MANAGE_CHANNELS
189    pub async fn delete(&self, http: &Http, reason: Option<&str>) -> Result<GuildChannel> {
190        let channel = self.id.widen().delete(http, reason).await?;
191        channel.guild().ok_or(Error::Model(ModelError::InvalidChannelType))
192    }

194    /// Edits the channel's settings.
195    ///
196    /// Refer to the documentation for [`EditChannel`] for a full list of methods.
197    ///
198    /// **Note**: Requires the [Manage Channels] permission. Modifying permissions via
199    /// [`EditChannel::permissions`] also requires the [Manage Roles] permission.
200    ///
201    /// # Examples
202    ///
203    /// Change a voice channels name and bitrate:
204    ///
205    /// ```rust,no_run
206    /// # use serenity::builder::EditChannel;
207    /// # use serenity::http::Http;
208    /// # use serenity::model::id::ChannelId;
209    /// # async fn run() {
210    /// # let http: Http = unimplemented!();
211    /// # let channel = ChannelId::new(1234);
212    /// let builder = EditChannel::new().name("test").bitrate(86400);
213    /// channel.edit(&http, builder).await;
214    /// # }
215    /// ```
216    ///
217    /// # Errors
218    ///
219    /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is given.
220    ///
221    /// [Manage Channels]: Permissions::MANAGE_CHANNELS
222    /// [Manage Roles]: Permissions::MANAGE_ROLES
223    pub async fn edit(&mut self, http: &Http, builder: EditChannel<'_>) -> Result<()> {
224        let channel = builder.execute(http, self.id).await?;
225        *self = channel;
226        Ok(())
227    }

229    /// Edits the voice state of a given user in a stage channel.
230    ///
231    /// **Note**: Requires the [Request to Speak] permission. Also requires the [Mute Members]
232    /// permission to suppress another user or unsuppress the current user. This is not required if
233    /// suppressing the current user.
234    ///
235    /// # Example
236    ///
237    /// Invite a user to speak.
238    ///
239    /// ```rust
240    /// # #[cfg(feature = "cache")]
241    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
242    /// # use std::sync::Arc;
243    /// # use serenity::{cache::Cache, http::Http, model::id::{GuildId, ChannelId, UserId}};
244    /// #
245    /// # let http: Http = unimplemented!();
246    /// # let cache = Cache::default();
247    /// # let (guild_id, channel_id, user_id) = (GuildId::new(1), ChannelId::new(1), UserId::new(1));
248    /// use serenity::builder::EditVoiceState;
249    /// use serenity::model::ModelError;
250    ///
251    /// let channel = {
252    ///     let guild = cache.guild(guild_id).ok_or(ModelError::ItemMissing)?;
253    ///     guild.channels.get(&channel_id).ok_or(ModelError::ItemMissing)?.clone()
254    /// };
255    ///
256    /// let builder = EditVoiceState::new().suppress(false);
257    /// channel.edit_voice_state(&http, user_id, builder).await?;
258    /// # Ok(())
259    /// # }
260    /// ```
261    ///
262    /// # Errors
263    ///
264    /// Returns a [`ModelError::InvalidChannelType`] if the channel is not a stage channel.
265    ///
266    /// Returns [`Error::Http`] if the user lacks permission, or if invalid data is given.
267    ///
268    /// [Request to Speak]: Permissions::REQUEST_TO_SPEAK
269    /// [Mute Members]: Permissions::MUTE_MEMBERS
270    pub async fn edit_voice_state(
271        &self,
272        http: &Http,
273        user_id: UserId,
274        builder: EditVoiceState,
275    ) -> Result<()> {
276        if self.base.kind != ChannelType::Stage {
277            return Err(Error::from(ModelError::InvalidChannelType));
278        }
279
280        builder.execute(http, self.base.guild_id, self.id, Some(user_id)).await
281    }

283    /// Edits the current user's voice state in a stage channel.
284    ///
285    /// **Note**: Requires the [Request to Speak] permission. The [Mute Members] permission is
286    /// **not** required.
287    ///
288    /// # Example
289    ///
290    /// Send a request to speak, then clear the request.
291    ///
292    /// ```rust
293    /// # #[cfg(feature = "cache")]
294    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
295    /// # use std::sync::Arc;
296    /// # use serenity::{cache::Cache, http::Http, model::id::{GuildId, ChannelId}};
297    /// #
298    /// # let http: Http = unimplemented!();
299    /// # let cache = Cache::default();
300    /// # let (guild_id, channel_id) = (GuildId::new(1), ChannelId::new(1));
301    /// use serenity::builder::EditVoiceState;
302    /// use serenity::model::ModelError;
303    ///
304    /// let channel = {
305    ///     let guild = cache.guild(guild_id).ok_or(ModelError::ItemMissing)?;
306    ///     guild.channels.get(&channel_id).ok_or(ModelError::ItemMissing)?.clone()
307    /// };
308    ///
309    /// // Send a request to speak
310    /// let builder = EditVoiceState::new().request_to_speak(true);
311    /// channel.edit_own_voice_state(&http, builder.clone()).await?;
312    ///
313    /// // Clear own request to speak
314    /// let builder = builder.request_to_speak(false);
315    /// channel.edit_own_voice_state(&http, builder).await?;
316    /// # Ok(())
317    /// # }
318    /// ```
319    ///
320    /// # Errors
321    ///
322    /// If the `cache` is enabled, returns a [`ModelError::InvalidChannelType`] if the channel is
323    /// not a stage channel.
324    ///
325    /// Returns [`Error::Http`] if the user lacks permission, or if invalid data is given.
326    ///
327    /// [Request to Speak]: Permissions::REQUEST_TO_SPEAK
328    /// [Mute Members]: Permissions::MUTE_MEMBERS
329    pub async fn edit_own_voice_state(&self, http: &Http, builder: EditVoiceState) -> Result<()> {
330        builder.execute(http, self.base.guild_id, self.id, None).await
331    }

333    /// Sends a message to the channel.
334    ///
335    /// Refer to the documentation for [`CreateMessage`] for information regarding content
336    /// restrictions and requirements.

338    /// # Errors

348    /// Retrieves [`Member`]s from the current channel.
349    ///
350    /// [`ChannelType::Voice`] and [`ChannelType::Stage`] returns [`Member`]s using the channel.
351    ///
352    /// [`ChannelType::Text`] and [`ChannelType::News`] return [`Member`]s that can read the
353    /// channel.
354    ///
355    /// # Errors
356    ///
357    /// Other [`ChannelType`]s lack the concept of [`Member`]s and will return:
358    /// [`ModelError::InvalidChannelType`].
359    #[cfg(feature = "cache")]
360    pub fn members(&self, cache: &Cache) -> Result<Vec<Member>> {
361        let guild = cache.guild(self.base.guild_id).ok_or(ModelError::GuildNotFound)?;
362
363        match self.base.kind {
364            ChannelType::Voice | ChannelType::Stage => Ok(guild
365                .voice_states
366                .iter()
367                .filter_map(|v| {
368                    v.channel_id.and_then(|c| {
369                        if c == self.id { guild.members.get(&v.user_id).cloned() } else { None }
370                    })
371                })
372                .collect()),
373            ChannelType::News | ChannelType::Text => Ok(guild
374                .members
375                .iter()
376                .filter(|member| {
377                    guild.user_permissions_in(self, member).contains(Permissions::VIEW_CHANNEL)
378                })
379                .cloned()
380                .collect::<Vec<Member>>()),
381            _ => Err(Error::from(ModelError::InvalidChannelType)),
382        }
383    }

385    /// Creates a webhook in the channel.
386    ///
387    /// # Errors
388    ///
389    /// Returns [`ModelError::InvalidChannelType`] if the corresponding channel is not of type
390    /// [`ChannelType::Text`] or [`ChannelType::News`].
391    ///
392    /// See [`CreateWebhook::execute`] for a detailed list of other
393    /// possible errors,
394    pub async fn create_webhook(&self, http: &Http, builder: CreateWebhook<'_>) -> Result<Webhook> {
395        // forum channels are not text-based, but webhooks can be created in them
396        // and used to send messages in their posts
397        if !self.is_text_based() && self.base.kind != ChannelType::Forum {
398            return Err(Error::Model(ModelError::InvalidChannelType));
399        }
400
401        self.id.create_webhook(http, builder).await
402    }
