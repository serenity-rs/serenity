9fn base64_bytes<'de, D>(deserializer: D) -> Result<Option<Vec<u8>>, D::Error>
10where
11    D: serde::Deserializer<'de>,
12{
13    use base64::Engine as _;
14    use serde::de::Error;
15
16    let base64 = <Option<CowStr<'de>>>::deserialize(deserializer)?;
17    let bytes = match base64 {
18        Some(CowStr(base64)) => {
19            Some(base64::prelude::BASE64_STANDARD.decode(&*base64).map_err(D::Error::custom)?)
20        },
21        None => None,
22    };
23    Ok(bytes)
24}

26/// A file uploaded with a message. Not to be confused with [`Embed`]s.
27///
28/// [Discord docs](https://docs.discord.com/developers/resources/message#attachment-object).
29///
30/// [`Embed`]: super::Embed
31#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
32#[derive(Clone, Debug, Deserialize, Serialize)]
33#[non_exhaustive]
34pub struct Attachment {
35    /// The unique ID given to this attachment.
36    pub id: AttachmentId,
37    /// The filename of the file that was uploaded. This is equivalent to what the uploader had
38    /// their file named.
39    pub filename: FixedString,
40    /// Description for the file (max 1024 characters).
41    pub description: Option<FixedString<u16>>,
42    /// If the attachment is an image, then the height of the image is provided.
43    pub height: Option<NonMaxU32>,
44    /// If the attachment is an image, then the width of the image is provided.
45    pub width: Option<NonMaxU32>,
46    /// The proxy URL.
47    pub proxy_url: FixedString,
48    /// The size of the file in bytes.
49    pub size: u32,
50    /// The URL of the uploaded attachment.
51    pub url: FixedString,
52    /// The attachment's [media type].
53    ///
54    /// [media type]: https://en.wikipedia.org/wiki/Media_type
55    pub content_type: Option<FixedString>,
56    /// Whether this attachment is ephemeral.
57    ///
58    /// Ephemeral attachments will automatically be removed after a set period of time.
59    ///
60    /// Ephemeral attachments on messages are guaranteed to be available as long as the message
61    /// itself exists.
62    #[serde(default, skip_serializing_if = "is_false")]
63    pub ephemeral: bool,
64    /// The duration of the audio file (present if [`MessageFlags::IS_VOICE_MESSAGE`]).
65    pub duration_secs: Option<f64>,
66    /// List of bytes representing a sampled waveform (present if
67    /// [`MessageFlags::IS_VOICE_MESSAGE`]).
68    ///
69    /// The waveform is intended to be a preview of the entire voice message, with 1 byte per
70    /// datapoint. Clients sample the recording at most once per 100 milliseconds, but will
71    /// downsample so that no more than 256 datapoints are in the waveform.
72    ///
73    /// The waveform details are a Discord implementation detail and may change without warning or
74    /// documentation.
75    #[serde(default, deserialize_with = "base64_bytes")]
76    pub waveform: Option<Vec<u8>>,
77}

80impl Attachment {

82    #[must_use]

87    /// Downloads the attachment, returning back a vector of bytes.
88    ///
89    /// # Examples
90    ///
91    /// Download all of the attachments associated with a [`Message`]:
92    ///
93    /// ```rust,no_run
94    /// use std::io::Write;
95    /// use std::path::Path;
96    ///
97    /// # use serenity::http::Http;
98    /// use serenity::model::prelude::*;
99    /// use serenity::prelude::*;
100    /// use tokio::fs::File;
101    /// use tokio::io::AsyncWriteExt;
102    ///
103    /// # async fn run() {
104    /// # let http: Http = unimplemented!();
105    /// # let message: Message = unimplemented!();
106    ///
107    /// for attachment in message.attachments {
108    ///     let content = match attachment.download().await {
109    ///         Ok(content) => content,
110    ///         Err(why) => {
111    ///             println!("Error downloading attachment: {:?}", why);
112    ///             let _ = message.channel_id.say(&http, "Error downloading attachment").await;
113    ///
114    ///             return;
115    ///         },
116    ///     };
117    ///
118    ///     let mut file = match File::create(&attachment.filename).await {
119    ///         Ok(file) => file,
120    ///         Err(why) => {
121    ///             println!("Error creating file: {:?}", why);
122    ///             let _ = message.channel_id.say(&http, "Error creating file").await;
123    ///
124    ///             return;
125    ///         },
126    ///     };
127    ///
128    ///     if let Err(why) = file.write_all(&content).await {
129    ///         println!("Error writing to file: {:?}", why);
130    ///
131    ///         return;
132    ///     }
133    ///
134    ///     let _ = message.channel_id.say(&http, format!("Saved {:?}", attachment.filename)).await;
135    /// }
136    ///
137    /// # }
138    /// ```
139    ///
140    /// # Errors
141    ///
142    /// Returns an [`Error::Io`] when there is a problem reading the contents of the HTTP response.
143    ///
144    /// Returns an [`Error::Http`] when there is a problem retrieving the attachment.
145    ///
146    /// [`Message`]: super::Message
147    pub async fn download(&self) -> Result<Vec<u8>> {
148        let reqwest = ReqwestClient::new();
149        let bytes = reqwest.get(&*self.url).send().await?.bytes().await?;
150        Ok(bytes.to_vec())
151    }
152}
