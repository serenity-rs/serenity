9enum_number! {
10    /// A representation of a type of webhook.
11    ///
12    /// [Discord docs](https://docs.discord.com/developers/resources/webhook#webhook-object-webhook-types).
13    #[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd, Deserialize, Serialize)]
14    #[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
15    #[non_exhaustive]
16    pub enum WebhookType {
17        /// An indicator that the webhook can post messages to channels with a token.
18        Incoming = 1,
19        /// An indicator that the webhook is managed by Discord for posting new messages to
20        /// channels without a token.
21        ChannelFollower = 2,
22        /// Application webhooks are webhooks used with Interactions.
23        Application = 3,
24        _ => Unknown(u8),
25    }
26}
27
28impl WebhookType {
29    #[must_use]
30    pub fn name(self) -> &'static str {
31        match self {
32            Self::Incoming => "incoming",
33            Self::ChannelFollower => "channel follower",
34            Self::Application => "application",
35            _ => "unknown",
36        }
37    }
38}

40/// A representation of a webhook, which is a low-effort way to post messages to channels. They do
41/// not necessarily require a bot user or authentication to use.
42///
43/// [Discord docs](https://docs.discord.com/developers/resources/webhook#webhook-object).
44#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
45#[derive(Debug, Clone, Deserialize, Serialize)]
46#[non_exhaustive]
47pub struct Webhook {
48    /// The unique Id.
49    ///
50    /// Can be used to calculate the creation date of the webhook.
51    pub id: WebhookId,
52    /// The type of the webhook.
53    #[serde(rename = "type")]
54    pub kind: WebhookType,
55    /// The Id of the guild that owns the webhook.
56    pub guild_id: Option<GuildId>,
57    /// The Id of the channel that owns the webhook.
58    pub channel_id: Option<ChannelId>,
59    /// The user that created the webhook.
60    ///
61    /// **Note**: This is not received when getting a webhook by its token.
62    pub user: Option<User>,
63    /// The default name of the webhook.
64    ///
65    /// This can be temporarily overridden via [`ExecuteWebhook::username`].
66    pub name: Option<FixedString<u8>>,
67    /// The default avatar.
68    ///
69    /// This can be temporarily overridden via [`ExecuteWebhook::avatar_url`].
70    pub avatar: Option<ImageHash>,
71    /// The webhook's secure token.
72    pub token: Option<SecretString>,
73    /// The bot/OAuth2 application that created this webhook.
74    pub application_id: Option<ApplicationId>,
75    /// The guild of the channel that this webhook is following (returned for
76    /// [`WebhookType::ChannelFollower`])
77    pub source_guild: Option<WebhookGuild>,
78    /// The channel that this webhook is following (returned for
79    /// [`WebhookType::ChannelFollower`]).
80    pub source_channel: Option<WebhookChannel>,
81    /// The url used for executing the webhook (returned by the webhooks OAuth2 flow).
82    pub url: Option<SecretString>,
83}
84
85impl ExtractKey<WebhookId> for Webhook {
86    fn extract_key(&self) -> &WebhookId {
87        &self.id
88    }
89}

117impl Webhook {
118    /// Retrieves a webhook given its Id.
119    ///
120    /// This method requires authentication, whereas [`Webhook::from_id_with_token`] and
121    /// [`Webhook::from_url`] do not.
122    ///
123    /// # Examples
124    ///
125    /// Retrieve a webhook by Id:
126    ///
127    /// ```rust,no_run
128    /// # use serenity::http::Http;
129    /// # use serenity::model::{webhook::Webhook, id::WebhookId};
130    /// #
131    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
132    /// # let http: Http = unimplemented!();
133    /// let id = WebhookId::new(245037420704169985);
134    /// let webhook = Webhook::from_id(&http, id).await?;
135    /// # Ok(())
136    /// # }
137    /// ```
138    ///
139    /// # Errors
140    ///
141    /// Returns an [`Error::Http`] if the current user is not authenticated, or if the webhook does
142    /// not exist.
143    ///
144    /// May also return an [`Error::Json`] if there is an error in deserialising Discord's
145    /// response.
146    pub async fn from_id(http: &Http, webhook_id: WebhookId) -> Result<Self> {
147        http.get_webhook(webhook_id).await
148    }
149
150    /// Retrieves a webhook given its Id and unique token.
151    ///
152    /// This method does _not_ require authentication.
153    ///
154    /// # Examples
155    ///
156    /// Retrieve a webhook by Id and its unique token:
157    ///
158    /// ```rust,no_run
159    /// # use serenity::http::Http;
160    /// # use serenity::model::{webhook::Webhook, id::WebhookId};
161    /// #
162    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
163    /// # let http: Http = unimplemented!();
164    /// let id = WebhookId::new(245037420704169985);
165    /// let token = "ig5AO-wdVWpCBtUUMxmgsWryqgsW3DChbKYOINftJ4DCrUbnkedoYZD0VOH1QLr-S3sV";
166    ///
167    /// let webhook = Webhook::from_id_with_token(&http, id, token).await?;
168    /// # Ok(())
169    /// # }
170    /// ```
171    ///
172    /// # Errors
173    ///
174    /// Returns an [`Error::Http`] if the webhook does not exist, or if the token is invalid.
175    ///
176    /// May also return an [`Error::Json`] if there is an error in deserialising Discord's
177    /// response.
178    pub async fn from_id_with_token(
179        http: &Http,
180        webhook_id: WebhookId,
181        token: &str,
182    ) -> Result<Self> {
183        http.get_webhook_with_token(webhook_id, token).await
184    }

186    /// Retrieves a webhook given its url.
187    ///
188    /// This method does _not_ require authentication.
189    ///
190    /// # Examples
191    ///
192    /// Retrieve a webhook by url:
193    ///
194    /// ```rust,no_run
195    /// # use serenity::http::Http;
196    /// # use serenity::model::webhook::Webhook;
197    /// #
198    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
199    /// # let http: Http = unimplemented!();
200    /// let url = "https://discord.com/api/webhooks/245037420704169985/ig5AO-wdVWpCBtUUMxmgsWryqgsW3DChbKYOINftJ4DCrUbnkedoYZD0VOH1QLr-S3sV";
201    /// let webhook = Webhook::from_url(&http, url).await?;
202    /// # Ok(())
203    /// # }
204    /// ```
205    ///
206    /// # Errors
207    ///
208    /// Returns an [`Error::Http`] if the url is malformed, or otherwise if the webhook does not
209    /// exist, or if the token is invalid.
210    ///
211    /// May also return an [`Error::Json`] if there is an error in deserialising Discord's
212    /// response.
213    pub async fn from_url(http: &Http, url: &str) -> Result<Self> {
214        http.get_webhook_from_url(url).await
215    }

217    /// Deletes the webhook.
218    ///
219    /// If [`Self::token`] is set, then authentication is _not_ required. Otherwise, if it is
220    /// [`None`], then authentication _is_ required.
221    ///
222    /// # Errors
223    ///
224    /// Returns [`Error::Http`] if the webhook does not exist, the token is invalid, or if the
225    /// webhook could not otherwise be deleted.
226    pub async fn delete(&self, http: &Http, reason: Option<&str>) -> Result<()> {
227        match &self.token {
228            Some(token) => {
229                http.delete_webhook_with_token(self.id, token.expose_secret(), reason).await
230            },
231            None => http.delete_webhook(self.id, reason).await,
232        }
233    }

235    /// Edits the webhook.
236    ///
237    /// If [`Self::token`] is set, then authentication is _not_ required. Otherwise, if it is
238    /// [`None`], then authentication _is_ required.
239    ///
240    /// # Examples
241    ///
242    /// ```rust,no_run
243    /// # use serenity::http::Http;
244    /// # use serenity::builder::EditWebhook;
245    /// # use serenity::model::webhook::Webhook;
246    /// #
247    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
248    /// # let http: Http = unimplemented!();
249    /// let url = "https://discord.com/api/webhooks/245037420704169985/ig5AO-wdVWpCBtUUMxmgsWryqgsW3DChbKYOINftJ4DCrUbnkedoYZD0VOH1QLr-S3sV";
250    /// let mut webhook = Webhook::from_url(&http, url).await?;
251    ///
252    /// let builder = EditWebhook::new().name("new name");
253    /// webhook.edit(&http, builder).await?;
254    /// # Ok(())
255    /// # }
256    /// ```
257    ///
258    /// # Errors
259    ///
260    /// Returns an [`Error::Model`] if [`Self::token`] is [`None`].
261    ///
262    /// May also return an [`Error::Http`] if the content is malformed, or if the token is invalid.
263    ///
264    /// Or may return an [`Error::Json`] if there is an error in deserialising Discord's response.
265    pub async fn edit(&mut self, http: &Http, builder: EditWebhook<'_>) -> Result<()> {
266        let token = self.token.as_ref().map(SecretString::expose_secret);
267        *self = builder.execute(http, self.id, token).await?;
268        Ok(())
269    }

271    /// Executes a webhook with the fields set via the given builder.
272    ///
273    /// # Examples
274    ///
275    /// Execute a webhook with message content of `test`:
276    ///
277    /// ```rust,no_run
278    /// # use serenity::builder::ExecuteWebhook;
279    /// # use serenity::http::Http;
280    /// # use serenity::model::webhook::Webhook;
281    /// #
282    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
283    /// # let http: Http = unimplemented!();
284    /// let url = "https://discord.com/api/webhooks/245037420704169985/ig5AO-wdVWpCBtUUMxmgsWryqgsW3DChbKYOINftJ4DCrUbnkedoYZD0VOH1QLr-S3sV";
285    /// let mut webhook = Webhook::from_url(&http, url).await?;
286    ///
287    /// let builder = ExecuteWebhook::new().content("test");
288    /// webhook.execute(&http, false, builder).await?;
289    /// # Ok(())
290    /// # }
291    /// ```
292    ///
293    /// Execute a webhook with message content of `test`, overriding the username to `serenity`,
294    /// and sending an embed:
295    ///
296    /// ```rust,no_run
297    /// # use serenity::http::Http;
298    /// # use serenity::model::webhook::Webhook;
299    /// #
300    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
301    /// # let http: Http = unimplemented!();
302    /// use serenity::builder::{CreateEmbed, ExecuteWebhook};
303    ///
304    /// let url = "https://discord.com/api/webhooks/245037420704169985/ig5AO-wdVWpCBtUUMxmgsWryqgsW3DChbKYOINftJ4DCrUbnkedoYZD0VOH1QLr-S3sV";
305    /// let mut webhook = Webhook::from_url(&http, url).await?;
306    ///
307    /// let embed = CreateEmbed::new()
308    ///     .title("Rust's website")
309    ///     .description(
310    ///         "Rust is a systems programming language that runs blazingly fast, prevents \
311    ///         segfaults, and guarantees thread safety.",
312    ///     )
313    ///     .url("https://rust-lang.org");
314    ///
315    /// let builder = ExecuteWebhook::new().content("test").username("serenity").embed(embed);
316    /// webhook.execute(&http, false, builder).await?;
317    /// # Ok(())
318    /// # }
319    /// ```
320    ///
321    /// # Errors
322    ///
323    /// Returns an [`Error::Model`] if [`Self::token`] is [`None`].
324    ///
325    /// May also return an [`Error::Http`] if the content is malformed, or if the webhook's token
326    /// is invalid.
327    ///
328    /// Or may return an [`Error::Json`] if there is an error deserialising Discord's response.
329    pub async fn execute(
330        &self,
331        http: &Http,
332        wait: bool,
333        builder: ExecuteWebhook<'_>,
334    ) -> Result<Option<Message>> {
335        let token = self.token.as_ref().ok_or(ModelError::NoTokenSet)?.expose_secret();
336        builder.execute(http, self.id, token, wait).await
337    }

400    /// Retrieves the latest information about the webhook, editing the webhook in-place.
401    ///
402    /// As this calls the [`Http::get_webhook_with_token`] function, authentication is not
403    /// required.
404    ///
405    /// # Errors
406    ///
407    /// Returns an [`Error::Model`] if the [`Self::token`] is [`None`].
408    ///
409    /// May also return an [`Error::Http`] if the http client errors or if Discord returns an
410    /// error. Such as if the [`Webhook`] was deleted.
411    ///
412    /// Or may return an [`Error::Json`] if there is an error deserialising Discord's response.
413    pub async fn refresh(&mut self, http: &Http) -> Result<()> {
414        let token = self.token.as_ref().ok_or(ModelError::NoTokenSet)?.expose_secret();
415        http.get_webhook_with_token(self.id, token).await.map(|replacement| {
416            *self = replacement;
417        })
418    }
419
420    /// Returns the url of the webhook.
421    ///
422    /// # Errors
423    ///
424    /// Returns an [`Error::Model`] if the [`Self::token`] is [`None`].
425    pub fn url(&self) -> Result<String> {
426        let token = self.token.as_ref().ok_or(ModelError::NoTokenSet)?.expose_secret();
427        Ok(format!("https://discord.com/api/webhooks/{}/{token}", self.id))
428    }
