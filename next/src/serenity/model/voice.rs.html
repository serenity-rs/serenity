8/// Information about an available voice region.
9///
10/// [Discord docs](https://docs.discord.com/developers/resources/voice#voice-region-object).
11#[bool_to_bitflags::bool_to_bitflags]
12#[derive(Clone, Debug, serde::Deserialize, serde::Serialize)]
13#[non_exhaustive]
14pub struct VoiceRegion {
15    /// Whether it is a custom voice region, which is used for events.
16    pub custom: bool,
17    /// Whether it is a deprecated voice region, which you should avoid using.
18    pub deprecated: bool,
19    /// The internal Id of the voice region.
20    pub id: FixedString,
21    /// A recognizable name of the location of the voice region.
22    pub name: FixedString,
23    /// Whether the voice region is optimal for use by the current user.
24    pub optimal: bool,
25}

27/// A user's state within a voice channel.
28///
29/// [Discord docs](https://docs.discord.com/developers/resources/voice#voice-state-object).
30#[bool_to_bitflags::bool_to_bitflags]
31#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
32#[derive(Clone, Debug, serde::Deserialize, serde::Serialize)]
33#[serde(remote = "Self")]
34#[non_exhaustive]
35pub struct VoiceState {
36    pub channel_id: Option<ChannelId>,
37    pub deaf: bool,
38    pub guild_id: Option<GuildId>,
39    pub member: Option<Member>,
40    pub mute: bool,
41    pub self_deaf: bool,
42    pub self_mute: bool,
43    pub self_stream: Option<bool>,
44    pub self_video: bool,
45    pub session_id: FixedString,
46    pub suppress: bool,
47    pub user_id: UserId,
48    /// When unsuppressed, non-bot users will have this set to the current time. Bot users will be
49    /// set to [`None`]. When suppressed, the user will have their
50    /// [`Self::request_to_speak_timestamp`] removed.
51    pub request_to_speak_timestamp: Option<Timestamp>,
52}

60// Manual impl needed to insert guild_id into Member
61impl<'de> Deserialize<'de> for VoiceStateGeneratedOriginal {
62    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
63        // calls #[serde(remote)]-generated inherent method
64        let mut state = Self::deserialize(deserializer)?;
65        if let (Some(guild_id), Some(member)) = (state.guild_id, state.member.as_mut()) {
66            member.guild_id = guild_id;
67        }
68        Ok(state)
69    }
70}

72impl Serialize for VoiceStateGeneratedOriginal {
73    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
74        // calls #[serde(remote)]-generated inherent method
75        Self::serialize(self, serializer)
76    }

79/// Information about an effect, such as an emoji reaction or a soundboard sound, sent in a voice
80/// channel.
81///
82/// [Discord docs](https://docs.discord.com/developers/events/gateway-events#voice-channel-effect-send).
83#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
84#[derive(Clone, Debug, Default, Deserialize, Serialize)]
85#[non_exhaustive]
86pub struct VoiceChannelEffect {
87    /// The Id of the channel the effect was sent in.
88    pub channel_id: ChannelId,
89    /// The Id of the guild the effect was sent in.
90    pub guild_id: GuildId,
91    /// The Id of the user who sent the effect.
92    pub user_id: UserId,
93    /// The emoji sent, for emoji reaction and soundboard effects.
94    pub emoji: Option<ReactionType>,
95    /// The type of emoji animation, for emoji reaction and soundboard effects.
96    pub animation_type: Option<AnimationType>,
97    /// The Id of the emoji animation, for emoji reaction and soundboard effects.
98    pub animation_id: Option<u32>,
99    /// The Id of the soundboard sound, for soundboard effects.
100    pub sound_id: Option<SoundId>,
101    /// The volume of the soundboard sound, from 0 to 1, for soundboard effects.
102    pub sound_volume: Option<f64>,
103}
