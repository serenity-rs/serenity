259    /// Returns either the [`GuildChannel`] or the [`GuildThread`] that this ID corresponds to.
260    #[must_use]
261    pub fn channel(&self, channel_id: GenericChannelId) -> Option<GenericGuildChannelRef<'_>> {
262        let (channel_id, thread_id) = channel_id.split();
263        let channel = self.channels.get(&channel_id).map(GenericGuildChannelRef::Channel);
264        let thread = || self.threads.get(&thread_id).map(GenericGuildChannelRef::Thread);
265
266        channel.or_else(thread)
267    }

416    ///
417    /// Returns None if the member has no roles or the member from this guild.

423    /// Helper function that can also be used from [`PartialGuild`].
424    pub(crate) fn _member_highest_role_in<'a>(
425        roles: &'a ExtractMap<RoleId, Role>,
426        member: &Member,
427    ) -> Option<&'a Role> {
428        let mut highest: Option<&Role> = None;
429
430        for role_id in &member.roles {
431            if let Some(role) = roles.get(role_id) {
432                // Skip this role if this role in iteration has:
433                // - a position less than the recorded highest
434                // - a position equal to the recorded, but a higher ID
435                if let Some(highest) = highest
436                    && (role.position < highest.position
437                        || (role.position == highest.position && role.id > highest.id))
438                {
439                    continue;
440                }
441
442                highest = Some(role);
443            }
444        }
445
446        highest
447    }

449    /// Returns which of two [`User`]s has a higher [`Member`] hierarchy.
450    ///
451    /// Hierarchy is essentially who has the [`Role`] with the highest [`position`].
452    ///
453    /// Returns [`None`] if at least one of the given users' member instances is not present.
454    /// Returns [`None`] if the users have the same hierarchy, as neither are greater than the
455    /// other.
456    ///
457    /// If both user IDs are the same, [`None`] is returned. If one of the users is the guild
458    /// owner, their ID is returned.
459    ///
460    /// [`position`]: Role::position
461    #[must_use]
462    pub fn greater_member_hierarchy(&self, lhs_id: UserId, rhs_id: UserId) -> Option<UserId> {
463        let lhs = self.members.get(&lhs_id)?;
464        let rhs = self.members.get(&rhs_id)?;
465        let lhs_highest_role = self.member_highest_role(lhs);
466        let rhs_highest_role = self.member_highest_role(rhs);
467
468        Self::_greater_member_hierarchy_in(
469            lhs_highest_role,
470            rhs_highest_role,
471            self.owner_id,
472            lhs,
473            rhs,
474        )
475    }

477    /// Helper function that can also be used from [`PartialGuild`].
478    #[must_use]
479    pub(crate) fn _greater_member_hierarchy_in(
480        lhs_highest_role: Option<&Role>,
481        rhs_highest_role: Option<&Role>,
482        owner_id: UserId,
483        lhs: &Member,
484        rhs: &Member,
485    ) -> Option<UserId> {
486        // Check that the IDs are the same. If they are, neither is greater.
487        if lhs.user.id == rhs.user.id {
488            return None;
489        }
490
491        // Check if either user is the guild owner.
492        if lhs.user.id == owner_id {
493            return Some(lhs.user.id);
494        } else if rhs.user.id == owner_id {
495            return Some(rhs.user.id);
496        }
497
498        let lhs_role = lhs_highest_role.map_or((RoleId::new(1), 0), |r| (r.id, r.position));
499
500        let rhs_role = rhs_highest_role.map_or((RoleId::new(1), 0), |r| (r.id, r.position));
501
502        // If LHS and RHS both have no top position or have the same role ID, then no one wins.
503        if (lhs_role.1 == 0 && rhs_role.1 == 0) || (lhs_role.0 == rhs_role.0) {
504            return None;
505        }
506
507        // If LHS's top position is higher than RHS, then LHS wins.
508        if lhs_role.1 > rhs_role.1 {
509            return Some(lhs.user.id);
510        }
511
512        // If RHS's top position is higher than LHS, then RHS wins.
513        if rhs_role.1 > lhs_role.1 {
514            return Some(rhs.user.id);
515        }
516
517        // If LHS and RHS both have the same position, but LHS has the lower role ID, then LHS
518        // wins.
519        //
520        // If RHS has the higher role ID, then RHS wins.
521        if lhs_role.1 == rhs_role.1 && lhs_role.0 < rhs_role.0 {
522            Some(lhs.user.id)
523        } else {
524            Some(rhs.user.id)
525        }
526    }

561    /// Retrieves the first [`Member`] found that matches the name - with an optional discriminator
562    /// - provided.
563    ///
564    /// Searching with a discriminator given is the most precise form of lookup, as no two people
565    /// can share the same username *and* discriminator.
566    ///
567    /// If a member can not be found by username or username#discriminator, then a search will be
568    /// done for the nickname. When searching by nickname, the hash (`#`) and everything after it
569    /// is included in the search.
570    ///
571    /// The following are valid types of searches:
572    /// - **username**: "zey"
573    /// - **username and discriminator**: "zey#5479"
574    ///
575    /// **Note**: This will only search members that are cached. If you want to search all members
576    /// in the guild via the Http API, use [`GuildId::search_members`].
577    #[must_use]
578    pub fn member_named(&self, name: &str) -> Option<&Member> {
579        let (username, discrim) = match crate::utils::parse_user_tag(name) {
580            Some((username, discrim)) => (username, Some(discrim)),
581            None => (name, None),
582        };
583
584        for member in &self.members {
585            if &*member.user.name == username
586                && discrim.is_none_or(|d| member.user.discriminator == d)
587            {
588                return Some(member);
589            }
590        }
591
592        self.members.iter().find(|member| member.nick.as_deref().is_some_and(|nick| nick == name))
593    }

595    /// Retrieves all [`Member`] that start with a given [`String`].
596    ///
597    /// `sorted` decides whether the best early match of the `prefix` should be the criteria to
598    /// sort the result.
599    ///
600    /// For the `prefix` "zey" and the unsorted result:
601    /// - "zeya", "zeyaa", "zeyla", "zeyzey", "zeyzeyzey"
602    ///
603    /// It would be sorted:
604    /// - "zeya", "zeyaa", "zeyla", "zeyzey", "zeyzeyzey"
605    ///
606    /// **Note**: This will only search members that are cached. If you want to search all members
607    /// in the guild via the Http API, use [`GuildId::search_members`].
608    #[must_use]
609    pub fn members_starting_with(
610        &self,
611        prefix: &str,
612        case_sensitive: bool,
613        sorted: bool,
614    ) -> Vec<(&Member, &str)> {
615        fn starts_with(name: &str, prefix: &str, case_sensitive: bool) -> bool {
616            if case_sensitive {
617                name.starts_with(prefix)
618            } else {
619                name.to_lowercase().starts_with(&prefix.to_lowercase())
620            }
621        }
622
623        let mut members = self
624            .members
625            .iter()
626            .filter_map(|member| {
627                let username = &member.user.name;
628
629                if starts_with(username, prefix, case_sensitive) {
630                    Some((member, username.as_str()))
631                } else {
632                    match &member.nick {
633                        Some(nick) => starts_with(nick, prefix, case_sensitive)
634                            .then(|| (member, nick.as_str())),
635                        None => None,
636                    }
637                }
638            })
639            .collect::<Vec<(&Member, &str)>>();
640
641        if sorted {
642            members.sort_by(|a, b| closest_to_origin(prefix, a.1, b.1));
643        }
644
645        members
646    }

649    /// priority on username.

652    /// - "zeyla", "meiyla", "yladenisyla"
653    ///
654    /// If 'case_sensitive' is false, the following are not found:
655    /// - "zeYLa", "meiyLa", "LYAdenislyA"
656    ///
657    /// `sorted` decides whether the best early match of the search-term should be the criteria to
658    /// sort the result. It will look at the account name first, if that does not fit the
659    /// search-criteria `substring`, the display-name will be considered.
660    ///
661    /// For the `substring` "zey" and the unsorted result:
662    /// - "azey", "zey", "zeyla", "zeylaa", "zeyzeyzey"
663    ///
664    /// It would be sorted:
665    /// - "zey", "azey", "zeyla", "zeylaa", "zeyzeyzey"
666    ///
667    /// **Note**: Due to two fields of a [`Member`] being candidates for the searched field,
668    /// setting `sorted` to `true` will result in an overhead, as both fields have to be considered
669    /// again for sorting.

672    /// in the guild via the Http API, use [`GuildId::search_members`].

675        &self,
676        substring: &str,
677        case_sensitive: bool,
678        sorted: bool,
679    ) -> Vec<(&Member, String)> {
680        let mut members = self
681            .members
682            .iter()
683            .filter_map(|member| {
684                let username = &member.user.name;
685
686                if contains(username, substring, case_sensitive) {
687                    Some((member, username.clone().into()))
688                } else {
689                    match &member.nick {
690                        Some(nick) => contains(nick, substring, case_sensitive)
691                            .then(|| (member, nick.clone().into())),

693                    }
694                }

876    /// Helper function that can also be used from [`PartialGuild`].
877    pub(crate) fn user_permissions_in_(
878        channel: Option<&GuildChannel>,
879        member_user_id: UserId,
880        member_roles: &[RoleId],
881        guild_id: GuildId,
882        guild_roles: &ExtractMap<RoleId, Role>,
883        guild_owner_id: UserId,
884    ) -> Permissions {
885        let mut everyone_allow_overwrites = Permissions::empty();
886        let mut everyone_deny_overwrites = Permissions::empty();
887        let mut roles_allow_overwrites = Vec::new();
888        let mut roles_deny_overwrites = Vec::new();
889        let mut member_allow_overwrites = Permissions::empty();
890        let mut member_deny_overwrites = Permissions::empty();
891
892        if let Some(channel) = channel {
893            for overwrite in &channel.permission_overwrites {
894                match overwrite.kind {
895                    PermissionOverwriteType::Member(user_id) => {
896                        if member_user_id == user_id {
897                            member_allow_overwrites = overwrite.allow;
898                            member_deny_overwrites = overwrite.deny;
899                        }
900                    },
901                    PermissionOverwriteType::Role(role_id) => {
902                        if role_id.get() == guild_id.get() {
903                            everyone_allow_overwrites = overwrite.allow;
904                            everyone_deny_overwrites = overwrite.deny;
905                        } else if member_roles.contains(&role_id) {
906                            roles_allow_overwrites.push(overwrite.allow);
907                            roles_deny_overwrites.push(overwrite.deny);
908                        }
909                    },
910                }
911            }
912        }
913
914        calculate_permissions(CalculatePermissions {
915            is_guild_owner: member_user_id == guild_owner_id,
916            everyone_permissions: if let Some(role) = guild_roles.get(&RoleId::new(guild_id.get()))
917            {
918                role.permissions
919            } else {
920                error!("@everyone role missing in {}", guild_id);
921                Permissions::empty()
922            },
923            user_roles_permissions: member_roles
924                .iter()
925                .map(|role_id| {
926                    if let Some(role) = guild_roles.get(role_id) {
927                        role.permissions
928                    } else {
929                        warn!(
930                            "{} on {} has non-existent role {:?}",
931                            member_user_id, guild_id, role_id
932                        );
933                        Permissions::empty()
934                    }
935                })
936                .collect(),
937            everyone_allow_overwrites,
938            everyone_deny_overwrites,
939            roles_allow_overwrites,
940            roles_deny_overwrites,
941            member_allow_overwrites,
942            member_deny_overwrites,
943        })
944    }

987struct CalculatePermissions {
988    /// Whether the guild member is the guild owner
989    pub is_guild_owner: bool,
990    /// Base permissions given to @everyone (guild level)
991    pub everyone_permissions: Permissions,
992    /// Permissions allowed to a user by their roles (guild level)
993    pub user_roles_permissions: Vec<Permissions>,
994    /// Overwrites that deny permissions for @everyone (channel level)
995    pub everyone_allow_overwrites: Permissions,
996    /// Overwrites that allow permissions for @everyone (channel level)
997    pub everyone_deny_overwrites: Permissions,
998    /// Overwrites that deny permissions for specific roles (channel level)
999    pub roles_allow_overwrites: Vec<Permissions>,
1000    /// Overwrites that allow permissions for specific roles (channel level)
1001    pub roles_deny_overwrites: Vec<Permissions>,
1002    /// Member-specific overwrites that deny permissions (channel level)
1003    pub member_allow_overwrites: Permissions,
1004    /// Member-specific overwrites that allow permissions (channel level)
1005    pub member_deny_overwrites: Permissions,
1006}

1025/// Translated from the pseudo code at https://docs.discord.com/developers/topics/permissions#permission-overwrites
1026///
1027/// The comments within this file refer to the above link
1028#[cfg(feature = "model")]
1029fn calculate_permissions(data: CalculatePermissions) -> Permissions {
1030    if data.is_guild_owner {
1031        return Permissions::all();
1032    }
1033
1034    // 1. Base permissions given to @everyone are applied at a guild level
1035    let mut permissions = data.everyone_permissions;
1036    // 2. Permissions allowed to a user by their roles are applied at a guild level
1037    for role_permission in data.user_roles_permissions {
1038        permissions |= role_permission;
1039    }
1040
1041    if permissions.contains(Permissions::ADMINISTRATOR) {
1042        return Permissions::all();
1043    }
1044
1045    // 3. Overwrites that deny permissions for @everyone are applied at a channel level
1046    permissions &= !data.everyone_deny_overwrites;
1047    // 4. Overwrites that allow permissions for @everyone are applied at a channel level
1048    permissions |= data.everyone_allow_overwrites;
1049
1050    // 5. Overwrites that deny permissions for specific roles are applied at a channel level
1051    let mut role_deny_permissions = Permissions::empty();
1052    for p in data.roles_deny_overwrites {
1053        role_deny_permissions |= p;
1054    }
1055    permissions &= !role_deny_permissions;
1056
1057    // 6. Overwrites that allow permissions for specific roles are applied at a channel level
1058    let mut role_allow_permissions = Permissions::empty();
1059    for p in data.roles_allow_overwrites {
1060        role_allow_permissions |= p;
1061    }
1062    permissions |= role_allow_permissions;
1063
1064    // 7. Member-specific overwrites that deny permissions are applied at a channel level
1065    permissions &= !data.member_deny_overwrites;
1066    // 8. Member-specific overwrites that allow permissions are applied at a channel level
1067    permissions |= data.member_allow_overwrites;
1068
1069    permissions
1070}

1082/// Takes a `&str` as `origin` and tests if either `word_a` or `word_b` is closer.
1083///
1084/// **Note**: Normally `word_a` and `word_b` are expected to contain `origin` as substring. If not,
1085/// using `closest_to_origin` would sort these the end.
1086#[cfg(feature = "model")]
1087fn closest_to_origin(origin: &str, word_a: &str, word_b: &str) -> std::cmp::Ordering {
1088    let value_a = match word_a.find(origin) {
1089        Some(value) => value + word_a.len(),
1090        None => return std::cmp::Ordering::Greater,
1091    };
1092
1093    let value_b = match word_b.find(origin) {
1094        Some(value) => value + word_b.len(),
1095        None => return std::cmp::Ordering::Less,
1096    };
1097
1098    value_a.cmp(&value_b)
1099}
