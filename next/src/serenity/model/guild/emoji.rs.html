6/// Represents a custom guild emoji, which can either be created using the API, or via an
7/// integration. Emojis created using the API only work within the guild it was created in.
8///
9/// [Discord docs](https://docs.discord.com/developers/resources/emoji#emoji-object).
10#[bool_to_bitflags::bool_to_bitflags]
11#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
12#[derive(Clone, Debug, serde::Deserialize, serde::Serialize)]
13#[non_exhaustive]
14pub struct Emoji {
15    /// Whether the emoji is animated.
16    #[serde(default)]
17    pub animated: bool,
18    /// Whether the emoji can be used. This may be false when the guild loses boosts, reducing the
19    /// emoji limit.
20    #[serde(default = "default_true")]
21    pub available: bool,
22    /// The Id of the emoji.
23    pub id: EmojiId,
24    /// The name of the emoji. It must be at least 2 characters long and can only contain
25    /// alphanumeric characters and underscores.
26    pub name: FixedString<u8>,
27    /// Whether the emoji is managed via an [`Integration`] service.
28    ///
29    /// [`Integration`]: super::Integration
30    #[serde(default)]
31    pub managed: bool,
32    /// Whether the emoji name needs to be surrounded by colons in order to be used by the client.
33    #[serde(default)]
34    pub require_colons: bool,
35    /// A list of [`Role`]s that are allowed to use the emoji. If there are no roles specified,
36    /// then usage is unrestricted.
37    ///
38    /// [`Role`]: super::Role
39    #[serde(default)]
40    pub roles: FixedArray<RoleId, u8>,
41    /// The user who created the emoji.
42    pub user: Option<User>,
43}

45#[cfg(feature = "model")]
46impl Emoji {
47    /// Generates a URL to the emoji's image.
48    ///
49    /// # Examples
50    ///
51    /// Print the direct link to the given emoji:
52    ///
53    /// ```rust,no_run
54    /// # use serenity::model::guild::Emoji;
55    /// #
56    /// # fn run(emoji: Emoji) {
57    /// // assuming emoji has been set already
58    /// println!("Direct link to emoji image: {}", emoji.url());
59    /// # }
60    /// ```
61    #[must_use]
62    pub fn url(&self) -> String {
63        let extension = if self.animated() { "gif" } else { "png" };
64        cdn!("/emojis/{}.{}", self.id, extension)
65    }
66}

68impl fmt::Display for Emoji {
69    /// Formats the emoji into a string that will cause Discord clients to render the emoji.
70    ///
71    /// This is in the format of either `<:NAME:EMOJI_ID>` for normal emojis, or
72    /// `<a:NAME:EMOJI_ID>` for animated emojis.
73    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
74        if self.animated() {
75            f.write_str("<a:")?;
76        } else {
77            f.write_str("<:")?;
78        }
79        f.write_str(&self.name)?;
80        fmt::Write::write_char(f, ':')?;
81        fmt::Display::fmt(&self.id, f)?;
82        fmt::Write::write_char(f, '>')
83    }
84}
