1/// Deserializes an optional string containing a valid integer as `Option<u64>`.
2///
3/// Used with `#[serde(with = "optional_string")]`.
4pub mod optional_string {
5    use std::fmt;
6    use std::marker::PhantomData;
7    use std::str::FromStr;
8
9    use serde::de::{Deserializer, Error, Visitor};
10    use serde::ser::Serializer;
11
12    // Workaround for https://github.com/LPGhatguy/nonmax/issues/17
13    pub(crate) trait TryFromU64
14    where
15        Self: Sized,
16    {
17        type Err: fmt::Display;
18        fn try_from_u64(value: u64) -> Result<Self, Self::Err>;
19    }
20
21    impl TryFromU64 for u64 {
22        type Err = std::convert::Infallible;
23        fn try_from_u64(value: u64) -> Result<Self, Self::Err> {
24            Ok(value)
25        }
26    }
27
28    impl TryFromU64 for nonmax::NonMaxU64 {
29        type Err = nonmax::TryFromIntError;
30        fn try_from_u64(value: u64) -> Result<Self, Self::Err> {
31            Self::try_from(value)
32        }
33    }
34
35    impl TryFromU64 for nonmax::NonMaxU32 {
36        type Err = nonmax::TryFromIntError;
37        fn try_from_u64(value: u64) -> Result<Self, Self::Err> {
38            Self::try_from(u32::try_from(value)?)
39        }
40    }
41
42    pub fn deserialize<'de, D, T>(deserializer: D) -> Result<Option<T>, D::Error>
43    where
44        D: Deserializer<'de>,
45        T: FromStr + TryFromU64,
46        <T as FromStr>::Err: fmt::Display,
47    {
48        deserializer.deserialize_option(OptionalStringVisitor::<T>(PhantomData))
49    }
50
51    #[expect(clippy::ref_option)]
52    pub fn serialize<S: Serializer>(
53        value: &Option<impl ToString>,
54        serializer: S,
55    ) -> Result<S::Ok, S::Error> {
56        match value {
57            Some(value) => serializer.serialize_some(&value.to_string()),
58            None => serializer.serialize_none(),
59        }
60    }
61
62    struct OptionalStringVisitor<T>(PhantomData<T>);
63
64    impl<'de, T> Visitor<'de> for OptionalStringVisitor<T>
65    where
66        T: FromStr + TryFromU64,
67        <T as FromStr>::Err: fmt::Display,
68    {
69        type Value = Option<T>;
70
71        fn expecting(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
72            formatter.write_str("an optional integer or a string with a valid number inside")
73        }
74
75        fn visit_some<D: Deserializer<'de>>(
76            self,
77            deserializer: D,
78        ) -> Result<Self::Value, D::Error> {
79            deserializer.deserialize_any(OptionalStringVisitor(PhantomData))
80        }
81
82        fn visit_none<E: Error>(self) -> Result<Self::Value, E> {
83            Ok(None)
84        }
85
86        fn visit_u64<E: Error>(self, val: u64) -> Result<Self::Value, E> {
87            T::try_from_u64(val).map(Some).map_err(Error::custom)
88        }
89
90        fn visit_str<E: Error>(self, string: &str) -> Result<Self::Value, E> {
91            string.parse().map(Some).map_err(Error::custom)
92        }
93    }
94}

97mod tests {
98    use serde_json::json;
99
100    use super::optional_string;
101    use crate::model::utils::assert_json;
102
103    #[test]
104    fn optional_string_module() {
105        #[derive(Debug, PartialEq, Deserialize, Serialize)]
106        struct T {
107            #[serde(with = "optional_string")]
108            opt: Option<u64>,
109        }
110
111        let value = T {
112            opt: Some(12345),
113        };
114
115        assert_json(&value, json!({"opt": "12345"}));
116    }
