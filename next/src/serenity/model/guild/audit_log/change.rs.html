23impl<'de> serde::Deserialize<'de> for EntityType {
24    fn deserialize<D: serde::de::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
25        Ok(StrOrInt::deserialize(deserializer)?.into_enum(Self::Str, Self::Int))

29macro_rules! generate_change {
30    ( $(
31        $( #[doc = $doc:literal] )?
32        $key:literal => $name:ident ($type:ty),
33    )* ) => {
34        #[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
35        #[derive(Debug, PartialEq, Eq, Serialize, Deserialize, Clone)]
36        #[non_exhaustive]
37        #[serde(tag = "key")]
38        #[serde(rename_all = "snake_case")]
39        pub enum Change {
40            $(
41                $( #[doc = $doc] )?
42                $name {
43                    #[serde(skip_serializing_if = "Option::is_none")]
44                    #[serde(rename = "old_value")]
45                    old: Option<$type>,
46                    #[serde(skip_serializing_if = "Option::is_none")]
47                    #[serde(rename = "new_value")]
48                    new: Option<$type>,
49                },
50            )*
51
52            /* These changes are special because their variant names do not match their keys. */
53
54            /// Role was added to a member.
55            #[serde(rename = "$add")]
56            RolesAdded {
57                #[serde(skip_serializing_if = "Option::is_none")]
58                #[serde(rename = "old_value")]
59                old: Option<FixedArray<AffectedRole>>,
60                #[serde(skip_serializing_if = "Option::is_none")]
61                #[serde(rename = "new_value")]
62                new: Option<FixedArray<AffectedRole>>,
63            },
64            /// Role was removed to a member.
65            #[serde(rename = "$remove")]
66            RolesRemove {
67                #[serde(skip_serializing_if = "Option::is_none")]
68                #[serde(rename = "old_value")]
69                old: Option<FixedArray<AffectedRole>>,
70                #[serde(skip_serializing_if = "Option::is_none")]
71                #[serde(rename = "new_value")]
72                new: Option<FixedArray<AffectedRole>>,
73            },
74
75            /// Unknown key was changed.
76            Other {
77                name: FixedString,
78                #[serde(skip_serializing_if = "Option::is_none")]
79                #[serde(rename = "old_value")]
80                old_value: Option<Value>,
81                #[serde(skip_serializing_if = "Option::is_none")]
82                #[serde(rename = "new_value")]
83                new_value: Option<Value>,
84            },
85
86            /// Unknown key was changed and was invalid
87            #[serde(other)]
88            Unknown
89        }
90
91        impl Change {
92            #[must_use]
93            pub fn key(&self) -> &str {
94                match self {
95                    $( Self::$name { .. } => $key, )*
96                    Self::RolesAdded { .. } => "$add",
97                    Self::RolesRemove { .. } => "$remove",
98                    Self::Other { name, .. } => name,
99                    Self::Unknown => "unknown",
100                }
101            }
102        }
103    };
104}

264    #[test]
265    fn afk_channel_id_variant() {
266        let value = Change::AfkChannelId {
267            old: Some(ChannelId::new(1)),
268            new: Some(ChannelId::new(2)),
269        };
270        assert_json(&value, json!({"key": "afk_channel_id", "old_value": "1", "new_value": "2"}));
271    }
272
273    #[test]
274    fn skip_serializing_if_none() {
275        let value = Change::AfkChannelId {
276            old: None,
277            new: Some(ChannelId::new(2)),
278        };
279        assert_json(&value, json!({"key": "afk_channel_id", "new_value": "2"}));
280        let value = Change::AfkChannelId {
281            old: Some(ChannelId::new(1)),
282            new: None,
283        };
284        assert_json(&value, json!({"key": "afk_channel_id", "old_value": "1"}));
285    }
286
287    #[test]
288    fn entity_type_variant() {
289        let value = Change::Type {
290            old: Some(EntityType::Int(123)),
291            new: Some(EntityType::Str(FixedString::from_static_trunc("discord"))),
292        };
293        assert_json(&value, json!({"key": "type", "old_value": 123, "new_value": "discord"}));
294    }
