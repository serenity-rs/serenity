14/// Determines the action that was done on a target.
15///
16/// [Discord docs](https://docs.discord.com/developers/resources/audit-log#audit-log-entry-object-audit-log-events).
17#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
18#[derive(Clone, Copy, Debug)]
19#[non_exhaustive]
20pub enum Action {
21    GuildUpdate,
22    Channel(ChannelAction),
23    ChannelOverwrite(ChannelOverwriteAction),
24    Member(MemberAction),
25    Role(RoleAction),
26    Invite(InviteAction),
27    Webhook(WebhookAction),
28    Emoji(EmojiAction),
29    Message(MessageAction),
30    Integration(IntegrationAction),
31    StageInstance(StageInstanceAction),
32    Sticker(StickerAction),
33    ScheduledEvent(ScheduledEventAction),
34    Thread(ThreadAction),
35    AutoMod(AutoModAction),
36    CreatorMonetization(CreatorMonetizationAction),
37    VoiceChannelStatus(VoiceChannelStatusAction),
38    Unknown(u16),
39}
40
41impl Action {
42    #[must_use]
43    pub const fn num(self) -> u16 {
44        match self {
45            Self::GuildUpdate => 1,
46            Self::Channel(x) => x as u16,
47            Self::ChannelOverwrite(x) => x as u16,
48            Self::Member(x) => x as u16,
49            Self::Role(x) => x as u16,
50            Self::Invite(x) => x as u16,
51            Self::Webhook(x) => x as u16,
52            Self::Emoji(x) => x as u16,
53            Self::Message(x) => x as u16,
54            Self::Integration(x) => x as u16,
55            Self::StageInstance(x) => x as u16,
56            Self::Sticker(x) => x as u16,
57            Self::ScheduledEvent(x) => x as u16,
58            Self::Thread(x) => x as u16,
59            Self::AutoMod(x) => x as u16,
60            Self::CreatorMonetization(x) => x as u16,
61            Self::VoiceChannelStatus(x) => x as u16,
62            Self::Unknown(x) => x,
63        }
64    }
65
66    #[must_use]
67    pub fn from_value(value: u16) -> Action {
68        match value {
69            1 => Action::GuildUpdate,
70            10 => Action::Channel(ChannelAction::Create),
71            11 => Action::Channel(ChannelAction::Update),
72            12 => Action::Channel(ChannelAction::Delete),
73            13 => Action::ChannelOverwrite(ChannelOverwriteAction::Create),
74            14 => Action::ChannelOverwrite(ChannelOverwriteAction::Update),
75            15 => Action::ChannelOverwrite(ChannelOverwriteAction::Delete),
76            20 => Action::Member(MemberAction::Kick),
77            21 => Action::Member(MemberAction::Prune),
78            22 => Action::Member(MemberAction::BanAdd),
79            23 => Action::Member(MemberAction::BanRemove),
80            24 => Action::Member(MemberAction::Update),
81            25 => Action::Member(MemberAction::RoleUpdate),
82            26 => Action::Member(MemberAction::MemberMove),
83            27 => Action::Member(MemberAction::MemberDisconnect),
84            28 => Action::Member(MemberAction::BotAdd),
85            30 => Action::Role(RoleAction::Create),
86            31 => Action::Role(RoleAction::Update),
87            32 => Action::Role(RoleAction::Delete),
88            40 => Action::Invite(InviteAction::Create),
89            41 => Action::Invite(InviteAction::Update),
90            42 => Action::Invite(InviteAction::Delete),
91            50 => Action::Webhook(WebhookAction::Create),
92            51 => Action::Webhook(WebhookAction::Update),
93            52 => Action::Webhook(WebhookAction::Delete),
94            60 => Action::Emoji(EmojiAction::Create),
95            61 => Action::Emoji(EmojiAction::Update),
96            62 => Action::Emoji(EmojiAction::Delete),
97            72 => Action::Message(MessageAction::Delete),
98            73 => Action::Message(MessageAction::BulkDelete),
99            74 => Action::Message(MessageAction::Pin),
100            75 => Action::Message(MessageAction::Unpin),
101            80 => Action::Integration(IntegrationAction::Create),
102            81 => Action::Integration(IntegrationAction::Update),
103            82 => Action::Integration(IntegrationAction::Delete),
104            83 => Action::StageInstance(StageInstanceAction::Create),
105            84 => Action::StageInstance(StageInstanceAction::Update),
106            85 => Action::StageInstance(StageInstanceAction::Delete),
107            90 => Action::Sticker(StickerAction::Create),
108            91 => Action::Sticker(StickerAction::Update),
109            92 => Action::Sticker(StickerAction::Delete),
110            100 => Action::ScheduledEvent(ScheduledEventAction::Create),
111            101 => Action::ScheduledEvent(ScheduledEventAction::Update),
112            102 => Action::ScheduledEvent(ScheduledEventAction::Delete),
113            110 => Action::Thread(ThreadAction::Create),
114            111 => Action::Thread(ThreadAction::Update),
115            112 => Action::Thread(ThreadAction::Delete),
116            140 => Action::AutoMod(AutoModAction::RuleCreate),
117            141 => Action::AutoMod(AutoModAction::RuleUpdate),
118            142 => Action::AutoMod(AutoModAction::RuleDelete),
119            143 => Action::AutoMod(AutoModAction::BlockMessage),
120            144 => Action::AutoMod(AutoModAction::FlagToChannel),
121            145 => Action::AutoMod(AutoModAction::UserCommunicationDisabled),
122            146 => Action::AutoMod(AutoModAction::QuarantineUser),
123            150 => Action::CreatorMonetization(CreatorMonetizationAction::RequestCreated),
124            151 => Action::CreatorMonetization(CreatorMonetizationAction::TermsAccepted),
125            192 => Action::VoiceChannelStatus(VoiceChannelStatusAction::StatusUpdate),
126            193 => Action::VoiceChannelStatus(VoiceChannelStatusAction::StatusDelete),
127            _ => Action::Unknown(value),
128        }
129    }
130}

132// Manual impl needed to emulate integer enum tags
133impl<'de> Deserialize<'de> for Action {
134    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> StdResult<Self, D::Error> {
135        let value = u16::deserialize(deserializer)?;
136        Ok(Action::from_value(value))
137    }
138}

140impl Serialize for Action {
141    fn serialize<S: Serializer>(&self, serializer: S) -> StdResult<S::Ok, S::Error> {
142        serializer.serialize_u16(self.num())
143    }
