5/// Information about a guild scheduled event.
6///
7/// [Discord docs](https://docs.discord.com/developers/resources/guild-scheduled-event#guild-scheduled-event-object).
8#[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
9#[derive(Clone, Debug, Deserialize, Serialize)]
10#[non_exhaustive]
11pub struct ScheduledEvent {
12    /// The Id of the scheduled event.
13    pub id: ScheduledEventId,
14    /// The Id of the guild that the event belongs to.
15    pub guild_id: GuildId,
16    /// The Id of the channel that the event belongs to, if any.
17    pub channel_id: Option<ChannelId>,
18    /// The Id of the User that created the scheduled event.
19    ///
20    /// Only `None` for events created before October 25th, 2021.
21    pub creator_id: Option<UserId>,
22    /// The name of the scheduled event.
23    pub name: FixedString,
24    /// The description of the scheduled event, if any.
25    pub description: Option<FixedString>,
26    /// The event's starting time.
27    #[serde(rename = "scheduled_start_time")]
28    pub start_time: Timestamp,
29    /// The event's ending time; optional.
30    #[serde(rename = "scheduled_end_time")]
31    pub end_time: Option<Timestamp>,
32    /// The privacy level of the scheduled event.
33    pub privacy_level: ScheduledEventPrivacyLevel,
34    /// The event's status; either Scheduled, Active, Completed, or Canceled.
35    pub status: ScheduledEventStatus,
36    /// The User that created the event.
37    ///
38    /// Only `None` for events created before October 25th, 2021.
39    pub creator: Option<User>,
40    /// The type of the event, indicating if it will take place in a Stage Instance, a Voice
41    /// Channel, or at some External location.
42    #[serde(rename = "entity_type")]
43    pub kind: ScheduledEventType,
44    /// The id of an entity associated with a guild scheduled event.
45    pub entity_id: Option<GenericId>,
46    /// Optional event location, only required for External events.
47    #[serde(rename = "entity_metadata")]
48    pub metadata: Option<ScheduledEventMetadata>,
49    /// Number of users interested in the event.
50    ///
51    /// Only populated if `with_user_count` is set to true provided when calling
52    /// [`GuildId::scheduled_event`] or [`GuildId::scheduled_events`].
53    pub user_count: Option<NonMaxU64>,
54    /// The hash of the event's cover image, if present.
55    pub image: Option<ImageHash>,
56}

58impl extract_map::ExtractKey<ScheduledEventId> for ScheduledEvent {
59    fn extract_key(&self) -> &ScheduledEventId {
60        &self.id
61    }
62}
