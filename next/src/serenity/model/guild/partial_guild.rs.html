164impl PartialGuild {
165    /// Edits the current guild with new data where specified.
166    ///
167    /// **Note**: Requires the [Manage Guild] permission.
168    ///
169    /// # Errors
170    ///
171    /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is given.
172    ///
173    /// [Manage Guild]: Permissions::MANAGE_GUILD
174    pub async fn edit(&mut self, http: &Http, builder: EditGuild<'_>) -> Result<()> {
175        *self = self.id.edit(http, builder).await?;
176        Ok(())
177    }
178
179    /// Gets a partial amount of guild data by its Id.
180    ///
181    /// # Errors
182    ///
183    /// Returns [`Error::Http`] if the current user is not
184    /// in the guild.
185    pub async fn get(cache_http: impl CacheHttp, guild_id: GuildId) -> Result<PartialGuild> {
186        guild_id.to_partial_guild(cache_http).await
187    }

189    /// Calculate a [`Member`]'s permissions in the guild.
190    ///
191    /// You likely want to use PartialGuild::user_permissions_in instead as this function does not
192    /// consider permission overwrites.
193    #[must_use]
194    pub fn member_permissions(&self, member: &Member) -> Permissions {
195        Guild::user_permissions_in_(
196            None,
197            member.user.id,
198            &member.roles,
199            self.id,
200            &self.roles,
201            self.owner_id,
202        )
203    }
204
205    /// Calculate a [`PartialMember`]'s permissions in the guild.
206    ///
207    /// You likely want to use PartialGuild::partial_member_permissions_in instead as this function
208    /// does not consider permission overwrites.
209    ///
210    /// # Panics
211    ///
212    /// Panics if the passed [`UserId`] does not match the [`PartialMember`] id, if user is Some.
213    #[must_use]
214    pub fn partial_member_permissions(
215        &self,
216        member_id: UserId,
217        member: &PartialMember,
218    ) -> Permissions {
219        if let Some(user) = &member.user {
220            assert_eq!(user.id, member_id, "User::id does not match provided PartialMember");
221        }
222
223        Guild::user_permissions_in_(
224            None,
225            member_id,
226            &member.roles,
227            self.id,
228            &self.roles,
229            self.owner_id,
230        )
231    }
232
233    /// Gets the highest role a [`Member`] of this Guild has.
234    ///
235    /// Returns None if the member has no roles or the member from this guild.
236    #[must_use]
237    pub fn member_highest_role(&self, member: &Member) -> Option<&Role> {
238        Guild::_member_highest_role_in(&self.roles, member)
239    }
240
241    /// See [`Guild::greater_member_hierarchy`] for more information.
242    ///
243    /// Note that unlike [`Guild::greater_member_hierarchy`], this method requires a [`Member`] as
244    /// member data is not available on a [`PartialGuild`].
245    #[must_use]
246    pub fn greater_member_hierarchy(&self, lhs: &Member, rhs: &Member) -> Option<UserId> {
247        let lhs_highest_role = self.member_highest_role(lhs);
248        let rhs_highest_role = self.member_highest_role(rhs);
249
250        Guild::_greater_member_hierarchy_in(
251            lhs_highest_role,
252            rhs_highest_role,
253            self.owner_id,
254            lhs,
255            rhs,
256        )
257    }

259    /// Calculate a [`PartialMember`]'s permissions in a given channel in a guild.
260    ///
261    /// # Panics
262    ///
263    /// Panics if the passed [`UserId`] does not match the [`PartialMember`] id, if user is Some.
264    #[must_use]
265    pub fn partial_member_permissions_in(
266        &self,
267        channel: &GuildChannel,
268        member_id: UserId,
269        member: &PartialMember,
270    ) -> Permissions {
271        if let Some(user) = &member.user {
272            assert_eq!(user.id, member_id, "User::id does not match provided PartialMember");
273        }
274
275        Guild::user_permissions_in_(
276            Some(channel),
277            member_id,
278            &member.roles,
279            self.id,
280            &self.roles,
281            self.owner_id,
282        )
283    }

285    /// Returns a formatted URL of the guild's icon, if the guild has an icon.
286    #[must_use]
287    pub fn icon_url(&self) -> Option<String> {
288        icon_url(self.id, self.icon.as_ref())

291    /// Returns a formatted URL of the guild's banner, if the guild has a banner.
292    #[must_use]
293    pub fn banner_url(&self) -> Option<String> {
294        self.banner.as_ref().map(|banner| cdn!("/banners/{}/{}.webp", self.id, banner))
295    }

297    /// Calculate a [`Member`]'s permissions in a given channel in the guild.
298    #[must_use]

300        Guild::user_permissions_in_(
301            Some(channel),

303            &member.roles,
304            self.id,
305            &self.roles,
306            self.owner_id,
307        )
308    }

310    /// Returns the formatted URL of the guild's splash image, if one exists.
311    #[must_use]
312    pub fn splash_url(&self) -> Option<String> {
313        self.splash.as_ref().map(|splash| cdn!("/splashes/{}/{}.webp?size=4096", self.id, splash))
314    }

316    /// Obtain a reference to a role by its name.
317    ///
318    /// **Note**: If two or more roles have the same name, obtained reference will be one of them.
319    ///
320    /// # Examples
321    ///
322    /// Obtain a reference to a [`Role`] by its name.
323    ///
324    /// ```rust,no_run
325    /// # use serenity::model::prelude::*;
326    /// # use serenity::prelude::*;
327    ///
328    /// # #[cfg(feature = "cache")]
329    /// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
330    /// # let cache: serenity::cache::Cache = unimplemented!();
331    /// # let msg: Message = unimplemented!();
332    ///
333    /// if let Some(guild_id) = msg.guild_id {
334    ///     if let Some(guild) = guild_id.to_guild_cached(&cache) {
335    ///         if let Some(role) = guild.role_by_name("role_name") {
336    ///             println!("Obtained role's reference: {:?}", role);
337    ///         }
338    ///     }
339    /// }
340    ///
341    /// # Ok(())
342    /// # }
343    /// ```
344    #[must_use]
345    pub fn role_by_name(&self, role_name: &str) -> Option<&Role> {
346        self.roles.iter().find(|role| role_name == &*role.name)
347    }
