3/// Preview [`Guild`] information.
4///
5/// [Discord docs](https://docs.discord.com/developers/resources/guild#guild-preview-object).
6///
7/// [`Guild`]: super::Guild
8#[derive(Clone, Debug, Deserialize, Serialize)]
9#[non_exhaustive]
10pub struct GuildPreview {
11    /// The guild Id.
12    pub id: GuildId,
13    /// The guild name.
14    pub name: FixedString,
15    /// The guild icon hash if it has one.
16    pub icon: Option<ImageHash>,
17    /// The guild splash hash if it has one.
18    pub splash: Option<ImageHash>,
19    /// The guild discovery splash hash it it has one.
20    pub discovery_splash: Option<ImageHash>,
21    /// The custom guild emojis.
22    pub emojis: FixedArray<Emoji>,
23    /// The guild features. See [`Guild::features`]
24    ///
25    /// [`Guild::features`]: super::Guild::features
26    pub features: FixedArray<String>,
27    /// Approximate number of members in this guild.
28    pub approximate_member_count: u64,
29    /// Approximate number of online members in this guild.
30    pub approximate_presence_count: u64,
31    /// The description for the guild, if the guild has the `DISCOVERABLE` feature.
32    pub description: Option<FixedString>,
33    /// Custom guild stickers.
34    pub stickers: FixedArray<Sticker>,
35}
