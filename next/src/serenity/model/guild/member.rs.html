16/// Information about a member of a guild.
17///
18/// [Discord docs](https://docs.discord.com/developers/resources/guild#guild-member-object),

40    pub premium_since: Option<Timestamp>,

79    /// Flags for a guild member.
80    ///
81    /// [Discord docs](https://docs.discord.com/developers/resources/guild#guild-member-object-guild-member-flags).
82    #[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
83    #[derive(Copy, Clone, Default, Debug, Eq, Hash, PartialEq)]
84    pub struct GuildMemberFlags: u32 {
85        /// Member has left and rejoined the guild. Not editable
86        const DID_REJOIN = 1 << 0;
87        /// Member has completed onboarding. Not editable
88        const COMPLETED_ONBOARDING = 1 << 1;
89        /// Member is exempt from guild verification requirements. Editable
90        const BYPASSES_VERIFICATION = 1 << 2;
91        /// Member has started onboarding. Not editable
92        const STARTED_ONBOARDING = 1 << 3;
93        /// Member is a guest and can only access the voice channel they were invited to. Not
94        /// editable
95        const IS_GUEST = 1 << 4;
96        /// Member has started Server Guide new member actions. Not editable
97        const STARTED_HOME_ACTIONS = 1 << 5;
98        /// Member has completed Server Guide new member actions. Not editable
99        const COMPLETED_HOME_ACTIONS = 1 << 6;
100        /// Member's username, display name, or nickname is blocked by AutoMod. Not editable
101        const AUTOMOD_QUARANTINED_USERNAME = 1 << 7;
102        /// Member has dismissed the DM settings upsell. Not editable
103        const DM_SETTINGS_UPSELL_ACKNOWLEDGED = 1 << 9;
104        /// Member's guild tag is blocked by AutoMod. Not editable
105        const AUTOMOD_QUARANTINED_GUILD_TAG = 1 << 10;
106    }

162    /// Determines the member's colour.
163    #[cfg(feature = "cache")]
164    pub fn colour(&self, cache: &Cache) -> Option<Colour> {
165        let guild = cache.guild(self.guild_id)?;
166
167        let mut roles = self
168            .roles
169            .iter()
170            .filter_map(|role_id| guild.roles.get(role_id))
171            .collect::<Vec<&Role>>();
172
173        roles.sort_by_key(|&b| Reverse(b));
174
175        let default = Colour::default();
176
177        roles.iter().find(|r| r.colour.0 != default.0).map(|r| r.colour)
178    }

180    /// Returns the "default channel" of the guild for the member. (This returns the first channel
181    /// that can be read by the member, if there isn't one returns [`None`])
182    #[cfg(feature = "cache")]
183    pub fn default_channel(&self, cache: &Cache) -> Option<GuildChannel> {
184        let guild = self.guild_id.to_guild_cached(cache)?;
185
186        let member = guild.members.get(&self.user.id)?;
187
188        for channel in &guild.channels {
189            if channel.base.kind != ChannelType::Category
190                && guild.user_permissions_in(channel, member).view_channel()
191            {
192                return Some(channel.clone());
193            }
194        }
195
196        None
197    }

518impl From<PartialMember> for Member {
519    fn from(partial: PartialMember) -> Self {
520        let (pending, deaf, mute) = (partial.pending(), partial.deaf(), partial.mute());
521        let mut member = Member {
522            __generated_flags: MemberGeneratedFlags::empty(),
523            user: partial.user.unwrap_or_default(),
524            nick: partial.nick,
525            avatar: partial.avatar,
526            banner: partial.banner,
527            roles: partial.roles,
528            joined_at: partial.joined_at,
529            premium_since: partial.premium_since,
530            flags: GuildMemberFlags::default(),
531            permissions: partial.permissions,
532            communication_disabled_until: None,
533            guild_id: partial.guild_id.unwrap_or_default(),
534            unusual_dm_activity_until: partial.unusual_dm_activity_until,
535            avatar_decoration_data: partial.avatar_decoration_data,
536            collectibles: partial.collectibles,
537        };
538
539        member.set_pending(pending);
540        member.set_deaf(deaf);
541        member.set_mute(mute);
542        member
543    }
544}
