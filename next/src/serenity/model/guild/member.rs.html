73    /// Flags for a guild member.
74    ///
75    /// [Discord docs](https://docs.discord.com/developers/resources/guild#guild-member-object-guild-member-flags).
76    #[cfg_attr(feature = "typesize", derive(typesize::derive::TypeSize))]
77    #[derive(Copy, Clone, Default, Debug, Eq, Hash, PartialEq)]
78    pub struct GuildMemberFlags: u32 {
79        /// Member has left and rejoined the guild. Not editable
80        const DID_REJOIN = 1 << 0;
81        /// Member has completed onboarding. Not editable
82        const COMPLETED_ONBOARDING = 1 << 1;
83        /// Member is exempt from guild verification requirements. Editable
84        const BYPASSES_VERIFICATION = 1 << 2;
85        /// Member has started onboarding. Not editable
86        const STARTED_ONBOARDING = 1 << 3;
87        /// Member is a guest and can only access the voice channel they were invited to. Not
88        /// editable
89        const IS_GUEST = 1 << 4;
90        /// Member has started Server Guide new member actions. Not editable
91        const STARTED_HOME_ACTIONS = 1 << 5;
92        /// Member has completed Server Guide new member actions. Not editable
93        const COMPLETED_HOME_ACTIONS = 1 << 6;
94        /// Member's username, display name, or nickname is blocked by AutoMod. Not editable
95        const AUTOMOD_QUARANTINED_USERNAME = 1 << 7;
96        /// Member has dismissed the DM settings upsell. Not editable
97        const DM_SETTINGS_UPSELL_ACKNOWLEDGED = 1 << 9;
98        /// Member's guild tag is blocked by AutoMod. Not editable
99        const AUTOMOD_QUARANTINED_GUILD_TAG = 1 << 10;
100    }

156    /// Determines the member's colour.
157    #[cfg(feature = "cache")]
158    pub fn colour(&self, cache: &Cache) -> Option<Colour> {
159        let guild = cache.guild(self.guild_id)?;
160
161        let mut roles = self
162            .roles
163            .iter()
164            .filter_map(|role_id| guild.roles.get(role_id))
165            .collect::<Vec<&Role>>();
166
167        roles.sort_by_key(|&b| Reverse(b));
168
169        let default = Colour::default();
170
171        roles.iter().find(|r| r.colour.0 != default.0).map(|r| r.colour)
172    }

174    /// Returns the "default channel" of the guild for the member. (This returns the first channel
175    /// that can be read by the member, if there isn't one returns [`None`])
176    #[cfg(feature = "cache")]
177    pub fn default_channel(&self, cache: &Cache) -> Option<GuildChannel> {
178        let guild = self.guild_id.to_guild_cached(cache)?;
179
180        let member = guild.members.get(&self.user.id)?;
181
182        for channel in &guild.channels {
183            if channel.base.kind != ChannelType::Category
184                && guild.user_permissions_in(channel, member).view_channel()
185            {
186                return Some(channel.clone());
187            }
188        }
189
190        None
191    }

507impl From<PartialMember> for Member {
508    fn from(partial: PartialMember) -> Self {
509        let (pending, deaf, mute) = (partial.pending(), partial.deaf(), partial.mute());
510        let mut member = Member {
511            __generated_flags: MemberGeneratedFlags::empty(),
512            user: partial.user.unwrap_or_default(),
513            nick: partial.nick,
514            avatar: partial.avatar,
515            banner: partial.banner,
516            roles: partial.roles,
517            joined_at: partial.joined_at,
518            premium_since: partial.premium_since,
519            flags: GuildMemberFlags::default(),
520            permissions: partial.permissions,
521            communication_disabled_until: None,
522            guild_id: partial.guild_id.unwrap_or_default(),
523            unusual_dm_activity_until: partial.unusual_dm_activity_until,
524            avatar_decoration_data: partial.avatar_decoration_data,
525        };
526
527        member.set_pending(pending);
528        member.set_deaf(deaf);
529        member.set_mute(mute);
530        member
531    }
532}
