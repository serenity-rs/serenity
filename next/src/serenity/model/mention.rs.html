13/// Allows something - such as a channel or role - to be mentioned in a message.
14pub trait Mentionable {
15    /// Creates a [`Mention`] that will be able to notify or create a link to the item.
16    ///
17    /// [`Mention`] implements [`Display`], so [`ToString::to_string()`] can be called on it, or
18    /// inserted directly into a [`format_args!`] type of macro.
19    ///
20    /// [`Display`]: fmt::Display
21    ///
22    /// # Examples
23    ///
24    /// ```
25    /// # #[cfg(feature = "client")] {
26    /// # use serenity::builder::CreateMessage;
27    /// # use serenity::model::guild::Member;
28    /// # use serenity::model::channel::GuildChannel;
29    /// # use serenity::model::id::ChannelId;
30    /// # use serenity::prelude::Context;
31    /// # use serenity::Error;
32    /// use serenity::model::mention::Mentionable;
33    /// async fn greet(
34    ///     ctx: Context,
35    ///     member: Member,
36    ///     to_channel: GuildChannel,
37    ///     rules_channel: ChannelId,
38    /// ) -> Result<(), Error> {
39    ///     let builder = CreateMessage::new().content(format!(
40    ///         "Hi {member}, welcome to the server! \
41    ///         Please refer to {rules} for our code of conduct, \
42    ///         and enjoy your stay.",
43    ///         member = member.mention(),
44    ///         rules = rules_channel.mention(),
45    ///     ));
46    ///     to_channel.id.send_message(&ctx.http, builder).await?;
47    ///     Ok(())
48    /// }
49    /// # }
50    /// ```
51    /// ```
52    /// # use serenity::model::id::{RoleId, ChannelId, UserId};
53    /// use serenity::model::mention::Mentionable;
54    /// let user = UserId::new(1);
55    /// let channel = ChannelId::new(2);
56    /// let role = RoleId::new(3);
57    /// assert_eq!(
58    ///     "<@1> <#2> <@&3>",
59    ///     format!("{} {} {}", user.mention(), channel.mention(), role.mention(),),
60    /// )
61    /// ```
62    fn mention(&self) -> Mention;
63}

65/// A struct that represents some way to insert a notification, link, or emoji into a message.
66///
67/// [`Display`] is the primary way of utilizing a [`Mention`], either in a [`format_args!`] type of
68/// macro or with [`ToString::to_string()`]. A [`Mention`] is created using
69/// [`Mentionable::mention()`], or with [`From`]/[`Into`].
70///
71/// [`Display`]: fmt::Display
72///
73/// # Examples
74///
75/// ```
76/// # use serenity::model::id::{RoleId, GenericChannelId, UserId};
77/// use serenity::model::mention::Mention;
78/// let user = UserId::new(1);
79/// let channel = GenericChannelId::new(2);
80/// let role = RoleId::new(3);
81/// assert_eq!(
82///     "<@1> <#2> <@&3>",
83///     format!("{} {} {}", Mention::from(user), Mention::from(channel), Mention::from(role),),
84/// )
85/// ```
86#[derive(Clone, Copy, Debug)]
87pub enum Mention {
88    Channel(GenericChannelId),
89    Role(RoleId),
90    User(UserId),
91}

93macro_rules! mention {

95        impl From<$t> for Mention {

97                $e
98            }
99        }
100    )*};
101}

109impl fmt::Display for Mention {
110    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
111        f.write_str(&self.to_arraystring())
112    }
113}

115impl ToArrayString for Mention {
116    const MAX_LENGTH: usize = 20 + 4;
117    type ArrayString = ArrayString<{ Self::MAX_LENGTH }>;
118
119    fn to_arraystring(self) -> Self::ArrayString {
120        let mut out = Self::ArrayString::new();
121
122        match self {
123            Self::Channel(id) => aformat_into!(out, "<#{id}>"),
124            Self::Role(id) => aformat_into!(out, "<@&{id}>"),
125            Self::User(id) => aformat_into!(out, "<@{id}>"),
126        }
127
128        out
129    }
130}

148#[cfg(all(feature = "model", feature = "utils"))]
149impl FromStr for Mention {
150    type Err = MentionParseError;
151
152    fn from_str(s: &str) -> StdResult<Self, Self::Err> {
153        let m = if let Some(id) = utils::parse_channel_mention(s) {
154            id.mention()
155        } else if let Some(id) = utils::parse_role_mention(s) {
156            id.mention()
157        } else if let Some(id) = utils::parse_user_mention(s) {
158            id.mention()
159        } else {
160            return Err(MentionParseError::InvalidMention);
161        };
162
163        Ok(m)
164    }
165}
