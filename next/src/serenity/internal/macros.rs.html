30/// The `enum_number!` macro generates `From` implementations to convert between values and the
31/// enum which can then be utilized by `serde` with `#[serde(from = "u8", into = "u8")]`.
32///
33/// When defining the enum like this:
34/// ```ignore
35/// enum_number! {
36///     /// The `Foo` enum
37///     #[derive(Clone, Copy, Deserialize, Serialize)]
38///     #[serde(from = "u8", into = "u8")]
39///     pub enum Foo {
40///         /// First
41///         Aah = 1,
42///         /// Second
43///         Bar = 2,
44///         _ => Unknown(u8),
45///     }
46/// }
47/// ```
48///
49/// Code like this will be generated:
50///
51/// ```
52/// # use serde::{Deserialize, Serialize};
53/// #
54/// /// The `Foo` enum
55/// #[derive(Clone, Copy, Deserialize, Serialize)]
56/// #[serde(from = "u8", into = "u8")]
57/// pub enum Foo {
58///     /// First
59///     Aah,
60///     /// Second,
61///     Bar,
62///     /// Variant value is unknown.
63///     Unknown(u8),
64/// }
65///
66/// impl From<u8> for Foo {
67///     fn from(value: u8) -> Self {
68///         match value {
69///             1 => Self::Aah,
70///             2 => Self::Bar,
71///             unknown => Self::Unknown(unknown),
72///         }
73///     }
74/// }
75///
76/// impl From<Foo> for u8 {
77///     fn from(value: Foo) -> Self {
78///         match value {
79///             Foo::Aah => 1,
80///             Foo::Bar => 2,
81///             Foo::Unknown(unknown) => unknown,
82///         }
83///     }
84/// }
85/// ```
86macro_rules! enum_number {
87    (
88        $(#[$outer:meta])*
89        $(#[<default> = $default:literal])?
90        $vis:vis enum $Enum:ident {
91            $(
92                $(#[doc = $doc:literal])*
93                $(#[cfg $($cfg:tt)*])?
94                $Variant:ident = $value:literal,
95            )*
96            _ => Unknown($T:ty),
97        }
98    ) => {
99        $(#[$outer])*
100        $vis struct $Enum (pub $T);
101
102        $(
103            impl Default for $Enum {
104                fn default() -> Self {
105                    Self($default)
106                }
107            }
108        )?
109
110        #[allow(non_snake_case, non_upper_case_globals)]
111        #[allow(clippy::allow_attributes, reason = "Does not always trigger due to macro")]
112        impl $Enum {
113            $(
114                $(#[doc = $doc])*
115                $(#[cfg $($cfg)*])?
116                $vis const $Variant: Self = Self($value);
117            )*
118
119            /// Variant value is unknown.
120            #[must_use]
121            $vis const fn Unknown(val: $T) -> Self {
122                Self(val)
123            }
124
125            /// Get a string representation
126            #[must_use]
127            $vis const fn as_str(&self) -> &'static str {
128                match &self.0 {
129                    $(
130                        $value => stringify!($Variant),
131                    )*
132                    _ => "Unknown",
133                }
134            }
135        }
136    };
137}

139/// The macro forwards the generation to the `bitflags::bitflags!` macro and implements the default
140/// (de)serialization for Discord's bitmask values.
141///
142/// The flags are created with `T::from_bits_truncate` for the deserialized integer value.
143///
144/// Use the `bitflags::bitflags! macro directly if a different serde implementation is required.
145macro_rules! bitflags {
146    (
147        $(#[$outer:meta])*
148        $vis:vis struct $BitFlags:ident: $T:ty {
149            $(
150                $(#[$inner:ident $($args:tt)*])*
151                const $Flag:ident = $value:expr;
152            )*
153        }
154    ) => {
155        $(#[$outer])*
156        #[repr(Rust, packed)]
157        $vis struct $BitFlags($T);
158
159        bitflags::bitflags! {
160            impl $BitFlags: $T {
161                $(
162                    $(#[$inner $($args)*])*
163                    const $Flag = $value;
164                )*
165            }
166        }
167
168        bitflags!(__impl_serde $BitFlags: $T);
169    };
170    (__impl_serde $BitFlags:ident: $T:tt) => {
171        impl<'de> serde::de::Deserialize<'de> for $BitFlags {
172            fn deserialize<D: serde::de::Deserializer<'de>>(deserializer: D) -> std::result::Result<Self, D::Error> {
173                Ok(Self::from_bits_truncate(u64::deserialize(deserializer)? as $T))
174            }
175        }
176
177        impl serde::ser::Serialize for $BitFlags {
178            fn serialize<S: serde::ser::Serializer>(&self, serializer: S) -> std::result::Result<S::Ok, S::Error> {
179                self.bits().serialize(serializer)
180            }
181        }
182    };
183}

185/// The `http_response_codes!` macro generates `const` definitions and methods for incorporating
186/// Discord-defined [`HttpResponseCode`]s and meanings into error messages.
187///
188/// [`HttpResponseCode`]: crate::http::HttpResponseCode
189macro_rules! http_response_codes {
190    (
191        $(
192            $(#[$docs:meta])*
193            ($code:expr, $konst:ident, $message:expr);
194        )+
195    ) => {
196        impl HttpResponseCode {
197        $(
198            $(#[$docs])*
199            pub const $konst: Self = Self($code);
200        )+
201
202        /// Returns the Discord-defined meaning of this HTTP response code, if available.
203        #[must_use]
204        pub const fn to_meaning(&self) -> Option<&'static str> {
205            Self::meaning_of(self.0)
206        }
207
208        /// Returns an HTTP response code returned by the API with its Discord-defined meaning,
209        /// if available.
210        #[must_use]
211        pub const fn meaning_of(code: u16) -> Option<&'static str> {
212            match code {
213                $(
214                $code => Some($message),
215                )+
216                503 | 504 => HttpResponseCode::meaning_of(500),
217                _ => None
218            }
219        }
220
221        }
222
223        impl From<StatusCode> for HttpResponseCode {
224            fn from(status_code: StatusCode) -> Self {
225                Self(status_code.as_u16())
226            }
227        }
228    }
229}

231#[cfg(test)]
232mod tests {
233    use serde_json::json;
234
235    use crate::model::assert_json;
236
237    #[test]
238    fn enum_number() {
239        enum_number! {
240            #[derive(Copy, Clone, Debug, PartialEq, Eq, Deserialize, Serialize)]
241            pub enum T {
242                /// AAA
243                A = 1,
244                /// BBB
245                B = 2,
246                /// CCC
247                C = 3,
248                _ => Unknown(u8),
249            }
250        }
251
252        assert_json(&T::A, json!(1));
253        assert_json(&T::B, json!(2));
254        assert_json(&T::C, json!(3));
255        assert_json(&T::Unknown(123), json!(123));
256
257        assert_eq!(T::A.as_str(), "A");
258        assert_eq!(T::B.as_str(), "B");
259        assert_eq!(T::C.as_str(), "C");
260        assert_eq!(T::Unknown(123).as_str(), "Unknown");
261    }
