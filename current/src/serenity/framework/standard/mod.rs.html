46/// An enum representing all possible fail conditions under which a command won't be executed.
47#[derive(Debug)]
48#[non_exhaustive]
49pub enum DispatchError {
50    /// When a custom function check has failed.
51    CheckFailed(&'static str, Reason),
52    /// When the command caller has exceeded a ratelimit bucket.
53    Ratelimited(RateLimitInfo),
54    /// When the requested command is disabled in bot configuration.
55    CommandDisabled,
56    /// When the user is blocked in bot configuration.
57    BlockedUser,
58    /// When the guild or its owner is blocked in bot configuration.
59    BlockedGuild,
60    /// When the channel blocked in bot configuration.
61    BlockedChannel,
62    /// When the requested command can only be used in a direct message or group
63    /// channel.
64    OnlyForDM,
65    /// When the requested command can only be ran in guilds, or the bot doesn't
66    /// support DMs.
67    OnlyForGuilds,
68    /// When the requested command can only be used by bot owners.
69    OnlyForOwners,
70    /// When the requested command requires one role.
71    LackingRole,
72    /// When the command requester lacks specific required permissions.
73    LackingPermissions(Permissions),
74    /// When there are too few arguments.
75    NotEnoughArguments { min: u16, given: usize },
76    /// When there are too many arguments.
77    TooManyArguments { max: u16, given: usize },
78}

171    /// Defines a bucket with `delay` between each command, and the `limit` of uses per
172    /// `time_span`.
173    ///
174    /// # Examples
175    ///
176    /// Create and use a bucket that limits a command to 3 uses per 10 seconds with a 2 second
177    /// delay in between invocations:

181    /// use serenity::framework::standard::{BucketBuilder, CommandResult, StandardFramework};

202    /// Whether the message should be ignored because it is from a bot or webhook.
203    fn should_ignore(&self, msg: &Message) -> bool {
204        let config = self.config.read();
205
206        (config.ignore_bots && msg.author.bot)
207            || (config.ignore_webhooks && msg.webhook_id.is_some())
208    }

210    async fn should_fail<'a>(
211        &'a self,
212        ctx: &'a Context,
213        msg: &'a Message,
214        args: &'a mut Args,
215        command: &'static CommandOptions,
216        group: &'static GroupOptions,
217    ) -> Option<DispatchError> {
218        if let Some(min) = command.min_args {
219            if args.len() < min as usize {
220                return Some(DispatchError::NotEnoughArguments {
221                    min,
222                    given: args.len(),
223                });
224            }
225        }
226
227        if let Some(max) = command.max_args {
228            if args.len() > max as usize {
229                return Some(DispatchError::TooManyArguments {
230                    max,
231                    given: args.len(),
232                });
233            }
234        }
235
236        {
237            let config = self.config.read();
238            if (group.owner_privilege && command.owner_privilege)
239                && config.owners.contains(&msg.author.id)
240            {
241                return None;
242            }
243
244            if config.blocked_users.contains(&msg.author.id) {
245                return Some(DispatchError::BlockedUser);
246            }
247
248            #[cfg(feature = "cache")]
249            {
250                if let Some(channel) = msg.channel_id.to_channel_cached(&ctx.cache) {
251                    let guild_id = channel.guild_id;
252
253                    if config.blocked_guilds.contains(&guild_id) {
254                        return Some(DispatchError::BlockedGuild);
255                    }
256
257                    if let Some(guild) = ctx.cache.guild(guild_id) {
258                        if config.blocked_users.contains(&guild.owner_id) {
259                            return Some(DispatchError::BlockedGuild);
260                        }
261                    }
262                }
263            }
264
265            if !config.allowed_channels.is_empty()
266                && !config.allowed_channels.contains(&msg.channel_id)
267            {
268                return Some(DispatchError::BlockedChannel);
269            }
270        }
271
272        // Try passing the command's bucket, exiting the loop if no command ratelimit has been hit
273        // or early-return when ratelimits cancel the framework invocation. Otherwise, delay and
274        // loop again to check if we passed the bucket.
275        loop {
276            let mut duration = None;
277
278            {
279                let mut buckets = self.buckets.lock().await;
280
281                if let Some(bucket) = command.bucket.and_then(|b| buckets.get_mut(b)) {
282                    if let Some(rate_limit_info) = bucket.take(ctx, msg).await {
283                        duration = match rate_limit_info.action {
284                            RateLimitAction::Cancelled | RateLimitAction::FailedDelay => {
285                                return Some(DispatchError::Ratelimited(rate_limit_info))
286                            },
287                            RateLimitAction::Delayed => Some(rate_limit_info.rate_limit),
288                        };
289                    }
290                }
291            }
292
293            match duration {
294                Some(duration) => sleep(duration).await,
295                None => break,
296            }
297        }
298
299        for check in group.checks.iter().chain(command.checks.iter()) {
300            let res = (check.function)(ctx, msg, args, command).await;
301
302            if let Result::Err(reason) = res {
303                return Some(DispatchError::CheckFailed(check.name, reason));
304            }
305        }
306
307        None
308    }

311    /// when using [`serenity::framework::standard::help_commands`].
312    ///
313    /// # Examples
314    ///
315    /// Add a group with ping and pong commands:
316    ///
317    /// ```rust,no_run
318    /// # use serenity::prelude::*;
319    /// # use std::error::Error as StdError;
320    /// # struct Handler;
321    /// #
322    /// # impl EventHandler for Handler {}
323    /// #
324    /// use serenity::client::{Client, Context};
325    /// use serenity::model::channel::Message;
326    /// use serenity::framework::standard::{
327    ///     StandardFramework,
328    ///     CommandResult,
329    ///     macros::{command, group},
330    /// };
331    ///
332    /// // For information regarding this macro, learn more about it in its documentation in `command_attr`.
333    /// #[command]
334    /// async fn ping(ctx: &Context, msg: &Message) -> CommandResult {
335    ///     msg.channel_id.say(&ctx.http, "pong!").await?;
336    ///
337    ///     Ok(())
338    /// }
339    ///
340    /// #[command]
341    /// async fn pong(ctx: &Context, msg: &Message) -> CommandResult {
342    ///     msg.channel_id.say(&ctx.http, "ping!").await?;
343    ///
344    ///     Ok(())
345    /// }
346    ///
347    /// #[group("bingbong")]
348    /// #[commands(ping, pong)]
349    /// struct BingBong;
350    ///
351    /// let framework = StandardFramework::new()
352    ///     // Groups' names are changed to all uppercase, plus appended with `_GROUP`.
353    ///     .group(&BINGBONG_GROUP);
354    /// ```
355    ///

357    #[must_use]
358    pub fn group(mut self, group: &'static CommandGroup) -> Self {
359        self.group_add(group);
360        self.initialized = true;
361
362        self
363    }

365    /// Adds a group to be used by the framework. Primary use-case is runtime modification of
366    /// groups in the framework; will _not_ mark the framework as initialized. Refer to
367    /// [`Self::group`] for adding groups in initial configuration.
368    ///
369    /// Note: does _not_ return [`Self`] like many other commands. This is because it's not
370    /// intended to be chained as the other commands are.
371    pub fn group_add(&mut self, group: &'static CommandGroup) {
372        let config = self.config.read();
373        let map = if group.options.prefixes.is_empty() {
374            Map::Prefixless(
375                GroupMap::new(group.options.sub_groups, &config),
376                CommandMap::new(group.options.commands, &config),
377            )
378        } else {
379            Map::WithPrefixes(GroupMap::new(&[group], &config))
380        };
381
382        self.groups.push((group, map));
383    }

385    /// Removes a group from being used in the framework. Primary use-case is runtime modification
386    /// of groups in the framework.
387    ///
388    /// Note: does _not_ return [`Self`] like many other commands. This is because it's not
389    /// intended to be chained as the other commands are.
390    pub fn group_remove(&mut self, group: &'static CommandGroup) {
391        // Iterates through the vector and if a given group _doesn't_ match, we retain it
392        self.groups.retain(|&(g, _)| g != group);
393    }

610impl Framework for StandardFramework {
611    #[instrument(skip(self, event))]
612    async fn dispatch(&self, mut ctx: Context, event: FullEvent) {
613        let FullEvent::Message {
614            new_message: msg,
615        } = event
616        else {
617            return;
618        };
619
620        if self.should_ignore(&msg) {
621            return;
622        }
623
624        let mut stream = Stream::new(&msg.content);
625
626        stream.take_while_char(char::is_whitespace);
627
628        let config = self.config.read().clone();
629
630        let prefix = parse::prefix(&ctx, &msg, &mut stream, &config).await;
631
632        if prefix.is_some() && stream.rest().is_empty() {
633            if let Some(prefix_only) = &self.prefix_only {
634                prefix_only(&mut ctx, &msg).await;
635            }
636
637            return;
638        }
639
640        if prefix.is_none() && !(config.no_dm_prefix && msg.is_private()) {
641            if let Some(normal) = &self.normal_message {
642                normal(&mut ctx, &msg).await;
643            }
644
645            return;
646        }
647
648        let invocation = parse::command(
649            &ctx,
650            &msg,
651            &mut stream,
652            &self.groups,
653            &config,
654            self.help.map(|h| h.options.names),
655        )
656        .await;
657
658        let invoke = match invocation {
659            Ok(i) => i,
660            Err(ParseError::UnrecognisedCommand(unreg)) => {
661                if let Some(unreg) = unreg {
662                    if let Some(unrecognised_command) = &self.unrecognised_command {
663                        unrecognised_command(&mut ctx, &msg, &unreg).await;
664                    }
665                }
666
667                if let Some(normal) = &self.normal_message {
668                    normal(&mut ctx, &msg).await;
669                }
670
671                return;
672            },
673            Err(ParseError::Dispatch {
674                error,
675                command_name,
676            }) => {
677                if let Some(dispatch) = &self.dispatch {
678                    dispatch(&mut ctx, &msg, error, &command_name).await;
679                }
680
681                return;
682            },
683        };
684
685        match invoke {
686            Invoke::Help(name) => {
687                if !config.allow_dm && msg.is_private() {
688                    return;
689                }
690
691                let args = Args::new(stream.rest(), &config.delimiters);
692
693                let groups = self.groups.iter().map(|(g, _)| *g).collect::<Vec<_>>();
694
695                // `parse_command` promises to never return a help invocation if
696                // `StandardFramework::help` is `None`.
697                #[allow(clippy::unwrap_used)]
698                let help = self.help.unwrap();
699
700                if let Some(before) = &self.before {
701                    if !before(&mut ctx, &msg, name).await {
702                        return;
703                    }
704                }
705
706                let res =
707                    (help.fun)(&mut ctx, &msg, args, help.options, &groups, config.owners).await;
708
709                if let Some(after) = &self.after {
710                    after(&mut ctx, &msg, name, res).await;
711                }
712            },
713            Invoke::Command {
714                command,
715                group,
716            } => {
717                let mut args = {
718                    use std::borrow::Cow;
719
720                    let mut delims = Cow::Borrowed(&config.delimiters);
721
722                    // If user has configured the command's own delimiters, use those instead.
723                    if !command.options.delimiters.is_empty() {
724                        // FIXME: Get rid of this allocation.
725                        let mut v = Vec::with_capacity(command.options.delimiters.len());
726
727                        for delim in command.options.delimiters {
728                            if delim.len() == 1 {
729                                // Should always be Some() in this case
730                                #[allow(clippy::unwrap_used)]
731                                v.push(Delimiter::Single(delim.chars().next().unwrap()));
732                            } else {
733                                // This too.
734                                v.push(Delimiter::Multiple((*delim).to_string()));
735                            }
736                        }
737
738                        delims = Cow::Owned(v);
739                    }
740
741                    Args::new(stream.rest(), &delims)
742                };
743
744                if let Some(error) =
745                    self.should_fail(&ctx, &msg, &mut args, command.options, group.options).await
746                {
747                    if let Some(dispatch) = &self.dispatch {
748                        let command_name = command.options.names[0];
749                        dispatch(&mut ctx, &msg, error, command_name).await;
750                    }
751
752                    return;
753                }
754
755                let name = command.options.names[0];
756
757                if let Some(before) = &self.before {
758                    if !before(&mut ctx, &msg, name).await {
759                        return;
760                    }
761                }
762
763                let res = (command.fun)(&mut ctx, &msg, args).await;
764
765                // Check if the command wants to revert the bucket by giving back a ticket.
766                if matches!(&res, Err(e) if e.is::<RevertBucket>()) {
767                    let mut buckets = self.buckets.lock().await;
768
769                    if let Some(bucket) = command.options.bucket.and_then(|b| buckets.get_mut(b)) {
770                        bucket.give(&ctx, &msg).await;
771                    }
772                }
773
774                if let Some(after) = &self.after {
775                    after(&mut ctx, &msg, name, res).await;
776                }
777            },
778        }
779    }
780}

852#[cfg(feature = "cache")]
853pub(crate) fn has_correct_permissions(
854    cache: impl AsRef<Cache>,
855    options: &impl CommonOptions,
856    message: &Message,
857) -> bool {
858    if options.required_permissions().is_empty() {
859        true
860    } else if let Some(permissions) = message.author_permissions(cache) {
861        permissions.contains(*options.required_permissions())
862    } else {
863        false
864    }
865}
866
867#[cfg(all(feature = "cache", feature = "http"))]
868pub(crate) fn has_correct_roles(
869    options: &impl CommonOptions,
870    roles: &HashMap<RoleId, Role>,
871    member: &Member,
872) -> bool {
873    if options.allowed_roles().is_empty() {
874        true
875    } else {
876        options
877            .allowed_roles()
878            .iter()
879            .filter_map(|r| roles.values().find(|role| *r == role.name))
880            .any(|g| member.roles.contains(&g.id))
881    }
882}
