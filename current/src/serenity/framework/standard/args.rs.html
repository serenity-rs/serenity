9/// Defines how an operation on an [`Args`] method failed.
10#[derive(Debug)]
11#[non_exhaustive]
12pub enum Error<E> {
13    /// "END-OF-STRING". We reached the end. There's nothing to parse anymore.
14    Eos,
15    /// Parsing operation failed. Contains how it did.
16    Parse(E),
17}
18
19impl<E> From<E> for Error<E> {
20    fn from(e: E) -> Self {
21        Error::Parse(e)
22    }
23}

25impl<E: fmt::Display> fmt::Display for Error<E> {
26    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
27        match self {
28            Self::Eos => f.write_str(r#"ArgError("end of string")"#),
29            Self::Parse(e) => write!(f, "ArgError(\"{e}\")"),
30        }
31    }
32}

106// A utility enum to handle an edge case with Apple OSs.
107//
108// By default, a feature called "Smart Quotes" is enabled on MacOS and iOS devices. This feature
109// automatically substitutes the lame, but simple `"` ASCII character for quotation with the cool
110// `”` Unicode character. It can be disabled, but users may not want to do that as it is a global
111// setting (i.e. they might not want to disable it just for properly invoking commands of bots on
112// Discord).
113#[derive(Clone, Copy)]
114enum QuoteKind {
115    Ascii,
116    Apple,
117}
118
119impl QuoteKind {
120    fn new(c: char) -> Option<Self> {
121        match c {
122            '"' => Some(QuoteKind::Ascii),
123            '\u{201C}' => Some(QuoteKind::Apple),
124            _ => None,
125        }
126    }
127
128    fn is_ending_quote(self, c: char) -> bool {
129        match self {
130            Self::Ascii => c == '"',
131            Self::Apple => c == '\u{201D}',
132        }
133    }
134}

136fn lex(stream: &mut Stream<'_>, delims: &[Cow<'_, str>]) -> Option<Token> {
137    if stream.is_empty() {
138        return None;
139    }
140
141    let start = stream.offset();
142    if let Some(kind) = QuoteKind::new(stream.current_char()?) {
143        stream.next_char();
144
145        let mut prev_was_backslash = false;
146        stream.take_until_char(|c| {
147            let result = kind.is_ending_quote(c) && !prev_was_backslash;
148            prev_was_backslash = c == '\\';
149            result
150        });
151
152        let is_quote = stream.current_char().is_some_and(|c| kind.is_ending_quote(c));
153        stream.next_char();
154
155        let end = stream.offset();
156
157        // Remove possible delimiters after the quoted argument.
158        for delim in delims {
159            stream.eat(delim);
160        }
161
162        return Some(if is_quote {
163            Token::new(TokenKind::QuotedArgument, start, end)
164        } else {
165            // We're missing an end quote. View this as a normal argument.
166            Token::new(TokenKind::Argument, start, stream.len())
167        });
168    }
169
170    let mut end = start;
171
172    'outer: while !stream.is_empty() {
173        for delim in delims {
174            end = stream.offset();
175
176            if stream.eat(delim) {
177                break 'outer;
178            }
179        }
180
181        stream.next_char();
182        end = stream.offset();
183    }
184
185    Some(Token::new(TokenKind::Argument, start, end))
186}

188fn is_surrounded_with(s: &str, begin: char, end: char) -> bool {
189    s.starts_with(begin) && s.ends_with(end)
190}
191
192fn is_quoted(s: &str) -> bool {
193    if s.len() < 2 {
194        return false;
195    }
196
197    // Refer to `QuoteKind` why we check for Unicode quote characters.
198    is_surrounded_with(s, '"', '"') || is_surrounded_with(s, '\u{201C}', '\u{201D}')
199}
200
201fn strip(s: &str, begin: char, end: char) -> Option<&str> {
202    let s = s.strip_prefix(begin)?;
203    s.strip_suffix(end)
204}
205
206fn remove_quotes(s: &str) -> &str {
207    if s.len() < 2 {
208        return s;
209    }
210
211    if let Some(s) = strip(s, '"', '"') {
212        return s;
213    }
214
215    // Refer to `QuoteKind` why we check for Unicode quote characters.
216    strip(s, '\u{201C}', '\u{201D}').unwrap_or(s)
217}

229/// A utility struct for handling "arguments" of a command.
230///
231/// An "argument" is a part of the message up that ends at one of the specified delimiters, or the
232/// end of the message.
233///
234/// # Example
235///
236/// ```rust
237/// use serenity::framework::standard::{Args, Delimiter};
238///
239/// let mut args = Args::new("hello world!", &[Delimiter::Single(' ')]); // A space is our delimiter.
240///
241/// // Parse our argument as a `String` and assert that it's the "hello" part of the message.
242/// assert_eq!(args.single::<String>().unwrap(), "hello");
243/// // Same here.
244/// assert_eq!(args.single::<String>().unwrap(), "world!");
245/// ```
246///
247/// We can also parse "quoted arguments" (no pun intended):
248///
249/// ```rust
250/// use serenity::framework::standard::{Args, Delimiter};
251///
252/// // Let us imagine this scenario:
253/// // You have a `photo` command that grabs the avatar url of a user. This command accepts names only.
254/// // Now, one of your users wants the avatar of a user named Princess Zelda.
255/// // Problem is, her name contains a space; our delimiter. This would result in two arguments, "Princess" and "Zelda".
256/// // So how shall we get around this? Through quotes! By surrounding her name in them we can perceive it as one single argument.
257/// let mut args = Args::new(r#""Princess Zelda""#, &[Delimiter::Single(' ')]);
258///
259/// // Hooray!
260/// assert_eq!(args.single_quoted::<String>().unwrap(), "Princess Zelda");
261/// ```
262///
263/// In case of a mistake, we can go back in time... er I mean, one step (or entirely):
264///
265/// ```rust
266/// use serenity::framework::standard::{Args, Delimiter};
267///
268/// let mut args = Args::new("4 2", &[Delimiter::Single(' ')]);
269///
270/// assert_eq!(args.single::<u32>().unwrap(), 4);
271///
272/// // Oh wait, oops, meant to double the 4.
273/// // But I won't able to access it now...
274/// // oh wait, I can `rewind`.
275/// args.rewind();
276///
277/// assert_eq!(args.single::<u32>().unwrap() * 2, 8);
278///
279/// // And the same for the 2
280/// assert_eq!(args.single::<u32>().unwrap() * 2, 4);
281///
282/// // WAIT, NO. I wanted to concatenate them into a "42" string...
283/// // Argh, what should I do now????
284/// // ....
285/// // oh, `restore`
286/// args.restore();
287///
288/// let res = format!("{}{}", args.single::<String>().unwrap(), args.single::<String>().unwrap());
289///
290/// // Yay.
291/// assert_eq!(res, "42");
292/// ```
293///
294/// Hmm, taking a glance at the prior example, it seems we have an issue with reading the same
295/// argument over and over. Is there a more sensible solution than rewinding...? Actually, there is!
296/// The [`Self::current`] and [`Self::parse`] methods:
297///
298/// ```rust
299/// use serenity::framework::standard::{Args, Delimiter};
300///
301/// let mut args = Args::new("trois cinq quatre six", &[Delimiter::Single(' ')]);
302///
303/// assert_eq!(args.parse::<String>().unwrap(), "trois");
304///
305/// // It might suggest we've lost the `trois`. But in fact, we didn't! And not only that, we can do it an infinite amount of times!
306/// assert_eq!(args.parse::<String>().unwrap(), "trois");
307/// assert_eq!(args.current(), Some("trois"));
308/// assert_eq!(args.parse::<String>().unwrap(), "trois");
309/// assert_eq!(args.current(), Some("trois"));
310///
311/// // Only if we use its brother method we'll then lose it.
312/// assert_eq!(args.single::<String>().unwrap(), "trois");
313/// assert_eq!(args.single::<String>().unwrap(), "cinq");
314/// assert_eq!(args.single::<String>().unwrap(), "quatre");
315/// assert_eq!(args.single::<String>().unwrap(), "six");
316/// ```
317#[derive(Clone, Debug)]

319    message: String,
320    tokens: Vec<Token>,
321    offset: usize,
322    state: State,
323}

326    /// Create a new instance of [`Args`] for parsing arguments.
327    ///
328    /// For more reference, look at [`Args`]'s struct documentation.
329    ///
330    /// # Example
331    ///
332    /// ```rust
333    /// use serenity::framework::standard::{Args, Delimiter};
334    ///
335    /// let mut args = Args::new(
336    /// // Our message from which we'll parse over.
337    /// "the quick brown fox jumps over the lazy",
338    ///
339    /// // The "delimiters", or aka the separators. They denote how we distinguish arguments as their own.
340    /// // For this example, we'll use one delimiter, the space (`0x20`), which will separate the message.
341    /// &[Delimiter::Single(' ')],
342    /// );
343    ///
344    /// assert_eq!(args.single::<String>().unwrap(), "the");
345    /// assert_eq!(args.single::<String>().unwrap(), "quick");
346    /// assert_eq!(args.single::<String>().unwrap(), "brown");
347    ///
348    /// // We shall not see `the quick brown` again.
349    /// assert_eq!(args.rest(), "fox jumps over the lazy");
350    /// ```
351    #[must_use]
352    pub fn new(message: &str, possible_delimiters: &[Delimiter]) -> Self {
353        let delims = possible_delimiters
354            .iter()
355            .filter(|d| match d {
356                Delimiter::Single(c) => message.contains(*c),
357                Delimiter::Multiple(s) => message.contains(s),
358            })
359            .map(Delimiter::to_str)
360            .collect::<Vec<_>>();
361
362        let tokens = if delims.is_empty() {
363            let msg = message.trim();
364            let kind = if is_quoted(msg) { TokenKind::QuotedArgument } else { TokenKind::Argument };
365
366            if msg.is_empty() {
367                Vec::new()
368            } else {
369                // If there are no delimiters, then the only possible argument is the whole
370                // message.
371                vec![Token::new(kind, 0, message.len())]
372            }
373        } else {
374            let mut args = Vec::new();
375            let mut stream = Stream::new(message);
376
377            while let Some(token) = lex(&mut stream, &delims) {
378                // Ignore empty arguments.
379                if message[token.span.0..token.span.1].is_empty() {
380                    continue;
381                }
382
383                args.push(token);
384            }
385
386            args
387        };
388
389        Args {
390            tokens,
391            message: message.to_string(),
392            offset: 0,
393            state: State::None,
394        }
395    }

442    fn apply<'a>(&self, s: &'a str) -> &'a str {
443        fn trim(s: &str) -> &str {
444            let trimmed = s.trim();
445
446            // Search where the argument starts and ends between the whitespace.
447            let start = s.find(trimmed).unwrap_or(0);
448            let end = start + trimmed.len();
449
450            &s[start..end]
451        }
452
453        let mut s = s;
454
455        match self.state {
456            State::None => {},
457            State::Quoted => {
458                s = remove_quotes(s);
459            },
460            State::Trimmed => {
461                s = trim(s);
462            },
463            State::QuotedTrimmed => {
464                s = remove_quotes(s);
465                s = trim(s);
466            },
467            State::TrimmedQuoted => {
468                s = trim(s);
469                s = remove_quotes(s);
470            },
471        }
472
473        s
474    }

476    /// Retrieve the current argument.
477    ///
478    /// Applies modifications set by [`Self::trimmed`] and [`Self::quoted`].
479    ///
480    /// # Note
481    ///
482    /// This borrows [`Args`] for the entire lifetime of the returned argument.
483    ///
484    /// # Examples
485    ///
486    /// ```rust
487    /// use serenity::framework::standard::{Args, Delimiter};
488    ///
489    /// let mut args = Args::new("4 2", &[Delimiter::Single(' ')]);
490    ///
491    /// assert_eq!(args.current(), Some("4"));

495    /// assert_eq!(args.current(), None);

500        if self.is_empty() {
501            return None;
502        }
503
504        let mut s = self.slice();
505        s = self.apply(s);
506
507        Some(s)
508    }

510    /// Apply trimming of whitespace to all arguments.
511    ///
512    /// # Examples
513    ///
514    /// ```rust
515    /// use serenity::framework::standard::Args;
516    ///
517    /// let mut args = Args::new("     42     ", &[]);

528        match self.state {

552    /// Remove quotations surrounding all arguments.
553    ///
554    /// Note that only the quotes of the argument are taken into account. The quotes in the message
555    /// are preserved.
556    ///
557    /// # Examples
558    ///
559    /// ```rust
560    /// use serenity::framework::standard::Args;
561    ///
562    /// let mut args = Args::new("\"42\"", &[]);
563    ///
564    /// // `quoted` lasts the whole lifetime of `Args`
565    /// args.quoted();
566    /// assert_eq!(args.current(), Some("42"));
567    /// // or until we decide
568    /// args.unquoted();
569    /// assert_eq!(args.current(), Some("\"42\""));
570    /// assert_eq!(args.message(), "\"42\"");
571    /// ```
572    pub fn quoted(&mut self) -> &mut Self {
573        if self.is_empty() {
574            return self;
575        }
576
577        let is_quoted = self.tokens[self.offset].kind == TokenKind::QuotedArgument;
578
579        if is_quoted {
580            match self.state {
581                State::None => self.state = State::Quoted,
582                State::Trimmed => self.state = State::TrimmedQuoted,
583                _ => {},
584            }
585        }
586
587        self
588    }

597            State::Quoted => self.state = State::None,
598            State::QuotedTrimmed | State::TrimmedQuoted => self.state = State::Trimmed,

742    /// Return an iterator over all arguments, stripped of their quotations if any were present.
743    ///
744    /// # Examples
745    ///
746    /// ```rust
747    /// use serenity::framework::standard::{Args, Delimiter};
748    ///
749    /// let args = Args::new("Saw \"The Mist\" \"A Quiet Place\"", &[Delimiter::Single(' ')]);
750    ///
751    /// let horror_movies = args.raw_quoted().collect::<Vec<&str>>();
752    ///
753    /// assert_eq!(&*horror_movies, &["Saw", "The Mist", "A Quiet Place"]);
754    /// ```
755    #[inline]
756    #[must_use]
757    pub fn raw_quoted(&self) -> RawArguments<'_> {
758        let mut raw = self.raw();
759        raw.quoted = true;
760        raw
761    }
762
763    /// Search for any available argument that can be parsed, and remove it from the arguments
764    /// queue.
765    ///
766    /// # Note
767    /// The removal is irreversible. And happens after the search *and* the parse were successful.
768    ///
769    /// # Note 2
770    /// "Arguments queue" is the list which contains all arguments that were deemed unique as
771    /// defined by quotations and delimiters. The 'removed' argument can be, likewise, still
772    /// accessed via [`Self::message`].
773    ///
774    /// # Examples
775    ///
776    /// ```rust
777    /// use serenity::framework::standard::{Args, Delimiter};
778    ///
779    /// let mut args = Args::new("c4 2", &[Delimiter::Single(' ')]);
780    ///
781    /// assert_eq!(args.find::<u32>().unwrap(), 2);
782    /// assert_eq!(args.single::<String>().unwrap(), "c4");
783    /// assert!(args.is_empty());
784    /// ```
785    ///
786    /// # Errors
787    ///
788    /// Returns [`Error::Eos`] if no argument can be parsed.
789    pub fn find<T: FromStr>(&mut self) -> Result<T, T::Err> {
790        if self.is_empty() {
791            return Err(Error::Eos);
792        }
793
794        let before = self.offset;
795        self.restore();
796
797        let Some(pos) = self.iter::<T>().quoted().position(|res| res.is_ok()) else {
798            self.offset = before;
799            return Err(Error::Eos);
800        };
801
802        self.offset = pos;
803        let parsed = self.single_quoted::<T>()?;
804
805        self.tokens.remove(pos);
806        self.offset = before;
807        self.rewind();
808
809        Ok(parsed)
810    }

812    /// Search for any available argument that can be parsed.
813    ///
814    /// # Examples
815    ///
816    /// ```rust
817    /// use serenity::framework::standard::{Args, Delimiter};
818    ///
819    /// let mut args = Args::new("c4 2", &[Delimiter::Single(' ')]);
820    ///
821    /// assert_eq!(args.find_n::<u32>().unwrap(), 2);
822    ///
823    /// // The `2` is still here, so let's parse it again.
824    /// assert_eq!(args.single::<String>().unwrap(), "c4");
825    /// assert_eq!(args.single::<u32>().unwrap(), 2);
826    /// assert!(args.is_empty());
827    /// ```
828    ///
829    /// # Errors
830    ///
831    /// Returns [`Error::Eos`] if no argument can be parsed.
832    pub fn find_n<T: FromStr>(&mut self) -> Result<T, T::Err> {
833        if self.is_empty() {
834            return Err(Error::Eos);
835        }
836
837        let before = self.offset;
838        self.restore();
839
840        let Some(pos) = self.iter::<T>().quoted().position(|res| res.is_ok()) else {
841            self.offset = before;
842            return Err(Error::Eos);
843        };
844
845        self.offset = pos;
846        let parsed = self.quoted().parse::<T>()?;
847
848        self.offset = before;
849
850        Ok(parsed)
851    }

853    /// Get the original, unmodified message passed to the command.
854    #[inline]
855    #[must_use]
856    pub fn message(&self) -> &str {
857        &self.message
858    }
859
860    /// Starting from the offset, return the remainder of available arguments.
861    #[inline]
862    #[must_use]
863    pub fn rest(&self) -> &str {
864        self.remains().unwrap_or_default()
865    }
866
867    /// Starting from the offset, return the remainder of available arguments.
868    ///
869    /// Returns [`None`] if there are no remaining arguments.
870    #[inline]
871    #[must_use]
872    pub fn remains(&self) -> Option<&str> {
873        if self.is_empty() {
874            return None;
875        }
876
877        let (start, _) = self.span();
878
879        Some(&self.message[start..])
880    }
881
882    /// Return the full amount of recognised arguments. The length of the "arguments queue".
883    ///
884    /// # Note
885    ///
886    /// The value returned is to be assumed to stay static. However, if [`Self::find`] was called
887    /// previously, and was successful, then the value is subtracted by one.
888    #[inline]
889    #[must_use]
890    pub fn len(&self) -> usize {
891        self.tokens.len()
892    }
893
894    /// Assert that there are no more arguments left.
895    #[inline]
896    #[must_use]
897    pub fn is_empty(&self) -> bool {
898        self.offset >= self.len()
899    }
900
901    /// Return the amount of arguments still available.
902    #[inline]
903    #[must_use]
904    pub fn remaining(&self) -> usize {
905        if self.is_empty() {
906            return 0;
907        }
908
909        self.len() - self.offset
910    }

913/// Parse each argument individually, as an iterator.
914pub struct Iter<'a, T: FromStr> {
915    args: &'a mut Args,
916    state: State,
917    _marker: PhantomData<T>,
918}
919
920#[allow(clippy::missing_errors_doc)]
921impl<T: FromStr> Iter<'_, T> {
922    /// Retrieve the current argument.
923    pub fn current(&mut self) -> Option<&str> {
924        self.args.state = self.state;
925        self.args.current()
926    }
927
928    /// Parse the current argument independently.
929    pub fn parse(&mut self) -> Result<T, T::Err> {
930        self.args.state = self.state;
931        self.args.parse::<T>()
932    }
933
934    /// Remove surrounding quotation marks from all of the arguments.
935    #[inline]
936    pub fn quoted(&mut self) -> &mut Self {
937        match self.state {
938            State::None => self.state = State::Quoted,
939            State::Trimmed => self.state = State::TrimmedQuoted,
940            _ => {},
941        }
942
943        self
944    }
945
946    /// Trim leading and trailing whitespace off all arguments.
947    #[inline]
948    pub fn trimmed(&mut self) -> &mut Self {
949        match self.state {
950            State::None => self.state = State::Trimmed,
951            State::Quoted => self.state = State::QuotedTrimmed,
952            _ => {},
953        }
954
955        self
956    }
957}
958
959impl<T: FromStr> Iterator for Iter<'_, T> {
960    type Item = Result<T, T::Err>;
961
962    fn next(&mut self) -> Option<Self::Item> {
963        if self.args.is_empty() {
964            None
965        } else {
966            let arg = self.parse();
967            self.args.advance();
968            Some(arg)
969        }
970    }
971}

973/// Access to all of the arguments, as an iterator.
974#[derive(Debug)]
975pub struct RawArguments<'a> {
976    msg: &'a str,
977    tokens: &'a [Token],
978    quoted: bool,
979}
980
981impl<'a> Iterator for RawArguments<'a> {
982    type Item = &'a str;
983
984    #[inline]
985    fn next(&mut self) -> Option<Self::Item> {
986        let (start, end) = self.tokens.first()?.span;
987
988        self.tokens = &self.tokens[1..];
989
990        let mut s = &self.msg[start..end];
991
992        if self.quoted {
993            s = remove_quotes(s);
994        }
995
996        Some(s)
997    }
998}
